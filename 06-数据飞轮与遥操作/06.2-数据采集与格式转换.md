## 1. 机器人数据格式

### 1.1 数据格式对比

| 格式 | 全称 | 特点 | 适用场景 | 工具支持 |
|------|------|------|---------|---------|
| HDF5 | Hierarchical Data Format v5 | 层次化存储、高效压缩、支持大规模数据 | ALOHA/Teleopit 原始数据 | h5py, PyTorch |
| LeRobot Dataset | LeRobot Dataset v2.1 | Parquet+图像、HF Hub 集成、版本管理 | LeRobot 框架训练 | LeRobot, datasets |
| ROS bag | ROS bagfile | 时间序列、话题记录、ROS 生态 | ROS 机器人数据 | rosbag2, rclpy |
| TFRecord | TensorFlow Record | TF 原生、protobuf 序列化 | TF 训练 | TensorFlow |
| WebDataset | WebDataset | 分片存储、流式加载、云原生 | 大规模分布式训练 | webdataset |
| Zarr | Zarr | 分块存储、并行读写、云存储 | 大规模数组 | zarr |

### 2.2 时间同步

```python
# 文件: data_sync.py
"""多传感器时间同步"""
import numpy as np
import time
from collections import defaultdict

class DataSynchronizer:
    """多源数据时间同步器"""
    
    def __init__(self, target_fps: int = 50, sync_tolerance_ms: float = 10.0):
        self.target_fps = target_fps
        self.target_dt = 1.0 / target_fps
        self.sync_tolerance = sync_tolerance_ms / 1000.0  # 转为秒
        
        # 数据缓冲区
        self.buffers = defaultdict(list)
        self.buffer_max = 1000
    
    def add_data(self, source: str, timestamp: float, data: dict):
        """添加数据到缓冲区"""
        self.buffers[source].append({
            'timestamp': timestamp,
            'data': data,
        })
        
        # 限制缓冲区大小
        if len(self.buffers[source]) > self.buffer_max:
            self.buffers[source] = self.buffers[source][-self.buffer_max:]
    
    def sync(self) -> list:
        """
        时间同步：将不同源的数据对齐到目标时间戳
        
        返回同步后的数据帧列表
        """
        if not self.buffers:
            return []
        
        # 获取所有数据源的时间范围
        all_timestamps = []
        for source, buffer in self.buffers.items():
            if buffer:
                all_timestamps.extend([d['timestamp'] for d in buffer])
        
        if not all_timestamps:
            return []
        
        # 创建目标时间序列
        t_start = min(all_timestamps)
        t_end = max(all_timestamps)
        target_times = np.arange(t_start, t_end, self.target_dt)
        
        synced_frames = []
        
        for t in target_times:
            frame = {'timestamp': t}
            valid = True
            
            for source, buffer in self.buffers.items():
                # 找最近的数据点
                closest = min(buffer, key=lambda x: abs(x['timestamp'] - t))
                
                # 检查是否在容差范围内
                if abs(closest['timestamp'] - t) > self.sync_tolerance:
                    valid = False
                    break
                
                frame[source] = closest['data']
            
            if valid:
                synced_frames.append(frame)
        
        return synced_frames
    
    def interpolate(self, t, buffer):
        """线性插值"""
        # 找到 t 前后的两个数据点
        before = None
        after = None
        
        for item in buffer:
            if item['timestamp'] <= t:
                before = item
            if item['timestamp'] >= t and after is None:
                after = item
        
        if before is None:
            return after['data'] if after else None
        if after is None:
            return before['data']
        
        # 线性插值
        alpha = (t - before['timestamp']) / (after['timestamp'] - before['timestamp'])
        
        result = {}
        for key in before['data']:
            if isinstance(before['data'][key], (int, float)):
                result[key] = before['data'][key] + alpha * (after['data'][key] - before['data'][key])
            elif isinstance(before['data'][key], np.ndarray):
                result[key] = before['data'][key] + alpha * (after['data'][key] - before['data'][key])
            else:
                result[key] = before['data'][key]  # 非数值类型不插值
        
        return result
```

```python
# 文件: calibration.py
"""传感器标定数据采集"""
import numpy as np
import cv2

class CalibrationDataCollector:
    """标定数据采集器"""
    
    def __init__(self):
        self.camera_images = []
        self.robot_poses = []
    
    def collect_camera_intrinsics(self, camera_id=0, num_images=20):
        """采集相机内参标定数据（棋盘格）"""
        cap = cv2.VideoCapture(camera_id)
        
        # 棋盘格参数
        checkerboard = (9, 6)
        criteria = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001)
        
        objpoints = []  # 3D 点
        imgpoints = []  # 2D 点
        
        objp = np.zeros((checkerboard[0] * checkerboard[1], 3), np.float32)
        objp[:, :2] = np.mgrid[0:checkerboard[0], 0:checkerboard[1]].T.reshape(-1, 2)
        objp *= 0.025  # 25mm 方格大小
        
        collected = 0
        while collected < num_images:
            ret, frame = cap.read()
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            
            ret, corners = cv2.findChessboardCorners(gray, checkerboard, None)
            
            if ret:
                corners2 = cv2.cornerSubPix(gray, corners, (11, 11), (-1, -1), criteria)
                imgpoints.append(corners2)
                objpoints.append(objp)
                collected += 1
                
                cv2.drawChessboardCorners(frame, checkerboard, corners2, ret)
            
            cv2.putText(frame, f"Collected: {collected}/{num_images}", (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)
            cv2.imshow('Calibration', frame)
            
            if cv2.waitKey(100) & 0xFF == ord('q'):
                break
        
        cap.release()
        cv2.destroyAllWindows()
        
        # 标定
        ret, mtx, dist, rvecs, tvecs = cv2.calibrateCamera(
            objpoints, imgpoints, gray.shape[::-1], None, None
        )
        
        print(f"标定完成")
        print(f"相机内参:\n{mtx}")
        print(f"畸变系数:\n{dist}")
        print(f"重投影误差: {ret}")
        
        return mtx, dist
    
    def collect_hand_eye(self, robot, camera, num_samples=15):
        """手眼标定数据采集"""
        samples = []
        
        for i in range(num_samples):
            # 移动机器人到不同姿态
            target_joints = np.random.uniform(-0.5, 0.5, 7)
            robot.move_to(target_joints)
            
            # 等待稳定
            time.sleep(2)
            
            # 获取机器人末端位姿
            ee_pose = robot.get_ee_pose()
            
            # 获取相机观测的标定板位姿
            image = camera.capture()
            board_pose = self._detect_board_pose(image)
            
            if board_pose is not None:
                samples.append({
                    'robot_pose': ee_pose,
                    'board_pose': board_pose,
                })
                print(f"采集样本 {i+1}/{num_samples}")
        
        return samples
```

### 3.2 完整转换代码

```python
# 文件: hdf5_to_lerobot.py
"""HDF5 -> LeRobot Dataset v2.1 转换器"""
import h5py
import numpy as np
import json
import os
import cv2
from pathlib import Path
import pyarrow as pa
import pyarrow.parquet as pq
from typing import Optional

class HDF5ToLeRobotConverter:
    """HDF5 数据集转换为 LeRobot Dataset v2.1 格式"""
    
    def __init__(
        self,
        hdf5_dir: str,
        output_dir: str,
        repo_id: str,
        fps: int = 50,
        robot_type: str = "g1",
    ):
        self.hdf5_dir = Path(hdf5_dir)
        self.output_dir = Path(output_dir)
        self.repo_id = repo_id
        self.fps = fps
        self.robot_type = robot_type
        
        # 创建输出目录结构
        self._create_output_structure()
    
    def _create_output_structure(self):
        """创建 LeRobot 目录结构"""
        dirs = [
            self.output_dir / "data",
            self.output_dir / "meta",
            self.output_dir / "videos",
        ]
        for d in dirs:
            d.mkdir(parents=True, exist_ok=True)
    
    def convert(self):
        """执行转换"""
        # 1. 查找所有 HDF5 文件
        hdf5_files = sorted(self.hdf5_dir.glob("*.hdf5"))
        print(f"找到 {len(hdf5_files)} 个 HDF5 文件")
        
        # 2. 解析第一个文件获取特征信息
        features = self._extract_features(hdf5_files[0])
        print(f"数据集特征: {list(features.keys())}")
        
        # 3. 转换每个 episode
        all_episodes_data = []
        episode_infos = []
        total_frames = 0
        
        for ep_idx, hdf5_file in enumerate(hdf5_files):
            print(f"转换 episode {ep_idx}: {hdf5_file.name}")
            
            episode_data = self._convert_episode(hdf5_file, ep_idx, features)
            all_episodes_data.append(episode_data)
            
            episode_infos.append({
                'episode_index': ep_idx,
                'tasks': episode_data['task'],
                'length': episode_data['num_frames'],
            })
            total_frames += episode_data['num_frames']
        
        # 4. 保存 Parquet 文件
        self._save_parquet(all_episodes_data, features)
        
        # 5. 保存视频
        self._save_videos(all_episodes_data, features)
        
        # 6. 保存元数据
        self._save_meta(features, episode_infos, total_frames)
        
        print(f"转换完成！共 {len(hdf5_files)} 集, {total_frames} 帧")
    
    def _extract_features(self, hdf5_file: Path) -> dict:
        """从 HDF5 文件提取特征信息"""
        features = {}
        
        with h5py.File(hdf5_file, 'r') as f:
            # 动作特征
            if 'action' in f:
                for key in f['action']:
                    data = f['action'][key]
                    features[f'action.{key}'] = {
                        'dtype': str(data.dtype),
                        'shape': data.shape[1:],
                    }
            
            # 观测特征
            if 'observation' in f:
                for key in f['observation']:
                    if key == 'images':
                        for img_key in f['observation/images']:
                            features[f'observation.images.{img_key}'] = {
                                'dtype': 'video',
                                'shape': f['observation/images'][img_key].shape[1:],
                            }
                    else:
                        data = f['observation'][key]
                        features[f'observation.{key}'] = {
                            'dtype': str(data.dtype),
                            'shape': data.shape[1:],
                        }
        
        return features
    
    def _convert_episode(self, hdf5_file: Path, ep_idx: int, features: dict) -> dict:
        """转换单个 episode"""
        with h5py.File(hdf5_file, 'r') as f:
            # 获取帧数
            if 'action' in f and len(f['action'].keys()) > 0:
                first_key = list(f['action'].keys())[0]
                num_frames = f['action'][first_key].shape[0]
            elif 'observation' in f:
                first_key = list(f['observation'].keys())[0]
                num_frames = f['observation'][first_key].shape[0]
            else:
                num_frames = 0
            
            # 提取数据
            episode_data = {
                'episode_index': ep_idx,
                'num_frames': num_frames,
                'task': f.attrs.get('task_description', 'unknown'),
                'frames': [],
                'images': {},
            }
            
            # 帧数据（非图像）
            for frame_idx in range(num_frames):
                frame = {
                    'frame_index': total_frame_idx := frame_idx,
                    'episode_index': ep_idx,
                    'timestamp': frame_idx / self.fps,
                    'episode_data_index_from': 0,
                    'episode_data_index_to': num_frames,
                }
                
                # 动作
                if 'action' in f:
                    for key in f['action']:
                        frame[f'action.{key}'] = f['action'][key][frame_idx]
                
                # 观测（非图像）
                if 'observation' in f:
                    for key in f['observation']:
                        if key != 'images':
                            frame[f'observation.{key}'] = f['observation'][key][frame_idx]
                
                episode_data['frames'].append(frame)
            
            # 图像数据
            if 'observation' in f and 'images' in f['observation']:
                for img_key in f['observation/images']:
                    images = f['observation/images'][img_key][:]
                    episode_data['images'][img_key] = images
        
        return episode_data
    
    def _save_parquet(self, all_episodes_data, features):
        """保存为 Parquet 格式"""
        # 合并所有帧
        all_frames = []
        for ep in all_episodes_data:
            all_frames.extend(ep['frames'])
        
        # 转换为 Arrow Table
        columns = {}
        for key in all_frames[0]:
            values = [frame[key] for frame in all_frames]
            if isinstance(values[0], np.ndarray):
                columns[key] = pa.array([v.tolist() for v in values])
            else:
                columns[key] = pa.array(values)
        
        table = pa.table(columns)
        
        # 保存
        parquet_path = self.output_dir / "data" / "train-00000-of-00001.parquet"
        pq.write_table(table, parquet_path)
        print(f"Parquet 数据已保存: {parquet_path}")
    
    def _save_videos(self, all_episodes_data, features):
        """保存图像为视频"""
        for ep in all_episodes_data:
            for img_key, images in ep['images'].items():
                video_dir = self.output_dir / "videos" / f"observation.images.{img_key}"
                video_dir.mkdir(parents=True, exist_ok=True)
                
                video_path = video_dir / f"episode_{ep['episode_index']:06d}.mp4"
                
                # 使用 H.264 编码
                height, width = images.shape[1:3]
                fourcc = cv2.VideoWriter_fourcc(*'mp4v')
                writer = cv2.VideoWriter(
                    str(video_path), fourcc, self.fps, (width, height)
                )
                
                for img in images:
                    # 确保是 BGR 格式（OpenCV）
                    if img.dtype != np.uint8:
                        img = (img * 255).astype(np.uint8)
                    writer.write(img)
                
                writer.release()
        
        print(f"视频数据已保存")
    
    def _save_meta(self, features, episode_infos, total_frames):
        """保存元数据"""
        # info.json
        info = {
            "codebase_version": "v2.1",
            "robot_type": self.robot_type,
            "fps": self.fps,
            "features": features,
            "total_episodes": len(episode_infos),
            "total_frames": total_frames,
            "data_dir": "data/",
        }
        
        with open(self.output_dir / "meta" / "info.json", 'w') as f:
            json.dump(info, f, indent=2)
        
        # episodes.jsonl
        with open(self.output_dir / "meta" / "episodes.jsonl", 'w') as f:
            for ep_info in episode_infos:
                f.write(json.dumps(ep_info) + '\n')
        
        # 计算统计信息
        stats = self._compute_stats(features)
        with open(self.output_dir / "meta" / "stats.json", 'w') as f:
            json.dump(stats, f, indent=2)
        
        print(f"元数据已保存")
    
    def _compute_stats(self, features):
        """计算数据集统计信息"""
        # 读取 Parquet 数据计算统计量
        parquet_path = self.output_dir / "data" / "train-00000-of-00001.parquet"
        table = pq.read_table(parquet_path)
        
        stats = {}
        for key in features:
            if features[key]['dtype'] in ['float32', 'float64']:
                column = table.column(key).to_pylist()
                arr = np.array(column)
                stats[key] = {
                    'mean': arr.mean(axis=0).tolist(),
                    'std': arr.std(axis=0).tolist(),
                    'min': arr.min(axis=0).tolist(),
                    'max': arr.max(axis=0).tolist(),
                }
        
        return stats


# === 使用示例 ===
if __name__ == "__main__":
```

```python
class ImageAugmentation:
    """图像数据增强"""
    
    def __init__(self, 
                 color_jitter: bool = True,
                 gaussian_noise: bool = True,
                 blur: bool = False,
                 brightness_range: float = 0.2,
                 contrast_range: float = 0.2):
        
        self.transforms = []
        
        if color_jitter:
            self.color_jitter = T.ColorJitter(
                brightness=brightness_range,
                contrast=contrast_range,
                saturation=0.2,
                hue=0.1,
            )
        
        self.gaussian_noise = gaussian_noise
        self.blur = blur
    
    def __call__(self, image: np.ndarray) -> np.ndarray:
        """应用增强"""
        # 转 PIL -> Tensor
        if isinstance(image, np.ndarray):
            image = torch.from_numpy(image).permute(2, 0, 1).float() / 255.0
        
        # 颜色抖动
        if hasattr(self, 'color_jitter'):
            image = self.color_jitter(image)
        
        # 高斯噪声
        if self.gaussian_noise and np.random.random() < 0.5:
            noise = torch.randn_like(image) * 0.01
            image = torch.clamp(image + noise, 0, 1)
        
        # 模糊
        if self.blur and np.random.random() < 0.3:
            kernel_size = np.random.choice([3, 5, 7])
            image = T.functional.gaussian_blur(image, kernel_size)
        
        # 转回 numpy
        image = (image.permute(1, 2, 0).numpy() * 255).astype(np.uint8)
        
        return image


class StateAugmentation:
    """状态数据增强"""
    
    def __init__(self,
                 joint_noise_std: float = 0.01,
                 position_noise_std: float = 0.005):
        self.joint_noise_std = joint_noise_std
        self.position_noise_std = position_noise_std
    
    def __call__(self, state: np.ndarray) -> np.ndarray:
        """应用增强"""
        augmented = state.copy()
        
        # 关节角度噪声
        noise = np.random.normal(0, self.joint_noise_std, augmented.shape)
        augmented += noise
        
        return augmented


class DomainRandomization:
    """域随机化（用于 Sim2Real）"""
    
    def __init__(self):
        self.light_range = (0.7, 1.3)
        self.shadow_prob = 0.3
        self.bg_colors = [
            (0.8, 0.8, 0.8), (0.6, 0.6, 0.6),
            (0.9, 0.9, 0.9), (0.5, 0.5, 0.5),
        ]
    
    def randomize_image(self, image: np.ndarray) -> np.ndarray:
        """图像域随机化"""
        # 光照变化
        light_factor = np.random.uniform(*self.light_range)
        image = np.clip(image * light_factor, 0, 255).astype(np.uint8)
        
        # 阴影
        if np.random.random() < self.shadow_prob:
            h, w = image.shape[:2]
            shadow = np.zeros((h, w), dtype=np.float32)
            x1, y1 = np.random.randint(0, w), np.random.randint(0, h)
            x2, y2 = np.random.randint(0, w), np.random.randint(0, h)
            cv2.rectangle(shadow, (min(x1,x2), min(y1,y2)), 
                         (max(x1,x2), max(y1,y2)), 0.5, -1)
            image = (image * shadow[:,:,np.newaxis]).astype(np.uint8)
        
        return image
    
    def randomize_physics(self, robot_params: dict) -> dict:
        """物理参数随机化"""
        randomized = robot_params.copy()
```

```python
"""数据质量评估"""
import numpy as np
from typing import Optional

class DataQualityAssessor:
    """数据质量评估器"""
    
    def __init__(self):
        self.metrics = {}
    
    def assess(self, dataset: dict) -> dict:
        """
        评估数据集质量
        
        参数:
            dataset: 包含 'states', 'actions', 'images' 的字典
        """
        self.metrics = {}
        
        # 1. 完整性检查
        self._check_completeness(dataset)
        
        # 2. 时间一致性
        self._check_temporal_consistency(dataset)
        
        # 3. 动作平滑性
        self._check_action_smoothness(dataset)
        
        # 4. 图像质量
        if 'images' in dataset:
            self._check_image_quality(dataset)
        
        # 5. 分布检查
        self._check_distribution(dataset)
        
        # 6. 异常检测
        self._check_anomalies(dataset)
        
        return self.metrics
    
    def _check_completeness(self, dataset: dict):
        """完整性检查"""
        total_frames = len(dataset.get('actions', []))
        
        completeness = {}
        for key, value in dataset.items():
            if isinstance(value, np.ndarray):
                # 检查 NaN 和 Inf
                nan_count = np.isnan(value).sum()
                inf_count = np.isinf(value).sum()
                
                completeness[key] = {
                    'total': len(value),
                    'nan_count': int(nan_count),
                    'inf_count': int(inf_count),
                    'completeness_rate': 1.0 - (nan_count + inf_count) / value.size,
                }
        
        self.metrics['completeness'] = completeness
    
    def _check_temporal_consistency(self, dataset: dict):
        """时间一致性检查"""
        if 'timestamps' not in dataset:
            return
        
        timestamps = np.array(dataset['timestamps'])
        dt = np.diff(timestamps)
        
        self.metrics['temporal'] = {
            'mean_dt': float(np.mean(dt)),
            'std_dt': float(np.std(dt)),
            'min_dt': float(np.min(dt)),
            'max_dt': float(np.max(dt)),
            'jitter_ms': float(np.std(dt) * 1000),
            'fps': float(1.0 / np.mean(dt)),
            'gaps': int(np.sum(dt > 2 * np.mean(dt))),  # 时间间隔异常
        }
    
    def _check_action_smoothness(self, dataset: dict):
        """动作平滑性检查"""
        if 'actions' not in dataset:
            return
        
        actions = np.array(dataset['actions'])
        
        # 计算加加速度（jerk）
        if len(actions) > 2:
            vel = np.diff(actions, axis=0)
            acc = np.diff(vel, axis=0)
            jerk = np.diff(acc, axis=0)
            
            jerk_magnitude = np.linalg.norm(jerk, axis=-1)
            
            self.metrics['smoothness'] = {
                'mean_jerk': float(np.mean(jerk_magnitude)),
                'max_jerk': float(np.max(jerk_magnitude)),
                'jerk_std': float(np.std(jerk_magnitude)),
                'smooth_score': float(1.0 / (1.0 + np.mean(jerk_magnitude))),
            }
    
    def _check_image_quality(self, dataset: dict):
        """图像质量检查"""
        images = dataset['images']
        if isinstance(images, dict):
            for cam_name, imgs in images.items():
                if len(imgs) == 0:
                    continue
                
                sample = imgs[0] if isinstance(imgs, list) else imgs[0]
                
                # 亮度
                brightness = np.mean(sample)
                
                # 对比度
                contrast = np.std(sample)
                
                # 模糊度（拉普拉斯方差）
                if sample.dtype != np.uint8:
                    sample = (sample * 255).astype(np.uint8)
                gray = cv2.cvtColor(sample, cv2.COLOR_RGB2GRAY)
                blur_score = cv2.Laplacian(gray, cv2.CV_64F).var()
                
                self.metrics.setdefault('image_quality', {})[cam_name] = {
                    'brightness': float(brightness),
                    'contrast': float(contrast),
                    'blur_score': float(blur_score),
                    'is_blurry': bool(blur_score < 100),
                }
    
    def _check_distribution(self, dataset: dict):
        """分布检查"""
        if 'actions' not in dataset:
            return
        
        actions = np.array(dataset['actions'])
        
        self.metrics['distribution'] = {
            'mean': actions.mean(axis=0).tolist(),
            'std': actions.std(axis=0).tolist(),
            'min': actions.min(axis=0).tolist(),
            'max': actions.max(axis=0).tolist(),
            'range': (actions.max(axis=0) - actions.min(axis=0)).tolist(),
        }
    
    def _check_anomalies(self, dataset: dict):
        """异常检测"""
        if 'actions' not in dataset:
            return
        
        actions = np.array(dataset['actions'])
        
        # 使用 Z-score 检测异常
        z_scores = np.abs((actions - actions.mean(axis=0)) / (actions.std(axis=0) + 1e-8))
        anomalies = z_scores > 3.0  # 3-sigma
        
        self.metrics['anomalies'] = {
            'total_anomalous': int(anomalies.sum()),
            'anomaly_rate': float(anomalies.sum() / anomalies.size),
            'anomalous_frames': np.where(anomalies.any(axis=1))[0].tolist(),
        }
    
    def get_report(self) -> str:
        """生成质量报告"""
        report = "=== 数据质量报告 ===\n\n"
        
        if 'completeness' in self.metrics:
            report += "1. 完整性\n"
            for key, info in self.metrics['completeness'].items():
                report += f"  {key}: 完整率 {info['completeness_rate']:.4f}\n"
        
        if 'temporal' in self.metrics:
            t = self.metrics['temporal']
            report += f"\n2. 时间一致性\n"
            report += f"  FPS: {t['fps']:.1f}\n"
            report += f"  抖动: {t['jitter_ms']:.2f}ms\n"
            report += f"  时间间隔异常: {t['gaps']}\n"
        
        if 'smoothness' in self.metrics:
            s = self.metrics['smoothness']
            report += f"\n3. 动作平滑性\n"
            report += f"  平滑度评分: {s['smooth_score']:.4f}\n"
            report += f"  平均加加速度: {s['mean_jerk']:.4f}\n"
        
        if 'anomalies' in self.metrics:
            a = self.metrics['anomalies']
            report += f"\n4. 异常检测\n"
            report += f"  异常率: {a['anomaly_rate']:.4f}\n"
            report += f"  异常帧数: {a['total_anomalous']}\n"
        
        return report
```

```python
import multiprocessing as mp
from multiprocessing import Pool, Queue, Process
import h5py

class ParallelDataConverter:
    """并行数据转换流水线"""
    
    def __init__(self, 
                 num_workers: int = 8,
                 batch_size: int = 10):
        self.num_workers = num_workers
        self.batch_size = batch_size
        
        # 任务队列
        self.task_queue = mp.Queue()
        self.result_queue = mp.Queue()
        
        # 统计
        self.stats = {
            'total_files': 0,
            'converted': 0,
            'failed': 0,
            'total_time': 0,
        }
    
    def convert_directory(self, input_dir: str, output_dir: str, converter_func: Callable):
        """
        并行转换目录中的所有文件
        
        参数:
            input_dir: 输入目录
            output_dir: 输出目录
            converter_func: 转换函数 (input_path, output_path) -> bool
        """
        input_dir = Path(input_dir)
        output_dir = Path(output_dir)
        output_dir.mkdir(parents=True, exist_ok=True)
        
        # 查找所有输入文件
        input_files = sorted(input_dir.glob("*.hdf5"))
        self.stats['total_files'] = len(input_files)
        
        print(f"开始并行转换: {len(input_files)} 个文件, {self.num_workers} 个工作进程")
        
        start_time = time.time()
        
        # 创建工作进程
        workers = []
        for i in range(self.num_workers):
            p = Process(
                target=self._worker_loop,
                args=(converter_func, self.task_queue, self.result_queue, i)
            )
            p.start()
            workers.append(p)
        
        # 分发任务
        for i, input_file in enumerate(input_files):
            output_file = output_dir / f"episode_{i:06d}"
            self.task_queue.put((str(input_file), str(output_file), i))
        
        # 发送结束信号
        for _ in range(self.num_workers):
            self.task_queue.put(None)
        
        # 收集结果
        results = []
        for _ in range(len(input_files)):
            result = self.result_queue.get()
            results.append(result)
            if result['success']:
                self.stats['converted'] += 1
            else:
                self.stats['failed'] += 1
            
            if (self.stats['converted'] + self.stats['failed']) % 10 == 0:
                print(f"进度: {self.stats['converted'] + self.stats['failed']}/{self.stats['total_files']}")
        
        # 等待工作进程结束
        for p in workers:
            p.join()
        
        self.stats['total_time'] = time.time() - start_time
        
        print(f"\n转换完成!")
        print(f"  成功: {self.stats['converted']}")
        print(f"  失败: {self.stats['failed']}")
        print(f"  耗时: {self.stats['total_time']:.1f}s")
        print(f"  速度: {self.stats['total_files'] / self.stats['total_time']:.1f} files/s")
    
    def _worker_loop(self, converter_func, task_queue, result_queue, worker_id):
        """工作进程主循环"""
        while True:
            task = task_queue.get()
            
            if task is None:  # 结束信号
                break
            
            input_path, output_path, episode_idx = task
            
            try:
                success = converter_func(input_path, output_path)
                result_queue.put({
                    'success': success,
                    'input': input_path,
                    'output': output_path,
                    'episode_idx': episode_idx,
                    'error': None,
                })
            except Exception as e:
                result_queue.put({
                    'success': False,
                    'input': input_path,
                    'output': output_path,
                    'episode_idx': episode_idx,
                    'error': str(e),
                })
```

```python
class StreamingDataPipeline:
    """流式数据处理管线"""
    
    def __init__(self, buffer_size: int = 100):
        self.buffer_size = buffer_size
        self.stages = []
    
    def add_stage(self, name: str, process_func: Callable, num_workers: int = 1):
        """添加处理阶段"""
        self.stages.append({
            'name': name,
            'func': process_func,
            'workers': num_workers,
            'queue': mp.Queue(maxsize=self.buffer_size),
        })
    
    def run(self, input_iterable):
        """运行流水线"""
        # 创建进程
        processes = []
        
        # 输入进程
        def input_producer():
            for item in input_iterable:
                self.stages[0]['queue'].put(item)
            self.stages[0]['queue'].put(None)  # 结束信号
        
        p = Process(target=input_producer)
        processes.append(p)
        
        # 中间处理进程
        for i, stage in enumerate(self.stages):
            def worker(stage=stage, next_queue=self.stages[i+1]['queue'] if i+1 < len(self.stages) else None):
                while True:
                    item = stage['queue'].get()
                    if item is None:
                        if next_queue:
                            next_queue.put(None)
                        break
                    result = stage['func'](item)
                    if next_queue and result is not None:
                        next_queue.put(result)
            
            for _ in range(stage['workers']):
                p = Process(target=worker)
                processes.append(p)
        
        # 启动
        for p in processes:
            p.start()
        
        # 等待完成
        for p in processes:
            p.join()
```

```python
# 文件: hdf5_io.py
"""HDF5 数据读写完整示例"""
import h5py
import numpy as np
import cv2
from pathlib import Path
from typing import Optional, Dict, List
import time


class HDF5DataWriter:
    """HDF5 数据写入器"""
    
    def __init__(self, file_path: str, compression: str = "gzip"):
        self.file_path = file_path
        self.compression = compression
        self.file = None
        self.episode_group = None
        self.frame_count = 0
    
    def create_episode(self, episode_id: int, metadata: dict):
        """创建新的 episode"""
        self.file = h5py.File(self.file_path, 'w')
        
        # 元数据
        for key, value in metadata.items():
            self.file.attrs[key] = value
        
        self.file.attrs['episode_id'] = episode_id
        self.file.attrs['created_at'] = time.strftime('%Y-%m-%d %H:%M:%S')
        
        # 创建组
        self.action_grp = self.file.create_group('action')
        self.obs_grp = self.file.create_group('observation')
        self.images_grp = self.obs_grp.create_group('images')
        
        # 时间戳数据集
        self.timestamps_ds = None
        
        self.frame_count = 0
        self.datasets_initialized = False
    
    def add_frame(self, 
                  action: dict,
                  observation: dict,
                  images: Optional[Dict[str, np.ndarray]] = None,
                  timestamp: Optional[float] = None):
        """
        添加一帧数据
        
        参数:
            action: 动作数据字典
            observation: 观测数据字典
            images: 图像字典 {camera_name: image_array}
            timestamp: 时间戳
        """
        if not self.datasets_initialized:
            self._init_datasets(action, observation, images)
            self.datasets_initialized = True
        
        # 写入动作
        for key, value in action.items():
            self.action_grp[key][self.frame_count] = value
        
        # 写入观测
        for key, value in observation.items():
            if key != 'images':
                self.obs_grp[key][self.frame_count] = value
        
        # 写入图像
        if images:
            for cam_name, img in images.items():
                self.images_grp[cam_name][self.frame_count] = img
        
        # 写入时间戳
        if self.timestamps_ds is not None:
            self.timestamps_ds[self.frame_count] = timestamp if timestamp else self.frame_count / 50.0
        
        self.frame_count += 1
    
    def _init_datasets(self, action: dict, observation: dict, images: Optional[dict]):
        """初始化数据集（可扩展）"""
        max_frames = 10000  # 最大帧数
        
        # 动作数据集
        for key, value in action.items():
            shape = (max_frames,) + np.array(value).shape
            self.action_grp.create_dataset(
                key, shape=shape, dtype='float32',
                compression=self.compression,
                chunks=True,  # 分块存储
            )
        
        # 观测数据集
        for key, value in observation.items():
            if key == 'images':
                continue
            shape = (max_frames,) + np.array(value).shape
            self.obs_grp.create_dataset(
                key, shape=shape, dtype='float32',
                compression=self.compression,
                chunks=True,
            )
        
        # 图像数据集
        if images:
            for cam_name, img in images.items():
                shape = (max_frames,) + img.shape
                self.images_grp.create_dataset(
                    cam_name, shape=shape, dtype='uint8',
                    compression=self.compression,
                    chunks=(1,) + img.shape,  # 每帧一个块
                )
        
        # 时间戳
        self.timestamps_ds = self.file.create_dataset(
            'timestamps', shape=(max_frames,), dtype='float64',
            compression=self.compression,
        )
    
    def close(self):
        """关闭文件"""
        if self.file:
            # 调整数据集大小为实际帧数
            for grp in [self.action_grp, self.obs_grp]:
                for key in grp:
                    ds = grp[key]
                    ds.resize((self.frame_count,) + ds.shape[1:])
            
            for cam_name in self.images_grp:
                ds = self.images_grp[cam_name]
                ds.resize((self.frame_count,) + ds.shape[1:])
            
            if self.timestamps_ds:
                self.timestamps_ds.resize((self.frame_count,))
            
            self.file.attrs['num_frames'] = self.frame_count
            self.file.close()
            self.file = None
            print(f"HDF5 文件已保存: {self.file_path} ({self.frame_count} 帧)")


class HDF5DataReader:
    """HDF5 数据读取器"""
    
    def __init__(self, file_path: str):
        self.file_path = file_path
        self.file = None
    
    def open(self):
        self.file = h5py.File(self.file_path, 'r')
    
    def get_metadata(self) -> dict:
        """获取元数据"""
        if not self.file:
            self.open()
        return dict(self.file.attrs)
    
    def get_frame(self, frame_idx: int) -> dict:
        """获取指定帧"""
        if not self.file:
            self.open()
        
        frame = {}
        
        # 动作
        if 'action' in self.file:
            frame['action'] = {}
            for key in self.file['action']:
                frame['action'][key] = self.file['action'][key][frame_idx]
        
        # 观测
        if 'observation' in self.file:
            frame['observation'] = {}
            for key in self.file['observation']:
                if key == 'images':
                    frame['observation']['images'] = {}
                    for cam in self.file['observation/images']:
                        frame['observation']['images'][cam] = self.file['observation/images'][cam][frame_idx]
                else:
                    frame['observation'][key] = self.file['observation'][key][frame_idx]
        
        # 时间戳
        if 'timestamps' in self.file:
            frame['timestamp'] = self.file['timestamps'][frame_idx]
        
        return frame
    
    def get_trajectory(self, start: int = 0, end: int = None) -> dict:
        """获取轨迹片段"""
        if not self.file:
            self.open()
        
        if end is None:
            end = self.file.attrs.get('num_frames', 0)
        
        trajectory = {}
        
        # 动作
        if 'action' in self.file:
            trajectory['action'] = {}
            for key in self.file['action']:
                trajectory['action'][key] = self.file['action'][key][start:end]
        
        # 观测
        if 'observation' in self.file:
            trajectory['observation'] = {}
            for key in self.file['observation']:
                if key == 'images':
                    trajectory['observation']['images'] = {}
                    for cam in self.file['observation/images']:
                        trajectory['observation']['images'][cam] = self.file['observation/images'][cam][start:end]
                else:
                    trajectory['observation'][key] = self.file['observation'][key][start:end]
        
        return trajectory
    
    def iterate_frames(self):
        """迭代所有帧"""
        if not self.file:
            self.open()
        
        num_frames = self.file.attrs.get('num_frames', 0)
        for i in range(num_frames):
            yield self.get_frame(i)
    
    def close(self):
        if self.file:
            self.file.close()
            self.file = None
```
