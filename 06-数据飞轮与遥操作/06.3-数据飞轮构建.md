## 1. 数据飞轮概念

### 1.1 什么是数据飞轮

数据飞轮（Data Flywheel）是一种持续改进的机器人学习范式，其核心思想是通过**采集→训练→部署→再采集**的循环，使机器人的能力随数据量增长而不断提升。

```
       ┌──────────┐
       │  采集数据  │ <─────────────┐
       └────┬─────┘               │
            │                      │
            v                      │
       ┌──────────┐               │
       │  训练模型  │               │
       └────┬─────┘               │
            │                      │
            v                      │
       ┌──────────┐               │
       │  部署执行  │               │
       └────┬─────┘               │
            │                      │
            v                      │
       ┌──────────┐               │
       │  收集反馈  │ ──────────────┘
       └──────────┘
```

### 1.2 飞轮的核心要素

| 要素 | 说明 | 关键指标 |
|------|------|---------|
| 数据采集 | 人类示教 + 自主采集 + 模型辅助 | 采集效率（条/小时） |
| 数据质量 | 标注准确性、轨迹平滑性、覆盖度 | 成功率、多样性 |
| 模型训练 | 策略学习、微调、迭代 | 成功率、泛化能力 |
| 部署评估 | 真实环境测试、失败分析 | Sim2Real Gap |
| 反馈循环 | 失败案例收集、改进策略 | 迭代速度 |

### 1.3 飞轮与传统的区别

| 维度 | 传统方法 | 数据飞轮 |
|------|---------|---------|
| 数据来源 | 一次性采集 | 持续采集 |
| 模型更新 | 手动重训 | 自动迭代 |
| 失败处理 | 忽略 | 主动收集 |
| 数据质量 | 静态 | 持续提升 |
| 扩展性 | 有限 | 无限循环 |
| 人力需求 | 高（全程人工） | 递减（模型辅助） |

```python
# 文件: data_flywheel.py
"""数据飞轮系统"""
import numpy as np
import time
import json
from pathlib import Path
from typing import Optional, Dict, List
from dataclasses import dataclass, field
from enum import Enum


class FlywheelStage(Enum):
    """飞轮阶段"""
    COLLECT = "collect"
    PROCESS = "process"
    TRAIN = "train"
    DEPLOY = "deploy"
    EVALUATE = "evaluate"


@dataclass
class FlywheelConfig:
    """飞轮配置"""
    # 数据采集
    collection_fps: int = 50
    auto_collection: bool = True
    model_assisted: bool = True
    
    # 数据处理
    augmentation: bool = True
    quality_threshold: float = 0.8
    
    # 训练
    training_framework: str = "lerobot"  # or "groot"
    policy_type: str = "diffusion"  # or "act", "tdmpc"
    fine_tune: bool = True
    
    # 部署
    sim2real: bool = True
    safety_mode: str = "assisted"  # or "autonomous"
    
    # 迭代
    max_iterations: int = 100
    min_new_data: int = 50  # 最少新数据量才触发训练


class DataFlywheel:
    """数据飞轮主控制器"""
    
    def __init__(self, config: FlywheelConfig, workspace: str = "./flywheel"):
        self.config = config
        self.workspace = Path(workspace)
        self.workspace.mkdir(parents=True, exist_ok=True)
        
        # 子目录
        self.data_dir = self.workspace / "data"
        self.models_dir = self.workspace / "models"
        self.logs_dir = self.workspace / "logs"
        
        for d in [self.data_dir, self.models_dir, self.logs_dir]:
            d.mkdir(exist_ok=True)
        
        # 状态
        self.current_iteration = 0
        self.current_model = None
        self.data_version = DataVersionManager(self.data_dir)
        self.stats = FlywheelStats()
    
    def run(self, task_description: str):
        """运行飞轮循环"""
        print(f"=== 数据飞轮启动 ===")
        print(f"任务: {task_description}")
        print(f"最大迭代: {self.config.max_iterations}")
        
        for iteration in range(self.config.max_iterations):
            self.current_iteration = iteration
            print(f"\n--- 迭代 {iteration + 1}/{self.config.max_iterations} ---")
            
            # 1. 数据采集
            print("[1/5] 数据采集...")
            new_data = self._collect_data(task_description)
            self.stats.record_collection(new_data)
            
            # 2. 数据处理
            print("[2/5] 数据处理...")
            processed = self._process_data(new_data)
            self.stats.record_processing(processed)
            
            # 3. 模型训练
            print("[3/5] 模型训练...")
            model = self._train_model(processed)
            self.stats.record_training(model)
            
            # 4. 部署评估
            print("[4/5] 部署评估...")
            eval_results = self._deploy_and_evaluate(model, task_description)
            self.stats.record_evaluation(eval_results)
            
            # 5. 反馈收集
            print("[5/5] 反馈收集...")
            feedback = self._collect_feedback(eval_results)
            self.stats.record_feedback(feedback)
            
            # 检查是否继续
            if self._should_continue(eval_results):
                self.current_model = model
                self._save_iteration(iteration, model, eval_results)
            else:
                print(f"飞轮在第 {iteration + 1} 次迭代后停止")
                break
        
        self._print_final_report()
    
    def _collect_data(self, task: str) -> List[dict]:
        """数据采集阶段"""
        episodes = []
        
        # 人类示教数据
        if self.current_iteration == 0:
            # 第一次：纯人类示教
            print("  模式: 人类示教")
            episodes = self._human_teleop_collection(task, num_episodes=20)
        
        elif self.config.model_assisted and self.current_model:
            # 模型辅助示教
            print("  模式: 模型辅助示教")
            episodes = self._model_assisted_collection(task, num_episodes=50)
        
        else:
            # 纯自主采集
            print("  模式: 自主采集")
            episodes = self._autonomous_collection(task, num_episodes=100)
        
        return episodes
    
    def _human_teleop_collection(self, task: str, num_episodes: int) -> List[dict]:
        """人类遥操作数据采集"""
        episodes = []
        
        for i in range(num_episodes):
            # 模拟遥操作采集
            episode = {
                'episode_id': f"human_{self.current_iteration}_{i}",
                'task': task,
                'source': 'human_teleop',
                'data': self._simulate_episode_data(task, success_rate=0.9),
                'success': True,
                'timestamp': time.time(),
            }
            episodes.append(episode)
        
        return episodes
    
    def _model_assisted_collection(self, task: str, num_episodes: int) -> List[dict]:
        """模型辅助示教"""
        episodes = []
        
        for i in range(num_episodes):
            # 模型执行
            model_output = self._run_model(task)
            
            # 人类监督和修正
            corrected = self._human_correction(model_output)
            
            episode = {
                'episode_id': f"assisted_{self.current_iteration}_{i}",
                'task': task,
                'source': 'model_assisted',
                'data': corrected,
                'model_output': model_output,
                'corrections': self._count_corrections(model_output, corrected),
                'success': True,
                'timestamp': time.time(),
            }
            episodes.append(episode)
        
        return episodes
    
    def _autonomous_collection(self, task: str, num_episodes: int) -> List[dict]:
        """自主数据采集"""
        episodes = []
        
        for i in range(num_episodes):
            # 模型自主执行
            result = self._run_model(task)
            
            success = np.random.random() < 0.7  # 模拟成功率
            
            episode = {
                'episode_id': f"auto_{self.current_iteration}_{i}",
                'task': task,
                'source': 'autonomous',
                'data': result,
                'success': success,
                'timestamp': time.time(),
            }
            episodes.append(episode)
        
        # 只保留成功的数据
        successful = [ep for ep in episodes if ep['success']]
        failed = [ep for ep in episodes if not ep['success']]
        
        print(f"  自主采集: {len(successful)} 成功, {len(failed)} 失败")
        
        return successful + failed  # 失败数据也保留用于分析
    
    def _process_data(self, raw_data: List[dict]) -> dict:
        """数据处理阶段"""
        processed = {
            'total_episodes': len(raw_data),
            'successful': sum(1 for d in raw_data if d.get('success', False)),
            'failed': sum(1 for d in raw_data if not d.get('success', False)),
            'augmented': 0,
        }
        
        # 数据增强
        if self.config.augmentation:
            for episode in raw_data:
                if episode.get('success'):
                    # 生成增强变体
                    augmented = self._augment_episode(episode)
                    processed['augmented'] += len(augmented)
        
        # 质量评估
        quality_scores = [self._assess_quality(ep) for ep in raw_data]
        processed['avg_quality'] = np.mean(quality_scores)
        processed['quality_threshold'] = self.config.quality_threshold
        
        # 版本管理
        version = self.data_version.save_version(raw_data, processed)
        processed['version'] = version
        
        return processed
    
    def _train_model(self, data_info: dict) -> dict:
        """模型训练阶段"""
        model_info = {
            'iteration': self.current_iteration,
            'framework': self.config.training_framework,
            'policy_type': self.config.policy_type,
            'data_version': data_info.get('version'),
            'training_data_size': data_info['total_episodes'] + data_info.get('augmented', 0),
        }
        
        if self.config.training_framework == "lerobot":
            model_info.update(self._train_with_lerobot(data_info))
        elif self.config.training_framework == "groot":
            model_info.update(self._train_with_groot(data_info))
        
        # 保存模型
        model_path = self.models_dir / f"model_iter_{self.current_iteration}"
        model_info['path'] = str(model_path)
        
        return model_info
    
    def _deploy_and_evaluate(self, model: dict, task: str) -> dict:
        """部署评估阶段"""
        results = {
            'model_iteration': model['iteration'],
            'task': task,
            'num_eval_episodes': 20,
            'success_rate': 0.0,
            'avg_completion_time': 0.0,
            'failure_modes': [],
        }
        
        # 模拟评估
        base_success = 0.3 + 0.05 * self.current_iteration  # 随迭代提升
        results['success_rate'] = min(0.95, base_success + np.random.normal(0, 0.05))
        results['avg_completion_time'] = 15.0 - 0.5 * self.current_iteration
        
        # 失败模式分析
        if results['success_rate'] < 0.9:
            results['failure_modes'] = self._analyze_failures()
        
        return results
    
    def _collect_feedback(self, eval_results: dict) -> dict:
        """反馈收集阶段"""
        feedback = {
            'success_rate': eval_results['success_rate'],
            'improvement': 0.0,
            'actionable_feedback': [],
        }
        
        if self.current_iteration > 0:
            prev_success = self.stats.history[-1]['evaluation']['success_rate'] if self.stats.history else 0
            feedback['improvement'] = eval_results['success_rate'] - prev_success
        
        # 生成可执行的改进建议
        if eval_results['success_rate'] < 0.5:
            feedback['actionable_feedback'].append("成功率过低，增加人类示教数据")
        if eval_results.get('failure_modes'):
            for mode in eval_results['failure_modes']:
                feedback['actionable_feedback'].append(f"针对 {mode} 增加专项训练数据")
        
        return feedback
    
    def _should_continue(self, eval_results: dict) -> bool:
        """判断是否继续迭代"""
        if eval_results['success_rate'] >= 0.95:
            return False  # 达到目标
        return True
    
    def _save_iteration(self, iteration: int, model: dict, eval_results: dict):
        """保存迭代结果"""
        log_path = self.logs_dir / f"iteration_{iteration}.json"
        log_data = {
            'iteration': iteration,
            'model': model,
            'evaluation': eval_results,
            'timestamp': time.time(),
        }
        with open(log_path, 'w') as f:
            json.dump(log_data, f, indent=2, default=str)
    
    def _print_final_report(self):
        """打印最终报告"""
        print(f"\n{'='*50}")
        print(f"数据飞轮最终报告")
        print(f"{'='*50}")
        print(f"总迭代次数: {self.current_iteration + 1}")
        print(f"最终成功率: {self.stats.history[-1]['evaluation']['success_rate']:.2%}" if self.stats.history else "N/A")
        print(f"总数据量: {self.stats.total_data}")
        print(f"总训练时间: {self.stats.total_training_time:.1f}s")


class FlywheelStats:
    """飞轮统计"""
    
    def __init__(self):
        self.history = []
        self.total_data = 0
        self.total_training_time = 0
    
    def record_collection(self, data):
        self.total_data += len(data)
    
    def record_processing(self, data):
        pass
    
    def record_training(self, model):
        pass
    
    def record_evaluation(self, results):
        self.history.append({'evaluation': results})
    
    def record_feedback(self, feedback):
        self.history[-1]['feedback'] = feedback if self.history else None


class DataVersionManager:
    """数据版本管理"""
    
    def __init__(self, data_dir: Path):
        self.data_dir = data_dir
        self.versions = []
        self.version_file = data_dir / "versions.json"
        
        if self.version_file.exists():
            with open(self.version_file) as f:
                self.versions = json.load(f)
    
    def save_version(self, data: List[dict], info: dict) -> str:
        """保存数据版本"""
        version = f"v{len(self.versions)}"
        
        version_info = {
            'version': version,
            'timestamp': time.time(),
            'num_episodes': len(data),
            'info': info,
        }
        
        self.versions.append(version_info)
        
        with open(self.version_file, 'w') as f:
            json.dump(self.versions, f, indent=2)
        
        return version
```

### 3.1 自动化采集系统

```python
# 文件: auto_collection.py
"""自动化数据采集系统"""
import numpy as np
import time
from typing import Optional
from dataclasses import dataclass

@dataclass
class CollectionTask:
    """采集任务定义"""
    task_name: str
    task_description: str
    target_episodes: int
    success_threshold: float = 0.7
    max_attempts: int = 200
    variation_params: dict = None  # 任务参数变化


class AutoCollectionSystem:
    """自动化数据采集系统"""
    
    def __init__(self, robot_interface, model_interface=None):
        self.robot = robot_interface
        self.model = model_interface
        
        # 采集模式
        self.modes = {
            'pure_teleop': self._pure_teleop,
            'model_assisted': self._model_assisted,
            'autonomous': self._autonomous,
            'hybrid': self._hybrid,
        }
        
        # 统计
        self.collection_stats = {
            'total_episodes': 0,
            'successful': 0,
            'failed': 0,
            'total_time': 0,
            'avg_episode_length': 0,
        }
    
    def collect(self, task: CollectionTask, mode: str = 'hybrid') -> list:
        """
        执行数据采集
        
        参数:
            task: 采集任务
            mode: 采集模式
                - pure_teleop: 纯人类遥操作
                - model_assisted: 模型辅助
                - autonomous: 纯自主
                - hybrid: 混合模式（推荐）
        """
        collector = self.modes.get(mode, self._hybrid)
        
        print(f"开始数据采集: {task.task_name}")
        print(f"  模式: {mode}")
        print(f"  目标: {task.target_episodes} 条成功数据")
        
        start_time = time.time()
        episodes = collector(task)
        elapsed = time.time() - start_time
        
        self.collection_stats['total_time'] += elapsed
        
        print(f"\n采集完成:")
        print(f"  总尝试: {self.collection_stats['total_episodes']}")
        print(f"  成功: {self.collection_stats['successful']}")
        print(f"  失败: {self.collection_stats['failed']}")
        print(f"  成功率: {self.collection_stats['successful'] / max(1, self.collection_stats['total_episodes']):.2%}")
        print(f"  耗时: {elapsed:.1f}s")
        print(f"  效率: {self.collection_stats['successful'] / max(1, elapsed):.1f} 条/秒")
        
        return episodes
    
    def _pure_teleop(self, task: CollectionTask) -> list:
        """纯遥操作采集"""
        episodes = []
        
        for i in range(task.target_episodes):
            print(f"  [{i+1}/{task.target_episodes}] 人类示教中...")
            
            # 等待人类操作员完成
            episode = self.robot.teleop_episode(task.task_description)
            
            if episode and episode.get('success', True):
                episodes.append(episode)
                self.collection_stats['successful'] += 1
            else:
                self.collection_stats['failed'] += 1
            
            self.collection_stats['total_episodes'] += 1
        
        return episodes
    
    def _model_assisted(self, task: CollectionTask) -> list:
        """模型辅助采集"""
        episodes = []
        attempts = 0
        
        while len(episodes) < task.target_episodes and attempts < task.max_attempts:
            attempts += 1
            
            # 1. 模型生成动作
            model_action = self.model.predict(task.task_description)
            
            # 2. 人类监督执行
            print(f"  [{len(episodes)+1}/{task.target_episodes}] 模型执行中... (尝试 {attempts})")
            
            episode = self.robot.execute_with_supervision(
                model_action,
                task.task_description,
                allow_human_correction=True,
            )
            
            # 3. 记录结果
            if episode['success']:
                episodes.append(episode)
                self.collection_stats['successful'] += 1
            else:
                # 失败数据也保留
                episodes.append(episode)
                self.collection_stats['failed'] += 1
            
            self.collection_stats['total_episodes'] += 1
        
        return episodes
    
    def _autonomous(self, task: CollectionTask) -> list:
        """纯自主采集"""
        episodes = []
        attempts = 0
        
        while len([e for e in episodes if e['success']]) < task.target_episodes:
            if attempts >= task.max_attempts:
                break
            attempts += 1
            
            print(f"  [{len(episodes)+1}] 自主执行中... (尝试 {attempts})")
            
            # 模型自主执行
            episode = self.robot.execute_autonomous(
                task.task_description,
                self.model,
            )
            
            episodes.append(episode)
            
            if episode['success']:
                self.collection_stats['successful'] += 1
            else:
                self.collection_stats['failed'] += 1
            
            self.collection_stats['total_episodes'] += 1
        
        # 只返回成功数据
        return [e for e in episodes if e['success']]
    
    def _hybrid(self, task: CollectionTask) -> list:
        """混合模式采集"""
        episodes = []
        
        # 阶段1: 人类示教（30%）
        teleop_count = int(task.target_episodes * 0.3)
        print(f"\n阶段1: 人类示教 ({teleop_count} 条)")
        episodes.extend(self._pure_teleop(CollectionTask(
            task_name=task.task_name,
            task_description=task.task_description,
            target_episodes=teleop_count,
        )))
        
        # 阶段2: 模型辅助（40%）
        if self.model:
            assisted_count = int(task.target_episodes * 0.4)
            print(f"\n阶段2: 模型辅助 ({assisted_count} 条)")
            episodes.extend(self._model_assisted(CollectionTask(
                task_name=task.task_name,
                task_description=task.task_description,
                target_episodes=assisted_count,
            )))
        
        # 阶段3: 自主采集（30%）
        if self.model:
            auto_count = task.target_episodes - len(episodes)
            print(f"\n阶段3: 自主采集 ({auto_count} 条)")
            episodes.extend(self._autonomous(CollectionTask(
                task_name=task.task_name,
                task_description=task.task_description,
                target_episodes=auto_count,
            )))
        
        return episodes
```

```python
# 文件: model_assisted_teleop.py
"""模型辅助示教系统"""
import numpy as np
import time
from typing import Optional

class ModelAssistedTeleop:
    """模型辅助遥操作"""
    
    def __init__(self, policy_model, robot_interface, vr_interface=None):
        self.policy = policy_model
        self.robot = robot_interface
        self.vr = vr_interface
        
        # 辅助参数
        self.assistance_level = 0.5  # 0=纯人类, 1=纯模型
        self.intervention_threshold = 0.1  # 人类干预阈值
        self.smoothing_factor = 0.3
        
        # 统计
        self.interventions = 0
        self.total_steps = 0
    
    def run_assisted_episode(self, task: str, max_steps: int = 300) -> dict:
        """
        运行一次模型辅助示教
        
        参数:
            task: 任务描述
            max_steps: 最大步数
        
        返回:
            完整的 episode 数据
        """
        episode_data = {
            'task': task,
            'steps': [],
            'model_actions': [],
            'human_actions': [],
            'final_actions': [],
            'interventions': [],
            'success': False,
        }
        
        # 重置环境
        obs = self.robot.reset()
        
        for step in range(max_steps):
            self.total_steps += 1
            
            # 1. 获取模型预测
            model_action = self.policy.predict(obs)
            episode_data['model_actions'].append(model_action)
            
            # 2. 获取人类输入（如果有）
            human_action = None
            if self.vr:
                human_input = self.vr.get_input()
                human_action = self._convert_vr_to_action(human_input)
            
            # 3. 融合动作
            if human_action is not None:
                # 检查是否需要干预
                action_diff = np.linalg.norm(
                    np.array(model_action) - np.array(human_action)
                )
                
                if action_diff > self.intervention_threshold:
                    # 人类干预
                    self.interventions += 1
                    episode_data['interventions'].append({
                        'step': step,
                        'diff': action_diff,
                    })
                    
                    # 融合：人类优先
                    final_action = (
                        self.assistance_level * np.array(model_action) +
                        (1 - self.assistance_level) * np.array(human_action)
                    )
                else:
                    # 模型与人类一致，使用模型动作
                    final_action = model_action
            else:
                # 无人类输入，纯模型
                final_action = model_action
            
            episode_data['human_actions'].append(human_action)
            episode_data['final_actions'].append(final_action)
            
            # 4. 执行动作
            obs, reward, done, info = self.robot.step(final_action)
            
            # 5. 记录步骤
            episode_data['steps'].append({
                'observation': obs,
                'action': final_action,
                'reward': reward,
                'step': step,
            })
            
            if done:
                episode_data['success'] = info.get('success', False)
                break
        
        # 统计
        episode_data['intervention_rate'] = len(episode_data['interventions']) / max(1, len(episode_data['steps']))
        episode_data['assistance_level'] = self.assistance_level
        
        return episode_data
    
    def adapt_assistance_level(self, success_rate: float):
        """根据成功率自适应调整辅助级别"""
        if success_rate > 0.8:
            # 成功率高，增加模型自主性
            self.assistance_level = min(1.0, self.assistance_level + 0.1)
        elif success_rate < 0.4:
            # 成功率低，增加人类控制
            self.assistance_level = max(0.0, self.assistance_level - 0.1)
        
        print(f"辅助级别调整为: {self.assistance_level:.1f}")
```

```python
# 文件: data_versioning.py
"""数据版本管理"""
import json
import hashlib
import shutil
from pathlib import Path
from datetime import datetime
from typing import Optional, List, Dict

class DataVersionManager:
    """数据版本管理器"""
    
    def __init__(self, root_dir: str):
        self.root = Path(root_dir)
        self.versions_dir = self.root / "versions"
        self.versions_dir.mkdir(parents=True, exist_ok=True)
        
        self.manifest_file = self.root / "manifest.json"
        self.manifest = self._load_manifest()
    
    def _load_manifest(self) -> dict:
        """加载版本清单"""
        if self.manifest_file.exists():
            with open(self.manifest_file) as f:
                return json.load(f)
        return {'versions': []}
    
    def _save_manifest(self):
        """保存版本清单"""
        with open(self.manifest_file, 'w') as f:
            json.dump(self.manifest, f, indent=2)
    
    def create_version(self, 
                       data_path: str,
                       version_tag: str,
                       description: str = "",
                       metadata: Optional[dict] = None) -> str:
        """
        创建数据版本
        
        参数:
            data_path: 数据目录路径
            version_tag: 版本标签 (如 "v1.0", "experiment_001")
            description: 版本描述
            metadata: 额外元数据
        
        返回:
            版本ID
        """
        # 计算数据哈希
        data_hash = self._compute_hash(data_path)
        
        # 版本ID
        version_id = f"{version_tag}_{data_hash[:8]}"
        
        # 复制数据到版本目录
        version_dir = self.versions_dir / version_id
        if version_dir.exists():
            print(f"版本 {version_id} 已存在")
            return version_id
        
        shutil.copytree(data_path, version_dir)
        
        # 记录版本信息
        version_info = {
            'version_id': version_id,
            'version_tag': version_tag,
            'description': description,
            'data_hash': data_hash,
            'created_at': datetime.now().isoformat(),
            'metadata': metadata or {},
            'path': str(version_dir),
        }
        
        self.manifest['versions'].append(version_info)
        self._save_manifest()
        
        print(f"创建版本: {version_id}")
        return version_id
    
    def _compute_hash(self, path: str) -> str:
        """计算目录哈希"""
        hasher = hashlib.md5()
        
        for file_path in sorted(Path(path).rglob("*")):
            if file_path.is_file():
                with open(file_path, 'rb') as f:
                    hasher.update(f.read())
        
        return hasher.hexdigest()
    
    def list_versions(self) -> List[dict]:
        """列出所有版本"""
        return self.manifest['versions']
    
    def get_version(self, version_id: str) -> Optional[dict]:
        """获取指定版本"""
        for v in self.manifest['versions']:
            if v['version_id'] == version_id:
                return v
        return None
    
    def compare_versions(self, v1_id: str, v2_id: str) -> dict:
        """比较两个版本"""
        v1 = self.get_version(v1_id)
        v2 = self.get_version(v2_id)
        
        if not v1 or not v2:
            return {'error': '版本不存在'}
        
        return {
            'v1': v1,
            'v2': v2,
            'same_data': v1['data_hash'] == v2['data_hash'],
            'time_diff': v2['created_at'] - v1['created_at'],
        }
    
    def rollback(self, version_id: str) -> str:
        """回滚到指定版本"""
        version = self.get_version(version_id)
        if not version:
            raise ValueError(f"版本 {version_id} 不存在")
        
        # 复制版本数据到当前目录
        current_data = self.root / "current"
        if current_data.exists():
            shutil.rmtree(current_data)
        
        shutil.copytree(version['path'], current_data)
        
        print(f"已回滚到版本: {version_id}")
        return str(current_data)
```

## 6. 缩放定律在机器人数据中的应用

### 6.1 机器人数据缩放定律

```
机器人策略性能 ~ α * D^β

其中:
  D = 数据量（轨迹数）
  α = 任务难度系数
  β = 缩放指数（通常 0.1-0.3）

经验值:
  - 简单任务（push, reach）: 50-100 条足够
  - 中等任务（pick-place）: 200-500 条
  - 复杂任务（双臂协作）: 1000-5000 条
  - 非常复杂任务（全身运动）: 5000-50000 条
```

### 6.2 数据效率分析

```python
# 文件: scaling_law.py
"""机器人数据缩放定律分析"""
import numpy as np
from dataclasses import dataclass

@dataclass
class ScalingLawConfig:
    """缩放定律配置"""
    alpha: float = 0.5      # 基础性能
    beta: float = 0.2       # 缩放指数
    saturation: float = 0.95  # 性能上限

class ScalingLawAnalyzer:
    """缩放定律分析器"""
    
    def __init__(self, config: ScalingLawConfig):
        self.config = config
    
    def predict_performance(self, num_episodes: int) -> float:
        """预测给定数据量下的性能"""
        raw = self.config.alpha * (num_episodes ** self.config.beta)
        # 使用 sigmoid 饱和
        performance = self.config.saturation * (
            1 - np.exp(-raw / self.config.saturation)
        )
        return min(performance, self.config.saturation)
    
    def estimate_data_needed(self, target_performance: float) -> int:
        """估算达到目标性能所需的数据量"""
        if target_performance >= self.config.saturation:
            return float('inf')
        
        # 反推
        raw = -self.config.saturation * np.log(
            1 - target_performance / self.config.saturation
        )
        num_episodes = (raw / self.config.alpha) ** (1 / self.config.beta)
        return int(num_episodes)
    
    def marginal_benefit(self, current_episodes: int, additional: int = 100) -> dict:
        """计算边际收益"""
        current_perf = self.predict_performance(current_episodes)
        new_perf = self.predict_performance(current_episodes + additional)
        
        return {
            'current_episodes': current_episodes,
            'current_performance': current_perf,
            'new_performance': new_perf,
            'improvement': new_perf - current_perf,
            'improvement_per_episode': (new_perf - current_perf) / additional,
            'diminishing_returns': self._is_diminishing(current_episodes),
        }
    
    def _is_diminishing(self, episodes: int) -> bool:
        """判断是否进入收益递减"""
        # 计算二阶导数
        eps = max(1, episodes // 10)
        p1 = self.predict_performance(episodes - eps)
        p2 = self.predict_performance(episodes)
        p3 = self.predict_performance(episodes + eps)
        
        second_derivative = (p3 - 2 * p2 + p1) / (eps ** 2)
        
        return second_derivative < 0
    
    def print_analysis(self):
        """打印缩放定律分析"""
        print("=== 数据缩放定律分析 ===\n")
        
        data_points = [10, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
        
        print(f"{'数据量':>8} | {'预期性能':>8} | {'边际收益':>10} | {'阶段':>8}")
        print("-" * 50)
        
        prev_perf = 0
        for n in data_points:
            perf = self.predict_performance(n)
            marginal = perf - prev_perf
            
            if n < 100:
                stage = "起步期"
            elif n < 500:
                stage = "成长期"
            elif n < 2000:
                stage = "成熟期"
            else:
                stage = "饱和期"
            
            print(f"{n:>8} | {perf:>8.2%} | {marginal:>10.4f} | {stage:>8}")
            prev_perf = perf
        
        print(f"\n目标性能 90% 需要约 {self.estimate_data_needed(0.9)} 条数据")
        print(f"目标性能 95% 需要约 {self.estimate_data_needed(0.95)} 条数据")


# 运行分析
if __name__ == "__main__":
    analyzer = ScalingLawAnalyzer(ScalingLawConfig(alpha=0.5, beta=0.2))
    analyzer.print_analysis()
```
