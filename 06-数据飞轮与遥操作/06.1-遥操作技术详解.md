## 1. 遥操作基本概念与分类

### 1.1 什么是遥操作

遥操作（Teleoperation）是指人类操作员通过远程接口控制机器人执行任务的技术。在具身智能领域，遥操作的核心目的是**采集高质量的人类示教数据**，用于训练机器人策略模型（如 VLA、Diffusion Policy 等）。

### 1.2 遥操作的核心价值

```
人类操作员 ──(输入设备)──> 遥操作系统 ──(网络)──> 机器人执行
    │                                                   │
    │              数据采集与录制                        │
    └───────────────────────────────────────────────────┘
                        │
                   策略训练
                        │
                   自主执行
```

- **数据采集**：收集人类操作机器人完成任务的轨迹数据
- **安全验证**：在危险环境中远程控制机器人
- **技能迁移**：将人类技能迁移到机器人策略模型
- **人机协作**：人类和机器人协同完成任务

### 1.3 遥操作分类

| 分类维度 | 类型 | 说明 | 典型代表 |
|---------|------|------|---------|
| **控制粒度** | 关节级 | 直接控制每个关节角度 | 键盘控制 |
| | 末端级 | 控制末端执行器位姿 | 空间鼠标 |
| | 任务级 | 高层任务指令 | 语音/自然语言 |
| **反馈方式** | 无反馈 | 仅视觉反馈 | 简单键盘控制 |
| | 视觉反馈 | 通过摄像头观察 | 大多数遥操作 |
| | 力触觉反馈 | 力反馈设备 | Phantom Omni |
| | VR 沉浸式 | VR 头显+手柄 | ALOHA、Teleopit |
| **同步方式** | 同步控制 | 实时控制，低延迟 | VR 遥操作 |
| | 异步示教 | 预录制动作回放 | 动捕数据回放 |
| **设备类型** | 外部设备 | 独立输入设备 | 键盘、鼠标、手柄 |
| | 可穿戴设备 | 穿戴在身上 | VR、动捕服 |
| | 同构设备 | 与机器人结构相同 | ALOHA 从臂 |

```python
            # 关节6
            'y': (5, 1), 'h': (5, -1),
            # 关节7
            'u': (6, 1), 'j': (6, -1),
            # 夹爪
            'o': 'gripper_open',
            'p': 'gripper_close',
        }
    
    def update(self):
        """更新控制命令"""
        for key, action in self.key_mapping.items():
            if keyboard.is_pressed(key):
                if isinstance(action, tuple):
                    joint_idx, direction = action
                    self.joint_commands[joint_idx] += direction * self.step_size
                elif action == 'gripper_open':
                    self.gripper = 1.0
                elif action == 'gripper_close':
                    self.gripper = 0.0
        
        return {
            'joint_commands': self.joint_commands.copy(),
            'gripper': getattr(self, 'gripper', 0.0),
        }
```

```python
"""SpaceMouse 遥操作控制器"""
import numpy as np
import spnav

class SpaceMouseController:
    """3Dconnexion SpaceMouse 控制器"""
    
    def __init__(self):
        spnav.open()
        self.translation_scale = 0.01   # 平移缩放
        self.rotation_scale = 0.02      # 旋转缩放
        self.deadzone = 50              # 死区
    
    def get_motion(self):
        """获取6DoF运动数据"""
        event = spnav.poll_event()
        
        if event is None:
            return None
        
        # 6自由度数据
        translation = np.array([
            event.translation[0],  # x
            event.translation[1],  # y
            event.translation[2],  # z
        ])
        rotation = np.array([
            event.rotation[0],     # rx
            event.rotation[1],     # ry
            event.rotation[2],     # rz
        ])
        
        # 死区处理
        translation[np.abs(translation) < self.deadzone] = 0
        rotation[np.abs(rotation) < self.deadzone] = 0
        
        # 缩放
        translation = translation * self.translation_scale / 350  # 归一化
        rotation = rotation * self.rotation_scale / 350
        
        # 按钮
        buttons = event.buttons  # [button0, button1]
        
        return {
            'translation': translation,  # [3] 末端平移增量
            'rotation': rotation,        # [3] 末端旋转增量
            'buttons': buttons,
            'gripper': 1.0 if buttons[0] else 0.0,
        }
    
    def close(self):
        spnav.close()
```

### 3.2 UDP 传输

```python
# 文件: udp_teleop.py
"""UDP 遥操作数据传输"""
import socket
import struct
import numpy as np
import pickle
import time

class UDPTeleopSender:
    """UDP 遥操作数据发送端"""
    
    def __init__(self, host: str = "0.0.0.0", port: int = 8888):
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.target = (host, port)
        self.sequence = 0
    
    def send(self, data: dict):
        """发送遥操作数据"""
        data['seq'] = self.sequence
        data['timestamp'] = time.time()
        
        # 序列化
        payload = pickle.dumps(data)
        
        # 发送
        self.sock.sendto(payload, self.target)
        self.sequence += 1
    
    def close(self):
        self.sock.close()

class UDPTeleopReceiver:
    """UDP 遥操作数据接收端"""
    
    def __init__(self, host: str = "0.0.0.0", port: int = 8888, buffer_size: int = 65536):
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_RCVBUF, buffer_size)
        self.sock.bind((host, port))
        self.sock.settimeout(1.0)  # 1秒超时
        
        self.last_data = None
        self.last_seq = -1
        self.stats = {
            'received': 0,
            'dropped': 0,
            'latency': [],
        }
    
    def receive(self) -> dict:
        """接收遥操作数据"""
        try:
            payload, addr = self.sock.recvfrom(65536)
            data = pickle.loads(payload)
            
            # 序列号检查
            if data['seq'] <= self.last_seq:
                self.stats['dropped'] += 1
                return self.last_data
            
            # 延迟计算
            latency = time.time() - data['timestamp']
            self.stats['latency'].append(latency)
            if len(self.stats['latency']) > 1000:
                self.stats['latency'] = self.stats['latency'][-1000:]
            
            self.stats['received'] += 1
            self.last_seq = data['seq']
            self.last_data = data
            
            return data
            
        except socket.timeout:
            return self.last_data
    
    def get_stats(self):
        """获取统计信息"""
        latencies = self.stats['latency']
        return {
            'received': self.stats['received'],
            'dropped': self.stats['dropped'],
            'avg_latency_ms': np.mean(latencies) * 1000 if latencies else 0,
            'max_latency_ms': np.max(latencies) * 1000 if latencies else 0,
            'packet_loss_rate': self.stats['dropped'] / max(1, self.stats['received'] + self.stats['dropped']),
        }
    
    def close(self):
        self.sock.close()
```

```python
# 文件: zmq_teleop.py
"""ZMQ 遥操作数据传输"""
import zmq
import numpy as np
import pickle
import time

class ZMQTeleopSender:
    """ZMQ PUB 端"""
    
    def __init__(self, port: int = 5555):
        self.context = zmq.Context()
        self.socket = self.context.socket(zmq.PUB)
        self.socket.set_hwm(10)  # 高水位标记
        self.socket.bind(f"tcp://*:{port}")
        # 等待订阅者连接
        time.sleep(0.5)
    
    def send(self, data: dict):
        """发送数据"""
        data['timestamp'] = time.time()
        self.socket.send_multipart([
            b"teleop",
            pickle.dumps(data),
        ])

class ZMQTeleopReceiver:
    """ZMQ SUB 端"""
    
    def __init__(self, host: str = "localhost", port: int = 5555):
        self.context = zmq.Context()
        self.socket = self.context.socket(zmq.SUB)
        self.socket.set_hwm(10)
        self.socket.connect(f"tcp://{host}:{port}")
        self.socket.subscribe(b"teleop")
    
    def receive(self, timeout_ms: int = 1000) -> dict:
        """接收数据"""
        poller = zmq.Poller()
        poller.register(self.socket, zmq.POLLIN)
        
        events = dict(poller.poll(timeout=timeout_ms))
        if self.socket in events:
            topic, payload = self.socket.recv_multipart()
            return pickle.loads(payload)
        return None
```

```python
    # 使用固定长度数组

class DDSTeleopPublisher:
    """DDS 发布者"""
    
    def __init__(self, domain_id: int = 0):
        self.participant = domain.DomainParticipant(domain_id)
        self.topic = core.Topic(self.participant, "TeleopData", TeleopData)
        self.publisher = pub.Publisher(self.participant)
        self.writer = pub.DataWriter(self.publisher, self.topic)
        self.seq = 0
    
    def publish(self, joint_positions, gripper, head_pos, head_rot):
        data = TeleopData(
            timestamp=time.time(),
            seq=self.seq,
            joint_positions=list(joint_positions),
            gripper=gripper,
            head_position=list(head_pos),
            head_rotation=list(head_rot),
        )
        self.writer.write(data)
        self.seq += 1

class DDSTeleopSubscriber:
    """DDS 订阅者"""
    
    def __init__(self, domain_id: int = 0):
        self.participant = domain.DomainParticipant(domain_id)
        self.topic = core.Topic(self.participant, "TeleopData", TeleopData)
        self.subscriber = sub.Subscriber(self.participant)
        self.reader = sub.DataReader(self.subscriber, self.topic)
    
    def read(self):
        samples = self.reader.take()
        if samples:
            return samples[-1]  # 返回最新数据
        return None
```

### 4.1 延迟来源分析

### 4.2 优化策略

```python
# 文件: latency_optimization.py
"""延迟优化策略实现"""
import numpy as np
import time
from collections import deque

class LatencyOptimizer:
    """延迟优化器"""
    
    def __init__(self):
        # 策略1: 数据压缩
        self.use_compression = True
        
        # 策略2: 预测补偿
        self.prediction_window = 5
        self.history = deque(maxlen=self.prediction_window)
        
        # 策略3: 平滑滤波
        self.smoothing_alpha = 0.3  # EMA 平滑系数
        
        # 策略4: 速度限制
        self.max_velocity = np.pi  # rad/s
        
        # 策略5: 死区
        self.deadzone = 0.01
    
    def compress_data(self, data: dict) -> bytes:
        """数据压缩（减少传输量）"""
        import zlib
        
        # 将 numpy 数组转为更紧凑的格式
        compact = {}
        for key, value in data.items():
            if isinstance(value, np.ndarray):
                compact[key] = {
                    'data': value.tobytes(),
                    'shape': value.shape,
                    'dtype': str(value.dtype),
                }
            else:
                compact[key] = value
        
        # 压缩
        import pickle
        raw = pickle.dumps(compact)
        compressed = zlib.compress(raw, level=1)  # 快速压缩
        
        return compressed
    
    def predict_future(self, current_data: dict, horizon: int = 3) -> dict:
        """
        基于历史数据预测未来状态（补偿延迟）
        
        使用线性外推法
        """
        self.history.append(current_data)
        
        if len(self.history) < 2:
            return current_data
        
        # 计算速度
        prev = self.history[-2]
        curr = self.history[-1]
        dt = curr.get('timestamp', 0) - prev.get('timestamp', 0)
        
        if dt <= 0:
            return current_data
        
        predicted = {}
        for key in ['joint_positions', 'head_position']:
            if key in curr and key in prev:
                velocity = (np.array(curr[key]) - np.array(prev[key])) / dt
                # 限速
                velocity = np.clip(velocity, -self.max_velocity, self.max_velocity)
                # 预测
                predicted[key] = np.array(curr[key]) + velocity * horizon * dt
        
        return {**current_data, **predicted}
    
    def smooth_signal(self, current: np.ndarray, previous: np.ndarray) -> np.ndarray:
        """指数移动平均平滑"""
        if previous is None:
            return current
        
        # 死区处理
        diff = current - previous
        diff[np.abs(diff) < self.deadzone] = 0
        
        # EMA 平滑
        smoothed = previous + self.smoothing_alpha * diff
        
        return smoothed
    
    def rate_limit(self, current: np.ndarray, previous: np.ndarray, dt: float) -> np.ndarray:
        """速率限制"""
        if previous is None:
            return current
        
        delta = current - previous
        max_delta = self.max_velocity * dt
        
        # 限制每步变化
        delta = np.clip(delta, -max_delta, max_delta)
        
        return previous + delta
```

### 4.3 延迟测量

```python
class LatencyMonitor:
    """延迟监控器"""
    
    def __init__(self, window_size: int = 100):
        self.window_size = window_size
        self.latencies = deque(maxlen=window_size)
        self.timestamps = deque(maxlen=window_size)
    
    def record(self, send_time: float, receive_time: float):
        """记录延迟"""
        latency = (receive_time - send_time) * 1000  # ms
        self.latencies.append(latency)
        self.timestamps.append(receive_time)
    
    def get_stats(self) -> dict:
        """获取统计信息"""
        if not self.latencies:
            return {}
        
        latencies = np.array(self.latencies)
        
        return {
            'mean_ms': np.mean(latencies),
            'median_ms': np.median(latencies),
            'p95_ms': np.percentile(latencies, 95),
            'p99_ms': np.percentile(latencies, 99),
            'max_ms': np.max(latencies),
            'min_ms': np.min(latencies),
            'std_ms': np.std(latencies),
            'jitter_ms': np.std(np.diff(latencies)),  # 抖动
        }
    
    def is_acceptable(self, threshold_ms: float = 50.0) -> bool:
        """判断延迟是否可接受"""
        stats = self.get_stats()
        if not stats:
            return False
        return stats['p95_ms'] < threshold_ms
```

```python
# ALOHA (Action Learning with Human Assist) 架构
"""
ALOHA 核心特点:
- 主从同构双臂系统
- 主臂（Leader）由人操作
- 从臂（Follower）跟随主臂运动
- 实时位置映射
- 支持双臂协作任务

硬件配置:
- 2x ViperX 300 机械臂（主臂）
- 2x ViperX 300 机械臂（从臂）
- 4x 相机（顶部+左右手腕）
"""

class ALOHATeleop:
    """ALOHA 遥操作框架"""
    
    def __init__(self):
        # 主臂（操作者控制）
        self.leader_left = LeaderArm(device_id="/dev/ttyUSB0")
        self.leader_right = LeaderArm(device_id="/dev/ttyUSB1")
        
        # 从臂（机器人执行）
        self.follower_left = FollowerArm(device_id="/dev/ttyUSB2")
        self.follower_right = FollowerArm(device_id="/dev/ttyUSB3")
        
        # 相机
        self.cameras = {
            'top': Camera(device_id=0),
            'left_wrist': Camera(device_id=1),
            'right_wrist': Camera(device_id=2),
        }
        
        # 数据录制
        self.recorder = ALOHADataRecorder()
    
    def teleop_step(self):
        """单步遥操作"""
        # 1. 读取主臂位置
        left_target = self.leader_left.read_position()
        right_target = self.leader_right.read_position()
        
        # 2. 从臂跟随（带平滑）
        self.follower_left.move_to(left_target)
        self.follower_right.move_to(right_target)
        
        # 3. 读取从臂实际位置
        left_actual = self.follower_left.read_position()
        right_actual = self.follower_right.read_position()
        
        # 4. 采集相机图像
        images = {name: cam.capture() for name, cam in self.cameras.items()}
        
        # 5. 记录数据
        self.recorder.record({
            'left_target': left_target,
            'right_target': right_target,
            'left_actual': left_actual,
            'right_actual': right_actual,
            'images': images,
        })
    
    def run(self, duration: float = 60.0, fps: int = 50):
        """运行遥操作"""
        import time
        dt = 1.0 / fps
        
        self.recorder.start_episode()
        
        start = time.time()
        while time.time() - start < duration:
            t0 = time.time()
            self.teleop_step()
            elapsed = time.time() - t0
            if elapsed < dt:
                time.sleep(dt - elapsed)
        
        self.recorder.end_episode()
```

```python
"""PICO VR 遥操作 Python SDK"""
import socket
import struct
import numpy as np
import time
from dataclasses import dataclass

@dataclass
class PICOState:
    """PICO VR 状态数据"""
    timestamp: float
    head_pos: np.ndarray        # [3]
    head_quat: np.ndarray       # [4]
    left_pos: np.ndarray        # [3]
    left_quat: np.ndarray       # [4]
    left_trigger: float
    left_grip: float
    left_thumbstick: np.ndarray # [2]
    left_buttons: int
    right_pos: np.ndarray       # [3]
    right_quat: np.ndarray      # [4]
    right_trigger: float
    right_grip: float
    right_thumbstick: np.ndarray # [2]
    right_buttons: int

class PICOReceiver:
    """PICO VR 数据接收器"""
    
    # 数据包格式: magic(4) + version(2) + seq(4) + timestamp(8)
    #           + head_pos(12) + head_quat(16)
    #           + left_pos(12) + left_quat(16) + left_trigger(4) + left_grip(4) + left_stick(8) + left_btn(4)
    #           + right_pos(12) + right_quat(16) + right_trigger(4) + right_grip(4) + right_stick(8) + right_btn(4)
    PACK_FORMAT = '<IHIddddddddddddddddddddd'
    PACK_SIZE = struct.calcsize(PACK_FORMAT)
    
    def __init__(self, port: int = 13131):
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_RCVBUF, 2**20)
        self.sock.bind(("0.0.0.0", port))
        self.sock.settimeout(1.0)
        
        self.latest_state = None
        self.running = False
        self.callback = None
    
    def receive_once(self) -> PICOState:
        """接收一次数据"""
        data, addr = self.sock.recvfrom(self.PACK_SIZE * 2)
        
        unpacked = struct.unpack(self.PACK_FORMAT, data[:self.PACK_SIZE])
        
        state = PICOState(
            timestamp=unpacked[3],
            head_pos=np.array(unpacked[4:7]),
            head_quat=np.array(unpacked[7:11]),
            left_pos=np.array(unpacked[11:14]),
            left_quat=np.array(unpacked[14:18]),
            left_trigger=unpacked[18],
            left_grip=unpacked[19],
            left_thumbstick=np.array(unpacked[20:22]),
            left_buttons=int(unpacked[22]),
            right_pos=np.array(unpacked[23:26]),
            right_quat=np.array(unpacked[26:30]),
            right_trigger=unpacked[30],
            right_grip=unpacked[31],
            right_thumbstick=np.array(unpacked[32:34]),
            right_buttons=int(unpacked[34]),
        )
        
        self.latest_state = state
        return state
    
    def run_continuous(self, callback=None):
        """持续接收数据"""
        self.running = True
        self.callback = callback
        
        while self.running:
            try:
                state = self.receive_once()
                if self.callback:
                    self.callback(state)
            except socket.timeout:
                continue
            except KeyboardInterrupt:
                break
    
    def stop(self):
        self.running = False
        self.sock.close()


# === 使用示例 ===
def on_pico_data(state: PICOState):
    """PICO 数据回调"""
    print(f"Head: pos={state.head_pos}, quat={state.head_quat}")
    print(f"Left: pos={state.left_pos}, trigger={state.left_trigger:.2f}")
    print(f"Right: pos={state.right_pos}, trigger={state.right_trigger:.2f}")

# 启动接收
receiver = PICOReceiver(port=13131)
receiver.run_continuous(callback=on_pico_data)
```

```python
# 文件: vr_to_g1_retarget.py
"""VR 到 G1 机器人的运动重定向"""
import numpy as np
from scipy.spatial.transform import Rotation as R

class VRToG1Retargeter:
    """VR 手柄数据 -> G1 机器人关节命令"""
    
    def __init__(self):
        # G1 关节配置
        self.g1_joint_names = [
            # 左臂 (7 DOF)
            'left_shoulder_pitch', 'left_shoulder_roll', 'left_shoulder_yaw',
            'left_elbow', 'left_wrist_roll', 'left_wrist_pitch', 'left_wrist_yaw',
            # 右臂 (7 DOF)
            'right_shoulder_pitch', 'right_shoulder_roll', 'right_shoulder_yaw',
            'right_elbow', 'right_wrist_roll', 'right_wrist_pitch', 'right_wrist_yaw',
            # 头部 (2 DOF)
            'head_pitch', 'head_yaw',
            # 腰部 (3 DOF)
            'waist_yaw', 'waist_roll', 'waist_pitch',
        ]
        
        # VR 到机器人的尺度缩放
        self.arm_scale = 0.8   # 手臂运动缩放
        self.head_scale = 0.5  # 头部运动缩放
    
    def retarget(self, pico_state) -> dict:
        """
        将 PICO VR 数据重定向为 G1 关节命令
        
        参数:
            pico_state: PICOState 对象
        
        返回:
            dict: 关节名 -> 角度 (rad)
        """
        commands = {}
        
        # === 头部重定向 ===
        head_euler = R.from_quat([
            pico_state.head_quat[1],  # x
            pico_state.head_quat[2],  # y
            pico_state.head_quat[3],  # z
            pico_state.head_quat[0],  # w
        ]).as_euler('XYZ')
        
        commands['head_pitch'] = np.clip(head_euler[0] * self.head_scale, -0.5, 0.5)
        commands['head_yaw'] = np.clip(head_euler[1] * self.head_scale, -0.87, 0.87)
        
        # === 左臂重定向 ===
        left_angles = self._retarget_arm(
            pico_state.left_pos, pico_state.left_quat, side='left'
        )
        for name, angle in left_angles.items():
            commands[name] = angle
        
        # === 右臂重定向 ===
        right_angles = self._retarget_arm(
            pico_state.right_pos, pico_state.right_quat, side='right'
        )
        for name, angle in right_angles.items():
            commands[name] = angle
        
        # === 夹爪/灵巧手控制 ===
        commands['left_gripper'] = pico_state.left_trigger
        commands['right_gripper'] = pico_state.right_trigger
        
        return commands
    
    def _retarget_arm(self, pos, quat, side='left'):
        """
        手臂重定向（简化版 IK）
        
        通过 VR 手柄位置和姿态，计算 G1 手臂关节角度
        """
        sign = 1 if side == 'left' else -1
        
        # 缩放位置
        scaled_pos = np.array(pos) * self.arm_scale
        
        # 肩关节：基于手柄相对肩部的位置
        shoulder_offset = np.array([0.0, sign * 0.2, 0.3])  # 肩部位置
        relative_pos = scaled_pos - shoulder_offset
        
        # 肩部俯仰
        shoulder_pitch = np.arctan2(-relative_pos[2], np.sqrt(relative_pos[0]**2 + relative_pos[1]**2))
        shoulder_pitch = np.clip(shoulder_pitch, -3.14, 1.57)
        
        # 肩部侧摆
        shoulder_roll = np.arctan2(sign * relative_pos[1], relative_pos[0])
        shoulder_roll = np.clip(shoulder_roll * sign, -0.43, 3.14)
        
        # 肘关节：基于手臂长度和距离
        arm_length = 0.6  # 上臂+前臂
        dist = np.linalg.norm(relative_pos)
        elbow = np.pi - 2 * np.arcsin(np.clip(dist / (2 * arm_length), 0, 1))
        elbow = np.clip(elbow, -1.57, 2.09)
        
        # 腕关节：从四元数提取
        rot = R.from_quat([quat[1], quat[2], quat[3], quat[0]])
        euler = rot.as_euler('XYZ')
        
        wrist_roll = np.clip(euler[2], -1.57, 1.57)
        wrist_pitch = np.clip(euler[0], -1.57, 1.57)
        wrist_yaw = np.clip(euler[1], -1.57, 1.57)
        
        return {
            f'{side}_shoulder_pitch': shoulder_pitch,
            f'{side}_shoulder_roll': shoulder_roll,
            f'{side}_shoulder_yaw': 0.0,  # 简化
            f'{side}_elbow': elbow,
            f'{side}_wrist_roll': wrist_roll,
            f'{side}_wrist_pitch': wrist_pitch,
            f'{side}_wrist_yaw': wrist_yaw,
        }
```

### 7.1 完整遥操作接收端

```python
# 文件: simple_udp_teleop.py
"""简单的 UDP 遥操作数据接收与机器人控制"""
import socket
import struct
import numpy as np
import time
import json
import h5py
from pathlib import Path

class SimpleTeleopReceiver:
    """简单遥操作数据接收器"""
    
    def __init__(self, listen_port: int = 9999):
        # UDP Socket
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_RCVBUF, 2**20)
        self.sock.bind(("0.0.0.0", listen_port))
        self.sock.settimeout(0.5)
        
        # 状态
        self.latest_data = None
        self.running = False
        self.stats = {
            'total_received': 0,
            'total_dropped': 0,
            'fps': 0,
            'latency_ms': 0,
        }
        
        # 数据录制
        self.recording = False
        self.recorded_data = []
        
        # FPS 计算
        self.frame_times = []
    
    def receive_loop(self):
        """接收循环"""
        self.running = True
        
        print(f"遥操作接收端已启动，监听端口 {self.sock.getsockname()[1]}")
        
        while self.running:
            try:
                # 接收数据
                data, addr = self.sock.recvfrom(65536)
                receive_time = time.time()
                
                # 解析
                parsed = self._parse_data(data)
                if parsed is None:
                    continue
                
                # 计算延迟
                latency = (receive_time - parsed['timestamp']) * 1000
                self.stats['latency_ms'] = latency
                
                # 更新最新数据
                self.latest_data = parsed
                self.stats['total_received'] += 1
                
                # FPS 计算
                self.frame_times.append(receive_time)
                if len(self.frame_times) > 100:
                    self.frame_times.pop(0)
                if len(self.frame_times) >= 2:
                    fps = len(self.frame_times) / (self.frame_times[-1] - self.frame_times[0])
                    self.stats['fps'] = fps
                
                # 录制
                if self.recording:
                    self.recorded_data.append({
                        **parsed,
                        'receive_time': receive_time,
                    })
                
                # 打印状态
                if self.stats['total_received'] % 100 == 0:
                    print(f"[{self.stats['total_received']}] "
                          f"FPS: {self.stats['fps']:.1f} | "
                          f"Latency: {self.stats['latency_ms']:.1f}ms | "
                          f"Data: {list(parsed.keys())}")
                
            except socket.timeout:
                continue
            except KeyboardInterrupt:
                print("\n停止接收")
                break
            except Exception as e:
                print(f"错误: {e}")
                self.stats['total_dropped'] += 1
        
        self.sock.close()
    
    def _parse_data(self, raw_data: bytes) -> dict:
        """解析接收到的数据"""
        try:
            # 使用 JSON 格式（简单但可读性好）
            text = raw_data.decode('utf-8')
            data = json.loads(text)
            
            # 转换 numpy 数组
            for key in ['joint_positions', 'head_pos', 'head_quat',
                        'left_pos', 'left_quat', 'right_pos', 'right_quat']:
                if key in data:
                    data[key] = np.array(data[key])
            
            return data
        except Exception as e:
            print(f"解析错误: {e}")
            return None
    
    def start_recording(self):
        """开始录制"""
        self.recording = True
        self.recorded_data = []
        print("开始录制遥操作数据...")
    
    def stop_recording(self, output_path: str):
        """停止录制并保存"""
        self.recording = False
        print(f"停止录制，共 {len(self.recorded_data)} 帧")
        
        # 保存为 HDF5
        self._save_hdf5(output_path)
        print(f"数据已保存到 {output_path}")
    
    def _save_hdf5(self, output_path: str):
        """保存为 HDF5 格式"""
        with h5py.File(output_path, 'w') as f:
            # 元数据
            f.attrs['num_frames'] = len(self.recorded_data)
            f.attrs['fps'] = self.stats['fps']
            f.attrs['avg_latency_ms'] = self.stats['latency_ms']
            f.attrs['recorded_at'] = time.strftime('%Y-%m-%d %H:%M:%S')
            
            # 创建组
            action_grp = f.create_group('action')
            obs_grp = f.create_group('observation')
            
            # 提取数据
            for i, frame in enumerate(self.recorded_data):
                # 动作数据
                if 'joint_positions' in frame:
                    if i == 0:
                        action_grp.create_dataset(
                            'joint_positions',
                            data=np.zeros((len(self.recorded_data), len(frame['joint_positions']))),
                        )
                    action_grp['joint_positions'][i] = frame['joint_positions']
                
                # 观测数据
                if 'head_pos' in frame:
                    if i == 0:
                        obs_grp.create_dataset(
                            'head_pos',
                            data=np.zeros((len(self.recorded_data), 3)),
                        )
                    obs_grp['head_pos'][i] = frame['head_pos']
            
            # 时间戳
            timestamps = [d['timestamp'] for d in self.recorded_data]
            f.create_dataset('timestamps', data=timestamps)
```
