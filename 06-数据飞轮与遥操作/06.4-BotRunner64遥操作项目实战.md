## 1. Teleopit 项目深度分析

### 1.1 项目概述

Teleopit 是 BotRunner64 项目中的核心遥操作模块，它实现了从人类动作到人形机器人（Unitree G1/H1）的全身运动重定向。项目支持 VR 遥操作和 BVH 动捕数据驱动两种模式。

### 1.2 架构设计详解

```
┌─────────────────────────────────────────────────────────┐
│                   Teleopit 架构                          │
├─────────────────────────────────────────────────────────┤
│                                                          │
│  ┌─────────────┐                                        │
│  │  输入层      │  PICO VR / BVH 动捕 / 键盘             │

"""Teleopit BVH → G1 运动重定向"""
import numpy as np
from scipy.spatial.transform import Rotation as R
from typing import Optional, Dict

class BVHToG1Retargeter:
    """BVH 动作数据到 G1 机器人的运动重定向"""
    
    # G1 关节配置 (29 DOF)
    G1_JOINTS = {

        # 头部 (2 DOF)
        'head_pitch': {'id': 15, 'range': (-0.87, 0.52), 'bvh': 'Head_X'},
        'head_yaw': {'id': 16, 'range': (-2.87, 2.87), 'bvh': 'Head_Y'},
        
        # 左臂 (6 DOF, 不含末端)
        'left_shoulder_pitch': {'id': 17, 'range': (-3.14, 1.57), 'bvh': 'LeftArm_X'},
        'left_shoulder_roll': {'id': 18, 'range': (-0.43, 3.14), 'bvh': 'LeftArm_Z'},
        'left_shoulder_yaw': {'id': 19, 'range': (-2.87, 2.87), 'bvh': 'LeftArm_Y'},
        'left_elbow': {'id': 20, 'range': (-1.57, 2.09), 'bvh': 'LeftForeArm_X'},
        'left_wrist_roll': {'id': 21, 'range': (-1.57, 1.57), 'bvh': 'LeftHand_Z'},
        'left_wrist_pitch': {'id': 22, 'range': (-1.57, 1.57), 'bvh': 'LeftHand_X'},
        
        # 右臂 (6 DOF)
        'right_shoulder_pitch': {'id': 23, 'range': (-3.14, 1.57), 'bvh': 'RightArm_X'},
        'right_shoulder_roll': {'id': 24, 'range': (-0.43, 3.14), 'bvh': 'RightArm_Z'},
        'right_shoulder_yaw': {'id': 25, 'range': (-2.87, 2.87), 'bvh': 'RightArm_Y'},
        'right_elbow': {'id': 26, 'range': (-1.57, 2.09), 'bvh': 'RightForeArm_X'},
        'right_wrist_roll': {'id': 27, 'range': (-1.57, 1.57), 'bvh': 'RightHand_Z'},
        'right_wrist_pitch': {'id': 28, 'range': (-1.57, 1.57), 'bvh': 'RightHand_X'},
    }
    
    def __init__(self, scale: float = 1.0, smoothing: float = 0.3):
        """
        参数:
            scale: BVH 到机器人的尺度缩放
            smoothing: 平滑系数 (0=无平滑, 1=完全平滑)
        """
        self.scale = scale
        self.smoothing = smoothing
        
        # 上一帧数据（用于平滑）
        self.prev_angles = np.zeros(29)
        
        # 根关节偏移
        self.root_offset = np.array([0.0, 0.0, 0.75])  # G1 站立高度
    
    def retarget(self, bvh_frame: dict) -> dict:
        """
        将 BVH 帧数据重定向为 G1 关节命令
        
        参数:
            bvh_frame: BVH 帧数据
                {
                    'root_position': [x, y, z],
                    'root_rotation': [w, x, y, z],
                    'joint_rotations': {joint_name: [rx, ry, rz]},
                }
        
        返回:
            G1 关节命令字典
        """
        # 1. 处理根关节（骨盆位置）
        root_pos = np.array(bvh_frame['root_position']) * self.scale
        root_pos += self.root_offset
        
        root_quat = np.array(bvh_frame['root_rotation'])
        root_euler = R.from_quat([root_quat[1], root_quat[2], root_quat[3], root_quat[0]]).as_euler('XYZ')
        
        # 2. 重定向各关节
        target_angles = np.zeros(29)
        
        for joint_name, config in self.G1_JOINTS.items():
            bvh_joint = config['bvh']
            
            if bvh_joint in bvh_frame['joint_rotations']:
                # 从 BVH 关节旋转中提取对应轴的角度
                bvh_rotation = bvh_frame['joint_rotations'][bvh_joint]
                
                # 提取指定轴角度
                axis = bvh_joint.split('_')[-1]  # X, Y, Z
                axis_idx = {'X': 0, 'Y': 1, 'Z': 2}[axis]
                angle = bvh_rotation[axis_idx]
                
                # 缩放
                angle *= self.scale
                
                # 限位
                angle = np.clip(angle, config['range'][0], config['range'][1])
                
                target_angles[config['id']] = angle
        
        # 3. 平滑滤波
        if self.smoothing > 0:
            target_angles = (
                (1 - self.smoothing) * target_angles +
                self.smoothing * self.prev_angles
            )
        
        self.prev_angles = target_angles.copy()
        
        # 4. 构建命令
        commands = {
            'root_position': root_pos.tolist(),
            'root_rotation': root_quat.tolist(),
            'joint_angles': target_angles.tolist(),
            'joint_names': list(self.G1_JOINTS.keys()),
        }
        
        return commands
    
    def retarget_sequence(self, bvh_frames: list) -> list:
        """重定向整个 BVH 序列"""
        self.prev_angles = np.zeros(29)  # 重置
        return [self.retarget(frame) for frame in bvh_frames]
```

### 1.4 PICO 4 VR 遥操作完整链路

```python
# 文件: teleopit_pico_pipeline.py
"""Teleopit PICO VR 遥操作完整链路"""
import numpy as np
import time
import h5py
from typing import Optional

class PICOVRPipeline:
    """PICO VR 遥操作完整链路"""
    
    def __init__(self, 
                 pico_port: int = 13131,
                 robot_type: str = "g1",
                 record: bool = True,
                 record_path: str = "./teleop_data.hdf5"):
        
        # 1. PICO 数据接收器
        from pico_teleop import PICOReceiver
        self.pico = PICOReceiver(port=pico_port)
        
        # 2. VR 到机器人重定向
        from vr_to_g1_retarget import VRToG1Retargeter
        self.retargeter = VRToG1Retargeter()
        
        # 3. 机器人接口
        self.robot_type = robot_type
        self.robot = None  # 按需初始化
        
        # 4. 安全策略
        self.safety = SafetyChecker(robot_type=robot_type)
        
        # 5. 数据录制
        self.record = record
        self.recorder = HDF5Recorder(record_path) if record else None
        
        # 6. 平滑滤波
        self.smoothing = SignalSmoother(alpha=0.3)
        
        # 运行状态
        self.running = False
        self.frame_count = 0
    
    def connect_robot(self, ip: str = "192.168.123.161"):
        """连接真实机器人"""
        if self.robot_type == "g1":
            from unitree_sdk import G1Robot
            self.robot = G1Robot(ip=ip)
        elif self.robot_type == "h1":
            from unitree_sdk import H1Robot
            self.robot = H1Robot(ip=ip)
        
        print(f"已连接 {self.robot_type} 机器人: {ip}")
    
    def connect_sim(self, model_path: str = "g1.xml"):
        """连接仿真环境"""
        import mujoco
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.data = mujoco.MjData(self.model)
        self.model.opt.timestep = 0.002
        print(f"已连接 MuJoCo 仿真: {model_path}")
    
    def run(self, duration: float = 60.0):
        """运行遥操作"""
        self.running = True
        start_time = time.time()
        
        print(f"=== PICO VR 遥操作启动 ===")
        print(f"  模式: {'真实机器人' if self.robot else '仿真'}")
        print(f"  录制: {'开启' if self.record else '关闭'}")
        print(f"  时长: {duration}s")
        
        if self.record:
            self.recorder.start_episode({
                'robot_type': self.robot_type,
                'input_device': 'pico_vr',
                'fps': 50,
            })
        
        try:
            while self.running and (time.time() - start_time) < duration:
                loop_start = time.time()
                
                # 1. 接收 PICO 数据
                pico_state = self.pico.receive_once()
                if pico_state is None:
                    continue
                
                # 2. 运动重定向
                robot_cmd = self.retargeter.retarget(pico_state)
                
                # 3. 平滑滤波
                robot_cmd = self.smoothing.smooth(robot_cmd)
                
                # 4. 安全检查
                if not self.safety.check(robot_cmd):
                    print(f"[WARNING] 安全检查失败，跳过此帧")
                    continue
                
                # 5. 执行
                if self.robot:
                    self._execute_real(robot_cmd)
                else:
                    self._execute_sim(robot_cmd)
                
                # 6. 获取反馈状态
                feedback = self._get_feedback()
                
                # 7. 录制数据
                if self.record:
                    self.recorder.add_frame({
                        'pico_state': pico_state,
                        'robot_cmd': robot_cmd,
                        'robot_state': feedback,
                        'timestamp': time.time(),
                    })
                
                self.frame_count += 1
                
                # FPS 控制
                elapsed = time.time() - loop_start
                target_dt = 1.0 / 50  # 50Hz
                if elapsed < target_dt:
                    time.sleep(target_dt - elapsed)
                
                # 打印状态
                if self.frame_count % 100 == 0:
                    fps = self.frame_count / (time.time() - start_time)
                    print(f"  Frame {self.frame_count} | FPS: {fps:.1f}")
        
        except KeyboardInterrupt:
            print("\n用户中断")
        
        finally:
            self.running = False
            if self.record:
                self.recorder.end_episode()
            
            elapsed = time.time() - start_time
            print(f"\n=== 遥操作结束 ===")
            print(f"  总帧数: {self.frame_count}")
            print(f"  耗时: {elapsed:.1f}s")
            print(f"  平均FPS: {self.frame_count / elapsed:.1f}")
    
    def _execute_real(self, cmd: dict):
        """执行到真实机器人"""
        # 发送关节命令
        self.robot.send_joint_command(
            cmd['joint_angles'],
            kp=80.0,
            kd=3.0,
        )
    
    def _execute_sim(self, cmd: dict):
        """执行到仿真"""
        import mujoco
        # 设置控制
        self.data.ctrl[:] = cmd['joint_angles']
        # 推进仿真
        for _ in range(10):  # 500Hz sim, 50Hz control
            mujoco.mj_step(self.model, self.data)
    
    def _get_feedback(self) -> dict:
        """获取反馈状态"""
        if self.robot:
            return self.robot.get_state()
        else:
            return {
                'joint_positions': self.data.qpos[7:].copy(),
                'joint_velocities': self.data.qvel[6:].copy(),
            }


class SafetyChecker:
    """安全检查器"""
    
    def __init__(self, robot_type: str = "g1"):
        # 关节限位
        self.joint_limits = {
            'g1': {
                'min': np.full(29, -3.14),
                'max': np.full(29, 3.14),
            }
        }
        
        # 速度限制
        self.max_velocity = 3.0  # rad/s
        
        # 自碰撞检查（简化）
        self.check_self_collision = True
    
    def check(self, cmd: dict) -> bool:
        """安全检查"""
        angles = np.array(cmd.get('joint_angles', []))
        
        if len(angles) == 0:
            return False
        
        # 1. 关节限位检查
        limits = self.joint_limits.get('g1')
        if limits:
            if np.any(angles < limits['min']) or np.any(angles > limits['max']):
                return False
        
        # 2. 速度检查（与上一帧比较）
        if hasattr(self, 'prev_angles'):
            velocity = np.abs(angles - self.prev_angles) / 0.02  # 50Hz
            if np.any(velocity > self.max_velocity):
                return False
        
        self.prev_angles = angles.copy()
        
        return True


class SignalSmoother:
    """信号平滑器"""
    
    def __init__(self, alpha: float = 0.3):
        self.alpha = alpha
        self.prev = None
    
    def smooth(self, cmd: dict) -> dict:
        """指数移动平均平滑"""
        if self.prev is None:
            self.prev = cmd
            return cmd
        
        smoothed = cmd.copy()
        angles = np.array(cmd['joint_angles'])
        prev_angles = np.array(self.prev['joint_angles'])
        
        smoothed['joint_angles'] = (
            (1 - self.alpha) * angles + self.alpha * prev_angles
        ).tolist()
        
        self.prev = smoothed
        return smoothed


class HDF5Recorder:
    """HDF5 数据录制器"""
    
    def __init__(self, file_path: str):
        self.file_path = file_path
        self.file = None
        self.frames = []
    
    def start_episode(self, metadata: dict):
        """开始录制"""
        self.file = h5py.File(self.file_path, 'w')
        for key, value in metadata.items():
            self.file.attrs[key] = value
        self.frames = []
        print(f"开始录制: {self.file_path}")
    
    def add_frame(self, frame_data: dict):
        """添加帧"""
        self.frames.append(frame_data)
    
    def end_episode(self):
        """结束录制"""
        if not self.file or not self.frames:
            return
        
        # 保存帧数据
        num_frames = len(self.frames)
        
        # 关节命令
        joint_angles = np.array([f['robot_cmd']['joint_angles'] for f in self.frames])
        self.file.create_dataset('action/joint_angles', data=joint_angles)
        
        # PICO 状态
        head_pos = np.array([f['pico_state'].head_pos for f in self.frames])
        self.file.create_dataset('observation/head_pos', data=head_pos)
        
        left_pos = np.array([f['pico_state'].left_pos for f in self.frames])
        self.file.create_dataset('observation/left_pos', data=left_pos)
        
        right_pos = np.array([f['pico_state'].right_pos for f in self.frames])
        self.file.create_dataset('observation/right_pos', data=right_pos)
        
        # 机器人状态
        robot_states = np.array([f['robot_state']['joint_positions'] for f in self.frames])
        self.file.create_dataset('observation/joint_positions', data=robot_states)
        
        # 时间戳
        timestamps = np.array([f['timestamp'] for f in self.frames])
        self.file.create_dataset('timestamps', data=timestamps)
        
        # 元数据
        self.file.attrs['num_frames'] = num_frames
        self.file.attrs['duration'] = timestamps[-1] - timestamps[0] if num_frames > 1 else 0
        self.file.attrs['fps'] = num_frames / max(0.1, self.file.attrs['duration'])
        
        self.file.close()
        print(f"录制完成: {num_frames} 帧, {self.file.attrs['fps']:.1f} FPS")
```

### 2.1 MediaPipe 手部追踪

```python
# 文件: somehand_mediapipe.py
"""somehand MediaPipe 手部追踪"""
import cv2
import mediapipe as mp
import numpy as np

class HandTracker:
    """MediaPipe 手部追踪"""
    
    def __init__(self, max_hands: int = 2, min_detection_conf: float = 0.5):
        self.mp_hands = mp.solutions.hands
        self.mp_drawing = mp.solutions.drawing_utils
        
        self.hands = self.mp_hands.Hands(
            static_image_mode=False,
            max_num_hands=max_hands,
            min_detection_confidence=min_detection_conf,
            min_tracking_confidence=0.5,
        )
    
    def process(self, image: np.ndarray) -> dict:
        """
        处理图像，提取手部关键点
        
        返回:
            {
                'left': {landmarks, angles, gesture},
                'right': {landmarks, angles, gesture},
            }
        """
        # 转换颜色空间
        rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        results = self.hands.process(rgb)
        
        output = {'left': None, 'right': None}
        
        if results.multi_hand_landmarks:
            for idx, (hand_landmarks, handedness) in enumerate(
                zip(results.multi_hand_landmarks, results.multi_handedness)
            ):
                # 判断左右手
                hand_label = handedness.label.lower()  # 'left' or 'right'
                
                # 提取关键点 (21个)
                landmarks = np.array([
                    [lm.x, lm.y, lm.z]
                    for lm in hand_landmarks.landmark
                ])  # [21, 3]
                
                # 计算关节角度
                angles = self._compute_finger_angles(landmarks)
                
                # 识别手势
                gesture = self._recognize_gesture(landmarks, angles)
                
                output[hand_label] = {
                    'landmarks': landmarks,
                    'angles': angles,
                    'gesture': gesture,
                    'confidence': handedness.score,
                }
        
        return output
    
    def _compute_finger_angles(self, landmarks: np.ndarray) -> dict:
        """
        计算各手指关节角度
        
        MediaPipe 21个关键点:
            0: 手腕
            1-4: 拇指 (CMC, MCP, IP, TIP)
            5-8: 食指 (MCP, PIP, DIP, TIP)
            9-12: 中指
            13-16: 无名指
            17-20: 小指
        """
        angles = {}
        
        # 拇指
        angles['thumb'] = self._compute_finger_curve(
            landmarks, [0, 1, 2, 3, 4], name='thumb'
        )
        
        # 食指
        angles['index'] = self._compute_finger_curve(
            landmarks, [0, 5, 6, 7, 8], name='index'
        )
        
        # 中指
        angles['middle'] = self._compute_finger_curve(
            landmarks, [0, 9, 10, 11, 12], name='middle'
        )
        
        # 无名指
        angles['ring'] = self._compute_finger_curve(
            landmarks, [0, 13, 14, 15, 16], name='ring'
        )
        
        # 小指
        angles['pinky'] = self._compute_finger_curve(
            landmarks, [0, 17, 18, 19, 20], name='pinky'
        )
        
        return angles
    
    def _compute_finger_curve(self, landmarks, indices, name):
        """计算单根手指的弯曲角度"""
        angles = {}
        
        for i in range(len(indices) - 2):
            p1 = landmarks[indices[i]]
            p2 = landmarks[indices[i + 1]]
            p3 = landmarks[indices[i + 2]]
            
            # 计算两段向量的夹角
            v1 = p2 - p1
            v2 = p3 - p2
            
            # 角度
            cos_angle = np.dot(v1, v2) / (
                np.linalg.norm(v1) * np.linalg.norm(v2) + 1e-8
            )
            angle = np.arccos(np.clip(cos_angle, -1, 1))
            
            angles[f'{name}_joint_{i+1}'] = angle
        
        return angles
    
    def _recognize_gesture(self, landmarks, angles) -> str:
        """识别手势"""
        # 简化的手势识别
        finger_extended = {}
        
        for finger in ['index', 'middle', 'ring', 'pinky']:
            # 指尖到手腕的距离 vs MCP到手腕的距离
            tip_idx = {'index': 8, 'middle': 12, 'ring': 16, 'pinky': 20}[finger]
            mcp_idx = {'index': 5, 'middle': 9, 'ring': 13, 'pinky': 17}[finger]
            
            tip_dist = np.linalg.norm(landmarks[tip_idx] - landmarks[0])
            mcp_dist = np.linalg.norm(landmarks[mcp_idx] - landmarks[0])
            
            finger_extended[finger] = tip_dist > mcp_dist * 1.5
        
        # 判断手势
        if all(not ext for ext in finger_extended.values()):
            return 'fist'
        elif all(finger_extended.values()):
            return 'open'
        elif finger_extended['index'] and not finger_extended['middle']:
            return 'point'
        elif finger_extended['index'] and finger_extended['middle']:
            return 'peace'
        else:
            return 'unknown'

import socket

from typing import Optional, Callable

@dataclass
class PICOVRData:
    """PICO VR 数据包"""
    timestamp: float
    head_pos: np.ndarray       # [3]
    head_quat: np.ndarray      # [4] (w, x, y, z)
    left_pos: np.ndarray       # [3]
    left_quat: np.ndarray      # [4]
    left_trigger: float        # [0, 1]
    left_grip: float           # [0, 1]
    left_thumbstick: np.ndarray  # [2]
    right_pos: np.ndarray      # [3]
    right_quat: np.ndarray     # [4]
    right_trigger: float       # [0, 1]
    right_grip: float          # [0, 1]
    right_thumbstick: np.ndarray  # [2]


class PICOBridgeSDK:
    """pico-bridge Python SDK"""
    
    def __init__(self, listen_port: int = 13131, timeout: float = 1.0):
        self.listen_port = listen_port
        self.timeout = timeout
        
        # UDP Socket
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_RCVBUF, 2**20)
        self.sock.bind(("0.0.0.0", listen_port))
        self.sock.settimeout(timeout)
        
        # 最新数据
        self.latest_data: Optional[PICOVRData] = None
        
        # 回调
        self.callback: Optional[Callable] = None
        
        # 统计
        self.frame_count = 0
        self.start_time = time.time()
        
        print(f"PICO Bridge SDK 已启动, 监听端口: {listen_port}")
    
    def receive(self) -> Optional[PICOVRData]:
        """接收一帧数据"""
        try:
            data, addr = self.sock.recvfrom(65536)
            parsed = json.loads(data.decode('utf-8'))
            
            vr_data = PICOVRData(
                timestamp=parsed['timestamp'],
                head_pos=np.array(parsed['head_pos']),
                head_quat=np.array(parsed['head_quat']),
                left_pos=np.array(parsed['left_pos']),
                left_quat=np.array(parsed['left_quat']),
                left_trigger=parsed['left_trigger'],
                left_grip=parsed['left_grip'],
                left_thumbstick=np.array(parsed['left_thumbstick']),
                right_pos=np.array(parsed['right_pos']),
                right_quat=np.array(parsed['right_quat']),
                right_trigger=parsed['right_trigger'],
                right_grip=parsed['right_grip'],
                right_thumbstick=np.array(parsed['right_thumbstick']),
            )
            
            self.latest_data = vr_data
            self.frame_count += 1
            
            return vr_data
            
        except socket.timeout:
            return None
        except Exception as e:
            print(f"接收错误: {e}")
            return None
    
    def run_continuous(self, callback: Optional[Callable] = None):
        """持续接收"""
        self.callback = callback
        print("开始持续接收 PICO 数据...")
        
        try:
            while True:
                data = self.receive()
                if data and self.callback:
                    self.callback(data)
        except KeyboardInterrupt:
            print("\n停止接收")
    
    def get_fps(self) -> float:
        """获取接收 FPS"""
        elapsed = time.time() - self.start_time
        return self.frame_count / max(0.1, elapsed)
    
    def close(self):
        """关闭"""
        self.sock.close()
        print(f"PICO Bridge SDK 已关闭, 总接收: {self.frame_count} 帧")

"""PICO 视频推流"""
import cv2
import numpy as np
import socket
import struct
import time
from threading import Thread

class PICOVideoStream:
    """PICO 视频推流（从 PC 到头显）"""
    
    def __init__(self, 
                 target_ip: str = "192.168.1.100",
                 target_port: int = 12345,
                 width: int = 1280,
                 height: int = 720,
                 fps: int = 30):
        self.target = (target_ip, target_port)
        self.width = width
        self.height = height
        self.fps = fps
        
        # UDP Socket
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_SNDBUF, 2**20)
        
        # 视频编码
        self.encoder = cv2.VideoWriter_fourcc(*'MJPG')
        
        self.running = False
    
    def send_frame(self, frame: np.ndarray):
        """发送一帧视频"""
        # 编码为 JPEG
        _, encoded = cv2.imencode('.jpg', frame, 
                                   [cv2.IMWRITE_JPEG_QUALITY, 80])
        
        # 分块发送（UDP 数据包大小限制）
        chunk_size = 65000  # 最大 UDP 数据包
        data = encoded.tobytes()
        total_size = len(data)
        num_chunks = (total_size + chunk_size - 1) // chunk_size
        
        # 发送头部
        header = struct.pack('III', total_size, num_chunks, time.time())
        self.sock.sendto(header, self.target)
        
        # 发送数据块
        for i in range(num_chunks):
            start = i * chunk_size
            end = min(start + chunk_size, total_size)
            chunk = data[start:end]
            
            # 添加序号
            packet = struct.pack('I', i) + chunk
            self.sock.sendto(packet, self.target)
    
    def stream_camera(self, camera_id: int = 0):
        """推流摄像头视频"""
        cap = cv2.VideoCapture(camera_id)
        cap.set(cv2.CAP_PROP_FRAME_WIDTH, self.width)
        cap.set(cv2.CAP_PROP_FRAME_HEIGHT, self.height)
        cap.set(cv2.CAP_PROP_FPS, self.fps)
        
        self.running = True
        print(f"视频推流已启动 -> {self.target}")
        
        frame_interval = 1.0 / self.fps
        
        while self.running:
            start = time.time()
            
            ret, frame = cap.read()
            if not ret:
                break
            
            self.send_frame(frame)
            
            elapsed = time.time() - start
            if elapsed < frame_interval:
                time.sleep(frame_interval - elapsed)
        
        cap.release()
        print("视频推流已停止")
    
    def stop(self):
        self.running = False
        self.sock.close()

import numpy as np
from scipy.spatial.transform import Rotation as R

class OpenNeckController:
    """头颈控制器"""
    
    def __init__(self, robot_type: str = "g1"):
        self.robot_type = robot_type
        
        # G1 头颈关节配置
        if robot_type == "g1":
            self.neck_joints = {
                'head_pitch': {
                    'id': 15,
                    'range': (-0.87, 0.52),  # rad
                    'max_velocity': 2.0,
                },
                'head_yaw': {
                    'id': 16,
                    'range': (-2.87, 2.87),
                    'max_velocity': 2.0,
                },
            }
        elif robot_type == "h1":
            self.neck_joints = {
                'head_pitch': {
                    'id': 17,
                    'range': (-0.87, 0.52),
                    'max_velocity': 2.0,
                },
                'head_yaw': {
                    'id': 18,
                    'range': (-2.87, 2.87),
                    'max_velocity': 2.0,
                },
            }
        
        # 平滑
        self.smoothing = 0.3
        self.prev_angles = np.zeros(len(self.neck_joints))
        
        # 死区
        self.deadzone = 0.05  # rad
    
    def control_from_vr(self, head_quat: np.ndarray) -> dict:
        """
        从 VR 头部姿态控制机器人头颈
        
        参数:
            head_quat: [w, x, y, z] VR 头部四元数
        """
        # 转换为欧拉角
        quat_xyzw = [head_quat[1], head_quat[2], head_quat[3], head_quat[0]]
        euler = R.from_quat(quat_xyzw).as_euler('XYZ')
        
        # 提取 pitch 和 yaw
        pitch = euler[0]  # 俯仰
        yaw = euler[2]    # 偏航
        
        # 死区处理
        if abs(pitch) < self.deadzone:
            pitch = 0.0
        if abs(yaw) < self.deadzone:
            yaw = 0.0
        
        # 限位
        pitch = np.clip(pitch, 
                       self.neck_joints['head_pitch']['range'][0],
                       self.neck_joints['head_pitch']['range'][1])
        yaw = np.clip(yaw,
                     self.neck_joints['head_yaw']['range'][0],
                     self.neck_joints['head_yaw']['range'][1])
        
        # 平滑
        target = np.array([pitch, yaw])
        target = (1 - self.smoothing) * target + self.smoothing * self.prev_angles
        self.prev_angles = target.copy()
        
        return {
            'head_pitch': target[0],
            'head_yaw': target[1],
        }
    
    def control_from_bvh(self, bvh_head_rotation: list) -> dict:
        """从 BVH 头部旋转控制"""
        # 类似 VR 控制
        return self.control_from_vr(np.array(bvh_head_rotation))
    
    def neutral_pose(self) -> dict:
        """返回中性姿态"""
        return {
            'head_pitch': 0.0,
            'head_yaw': 0.0,
        }
```
