## 一、Sim2Real Gap 问题分析

### 1.1 什么是 Sim2Real Gap

```
Sim2Real Gap = 仿真环境与真实环境之间的差异

仿真环境 ≠ 真实环境，导致在仿真中训练的策略在真机上失效

主要差异来源：

1. 物理差异
   - 摩擦系数、质量、惯性不同
   - 接触动力学简化
   - 空气阻力等被忽略

2. 视觉差异
   - 渲染图像 vs 真实相机图像
   - 光照、纹理、色彩不同
   - 传感器噪声不同

3. 动力学差异
   - 控制延迟不同
   - 电机响应不同
   - 关节摩擦/回差不同

4. 传感器差异
   - 噪声特性不同
   - 采样率不同
   - 标定误差

5. 环境差异
   - 温度/湿度影响
   - 随机干扰
   - 磨损
```

### 1.2 Gap 的影响

```python
"""
Sim2Real Gap 的影响示例

仿真训练成功率：95%
真机成功率：10-30%（没有 Sim2Real 技术时）

常见的失败模式：
  1. 策略在真机上过于保守（不接触物体）
  2. 策略过于激进（用力过猛）
  3. 视觉感知错误（无法识别物体）
  4. 抓取位置偏移
  5. 摔倒（运动控制场景）
"""
```

## 二、域随机化（Domain Randomization）

### 2.1 核心思想

```
域随机化（Domain Randomization, DR）：
  在训练时随机化仿真参数，
  使策略对参数变化具有鲁棒性，
  从而在真实环境（被视为随机化的一个样本）中工作。

核心假设：
  如果策略能在参数大范围随机化的仿真中工作，
  那么真实环境参数就在这个范围内，
  策略也能在真机工作。
```

### 2.2 域随机化配置示例

```python
import numpy as np

class DomainRandomizationConfig:
    """完整的域随机化配置"""
    
    def __init__(self):
        self.physics = {
            # 摩擦系数
            "friction": {"range": (0.3, 1.5), "distribution": "uniform"},
            # 质量缩放
            "mass_scale": {"range": (0.8, 1.2), "distribution": "uniform"},
            # 质心偏移
            "com_offset": {"range": (-0.02, 0.02), "distribution": "uniform"},
            # 阻尼
            "damping": {"range": (0.1, 0.5), "distribution": "uniform"},
            # 刚度
            "stiffness": {"range": (50, 150), "distribution": "uniform"},
            # 恢复系数
            "restitution": {"range": (0.0, 0.3), "distribution": "uniform"},
        }
        
        self.visual = {
            # 光照
            "light_position": {"range": (-3, 3), "distribution": "uniform", "dim": 3},
            "light_intensity": {"range": (0.5, 1.5), "distribution": "uniform"},
            "light_color": {"range": (0.8, 1.0), "distribution": "uniform", "dim": 3},
            
            # 纹理
            "texture_color": {"range": (0, 1), "distribution": "uniform", "dim": 3},
            "texture_noise": {"range": (0, 0.1), "distribution": "uniform"},
            
            # 相机
            "camera_position_noise": {"range": (-0.02, 0.02), "distribution": "uniform"},
            "camera_angle_noise": {"range": (-0.02, 0.02), "distribution": "uniform"},
            "camera_fov": {"range": (45, 75), "distribution": "uniform"},
        }
        
        self.dynamics = {
            # 控制延迟
            "control_delay": {"range": (0, 3), "distribution": "uniform_int"},
            # 动作噪声
            "action_noise_std": {"range": (0, 0.05), "distribution": "uniform"},
            # 观测噪声
            "observation_noise_std": {"range": (0, 0.02), "distribution": "uniform"},
            # 电机强度
            "motor_strength": {"range": (0.9, 1.1), "distribution": "uniform"},
            # 关节摩擦
            "joint_friction": {"range": (0, 0.05), "distribution": "uniform"},
        }
    
    def sample(self, category=None):
        """随机采样参数"""
        if category:
            params = getattr(self, category)
        else:
            params = {**self.physics, **self.visual, **self.dynamics}
        
        sampled = {}
        for key, config in params.items():
            if config["distribution"] == "uniform":
                if "dim" in config:
                    sampled[key] = np.random.uniform(
                        config["range"][0], config["range"][1], config["dim"]
                    )
                else:
                    sampled[key] = np.random.uniform(*config["range"])
            elif config["distribution"] == "uniform_int":
                sampled[key] = np.random.randint(*config["range"])
        
        return sampled
    
    def apply_to_env(self, env, params=None):
        """将随机化参数应用到仿真环境"""
        if params is None:
            params = self.sample()
        
        # 物理参数
        if "friction" in params:
            env.set_friction(params["friction"])
        if "mass_scale" in params:
            env.set_mass_scale(params["mass_scale"])
        
        # 视觉参数
        if "light_position" in params:
            env.set_light_position(params["light_position"])
        if "light_intensity" in params:
            env.set_light_intensity(params["light_intensity"])
        
        # 动力学参数
        if "control_delay" in params:
            env.set_control_delay(params["control_delay"])
        if "action_noise_std" in params:
            env.set_action_noise(params["action_noise_std"])
        
        return params

# 使用
dr_config = DomainRandomizationConfig()
sampled_params = dr_config.sample()
print("随机化参数示例:")
for k, v in sampled_params.items():
    print(f"  {k}: {v:.4f}")
```

### 2.3 渐进域随机化

```python
"""
渐进域随机化 (Progressive Domain Randomization)

从窄随机化范围开始，逐步增大范围

优势：
  - 训练初期更容易收敛
  - 逐步增加鲁棒性
  - 避免一开始就因为过大的随机化导致训练失败
"""

class ProgressiveDR:
    """渐进域随机化"""
    def __init__(self, base_config, max_config, num_stages=5):
        self.base = base_config  # 最小随机化
        self.max = max_config    # 最大随机化
        self.num_stages = num_stages
        self.current_stage = 0
    
    def get_current_range(self):
        """获取当前阶段的随机化范围"""
        progress = self.current_stage / (self.num_stages - 1)
        
        current = {}
        for key in self.base:
            base_val = self.base[key]
            max_val = self.max[key]
            # 线性插值
            if isinstance(base_val, tuple):
                current[key] = tuple(
                    b + (m - b) * progress for b, m in zip(base_val, max_val)
                )
            else:
                current[key] = base_val + (max_val - base_val) * progress
        return current
    
    def advance_stage(self, success_rate, threshold=0.7):
        """根据成功率推进阶段"""
        if success_rate > threshold and self.current_stage < self.num_stages - 1:
            self.current_stage += 1
            print(f"DR 阶段升级: {self.current_stage}/{self.num_stages}")
```

```python
"""
域适应 (Domain Adaptation):
  减少仿真域和真实域之间的分布差异

方法3：风格迁移
  - 用 CycleGAN 进行仿真→真实风格迁移
  - 保留内容（物体位置等）改变风格
"""
```

## 四、系统辨识（System ID）

```python
"""
系统辨识 (System Identification):
  测量真实机器人的物理参数，使仿真更接近真实

流程：
  1. 在真机上执行已知运动
  2. 记录传感器数据
  3. 优化仿真参数使仿真轨迹匹配真实轨迹
  
可辨识的参数：
  - 关节摩擦
  - 连杆质量和惯量
  - 电机力矩常数
  - 控制延迟
"""

class SystemIdentification:
    """系统辨识"""
    def __init__(self, sim_model):
        self.sim = sim_model
    
    def identify(self, real_trajectories, param_names, initial_params):
        """
        real_trajectories: 真机记录的轨迹
        param_names: 要辨识的参数名
        initial_params: 参数初始猜测
        """
        from scipy.optimize import minimize
        
        def objective(params):
            # 设置仿真参数
            for name, val in zip(param_names, params):
                self.sim.set_param(name, val)
            
            # 运行仿真，计算与真实轨迹的差异
            total_error = 0
            for real_traj in real_trajectories:
                sim_traj = self.sim.run(real_traj['initial_state'], 
                                        real_traj['actions'])
                total_error += np.sum((sim_traj - real_traj['states'])**2)
            
            return total_error
        
        result = minimize(objective, initial_params, method='Nelder-Mead')
        identified_params = dict(zip(param_names, result.x))
        
        return identified_params
```

## 五、Teacher-Student 蒸馏

### 5.1 框架详解

```python
"""
Teacher-Student 蒸馏在 Sim2Real 中的应用

Teacher（仿真训练时）:
  - 输入：本体感知 + 特权信息（地形高度图、精确摩擦等）
  - 特权信息使训练更容易
  - 在仿真中用 RL 训练

Student（真机部署时）:
  - 输入：仅本体感知 + 历史信息
  - 不需要特权传感器
  - 通过蒸馏学习 Teacher 的行为
  - 用 RNN/Transformer 从历史推断隐含参数

关键：Student 从历史信息中"推断"特权信息
  例如：通过最近几步的滑移情况推断摩擦系数
"""
```

### 5.2 实现代码

```python
import torch
import torch.nn as nn

class TeacherPolicy(nn.Module):
    """Teacher: 有特权信息"""
    def __init__(self, obs_dim, priv_dim, action_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim + priv_dim, 512),
            nn.ELU(),
            nn.Linear(512, 256),
            nn.ELU(),
            nn.Linear(256, action_dim)
        )
    
    def forward(self, obs, priv_info):
        x = torch.cat([obs, priv_info], dim=-1)
        return self.net(x)

class StudentPolicy(nn.Module):
    """Student: 仅可观测信息 + 历史编码"""
    def __init__(self, obs_dim, action_dim, hidden_dim=256, history_len=15):
        super().__init__()
        self.history_len = history_len
        
        # 历史编码器（从历史推断特权信息）
        self.history_encoder = nn.LSTM(obs_dim, hidden_dim, batch_first=True)
        
        # 策略网络
        self.policy = nn.Sequential(
            nn.Linear(obs_dim + hidden_dim, 512),
            nn.ELU(),
            nn.Linear(512, 256),
            nn.ELU(),
            nn.Linear(256, action_dim)
        )
    
    def forward(self, obs_history):
        """
        obs_history: (B, T, obs_dim) 历史观测序列
        """
        # 编码历史
        _, (hidden, _) = self.history_encoder(obs_history)
        history_feat = hidden.squeeze(0)  # (B, hidden_dim)
        
        # 当前观测
        current_obs = obs_history[:, -1, :]  # (B, obs_dim)
        
        # 拼接并决策
        combined = torch.cat([current_obs, history_feat], dim=-1)
        return self.policy(combined)

def distillation_training(teacher, student, dataset, epochs=200):
    """蒸馏训练"""
    optimizer = torch.optim.Adam(student.parameters(), lr=1e-3)
    criterion = nn.MSELoss()
    
    for epoch in range(epochs):
        for batch in dataset:
            obs_history, current_obs, priv_info = batch
            
            # Teacher 预测
            with torch.no_grad():
                teacher_action = teacher(current_obs, priv_info)
            
            # Student 预测
            student_action = student(obs_history)
            
            # 蒸馏损失
            loss = criterion(student_action, teacher_action)
            
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
        
        if (epoch + 1) % 20 == 0:
            print(f"Distillation Epoch {epoch+1}, Loss: {loss.item():.6f}")
```

```python
Sim2Real 完整流程

   - 安全监测

   - 系统辨识更新
   - 重新训练
"""
```

### 6.1 模型导出

```python
import torch

def export_for_deployment(model, example_input, output_path="policy.onnx"):
    """导出模型用于部署"""
    model.eval()
    
    torch.onnx.export(
        model,
        example_input,
        output_path,
        input_names=["observation"],
        output_names=["action"],
        dynamic_axes={
            "observation": {0: "batch_size"},
            "action": {0: "batch_size"}
        },
        opset_version=14
    )
    print(f"模型导出到 {output_path}")

# ONNX 推理
def deploy_with_onnx(onnx_path):
    import onnxruntime as ort
    import numpy as np
    
    session = ort.InferenceSession(onnx_path)
    
    def inference(obs):
        input_name = session.get_inputs()[0].name
        action = session.run(None, {input_name: obs.astype(np.float32)})[0]
        return action
    
    return inference
```
