## 一、坐标系与变换

### 1.1 坐标系基础

机器人学中常用的坐标系：
- **世界坐标系（World Frame）**：固定的全局参考坐标系
- **基座坐标系（Base Frame）**：固定在机器人底座的坐标系
- **工具坐标系（Tool Frame）**：固定在末端执行器的坐标系
- **关节坐标系（Joint Frame）**：每个关节的局部坐标系

### 1.2 旋转表示

#### 旋转矩阵

```python
import numpy as np

def rotation_matrix_x(angle):
    """绕 X 轴旋转"""
    c, s = np.cos(angle), np.sin(angle)
    return np.array([
        [1, 0, 0],
        [0, c, -s],
        [0, s, c]
    ])

def rotation_matrix_y(angle):
    """绕 Y 轴旋转"""
    c, s = np.cos(angle), np.sin(angle)
    return np.array([
        [c, 0, s],
        [0, 1, 0],
        [-s, 0, c]
    ])

def rotation_matrix_z(angle):
    """绕 Z 轴旋转"""
    c, s = np.cos(angle), np.sin(angle)
    return np.array([
        [c, -s, 0],
        [s, c, 0],
        [0, 0, 1]
    ])

# 组合旋转：先绕Z旋转90°，再绕Y旋转30°
R = rotation_matrix_z(np.pi/2) @ rotation_matrix_y(np.pi/6)
print("组合旋转矩阵:")
print(np.round(R, 4))
```

#### 四元数

```python
def quaternion_from_euler(roll, pitch, yaw):
    """欧拉角转四元数"""
    cr, sr = np.cos(roll/2), np.sin(roll/2)
    cp, sp = np.cos(pitch/2), np.sin(pitch/2)
    cy, sy = np.cos(yaw/2), np.sin(yaw/2)
    
    w = cr * cp * cy + sr * sp * sy
    x = sr * cp * cy - cr * sp * sy
    y = cr * sp * cy + sr * cp * sy
    z = cr * cp * sy - sr * sp * cy
    
    return np.array([w, x, y, z])

def quaternion_to_matrix(q):
    """四元数转旋转矩阵"""
    w, x, y, z = q
    return np.array([
        [1 - 2*(y*y + z*z), 2*(x*y - w*z),     2*(x*z + w*y)],
        [2*(x*y + w*z),     1 - 2*(x*x + z*z), 2*(y*z - w*x)],
        [2*(x*z - w*y),     2*(y*z + w*x),     1 - 2*(x*x + y*y)]
    ])

# 测试

print(f"四元数: {np.round(q, 4)}")
R = quaternion_to_matrix(q)
print(f"旋转矩阵:\n{np.round(R, 4)}")
```

def euler_from_matrix(R):
    """旋转矩阵转欧拉角 (RPY)"""
    pitch = np.arctan2(-R[2, 0], np.sqrt(R[0, 0]**2 + R[1, 0]**2))
    
    if abs(R[2, 0]) < 0.9999:
        roll = np.arctan2(R[2, 1], R[2, 2])
        yaw = np.arctan2(R[1, 0], R[0, 0])
    else:
        # 万向锁情况
        roll = 0
        yaw = np.arctan2(-R[0, 1], R[1, 1])
    
    return np.array([roll, pitch, yaw])

### 1.3 齐次变换矩阵

def homogeneous_transform(R, t):
    """构造齐次变换矩阵"""
    T = np.eye(4)
    T[:3, :3] = R
    T[:3, 3] = t
    return T

# 示例：从基座到工具坐标系的变换

## 二、正运动学

### 2.1 DH 参数法

DH（Denavit-Hartenberg）参数法是描述串联机械臂运动学的标准方法：

```python
def dh_transform(theta, d, a, alpha):
    """
    DH 参数变换矩阵
    theta: 关节角度
    d: 连杆偏移
    a: 连杆长度
    alpha: 连杆扭转角
    """
    ct, st = np.cos(theta), np.sin(theta)
    ca, sa = np.cos(alpha), np.sin(alpha)
    
    return np.array([
        [ct, -st*ca,  st*sa, a*ct],
        [st,  ct*ca, -ct*sa, a*st],
        [0,   sa,     ca,    d],
        [0,   0,      0,     1]
    ])

def forward_kinematics_dh(dh_params, joint_angles):
    """
    使用 DH 参数计算正运动学
    dh_params: [(d1, a1, alpha1), (d2, a2, alpha2), ...]
    joint_angles: [theta1, theta2, ...]
    """
    T = np.eye(4)
    for i, (theta, (d, a, alpha)) in enumerate(zip(joint_angles, dh_params)):
        T_i = dh_transform(theta, d, a, alpha)
        T = T @ T_i
    return T

# 示例：2-DOF 平面机械臂
# DH 参数: (d=0, a=l, alpha=0) for each joint
l1, l2 = 1.0, 0.8
dh_params = [(0, l1, 0), (0, l2, 0)]
joint_angles = [np.pi/4, np.pi/6]  # 45° and 30°

T = forward_kinematics_dh(dh_params, joint_angles)
end_effector_pos = T[:2, 3]  # 平面机械臂只取 x, y
print(f"关节角度: {np.degrees(joint_angles)}")
print(f"末端位置: {np.round(end_effector_pos, 4)}")
```

### 2.2 2-DOF 机械臂正运动学（完整版）

```python
import numpy as np
import matplotlib.pyplot as plt

class TwoLinkArm:
    """2-DOF 平面机械臂"""
    def __init__(self, l1=1.0, l2=0.8):
        self.l1 = l1
        self.l2 = l2
    
    def forward_kinematics(self, theta1, theta2):
        """正运动学：关节角度 → 末端位置"""
        x = self.l1 * np.cos(theta1) + self.l2 * np.cos(theta1 + theta2)
        y = self.l1 * np.sin(theta1) + self.l2 * np.sin(theta1 + theta2)
        return np.array([x, y])
    
    def inverse_kinematics(self, x, y, elbow="up"):
        """
        逆运动学：末端位置 → 关节角度
        elbow: "up" 或 "down" 选择肘部朝向
        """
        d_sq = x**2 + y**2
        cos_theta2 = (d_sq - self.l1**2 - self.l2**2) / (2 * self.l1 * self.l2)
        
        if abs(cos_theta2) > 1:
            raise ValueError(f"目标 ({x}, {y}) 超出工作空间")
        
        if elbow == "up":
            theta2 = np.arccos(cos_theta2)
        else:
            theta2 = -np.arccos(cos_theta2)
        
        theta1 = np.arctan2(y, x) - np.arctan2(
            self.l2 * np.sin(theta2),
            self.l1 + self.l2 * np.cos(theta2)
        )
        
        return np.array([theta1, theta2])
    
    def get_joint_positions(self, theta1, theta2):
        """获取各关节位置（用于可视化）"""
        x1 = self.l1 * np.cos(theta1)
        y1 = self.l1 * np.sin(theta1)
        x2 = x1 + self.l2 * np.cos(theta1 + theta2)
        y2 = y1 + self.l2 * np.sin(theta1 + theta2)
        return np.array([[0, x1, x2], [0, y1, y2]])
    
    def visualize(self, theta1, theta2, target=None):
        """可视化机械臂"""
        positions = self.get_joint_positions(theta1, theta2)
        
        plt.figure(figsize=(8, 8))
        plt.plot(positions[0], positions[1], 'b-o', linewidth=3, markersize=10)
        plt.plot(0, 0, 'ks', markersize=12, label='Base')
        
        if target is not None:
            plt.plot(target[0], target[1], 'r*', markersize=15, label='Target')
        
        # 工作空间
        theta_range = np.linspace(0, 2*np.pi, 100)
        r_outer = self.l1 + self.l2
        r_inner = abs(self.l1 - self.l2)
        plt.plot(r_outer*np.cos(theta_range), r_outer*np.sin(theta_range), 
                 'g--', alpha=0.3, label='Outer workspace')
        plt.plot(r_inner*np.cos(theta_range), r_inner*np.sin(theta_range), 
                 'g--', alpha=0.3, label='Inner workspace')
        
        plt.xlim(-2, 2)
        plt.ylim(-2, 2)
        plt.gca().set_aspect('equal')
        plt.grid(True, alpha=0.3)
        plt.legend()
        plt.title(f'2-DOF Arm: θ1={np.degrees(theta1):.1f}°, θ2={np.degrees(theta2):.1f}°')
        plt.savefig('arm_visualization.png', dpi=100, bbox_inches='tight')
        plt.show()

# 测试
arm = TwoLinkArm(l1=1.0, l2=0.8)

# 正运动学
theta = np.array([np.pi/4, np.pi/6])
pos = arm.forward_kinematics(*theta)
print(f"正运动学: θ={np.degrees(theta)}° → pos={np.round(pos, 4)}")

# 逆运动学
target = np.array([1.2, 0.8])
angles_up = arm.inverse_kinematics(*target, elbow="up")
angles_down = arm.inverse_kinematics(*target, elbow="down")
print(f"逆运动学 (elbow up): pos={target} → θ={np.degrees(angles_up)}°")
print(f"逆运动学 (elbow down): pos={target} → θ={np.degrees(angles_down)}°")

# 验证
pos_up = arm.forward_kinematics(*angles_up)
print(f"验证: {np.round(pos_up, 4)} (误差: {np.linalg.norm(pos_up - target):.2e})")
```

## 三、逆运动学

### 3.1 数值解法 - 雅可比矩阵法

```python
class NumericalIK:
    """数值逆运动学（雅可比矩阵法）"""
    def __init__(self, arm):
        self.arm = arm
    
    def jacobian(self, theta1, theta2):
        """计算雅可比矩阵"""
        l1, l2 = self.arm.l1, self.arm.l2
        j11 = -l1 * np.sin(theta1) - l2 * np.sin(theta1 + theta2)
        j12 = -l2 * np.sin(theta1 + theta2)
        j21 = l1 * np.cos(theta1) + l2 * np.cos(theta1 + theta2)
        j22 = l2 * np.cos(theta1 + theta2)
        return np.array([[j11, j12], [j21, j22]])
    
    def solve(self, target, theta_init=None, max_iter=100, tol=1e-6):
        """迭代求解 IK"""
        if theta_init is None:
            theta = np.array([0.1, 0.1])
        else:
            theta = theta_init.copy()
        
        for i in range(max_iter):
            current_pos = self.arm.forward_kinematics(*theta)
            error = target - current_pos
            
            if np.linalg.norm(error) < tol:
                print(f"收敛于第 {i} 次迭代")
                return theta
            
            J = self.jacobian(*theta)
            # 使用伪逆解决非方阵情况
            dtheta = np.linalg.pinv(J) @ error * 0.5  # 阻尼系数 0.5
            theta += dtheta
        
        print(f"未收敛，最终误差: {np.linalg.norm(error):.6f}")
        return theta

# 测试数值 IK
arm = TwoLinkArm(l1=1.0, l2=0.8)
ik = NumericalIK(arm)
target = np.array([1.2, 0.8])
theta_solution = ik.solve(target)
print(f"数值 IK 解: θ={np.degrees(theta_solution)}°")
print(f"验证位置: {np.round(arm.forward_kinematics(*theta_solution), 6)}")
```

### 3.2 IK 优化方法

```python
from scipy.optimize import minimize

def ik_optimization(arm, target, theta_init=None):
    """使用 scipy 优化求解 IK"""
    if theta_init is None:
        theta_init = np.array([0.1, 0.1])
    
    def objective(theta):
        pos = arm.forward_kinematics(*theta)
        return np.sum((pos - target)**2)
    
    result = minimize(objective, theta_init, method='L-BFGS-B')
    return result.x

# 测试
theta_opt = ik_optimization(arm, target)
print(f"优化 IK 解: θ={np.degrees(theta_opt)}°")
print(f"验证位置: {np.round(arm.forward_kinematics(*theta_opt), 6)}")
```

## 四、动力学建模

### 4.1 拉格朗日方法

```python
"""
2-DOF 机械臂拉格朗日动力学
"""
class TwoLinkDynamics:
    def __init__(self, l1=1.0, l2=0.8, m1=1.0, m2=1.0):
        self.l1, self.l2 = l1, l2
        self.m1, self.m2 = m1, m2
        # 假设质量集中在连杆末端
        self.lc1, self.lc2 = l1, l2  # 质心距离
        self.I1 = m1 * l1**2 / 3  # 转动惯量
        self.I2 = m2 * l2**2 / 3
        self.g = 9.81
    
    def mass_matrix(self, theta1, theta2, dtheta1, dtheta2):
        """质量矩阵 M(q)"""
        m11 = self.I1 + self.I2 + self.m2 * self.l1**2 + \
              2 * self.m2 * self.l1 * self.l2 * np.cos(theta2)
        m12 = self.I2 + self.m2 * self.l1 * self.l2 * np.cos(theta2)
        m21 = m12
        m22 = self.I2
        return np.array([[m11, m12], [m21, m22]])
    
    def coriolis_matrix(self, theta1, theta2, dtheta1, dtheta2):
        """科里奥利矩阵 C(q, dq)"""
        c = self.m2 * self.l1 * self.l2 * np.sin(theta2)
        C = np.array([
            [-c * dtheta2, -c * (dtheta1 + dtheta2)],
            [c * dtheta1, 0]
        ])
        return C
    
    def gravity_vector(self, theta1, theta2):
        """重力向量 G(q)"""
        g1 = (self.m1 * self.lc1 + self.m2 * self.l1) * self.g * np.cos(theta1) + \
             self.m2 * self.g * self.lc2 * np.cos(theta1 + theta2)
        g2 = self.m2 * self.g * self.lc2 * np.cos(theta1 + theta2)
        return np.array([g1, g2])
    
    def compute_torque(self, theta, dtheta, ddtheta):
        """计算所需力矩: τ = M(q)q̈ + C(q,q̇)q̇ + G(q)"""
        M = self.mass_matrix(*theta, *dtheta)
        C = self.coriolis_matrix(*theta, *dtheta)
        G = self.gravity_vector(*theta)
        tau = M @ ddtheta + C @ dtheta + G
        return tau

# 测试
dyn = TwoLinkDynamics()
theta = np.array([0.3, 0.5])
dtheta = np.array([0.1, 0.0])
ddtheta = np.array([0.0, 0.0])

tau = dyn.compute_torque(theta, dtheta, ddtheta)
print(f"所需力矩: τ = {np.round(tau, 4)} Nm")
```

## 五、控制基础

### 5.1 PID 控制

```python
class PIDController:
    """PID 控制器"""
    def __init__(self, kp=1.0, ki=0.0, kd=0.0, output_limit=None):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.output_limit = output_limit
        self.integral = 0
        self.prev_error = 0
    
    def compute(self, setpoint, measured, dt):
        """计算控制输出"""
        error = setpoint - measured
        
        # 比例项
        p = self.kp * error
        
        # 积分项
        self.integral += error * dt
        i = self.ki * self.integral
        
        # 微分项
        derivative = (error - self.prev_error) / dt if dt > 0 else 0
        d = self.kd * derivative
        
        self.prev_error = error
        
        output = p + i + d
        
        # 输出限幅
        if self.output_limit is not None:
            output = np.clip(output, -self.output_limit, self.output_limit)
        
        return output
    
    def reset(self):
        self.integral = 0
        self.prev_error = 0

# 测试：关节角度 PID 控制
pid = PIDController(kp=10.0, ki=0.5, kd=0.1, output_limit=5.0)
target_angle = 1.0  # 目标角度
current_angle = 0.0
dt = 0.01

print("PID 控制仿真:")
for step in range(200):
    torque = pid.compute(target_angle, current_angle, dt)
    # 简化的动力学模型：θ̈ = τ / m
    current_angle += torque * dt * 0.1
    if step % 50 == 0:
        print(f"  Step {step}: angle={current_angle:.4f}, torque={torque:.4f}")
```

### 5.2 阻抗控制

```python
class ImpedanceController:
    """
    阻抗控制：控制机器人与环境交互时的力/位置关系
    M * ẍ + D * ẋ + K * (x - x_d) = F_ext
    """
    def __init__(self, K=100.0, D=20.0, M=1.0):
        self.K = K  # 刚度
        self.D = D  # 阻尼
        self.M = M  # 惯性
    
    def compute(self, x, dx, x_desired, dx_desired, F_ext=0):
        """计算控制力"""
        # 阻抗方程
        F = self.K * (x_desired - x) + self.D * (dx_desired - dx) + F_ext
        return F

# 阻抗控制示例：机械臂接触墙面
impedance = ImpedanceController(K=200, D=30, M=1.0)
x, dx = 0.0, 0.0  # 当前位置和速度
x_desired = 0.5   # 目标位置

print("阻抗控制仿真:")
for step in range(100):
    # 模拟环境：当 x > 0.4 时有墙
    F_ext = -1000 * max(0, x - 0.4) if x > 0.4 else 0
    
    F = impedance.compute(x, dx, x_desired, 0, F_ext)
    
    # 简化动力学
    ddx = F / impedance.M
    dx += ddx * 0.01
    x += dx * 0.01
    
    if step % 20 == 0:
        print(f"  Step {step}: x={x:.4f}, F_ext={F_ext:.2f}, F={F:.2f}")
```

### 5.3 导纳控制

```python
class AdmittanceController:
    """
    导纳控制：根据外力调整位置输出
    与阻抗控制互补：阻抗控制输出力，导纳控制输出位置
    """
    def __init__(self, M=1.0, D=20.0, K=100.0):
        self.M = M
        self.D = D
        self.K = K
        self.x = 0.0  # 导纳位置
        self.dx = 0.0
    
    def compute(self, x_ref, F_ext, dt=0.01):
        """根据参考位置和外力计算输出位置"""
        # 导纳动力学
        ddx = (F_ext - self.D * self.dx - self.K * (self.x - 0)) / self.M
        self.dx += ddx * dt
        self.x += self.dx * dt
        
        # 最终输出位置 = 参考位置 + 导纳偏移
        return x_ref + self.x
```
