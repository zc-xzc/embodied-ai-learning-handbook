## 一、Self-Attention 机制详解

### 1.1 注意力机制直觉

注意力机制（Attention Mechanism）的核心思想：在处理序列时，不是对所有位置一视同仁，而是"关注"与当前位置最相关的部分。

```
类比：
  你在阅读一段文字时，并不是逐字匀速阅读，
  而是在关键词上停留更久，忽略无关词汇。
  
  Self-Attention 就是让序列中的每个位置
  自动"决定"应该关注其他哪些位置。
```

### 1.2 QKV 计算过程

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class SelfAttention(nn.Module):
    """完整的 Self-Attention 实现"""
    def __init__(self, embed_dim, num_heads=8, dropout=0.1):
        super().__init__()
        self.embed_dim = embed_dim
        self.num_heads = num_heads
        self.head_dim = embed_dim // num_heads
        
        assert self.head_dim * num_heads == embed_dim, \
            "embed_dim 必须能被 num_heads 整除"
        
        # Q, K, V 投影矩阵
        self.q_proj = nn.Linear(embed_dim, embed_dim)
        self.k_proj = nn.Linear(embed_dim, embed_dim)
        self.v_proj = nn.Linear(embed_dim, embed_dim)
        
        # 输出投影
        self.out_proj = nn.Linear(embed_dim, embed_dim)
        
        self.dropout = nn.Dropout(dropout)
        self.scale = math.sqrt(self.head_dim)
    
    def forward(self, x, mask=None):
        """
        x: (batch_size, seq_len, embed_dim)
        mask: (batch_size, seq_len, seq_len) 或 None
        """
        B, N, C = x.shape
        
        # Step 1: 线性投影 Q, K, V
        Q = self.q_proj(x)  # (B, N, C)
        K = self.k_proj(x)  # (B, N, C)
        V = self.v_proj(x)  # (B, N, C)
        
        # Step 2: 分头 (reshape for multi-head)
        Q = Q.view(B, N, self.num_heads, self.head_dim).transpose(1, 2)  # (B, H, N, D)
        K = K.view(B, N, self.num_heads, self.head_dim).transpose(1, 2)  # (B, H, N, D)
        V = V.view(B, N, self.num_heads, self.head_dim).transpose(1, 2)  # (B, H, N, D)
        
        # Step 3: 计算注意力分数
        # scores = Q @ K^T / sqrt(d)
        scores = torch.matmul(Q, K.transpose(-2, -1)) / self.scale  # (B, H, N, N)
        
        # Step 4: 应用 mask（可选）
        if mask is not None:
            scores = scores.masked_fill(mask == 0, float('-inf'))
        
        # Step 5: Softmax 归一化
        attn_weights = F.softmax(scores, dim=-1)  # (B, H, N, N)
        attn_weights = self.dropout(attn_weights)
        
        # Step 6: 加权求和
        out = torch.matmul(attn_weights, V)  # (B, H, N, D)
        
        # Step 7: 合并多头
        out = out.transpose(1, 2).contiguous().view(B, N, C)  # (B, N, C)
        
        # Step 8: 输出投影
        out = self.out_proj(out)
        
        return out, attn_weights

# 测试
attn = SelfAttention(embed_dim=256, num_heads=8)
x = torch.randn(2, 10, 256)  # batch=2, seq_len=10, dim=256
out, weights = attn(x)
print(f"输入: {x.shape}")
print(f"输出: {out.shape}")
print(f"注意力权重: {weights.shape}")
print(f"注意力权重和（每行=1）: {weights[0, 0, 0, :].sum().item():.4f}")
```

### 1.3 计算过程图解

```
Self-Attention 计算流程

输入: X (B, N, C)
       │
       ├──→ W_Q ──→ Q (B, N, C)
       ├──→ W_K ──→ K (B, N, C)
       └──→ W_V ──→ V (B, N, C)
                         │
                    分头 reshape
                         │
                    Q: (B, H, N, D)
                    K: (B, H, N, D)
                    V: (B, H, N, D)
                         │
          scores = Q @ K^T / sqrt(D)   ← (B, H, N, N)
                         │
          attn = softmax(scores)        ← (B, H, N, N)
                         │
          out = attn @ V                ← (B, H, N, D)
                         │
                    合并头
                         │
          out: (B, N, C)
                         │
                    W_O 投影
                         │
          最终输出: (B, N, C)
```

## 二、Multi-Head Attention

### 2.1 为什么要多头

```python
"""
多头注意力的直觉：

单头注意力只能学习一种"关注模式"，
多头注意力让模型同时从多个子空间学习不同的关注模式。

例如：
  Head 1: 关注语法关系（主谓宾）
  Head 2: 关注语义相似性
  Head 3: 关注位置邻接
  Head 4: 关注长距离依赖
  ...
"""

# Multi-Head Attention 已在 SelfAttention 类中实现
# 核心是 reshape: (B, N, C) → (B, H, N, D) where D = C / H

# 多头的好处：
multi_head_benefits = {
    "多子空间": "不同头学习不同的表示子空间",
    "并行计算": "多个头可以并行计算，效率高",
    "丰富表达": "比单头有更强的表达能力",
    "正则化": "每个头看到更小的维度，有正则化效果",
}
```

### 2.2 交叉注意力（Cross-Attention）

```python
class CrossAttention(nn.Module):
    """交叉注意力：Q 来自一个序列，K/V 来自另一个序列"""
    def __init__(self, embed_dim, num_heads=8):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = embed_dim // num_heads
        
        self.q_proj = nn.Linear(embed_dim, embed_dim)
        self.k_proj = nn.Linear(embed_dim, embed_dim)
        self.v_proj = nn.Linear(embed_dim, embed_dim)
        self.out_proj = nn.Linear(embed_dim, embed_dim)
    
    def forward(self, query_seq, kv_seq):
        """
        query_seq: (B, N_q, C) - 查询序列（如解码器）
        kv_seq: (B, N_kv, C) - 键值序列（如编码器输出）
        """
        B, Nq, C = query_seq.shape
        Nkv = kv_seq.shape[1]
        
        Q = self.q_proj(query_seq).view(B, Nq, self.num_heads, self.head_dim).transpose(1, 2)
        K = self.k_proj(kv_seq).view(B, Nkv, self.num_heads, self.head_dim).transpose(1, 2)
        V = self.v_proj(kv_seq).view(B, Nkv, self.num_heads, self.head_dim).transpose(1, 2)
        
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.head_dim)
        attn = F.softmax(scores, dim=-1)
        out = torch.matmul(attn, V)
        
        out = out.transpose(1, 2).contiguous().view(B, Nq, C)
        return self.out_proj(out)

# 在 VLA 模型中的应用：
# query = 语言指令的文本特征
# kv = 视觉图像的 patch 特征
# 交叉注意力让语言"查询"视觉信息
```

## 三、位置编码

### 3.1 正弦位置编码

```python
class SinusoidalPositionalEncoding(nn.Module):
    """正弦余弦位置编码（原始 Transformer）"""
    def __init__(self, embed_dim, max_len=5000):
        super().__init__()
        
        pe = torch.zeros(max_len, embed_dim)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        
        # 频率：1 / 10000^(2i/d)
        div_term = torch.exp(
            torch.arange(0, embed_dim, 2).float() * 
            (-math.log(10000.0) / embed_dim)
        )
        
        pe[:, 0::2] = torch.sin(position * div_term)  # 偶数位用sin
        pe[:, 1::2] = torch.cos(position * div_term)  # 奇数位用cos
        
        pe = pe.unsqueeze(0)  # (1, max_len, embed_dim)
        self.register_buffer('pe', pe)
    
    def forward(self, x):
        """x: (B, N, C)"""
        return x + self.pe[:, :x.size(1), :]

# 测试
```

### 3.2 RoPE（旋转位置编码）

```python
class RotaryPositionalEmbedding(nn.Module):
    """RoPE: 旋转位置编码（LLaMA 使用）"""
    def __init__(self, head_dim, max_seq_len=2048):
        super().__init__()
        
        # 计算频率
        inv_freq = 1.0 / (10000 ** (torch.arange(0, head_dim, 2).float() / head_dim))
        self.register_buffer('inv_freq', inv_freq)
    
    def forward(self, seq_len, device):
        """生成旋转矩阵"""
        t = torch.arange(seq_len, device=device).float()
        freqs = torch.einsum('i,j->ij', t, self.inv_freq)
        # cos 和 sin
        cos = freqs.cos()
        sin = freqs.sin()
        return cos, sin

def apply_rotary_emb(x, cos, sin):
    """应用旋转位置编码"""
    # x: (B, H, N, D)
    # cos, sin: (N, D/2)
    d = x.shape[-1]
    x1, x2 = x[..., :d//2], x[..., d//2:]
    
    cos = cos.unsqueeze(0).unsqueeze(0)  # (1, 1, N, D/2)
    sin = sin.unsqueeze(0).unsqueeze(0)
    
    # 旋转
    rotated = torch.cat([x1 * cos - x2 * sin, x1 * sin + x2 * cos], dim=-1)
    return rotated

# RoPE 的优势：
# 1. 相对位置编码（自动学习相对距离）
# 2. 长序列外推性好
# 3. 计算高效
```

## 四、Encoder-Decoder 架构

### 4.1 完整 Transformer

```python
class TransformerEncoderLayer(nn.Module):
    """Transformer Encoder Layer"""
    def __init__(self, embed_dim=512, num_heads=8, ff_dim=2048, dropout=0.1):
        super().__init__()
        self.self_attn = SelfAttention(embed_dim, num_heads, dropout)
        self.norm1 = nn.LayerNorm(embed_dim)
        self.norm2 = nn.LayerNorm(embed_dim)
        
        self.ffn = nn.Sequential(
            nn.Linear(embed_dim, ff_dim),
            nn.GELU(),
            nn.Linear(ff_dim, embed_dim),
        )
        self.dropout = nn.Dropout(dropout)
    
    def forward(self, x):
        # Pre-LN 结构（更稳定）
        normed = self.norm1(x)
        attn_out, _ = self.self_attn(normed)
        x = x + self.dropout(attn_out)
        
        normed = self.norm2(x)
        ff_out = self.ffn(normed)
        x = x + self.dropout(ff_out)
        return x

class TransformerDecoderLayer(nn.Module):
    """Transformer Decoder Layer"""
    def __init__(self, embed_dim=512, num_heads=8, ff_dim=2048, dropout=0.1):
        super().__init__()
        self.self_attn = SelfAttention(embed_dim, num_heads, dropout)
        self.cross_attn = CrossAttention(embed_dim, num_heads)
        self.norm1 = nn.LayerNorm(embed_dim)
        self.norm2 = nn.LayerNorm(embed_dim)
        self.norm3 = nn.LayerNorm(embed_dim)
        
        self.ffn = nn.Sequential(
            nn.Linear(embed_dim, ff_dim),
            nn.GELU(),
            nn.Linear(ff_dim, embed_dim),
        )
        self.dropout = nn.Dropout(dropout)
    
    def forward(self, x, enc_output, tgt_mask=None):
        # Self-attention
        normed = self.norm1(x)
        attn_out, _ = self.self_attn(normed, mask=tgt_mask)
        x = x + self.dropout(attn_out)
        
        # Cross-attention
        normed = self.norm2(x)
        cross_out = self.cross_attn(normed, enc_output)
        x = x + self.dropout(cross_out)
        
        # FFN
        normed = self.norm3(x)
        ff_out = self.ffn(normed)
        x = x + self.dropout(ff_out)
        return x

class SimpleTransformer(nn.Module):
    """简化版完整 Transformer"""
    def __init__(self, vocab_size=10000, embed_dim=256, num_heads=8, 
                 num_layers=6, ff_dim=1024, max_len=512, dropout=0.1):
        super().__init__()
        self.token_embed = nn.Embedding(vocab_size, embed_dim)
        self.pos_enc = SinusoidalPositionalEncoding(embed_dim, max_len)
        
        self.encoder_layers = nn.ModuleList([
            TransformerEncoderLayer(embed_dim, num_heads, ff_dim, dropout)
            for _ in range(num_layers)
        ])
        self.decoder_layers = nn.ModuleList([
            TransformerDecoderLayer(embed_dim, num_heads, ff_dim, dropout)
            for _ in range(num_layers)
        ])
        
        self.norm = nn.LayerNorm(embed_dim)
        self.output_proj = nn.Linear(embed_dim, vocab_size)
    
    def encode(self, src):
        x = self.token_embed(src)
        x = self.pos_enc(x)
        for layer in self.encoder_layers:
            x = layer(x)
        return x
    
    def decode(self, tgt, enc_output):
        x = self.token_embed(tgt)
        x = self.pos_enc(x)
        for layer in self.decoder_layers:
            x = layer(x, enc_output)
        return x
    
    def forward(self, src, tgt):
        enc_output = self.encode(src)
        dec_output = self.decode(tgt, enc_output)
        dec_output = self.norm(dec_output)
        logits = self.output_proj(dec_output)
        return logits

# 测试
model = SimpleTransformer(vocab_size=1000, embed_dim=128, num_heads=4, num_layers=2)
src = torch.randint(0, 1000, (2, 10))  # batch=2, seq_len=10
tgt = torch.randint(0, 1000, (2, 8))

logits = model(src, tgt)
print(f"源序列: {src.shape}")
print(f"目标序列: {tgt.shape}")
print(f"输出 logits: {logits.shape}")

# 统计参数量
total_params = sum(p.numel() for p in model.parameters())
print(f"模型参数量: {total_params:,}")
```

## 五、Transformer 在机器人中的应用

### 5.1 VLA 模型中的 Transformer

```
VLA 模型中 Transformer 的角色：

1. 视觉编码器（Vision Encoder）
   - 通常是 ViT（Vision Transformer）
   - 将图像分割为 patches，每个 patch 作为一个 token
   - 通过 Self-Attention 提取视觉特征

2. 语言模型骨干（Language Model Backbone）
   - 通常是 Pre-trained LLM（如 LLaMA）
   - 处理语言指令 + 视觉 token
   - 通过 Self-Attention 融合多模态信息

3. 动作头（Action Head）
   - 将 LLM 输出映射为机器人动作
   - 可以是简单的线性层或更复杂的结构
```

### 5.2 ACT 中的 Transformer

```python
class ACTBackbone(nn.Module):
    """ACT (Action Chunking Transformer) 的简化骨干"""
    def __init__(self, obs_dim, action_dim, chunk_size=10, 
                 embed_dim=256, num_heads=8, num_layers=4):
        super().__init__()
        self.chunk_size = chunk_size
        
        # 输入编码
        self.obs_embed = nn.Linear(obs_dim, embed_dim)
        self.action_embed = nn.Linear(action_dim, embed_dim)
        
        # Encoder（处理观测）
        self.encoder = nn.TransformerEncoder(
            nn.TransformerEncoderLayer(
                d_model=embed_dim, nhead=num_heads,
                dim_feedforward=embed_dim*4, batch_first=True
            ),
            num_layers=num_layers
        )
        
        # Decoder（生成动作序列）
        self.decoder = nn.TransformerDecoder(
            nn.TransformerDecoderLayer(
                d_model=embed_dim, nhead=num_heads,
                dim_feedforward=embed_dim*4, batch_first=True
            ),
            num_layers=num_layers
        )
        
        # VAE 隐变量
        self.latent_dim = 32
        self.encoder_mu = nn.Linear(embed_dim, self.latent_dim)
        self.encoder_logvar = nn.Linear(embed_dim, self.latent_dim)
        self.decoder_latent = nn.Linear(self.latent_dim, embed_dim)
        
        # 动作输出
        self.action_head = nn.Linear(embed_dim, action_dim)
    
    def forward(self, obs, actions=None, z=None):
        """
        训练时: 输入 obs 和 actions（teacher forcing）
        推理时: 输入 obs, z 从先验采样
        """
        obs_embed = self.obs_embed(obs)  # (B, 1, C)
        
        # VAE
        if z is None:
            enc_out = self.encoder(obs_embed)
            mu = self.encoder_mu(enc_out[:, 0])
            logvar = self.encoder_logvar(enc_out[:, 0])
            z = mu + torch.exp(logvar / 2) * torch.randn_like(mu)
        else:
            mu = logvar = None
        
        z_embed = self.decoder_latent(z).unsqueeze(1)  # (B, 1, C)
        
        # Decoder 生成动作序列
        if actions is not None:
            # Teacher forcing
            action_embed = self.action_embed(actions)
            dec_input = torch.cat([z_embed, action_embed[:, :-1]], dim=1)
        else:
            # 自回归生成
            dec_input = z_embed.expand(-1, self.chunk_size, -1)
        
        dec_output = self.decoder(dec_input, enc_out if z is None else obs_embed)
        pred_actions = self.action_head(dec_output)
        
        return pred_actions, mu, logvar
```
