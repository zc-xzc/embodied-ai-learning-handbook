## 一、扩散模型原理

### 1.1 核心思想

扩散模型（Diffusion Model）的核心思想：
1. **前向过程（Forward Process）**：逐步向数据中添加噪声，直到变成纯噪声
2. **反向过程（Reverse Process）**：训练一个神经网络，从纯噪声逐步去噪，恢复出数据

```
前向过程（加噪）：
  x_0 → x_1 → x_2 → ... → x_T
  数据   逐步加噪      纯噪声

反向过程（去噪）：
  x_T → x_{T-1} → ... → x_1 → x_0
  纯噪声  逐步去噪      生成的数据
```

### 1.2 前向过程

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import numpy as np
import matplotlib.pyplot as plt

class DiffusionForward:
    """前向扩散过程"""
    def __init__(self, num_timesteps=1000, beta_start=1e-4, beta_end=0.02):
        self.num_timesteps = num_timesteps
        
        # 线性噪声调度
        self.betas = torch.linspace(beta_start, beta_end, num_timesteps)
        
        # 计算 alphas
        self.alphas = 1.0 - self.betas
        self.alphas_cumprod = torch.cumprod(self.alphas, dim=0)
        self.alphas_cumprod_prev = F.pad(self.alphas_cumprod[:-1], (1, 0), value=1.0)
        
        # 计算用于采样的系数
        self.sqrt_alphas_cumprod = torch.sqrt(self.alphas_cumprod)
        self.sqrt_one_minus_alphas_cumprod = torch.sqrt(1.0 - self.alphas_cumprod)
    
    def q_sample(self, x_0, t, noise=None):
        """
        前向采样：从 x_0 直接采样 x_t
        x_t = sqrt(alpha_bar_t) * x_0 + sqrt(1 - alpha_bar_t) * noise
        """
        if noise is None:
            noise = torch.randn_like(x_0)
        
        sqrt_alpha = self.sqrt_alphas_cumprod[t].view(-1, 1)
        sqrt_one_minus = self.sqrt_one_minus_alphas_cumprod[t].view(-1, 1)
        
        x_t = sqrt_alpha * x_0 + sqrt_one_minus * noise
        return x_t

# 可视化前向过程
diffusion = DiffusionForward(num_timesteps=1000)

# 创建一个简单的数据点
x_0 = torch.randn(1, 2) * 0.5 + torch.tensor([2.0, 1.0])

# 观察不同时间步的加噪效果
print("前向扩散过程:")
for t in [0, 100, 300, 500, 700, 999]:
    x_t = diffusion.q_sample(x_0, torch.tensor([t]))
    print(f"  t={t:4d}: x_t = {x_t[0].numpy()}")

# 可视化噪声调度
fig, axes = plt.subplots(1, 2, figsize=(12, 4))
axes[0].plot(diffusion.betas.numpy())
axes[0].set_title('Beta Schedule')
axes[0].set_xlabel('Timestep')
axes[1].plot(diffusion.alphas_cumprod.numpy())
axes[1].set_title('Cumulative Alpha (alpha_bar)')
axes[1].set_xlabel('Timestep')
plt.tight_layout()
plt.savefig('diffusion_schedule.png', dpi=100)
plt.show()
```

### 1.3 反向过程

```python
class DiffusionReverse:
    """反向去噪过程"""
    def __init__(self, forward: DiffusionForward):
        self.fwd = forward
        self.betas = forward.betas
        self.alphas = forward.alphas
        self.alphas_cumprod = forward.alphas_cumprod
        self.alphas_cumprod_prev = forward.alphas_cumprod_prev
    
    def p_sample(self, model, x_t, t, t_index):
        """
        单步去噪采样
        model: 训练好的去噪网络
        """
        betas_t = self.betas[t]
        sqrt_one_minus_alphas_cumprod_t = self.fwd.sqrt_one_minus_alphas_cumprod[t]
        sqrt_recip_alphas_t = torch.sqrt(1.0 / self.alphas[t])
        
        # 模型预测噪声
        predicted_noise = model(x_t, t)
        
        # 计算均值
        model_mean = sqrt_recip_alphas_t * (
            x_t - betas_t * predicted_noise / sqrt_one_minus_alphas_cumprod_t
        )
        
        if t_index == 0:
            return model_mean
        else:
            # 添加随机噪声（除了最后一步）
            posterior_variance = self.betas[t] * (1 - self.alphas_cumprod_prev[t]) / (1 - self.alphas_cumprod[t])
            noise = torch.randn_like(x_t)
            return model_mean + torch.sqrt(posterior_variance) * noise
    
    def sample(self, model, shape, device='cpu'):
        """完整采样流程：从噪声生成数据"""
        b = shape[0]
        
        # 从纯噪声开始
        x = torch.randn(shape, device=device)
        
        # 逐步去噪
        for i in reversed(range(self.fwd.num_timesteps)):
            t = torch.full((b,), i, device=device, dtype=torch.long)
            x = self.p_sample(model, x, t, i)
        
        return x
```

## 二、DDPM vs DDIM

### 2.1 DDPM（去噪扩散概率模型）

```python
class DDPM:
    """DDPM: 经典扩散模型"""
    def __init__(self, num_timesteps=1000):
        self.num_timesteps = num_timesteps
    
    def training_loss(self, model, x_0, t, noise=None):
        """
        DDPM 训练损失
        核心思想：预测添加的噪声
        """
        if noise is None:
            noise = torch.randn_like(x_0)
        
        # 前向加噪
        x_t = self.q_sample(x_0, t, noise)
        
        # 模型预测噪声
        predicted_noise = model(x_t, t)
        
        # 损失：预测噪声 vs 实际噪声
        loss = F.mse_loss(noise, predicted_noise)
        return loss
    
    def sample(self, model, shape, num_steps=None):
        """
        DDPM 采样：需要 num_timesteps 步去噪
        缺点：采样慢（通常需要 1000 步）
        """
        num_steps = num_steps or self.num_timesteps
        x = torch.randn(shape)
        
        for i in reversed(range(num_steps)):
            t = torch.full((shape[0],), i)
            x = self.p_sample(model, x, t, i)
        
        return x
```

### 2.2 DDIM（去噪扩散隐式模型）

```python
class DDIM:
    """DDIM: 加速采样的扩散模型"""
    def __init__(self, num_timesteps=1000):
        self.num_timesteps = num_timesteps
    
    def sample(self, model, shape, num_steps=50, eta=0.0):
        """
        DDIM 采样：可以用更少的步数（如 50 步）生成
        eta=0: 确定性采样（DDIM）
        eta=1: 等价于 DDPM
        """
        # 选择子序列
        step_ratio = self.num_timesteps // num_steps
        timesteps = list(range(0, self.num_timesteps, step_ratio))
        timesteps = list(reversed(timesteps))
        
        x = torch.randn(shape)
        
        for i, t in enumerate(timesteps):
            t_tensor = torch.full((shape[0],), t)
            
            # 预测噪声
            predicted_noise = model(x, t_tensor)
            
            # DDIM 更新公式
            alpha_t = self.alphas_cumprod[t]
            if i < len(timesteps) - 1:
                alpha_prev = self.alphas_cumprod[timesteps[i + 1]]
            else:
                alpha_prev = torch.tensor(1.0)
            
            # 预测 x_0
            x0_pred = (x - torch.sqrt(1 - alpha_t) * predicted_noise) / torch.sqrt(alpha_t)
            
            # DDIM 一步
            if eta == 0:
                # 确定性
                x = torch.sqrt(alpha_prev) * x0_pred + torch.sqrt(1 - alpha_prev) * predicted_noise
            else:
                # 带随机性
                sigma = eta * torch.sqrt((1 - alpha_prev) / (1 - alpha_t) * (1 - alpha_t / alpha_prev))
                noise = torch.randn_like(x)
                x = torch.sqrt(alpha_prev) * x0_pred + torch.sqrt(1 - alpha_prev - sigma**2) * predicted_noise + sigma * noise
        
        return x

# 对比
print("DDPM vs DDIM:")
print("  DDPM: 1000 步采样，有随机性")
print("  DDIM: 50 步采样，可确定性，质量接近 DDPM")
print("  DDIM 速度提升: ~20x")
```

### 2.3 噪声调度对比

```python
def cosine_beta_schedule(timesteps, s=0.008):
    """余弦噪声调度（改进版）"""
    steps = timesteps + 1
    x = torch.linspace(0, timesteps, steps)
    alphas_cumprod = torch.cos(((x / timesteps) + s) / (1 + s) * torch.pi * 0.5) ** 2
    alphas_cumprod = alphas_cumprod / alphas_cumprod[0]
    betas = 1 - (alphas_cumprod[1:] / alphas_cumprod[:-1])
    return torch.clip(betas, 0.0001, 0.9999)

def linear_beta_schedule(timesteps, beta_start=1e-4, beta_end=0.02):
    """线性噪声调度（原始 DDPM）"""
    return torch.linspace(beta_start, beta_end, timesteps)

# 余弦调度通常在机器人策略中表现更好
# 因为它在前向过程中更平缓地添加噪声
```

## 三、在机器人策略中的应用：Diffusion Policy

### 3.1 Diffusion Policy 核心思想

```
Diffusion Policy 的核心思想：

传统 BC 策略：
  观测 → 策略网络 → 动作（单步）

Diffusion Policy：
  观测 → 条件信息
  随机噪声 → [去噪网络（条件=观测）] → 动作序列（多步）

关键区别：
  1. 生成动作序列而非单步动作
  2. 扩散过程以观测为条件
  3. 能建模多模态动作分布
```

### 3.2 简化 Diffusion Policy 实现

```python
class DiffusionPolicy(nn.Module):
    """简化的 Diffusion Policy"""
    def __init__(self, obs_dim, action_dim, horizon=16, 
                 num_diffusion_steps=100, embed_dim=256):
        super().__init__()
        self.action_dim = action_dim
        self.horizon = horizon
        self.num_diffusion_steps = num_diffusion_steps
        
        # 观测编码器
        self.obs_encoder = nn.Sequential(
            nn.Linear(obs_dim, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim),
        )
        
        # 时间步编码
        self.time_embed = nn.Embedding(num_diffusion_steps, embed_dim)
        
        # 去噪网络（简化版 U-Net）
        self.denoise_net = nn.Sequential(
            nn.Linear(action_dim * horizon + embed_dim * 2, 512),
            nn.ReLU(),
            nn.Linear(512, 512),
            nn.ReLU(),
            nn.Linear(512, 512),
            nn.ReLU(),
            nn.Linear(512, action_dim * horizon),
        )
        
        # 噪声调度
        betas = torch.linspace(1e-4, 0.02, num_diffusion_steps)
        alphas = 1.0 - betas
        alphas_cumprod = torch.cumprod(alphas, dim=0)
        
        self.register_buffer('betas', betas)
        self.register_buffer('alphas_cumprod', alphas_cumprod)
        self.register_buffer('sqrt_alphas_cumprod', torch.sqrt(alphas_cumprod))
        self.register_buffer('sqrt_one_minus_alphas_cumprod', torch.sqrt(1 - alphas_cumprod))
    
    def predict_noise(self, x_t, t, obs_embed):
        """预测噪声"""
        # x_t: (B, horizon, action_dim)
        # t: (B,)
        # obs_embed: (B, embed_dim)
        
        B = x_t.shape[0]
        x_flat = x_t.reshape(B, -1)  # (B, horizon * action_dim)
        t_embed = self.time_embed(t)  # (B, embed_dim)
        
        # 拼接输入
        inp = torch.cat([x_flat, t_embed, obs_embed], dim=-1)
        
        # 预测噪声
        noise_pred = self.denoise_net(inp)
        return noise_pred.reshape(B, self.horizon, self.action_dim)
    
    def compute_loss(self, actions, obs):
        """训练损失"""
        B = actions.shape[0]
        
        # 编码观测
        obs_embed = self.obs_encoder(obs)
        
        # 随机采样时间步
        t = torch.randint(0, self.num_diffusion_steps, (B,), device=actions.device)
        
        # 随机噪声
        noise = torch.randn_like(actions)
        
        # 前向加噪
        sqrt_alpha = self.sqrt_alphas_cumprod[t].view(B, 1, 1)
        sqrt_one_minus = self.sqrt_one_minus_alphas_cumprod[t].view(B, 1, 1)
        x_t = sqrt_alpha * actions + sqrt_one_minus * noise
        
        # 预测噪声
        predicted_noise = self.predict_noise(x_t, t, obs_embed)
        
        # MSE 损失
        loss = F.mse_loss(predicted_noise, noise)
        return loss
    
    def sample(self, obs, num_inference_steps=50):
        """推理：从噪声生成动作序列"""
        B = obs.shape[0]
        device = obs.device
        
        # 编码观测
        obs_embed = self.obs_encoder(obs)
        
        # 选择时间步子序列
        step_ratio = self.num_diffusion_steps // num_inference_steps
        timesteps = list(range(0, self.num_diffusion_steps, step_ratio))
        timesteps = list(reversed(timesteps))
        
        # 从纯噪声开始
        x = torch.randn(B, self.horizon, self.action_dim, device=device)
        
        # 逐步去噪
        for t in timesteps:
            t_tensor = torch.full((B,), t, device=device)
            
            # 预测噪声
            noise_pred = self.predict_noise(x, t_tensor, obs_embed)
            
            # DDIM 更新
            alpha_t = self.alphas_cumprod[t]
            x0_pred = (x - torch.sqrt(1 - alpha_t) * noise_pred) / torch.sqrt(alpha_t)
            x = torch.sqrt(alpha_t) * x0_pred + torch.sqrt(1 - alpha_t) * noise_pred
        
        return x  # (B, horizon, action_dim)

# 测试
policy = DiffusionPolicy(obs_dim=10, action_dim=7, horizon=16, num_diffusion_steps=100)

# 训练步骤
obs = torch.randn(8, 10)
actions = torch.randn(8, 16, 7)
loss = policy.compute_loss(actions, obs)
print(f"训练损失: {loss.item():.4f}")

# 推理
sampled_actions = policy.sample(obs, num_inference_steps=20)
print(f"采样动作形状: {sampled_actions.shape}")  # (8, 16, 7)
```

## 四、条件生成

### 4.1 分类器引导（Classifier Guidance）

```python
class ClassifierGuidance:
    """
    分类器引导：训练一个分类器 p(y|x_t)，
    在采样时用分类器的梯度引导生成
    """
    def __init__(self, diffusion_model, classifier):
        self.model = diffusion_model
        self.classifier = classifier
    
    def guided_sample(self, x_t, t, y, guidance_scale=1.0):
        """带分类器引导的采样"""
        x_t.requires_grad_(True)
        
        # 分类器预测
        logits = self.classifier(x_t, t)
        log_prob = F.log_softmax(logits, dim=-1)
        
        # 计算梯度
        grad = torch.autograd.grad(log_prob[:, y].sum(), x_t)[0]
        
        # 引导扩散
        x_t = x_t - guidance_scale * grad
        
        return x_t.detach()
```

### 4.2 无分类器引导（Classifier-Free Guidance, CFG）

```python
class ClassifierFreeGuidance:
    """
    CFG: 无需额外分类器
    训练时：随机丢弃条件（用无条件训练）
    推理时：结合条件和无条件预测
    """
    def __init__(self, guidance_scale=3.0):
        self.scale = guidance_scale
    
    def train_step(self, model, x_t, t, condition, p_uncond=0.1):
        """
        训练时：以 p_uncond 概率丢弃条件
        """
        # 随机丢弃条件
        mask = torch.rand(len(condition)) > p_uncond
        condition_masked = condition * mask.view(-1, 1)
        
        # 预测
        noise_pred = model(x_t, t, condition_masked)
        return noise_pred
    
    def guided_predict(self, model, x_t, t, condition):
        """
        推理时：结合条件和无条件预测
        ε_guided = ε_uncond + s * (ε_cond - ε_uncond)
        """
        # 无条件预测
        noise_uncond = model(x_t, t, torch.zeros_like(condition))
        
        # 条件预测
        noise_cond = model(x_t, t, condition)
        
        # 引导
        noise_guided = noise_uncond + self.scale * (noise_cond - noise_uncond)
        return noise_guided

# CFG 在 Diffusion Policy 中的应用：
# 条件 = 观测信息
# 无条件 = 零向量（或学习到的空条件）
# guidance_scale 控制观测条件的影响强度
```
