### 1.1 CLIP（Contrastive Language-Image Pre-training）

```python
from transformers import CLIPProcessor, CLIPModel, CLIPTokenizer, CLIPImageProcessor
from PIL import Image
import torch
import numpy as np

class CLIPEncoder:
    """CLIP 视觉-语言编码器"""
    def __init__(self, model_name="openai/clip-vit-base-patch32", device="cuda"):
        self.device = device
        self.model = CLIPModel.from_pretrained(model_name).to(device)
        self.processor = CLIPProcessor.from_pretrained(model_name)
        self.model.eval()
    
    def encode_image(self, image):
        """编码图像为特征向量"""
        inputs = self.processor(images=image, return_tensors="pt").to(self.device)
        with torch.no_grad():
            features = self.model.get_image_features(**inputs)
        # L2 归一化
        features = features / features.norm(dim=-1, keepdim=True)
        return features
    
    def encode_text(self, text):
        """编码文本为特征向量"""
        inputs = self.processor(text=text, return_tensors="pt", padding=True).to(self.device)
        with torch.no_grad():
            features = self.model.get_text_features(**inputs)
        features = features / features.norm(dim=-1, keepdim=True)
        return features
    
    def compute_similarity(self, image, texts):
        """计算图像与多个文本的相似度"""
        img_feat = self.encode_image(image)
        text_feat = self.encode_text(texts)
        
        # 余弦相似度
        similarity = (img_feat @ text_feat.T).squeeze(0)
        return similarity

# 使用示例
encoder = CLIPEncoder()

# 创建示例图像（实际使用时替换为真实图像）
image = Image.new('RGB', (224, 224), color=(128, 0, 0))

# 图像-文本匹配
texts = [
    "a red square",
    "a robot arm picking up an object",
    "a kitchen scene",
    "a blue circle",
]
similarity = encoder.compute_similarity(image, texts)
print("图像-文本相似度:")
for text, sim in zip(texts, similarity):
    print(f"  '{text}': {sim:.4f}")

# 提取图像特征（用于 VLA 模型）
img_features = encoder.encode_image(image)
print(f"\n图像特征维度: {img_features.shape}")  # (1, 512)
```

### 1.3 DINOv2（自监督视觉编码器）

```python
from transformers import AutoImageProcessor, AutoModel

class DINOv2Encoder:
    """DINOv2: 自监督学习的视觉编码器"""
    def __init__(self, model_name="facebook/dinov2-base", device="cuda"):
        self.device = device
        self.processor = AutoImageProcessor.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name).to(device)
        self.model.eval()
    
    def encode_image(self, image):
        """提取图像特征"""
        inputs = self.processor(images=image, return_tensors="pt").to(self.device)
        with torch.no_grad():
            outputs = self.model(**inputs)
        
        # 获取 CLS token 特征
        cls_features = outputs.last_hidden_state[:, 0, :]  # (1, 768)
        return cls_features
    
    def get_patch_features(self, image):
        """获取所有 patch 的特征（用于密集预测）"""
        inputs = self.processor(images=image, return_tensors="pt").to(self.device)
        with torch.no_grad():
            outputs = self.model(**inputs)
        
        # 所有 patch 特征
        patch_features = outputs.last_hidden_state  # (1, num_patches+1, 768)
        return patch_features

# DINOv2 优势：
# 1. 自监督训练（无需标注）
# 2. 强大的密集特征表示
# 3. 在 3D 视觉中表现优秀（DP3 使用 DINOv2）
# 4. 不需要语言配对
```

### 1.4 视觉编码器对比

| 编码器 | 预训练方式 | 特征维度 | 优势 | 在 VLA 中的使用 |
|--------|-----------|---------|------|----------------|
| CLIP | 对比学习（图文对） | 512/768 | 图文对齐 | RT-1, RT-2 |
| SigLIP | 对比学习（改进） | 768 | 小batch高效 | OpenVLA |
| DINOv2 | 自监督 | 768/1024 | 密集特征强 | DP3 |
| ViT | 监督/自监督 | 768/1024 | 通用 | 各种模型 |

## 二、语言模型基础

### 2.1 LLM 原理简介

```
大语言模型（LLM）核心组件：

1. Tokenizer：将文本切分为 token
   "pick up the red block" → [pick, up, the, red, block] → token_ids

2. Embedding：将 token_id 映射为向量
   token_id → embedding_vector (维度 4096)

3. Transformer Layers：处理 token 序列
   多层 Self-Attention + FFN

4. LM Head：输出下一个 token 的概率分布
   最后一个隐藏状态 → logits → softmax → token 概率
```

```python
from transformers import AutoTokenizer, AutoModelForCausalLM

class LLMWrapper:
    """LLM 基础使用"""
    def __init__(self, model_name="meta-llama/Llama-3.2-1B", device="cuda"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForCausalLM.from_pretrained(
            model_name,
            torch_dtype=torch.bfloat16,
            device_map="auto"
        )
    
    def generate(self, prompt, max_new_tokens=100):
        """文本生成"""
        inputs = self.tokenizer(prompt, return_tensors="pt").to(self.model.device)
        
        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=max_new_tokens,
                do_sample=True,
                temperature=0.7,
                top_p=0.9,
            )
        
        response = self.tokenizer.decode(outputs[0], skip_special_tokens=True)
        return response
    
    def get_hidden_states(self, text):
        """获取隐藏状态（用于 VLA 模型）"""
        inputs = self.tokenizer(text, return_tensors="pt").to(self.model.device)
        
        with torch.no_grad():
            outputs = self.model(**inputs, output_hidden_states=True)
        
        # 最后一层隐藏状态
        hidden = outputs.hidden_states[-1]  # (1, seq_len, hidden_dim)
        return hidden
```

### 2.2 VLM（视觉语言模型）

```python
"""
VLM = 视觉编码器 + 语言模型

工作流程：
1. 图像 → 视觉编码器 → 视觉 token
2. 文本 → tokenizer → 文本 token
3. [视觉 token + 文本 token] → LLM → 输出

代表模型：
- LLaVA: CLIP ViT + LLaMA
- Prismatic VLM: SigLIP + LLaMA (OpenVLA 的骨干)

"""
```

## 三、视觉-语言对齐

### 3.1 对齐方法

```python
"""
视觉-语言对齐的核心方法

1. 对比学习（CLIP/SigLIP）
   - 图文对中正样本拉近，负样本推远
   - 对齐到共享的嵌入空间

2. 交叉注意力（Cross-Attention）
   - 文本作为 Query 查询视觉信息
   - 用于 VQA、Image Captioning

3. 投影层（Projection Layer）
   - 将视觉特征投影到语言模型的嵌入空间
   - LLaVA 使用简单的 MLP 投影

4. Q-Former（BLIP-2）
   - 可学习的 Query 提取视觉信息
   - 更灵活但对齐训练更复杂
"""
```

### 3.2 简单的视觉-语言对齐示例

```python
import torch.nn as nn

class VisionLanguageAligner(nn.Module):
    """视觉-语言对齐模块"""
    def __init__(self, vision_dim=768, lang_dim=4096, hidden_dim=2048):
        super().__init__()
        # 视觉特征投影到语言空间
        self.vision_proj = nn.Sequential(
            nn.Linear(vision_dim, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, lang_dim),
        )
        # LayerNorm
        self.norm = nn.LayerNorm(lang_dim)
    
    def forward(self, vision_features, text_embeddings):
        """
        vision_features: (B, num_patches, vision_dim)
        text_embeddings: (B, seq_len, lang_dim)
        """
        # 投影视觉特征
        vision_proj = self.vision_proj(vision_features)  # (B, num_patches, lang_dim)
        vision_proj = self.norm(vision_proj)
        
        # 拼接视觉和文本特征
        # [V_1, V_2, ..., V_n, T_1, T_2, ..., T_m]
        combined = torch.cat([vision_proj, text_embeddings], dim=1)
        
        return combined

# 使用
aligner = VisionLanguageAligner(vision_dim=768, lang_dim=4096)
vision_feat = torch.randn(1, 196, 768)   # 14x14 patches
text_emb = torch.randn(1, 10, 4096)       # 10 个文本 token
combined = aligner(vision_feat, text_emb)
print(f"对齐后序列长度: {combined.shape}")  # (1, 206, 4096)
```

## 四、多模态输入到动作输出的完整流程

### 4.1 VLA 模型数据流

```
完整 VLA 数据流：

输入：
  1. RGB 图像 (H, W, 3)
  2. 语言指令 "pick up the red block"
  3. 本体感知 (joint_positions, gripper_state)

处理流程：
  图像 → 视觉编码器(CLIP/SigLIP) → 视觉 token (n_patches, vision_dim)
                                           ↓
                                     投影层 (vision_dim → lang_dim)
                                           ↓
  指令 → Tokenizer → token_ids → Embedding → 文本 token (seq_len, lang_dim)
                                           ↓
  本体感知 → MLP → 本体 token (1, lang_dim)
                                           ↓
  [本体 token, 视觉 token, 文本 token] → LLM (Transformer)
                                           ↓
                                    隐藏状态 (total_len, lang_dim)
                                           ↓
                              动作头 (lang_dim → action_dim)
                                           ↓
                                    机器人动作 (7,)
                                    [dx, dy, dz, droll, dpitch, dyaw, gripper]
```

### 4.2 简化 VLA 模型实现

```python
import torch
import torch.nn as nn

class SimpleVLA(nn.Module):
    """简化的 VLA 模型"""
    def __init__(self, vision_model, llm, action_dim=7):
        super().__init__()
        self.vision_model = vision_model  # CLIP/SigLIP
        self.llm = llm                    # 预训练 LLM
        
        # 视觉到语言的投影层
        vision_dim = 768  # CLIP 输出维度
        lang_dim = 4096   # LLM 隐藏维度
        self.vision_proj = nn.Sequential(
            nn.Linear(vision_dim, lang_dim),
            nn.GELU(),
            nn.Linear(lang_dim, lang_dim),
            nn.LayerNorm(lang_dim),
        )
        
        # 本体感知编码
        proprio_dim = 7  # 6关节 + 1夹爪
        self.proprio_embed = nn.Linear(proprio_dim, lang_dim)
        
        # 动作头
        self.action_head = nn.Sequential(
            nn.Linear(lang_dim, 256),
            nn.ReLU(),
            nn.Linear(256, action_dim),
            nn.Tanh(),  # 输出限制在 [-1, 1]
        )
    
    def forward(self, image, instruction, proprioception):
        """
        image: (B, 3, H, W) 输入图像
        instruction: (B, seq_len) 语言指令 token ids
        proprioception: (B, proprio_dim) 本体感知
        """
        # 1. 视觉编码
        with torch.no_grad():
            vision_features = self.vision_model.encode_image(image)  # (B, vision_dim)
        
        # 2. 投影到语言空间
        vision_tokens = self.vision_proj(vision_features).unsqueeze(1)  # (B, 1, lang_dim)
        
        # 3. 本体感知编码
        proprio_token = self.proprio_embed(proprioception).unsqueeze(1)  # (B, 1, lang_dim)
        
        # 4. 文本 embedding
        text_embed = self.llm.get_input_embeddings()(instruction)  # (B, seq_len, lang_dim)
        
        # 5. 拼接: [proprio, vision, text]
        input_embed = torch.cat([proprio_token, vision_tokens, text_embed], dim=1)
        
        # 6. LLM 前向
        outputs = self.llm(inputs_embeds=input_embed)
        hidden = outputs.last_hidden_state  # (B, total_len, lang_dim)
        
        # 7. 动作预测（取最后一个 token 的隐藏状态）
        action = self.action_head(hidden[:, -1, :])  # (B, action_dim)
        
        return action

# 动作空间说明
action_space = {
    "dim": 7,
    "含义": ["dx", "dy", "dz", "droll", "dpitch", "dyaw", "gripper"],
    "范围": [-1, 1],
    "说明": [
        "dx/dy/dz: 末端执行器位置增量",
        "droll/dpitch/dyaw: 末端执行器旋转增量",
        "gripper: 夹爪动作 (1=闭合, -1=张开)"
    ]
}
```

## 五、实践：使用 CLIP 进行场景理解

```python
class SceneUnderstanding:
    """使用 CLIP 进行场景理解"""
    def __init__(self):
        self.encoder = CLIPEncoder()
    
    def identify_objects(self, image, candidate_objects):
        """识别图像中的物体"""
        # 为每个候选物体生成描述
        descriptions = [f"a photo of a {obj}" for obj in candidate_objects]
        similarity = self.encoder.compute_similarity(image, descriptions)
        
        # 排序
        results = list(zip(candidate_objects, similarity.tolist()))
        results.sort(key=lambda x: x[1], reverse=True)
        
        return results
    
    def estimate_scene(self, image):
        """估计场景类型"""
        scenes = [
            "a kitchen scene",
            "a living room scene",
            "a workshop scene",
            "a laboratory scene",
            "an office desk",
            "a table with objects",
        ]
        similarity = self.encoder.compute_similarity(image, scenes)
        
        best_idx = similarity.argmax().item()
        return scenes[best_idx], similarity[best_idx].item()
    
    def generate_action_prompt(self, image, task):
        """生成动作提示"""
        prompt = f"In: What action should the robot take to {task}?\nOut:"
        return prompt
```
