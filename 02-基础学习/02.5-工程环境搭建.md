## 一、Conda 环境管理最佳实践

### 1.1 安装 Miniconda

```bash
# Linux
wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh
bash Miniconda3-latest-Linux-x86_64.sh
source ~/.bashrc

# 验证
conda --version
```

### 1.2 创建具身智能开发环境

```bash
# 创建新环境
conda create -n embodied python=3.10 -y
conda activate embodied

# 配置 pip 镜像（国内加速）
pip config set global.index-url https://pypi.tuna.tsinghua.edu.cn/simple

# 配置 HuggingFace 镜像
export HF_ENDPOINT=https://hf-mirror.com
# 写入 ~/.bashrc 永久生效
echo 'export HF_ENDPOINT=https://hf-mirror.com' >> ~/.bashrc
```

### 1.3 环境管理常用命令

```bash
# 查看所有环境
conda env list

# 克隆环境
conda create -n embodied_v2 --clone embodied

# 导出环境
conda env export > environment.yml

# 从文件创建环境
conda env create -f environment.yml

# 删除环境
conda env remove -n old_env

# 查看已安装的包
conda list
pip list
```

### 1.4 多环境管理策略

```bash
# 建议：为不同项目创建独立环境
conda create -n lerobot python=3.10 -y      # LeRobot 环境
conda create -n openvla python=3.10 -y       # OpenVLA 环境
conda create -n isaac python=3.10 -y         # Isaac Lab 环境
conda create -n mujoco python=3.10 -y        # MuJoCo 环境

# 使用 conda-forge 渠道（包更全更新）
conda config --add channels conda-forge
```

## 二、CUDA / cuDNN 版本管理

### 2.1 查看当前 CUDA 版本

```bash
# 查看 NVIDIA 驱动支持的最高 CUDA 版本
nvidia-smi

# 查看当前安装的 CUDA 版本
nvcc --version

# 查看 cuDNN 版本
cat /usr/local/cuda/include/cudnn_version.h | grep CUDNN_MAJOR -A 2
# 或
dpkg -l | grep cudnn
```

### 2.2 PyTorch 与 CUDA 版本对应

```bash
# 查看 PyTorch 编译时使用的 CUDA 版本
python -c "import torch; print(torch.version.cuda)"

# 根据 CUDA 版本安装 PyTorch
# CUDA 11.8
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu118

# CUDA 12.1
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121

# CUDA 12.4
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu124

# CPU 版本（无GPU时）
pip install torch torchvision torchaudio
```

### 2.3 版本兼容性检查脚本

```python
import torch
import sys

def check_environment():
    """检查深度学习环境"""
    print("=" * 50)
    print("环境检查")
    print("=" * 50)
    
    # Python 版本
    print(f"Python: {sys.version}")
    
    # PyTorch
    print(f"PyTorch: {torch.__version__}")
    print(f"CUDA available: {torch.cuda.is_available()}")
    
    if torch.cuda.is_available():
        print(f"CUDA version (PyTorch): {torch.version.cuda}")
        print(f"cuDNN version: {torch.backends.cudnn.version()}")
        
        # GPU 信息
        for i in range(torch.cuda.device_count()):
            gpu_name = torch.cuda.get_device_name(i)
            gpu_mem = torch.cuda.get_device_properties(i).total_mem / 1e9
            print(f"GPU {i}: {gpu_name} ({gpu_mem:.1f} GB)")
    
    # 检查其他关键库
    try:
        import numpy; print(f"NumPy: {numpy.__version__}")
    except: print("NumPy: 未安装")
    
    try:
        import transformers; print(f"Transformers: {transformers.__version__}")
    except: print("Transformers: 未安装")
    
    try:
        import mujoco; print(f"MuJoCo: {mujoco.__version__}")
    except: print("MuJoCo: 未安装")

check_environment()
```

## 三、Docker 容器化

### 3.1 Docker 基础

```bash
# 安装 Docker（Ubuntu）
sudo apt-get update
sudo apt-get install docker.io
sudo systemctl start docker
sudo systemctl enable docker

# 将当前用户加入 docker 组（免sudo）
sudo usermod -aG docker $USER
# 重新登录生效

# NVIDIA Docker（GPU 支持）
distribution=$(. /etc/os-release;echo $ID$VERSION_ID)
curl -s -L https://nvidia.github.io/nvidia-docker/gpgkey | sudo apt-key add -
curl -s -L https://nvidia.github.io/nvidia-docker/$distribution/nvidia-docker.list | sudo tee /etc/apt/sources.list.d/nvidia-docker.list
sudo apt-get update
sudo apt-get install -y nvidia-container-toolkit
sudo systemctl restart docker
```

### 3.2 具身智能 Dockerfile

```dockerfile
# Dockerfile for Embodied AI Development
FROM nvidia/cuda:12.1.0-cudnn8-devel-ubuntu22.04

# 设置环境变量
ENV DEBIAN_FRONTEND=noninteractive
ENV PYTHON_VERSION=3.10
ENV CONDA_DIR=/opt/conda

# 安装系统依赖
RUN apt-get update && apt-get install -y \
    python3.10 \
    python3-pip \
    git \
    wget \
    curl \
    vim \
    build-essential \
    libgl1-mesa-glx \
    libglib2.0-0 \
    && rm -rf /var/lib/apt/lists/*

# 安装 PyTorch
RUN pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121

# 安装具身智能相关库
RUN pip install \
    numpy \
    scipy \
    matplotlib \
    opencv-python \
    transformers \
    diffusers \
    peft \
    accelerate \
    wandb \
    mujoco \
    gymnasium \
    stable-baselines3

WORKDIR /workspace
CMD ["bash"]
```

### 3.3 构建和使用

```bash
# 构建镜像
docker build -t embodied-ai:latest .

# 运行容器（带GPU）
docker run --gpus all -it --rm \
    -v $(pwd):/workspace \
    -v /tmp/.X11-unix:/tmp/.X11-unix \
    -e DISPLAY=$DISPLAY \
    --shm-size=16g \
    embodied-ai:latest

# 使用 Docker Compose
# docker-compose.yml
```

```yaml
# docker-compose.yml
version: '3.8'
services:
  embodied:
    image: embodied-ai:latest
    deploy:
      resources:
        reservations:
          devices:
            - driver: nvidia
              count: all
              capabilities: [gpu]
    volumes:
      - ./:/workspace
      - /tmp/.X11-unix:/tmp/.X11-unix
    environment:
      - DISPLAY=${DISPLAY}
    shm_size: 16g
    stdin_open: true
    tty: true
```

## 四、Git 工作流

### 4.1 基本配置

```bash
# 安装 Git
sudo apt-get install git

# 配置用户信息
git config --global user.name "Your Name"
git config --global user.email "your.email@example.com"

# 配置默认编辑器
git config --global core.editor vim

# 配置行尾（跨平台）
git config --global core.autocrlf input  # Linux/Mac
git config --global core.autocrlf true   # Windows

# 设置别名（提高效率）
git config --global alias.st status
git config --global alias.co checkout
git config --global alias.br branch
git config --global alias.ci commit
git config --global alias.lg "log --oneline --graph --all"
```

### 4.2 研究项目 Git 工作流

```bash
# 克隆项目
git clone https://github.com/huggingface/lerobot.git
cd lerobot

# 创建特性分支
git checkout -b feature/my-experiment

# 提交更改
git add .
git commit -m "feat: add new diffusion policy experiment"

# 同步上游
git remote add upstream https://github.com/huggingface/lerobot.git
git fetch upstream
git rebase upstream/main  # 或 merge

# 推送
git push origin feature/my-experiment
```

## 五、常见环境冲突排查

### 5.1 常见问题与解决

```bash
# 问题1: CUDA 版本不匹配
# 症状: RuntimeError: CUDA error
# 解决:
pip uninstall torch torchvision torchaudio
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121

# 问题2: MuJoCo 渲染失败
# 症状: RuntimeError: Could not create GLFW window
# 解决:
export MUJOCO_GL=egl    # 无头渲染（服务器）
# 或
export MUJOCO_GL=osmesa  # 软件渲染

# 问题3: gym vs gymnasium 冲突
# 症状: ImportError: cannot import name '...'
# 解决: 统一使用 gymnasium
pip uninstall gym
pip install gymnasium

# 问题4: 包版本冲突
# 症状: pip install 报依赖冲突
# 解决:
pip install package_name --no-deps  # 跳过依赖检查（慎用）
# 或创建新环境
conda create -n fresh_env python=3.10

# 问题5: 共享内存不足
# 症状: RuntimeError: DataLoader worker (pid X) is killed by signal
# 解决:
# Docker 中增加 shm_size
docker run --shm-size=16g ...
# 或在 DataLoader 中设置 num_workers=0
```

### 5.2 环境诊断脚本

```python
import subprocess
import sys

def diagnose_environment():
    """诊断环境问题"""
    print("=" * 60)
    print("环境诊断")
    print("=" * 60)
    
    # 1. 检查 Python
    print(f"\n[1] Python: {sys.version}")
    
    # 2. 检查关键包
    packages = ['torch', 'numpy', 'transformers', 'mujoco', 
                'gymnasium', 'stable_baselines3', 'cv2']
    print("\n[2] 包版本检查:")
    for pkg in packages:
        try:
            mod = __import__(pkg)
            version = getattr(mod, '__version__', 'unknown')
            print(f"  {pkg}: {version}")
        except ImportError:
            print(f"  {pkg}: 未安装")
    
    # 3. 检查 GPU
    print("\n[3] GPU 检查:")
    try:
        import torch
        if torch.cuda.is_available():
            print(f"  CUDA: {torch.version.cuda}")
            print(f"  GPU: {torch.cuda.get_device_name(0)}")
            mem = torch.cuda.get_device_properties(0).total_mem / 1e9
            print(f"  显存: {mem:.1f} GB")
        else:
            print("  CUDA 不可用")
    except:
        print("  PyTorch 未安装")
    
    # 4. 检查显示
    print("\n[4] 显示检查:")
    import os
    display = os.environ.get('DISPLAY', '未设置')
    mujoco_gl = os.environ.get('MUJOCO_GL', '未设置')
    print(f"  DISPLAY: {display}")
    print(f"  MUJOCO_GL: {mujoco_gl}")
    
    print("\n" + "=" * 60)
    print("诊断完成")
    print("=" * 60)

diagnose_environment()
```

## 六、完整环境搭建脚本

### 6.1 一键安装脚本

```bash
#!/bin/bash
# setup_embodied_ai.sh - 具身智能开发环境一键安装

set -e  # 出错即退出

echo "=========================================="
echo "具身智能开发环境安装脚本"
echo "=========================================="

# 1. 创建 Conda 环境
ENV_NAME="embodied"
echo "[1/6] 创建 Conda 环境: $ENV_NAME"
conda create -n $ENV_NAME python=3.10 -y
source activate $ENV_NAME

# 2. 安装 PyTorch
echo "[2/6] 安装 PyTorch (CUDA 12.1)"
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121

# 3. 安装基础科学计算
echo "[3/6] 安装基础库"
pip install numpy scipy matplotlib pandas scikit-learn
pip install opencv-python pillow
pip install tqdm wandb tensorboard

# 4. 安装 HuggingFace 生态
echo "[4/6] 安装 HuggingFace 生态"
pip install transformers accelerate datasets
pip install diffusers peft bitsandbytes

# 5. 安装机器人仿真
echo "[5/6] 安装仿真环境"
pip install mujoco dm_control
pip install gymnasium
pip install stable-baselines3[extra]
pip install pybullet

# 6. 安装 LeRobot
echo "[6/6] 安装 LeRobot"
pip install lerobot

# 验证
echo ""
echo "=========================================="
echo "验证安装"
echo "=========================================="
python -c "
import torch
import numpy
import transformers
import mujoco
import gymnasium
print(f'PyTorch: {torch.__version__}')
print(f'CUDA: {torch.cuda.is_available()}')
print(f'NumPy: {numpy.__version__}')
print(f'Transformers: {transformers.__version__}')
print(f'MuJoCo: {mujoco.__version__}')
print('安装成功！')
"

echo ""
echo "环境搭建完成！"
echo "激活环境: conda activate $ENV_NAME"
```

### 6.2 使用方法

```bash
# 赋予执行权限
chmod +x setup_embodied_ai.sh

# 运行
./setup_embodied_ai.sh

# 激活环境
conda activate embodied
```
