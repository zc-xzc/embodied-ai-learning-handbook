## 2. BVH 文件获取与处理

### 2.1 BVH 文件格式

BVH (Biovision Hierarchy) 是动作捕捉领域的标准文件格式，包含两部分：

1. **层次结构（Hierarchy）**：定义骨骼的父子关系
2. **运动数据（Motion）**：每帧的旋转和平移数据

```bvh
# BVH 文件示例
HIERARCHY
ROOT Hips
{
    OFFSET 0.000000 0.000000 0.000000
    CHANNELS 6 Xposition Yposition Zposition Zrotation Xrotation Yrotation
    JOINT Spine
    {
        OFFSET 0.000000 0.000000 0.000000
        CHANNELS 3 Zrotation Xrotation Yrotation
        JOINT LeftArm
        {
            OFFSET 0.000000 0.000000 0.000000
            CHANNELS 3 Zrotation Xrotation Yrotation
            ...
        }
    }
}

MOTION
Frames: 100
Frame Time: 0.033333
0.000000 0.000000 0.000000 0.000000 0.000000 0.000000 ...
...
```

### 2.3 BVH 解析器

```python
#!/usr/bin/env python3
"""BVH 文件解析器"""

import numpy as np
from dataclasses import dataclass, field
from typing import List, Dict, Optional

@dataclass
class BVHJoint:
    """BVH 关节"""
    name: str
    offset: np.ndarray  # [3]
    channels: List[str]  # 通道名列表
    parent: Optional['BVHJoint'] = None
    children: List['BVHJoint'] = field(default_factory=list)
    
    # 末端效应器（如果是 End Site）
    is_end_site: bool = False
    end_offset: Optional[np.ndarray] = None

@dataclass
class BVHData:
    """BVH 数据"""
    root: BVHJoint
    joints: List[BVHJoint]  # 所有关节（按顺序）
    joint_map: Dict[str, BVHJoint]  # 名称到关节的映射
    num_frames: int
    frame_time: float
    motion_data: np.ndarray  # [num_frames, num_channels]


class BVHParser:
    """BVH 文件解析器"""
    
    def __init__(self):
        self.joints = []
        self.joint_map = {}
        self.channel_order = []
    
    def parse(self, filepath):
        """
        解析 BVH 文件
        
        Args:
            filepath: BVH 文件路径
        
        Returns:
            BVHData 对象
        """
        with open(filepath, 'r') as f:
            content = f.read()
        
        # 解析层次结构
        hierarchy_end = content.find('MOTION')
        hierarchy_str = content[:hierarchy_end]
        motion_str = content[hierarchy_end:]
        
        root = self._parse_hierarchy(hierarchy_str)
        
        # 解析运动数据
        num_frames, frame_time, motion_data = self._parse_motion(motion_str)
        
        return BVHData(
            root=root,
            joints=self.joints,
            joint_map=self.joint_map,
            num_frames=num_frames,
            frame_time=frame_time,
            motion_data=motion_data,
        )
    
    def _parse_hierarchy(self, content):
        """解析层次结构"""
        lines = content.strip().split('\n')
        idx = 0
        
        # 跳过 HIERARCHY 关键字
        while idx < len(lines) and 'HIERARCHY' not in lines[idx]:
            idx += 1
        idx += 1
        
        return self._parse_joint(lines, idx, parent=None)
    
    def _parse_joint(self, lines, idx, parent):
        """递归解析关节"""
        line = lines[idx].strip()
        
        # 判断 ROOT 或 JOINT
        if line.startswith('ROOT'):
            joint_name = line.split()[1]
        elif line.startswith('JOINT'):
            joint_name = line.split()[1]
        elif line.startswith('End'):
            # End Site
            joint_name = parent.name + '_end'
            idx += 1  # 跳过 {
            idx += 1  # 跳过 OFFSET 行
            offset_parts = lines[idx].strip().split()
            offset = np.array([float(offset_parts[1]), 
                              float(offset_parts[2]), 
                              float(offset_parts[3])])
            
            end_joint = BVHJoint(
                name=joint_name,
                offset=offset,
                channels=[],
                parent=parent,
                is_end_site=True,
                end_offset=offset,
            )
            parent.children.append(end_joint)
            self.joints.append(end_joint)
            self.joint_map[joint_name] = end_joint
            
            idx += 1  # 跳过 OFFSET 行
            idx += 1  # 跳过 }
            return end_joint, idx
        
        idx += 1  # 跳过 {
        
        # OFFSET
        idx += 1
        offset_parts = lines[idx].strip().split()
        offset = np.array([float(offset_parts[1]), 
                          float(offset_parts[2]), 
                          float(offset_parts[3])])
        
        # CHANNELS
        idx += 1
        channel_parts = lines[idx].strip().split()
        num_channels = int(channel_parts[1])
        channels = channel_parts[2:2+num_channels]
        
        joint = BVHJoint(
            name=joint_name,
            offset=offset,
            channels=channels,
            parent=parent,
        )
        
        if parent:
            parent.children.append(joint)
        
        self.joints.append(joint)
        self.joint_map[joint_name] = joint
        self.channel_order.extend(channels)
        
        idx += 1
        
        # 递归解析子关节
        while idx < len(lines):
            line = lines[idx].strip()
            if line == '}':
                idx += 1
                break
            elif line.startswith('JOINT') or line.startswith('End'):
                child, idx = self._parse_joint(lines, idx, joint)
            else:
                idx += 1
        
        return joint, idx
    
    def _parse_motion(self, content):
        """解析运动数据"""
        lines = content.strip().split('\n')
        
        num_frames = 0
        frame_time = 0.0
        
        idx = 0
        while idx < len(lines):
            line = lines[idx].strip()
            if line.startswith('Frames:'):
                num_frames = int(line.split(':')[1].strip())
            elif line.startswith('Frame Time:'):
                frame_time = float(line.split(':')[1].strip())
                idx += 1
                break
            idx += 1
        
        # 解析运动数据
        motion_lines = lines[idx:]
        motion_data = np.zeros((num_frames, len(self.channel_order)))
        
        for i, line in enumerate(motion_lines):
            if i >= num_frames:
                break
            values = [float(v) for v in line.strip().split()]
            motion_data[i] = values
        
        return num_frames, frame_time, motion_data
    
    def get_frame_data(self, bvh_data, frame_idx):
        """获取指定帧的数据"""
        return bvh_data.motion_data[frame_idx]


# 使用示例
if __name__ == "__main__":
    parser = BVHParser()
    bvh_data = parser.parse("examples/bvh/walking.bvh")
    
    print(f"关节数: {len(bvh_data.joints)}")
    print(f"帧数: {bvh_data.num_frames}")
    print(f"帧时间: {bvh_data.frame_time}")
    print(f"通道数: {len(bvh_data.motion_data[0])}")
    
    # 打印关节层次
    def print_hierarchy(joint, indent=0):
        prefix = "  " * indent
        print(f"{prefix}- {joint.name} (channels: {joint.channels})")
        for child in joint.children:
            print_hierarchy(child, indent + 1)
    
    print("\n关节层次结构:")
    print_hierarchy(bvh_data.root)
```

### 2.4 BVH 数据预处理

```python
#!/usr/bin/env python3
"""BVH 数据预处理"""

import numpy as np

class BVHPreprocessor:
    """BVH 数据预处理器"""
    
    def __init__(self, bvh_data):
        self.bvh = bvh_data
    
    def extract_joint_rotations(self, frame_idx):
        """
        提取指定帧的关节旋转数据
        
        Returns:
            dict: {joint_name: rotation_euler (3,)} in degrees
        """
        frame_data = self.bvh.motion_data[frame_idx]
        rotations = {}
        
        channel_idx = 0
        for joint in self.bvh.joints:
            if joint.is_end_site or len(joint.channels) == 0:
                continue
            
            # 提取旋转通道
            rotation = np.zeros(3)
            for i, ch in enumerate(joint.channels):
                if 'rotation' in ch.lower():
                    # 确定旋转轴
                    if 'X' in ch:
                        rotation[0] = frame_data[channel_idx]
                    elif 'Y' in ch:
                        rotation[1] = frame_data[channel_idx]
                    elif 'Z' in ch:
                        rotation[2] = frame_data[channel_idx]
                channel_idx += 1
            
            rotations[joint.name] = np.radians(rotation)  # 转为弧度
        
        return rotations
    
    def extract_root_position(self, frame_idx):
        """提取根节点位置"""
        frame_data = self.bvh.motion_data[frame_idx]
        root_channels = self.bvh.root.channels
        
        position = np.zeros(3)
        for i, ch in enumerate(root_channels):
            if 'position' in ch.lower():
                if 'X' in ch:
                    position[0] = frame_data[i]
                elif 'Y' in ch:
                    position[1] = frame_data[i]
                elif 'Z' in ch:
                    position[2] = frame_data[i]
        
        return position
    
    def downsample(self, target_fps=30):
        """降采样到目标帧率"""
        original_fps = 1.0 / self.bvh.frame_time
        if original_fps <= target_fps:
            return self.bvh
        
        step = int(original_fps / target_fps)
        self.bvh.motion_data = self.bvh.motion_data[::step]
        self.bvh.num_frames = len(self.bvh.motion_data)
        self.bvh.frame_time = 1.0 / target_fps
        
        return self.bvh
    
    def normalize(self):
        """归一化（将根节点高度归零）"""
        for i in range(self.bvh.num_frames):
            # 将 Y 位置归零（BVH 通常 Y 轴朝上）
            root_y_idx = None
            for j, ch in enumerate(self.bvh.root.channels):
                if ch == 'Yposition':
                    root_y_idx = j
                    break
            
            if root_y_idx is not None:
                self.bvh.motion_data[i, root_y_idx] = 0.0
        
        return self.bvh
    
    def smooth(self, window_size=5):
        """平滑滤波"""
        if window_size < 3:
            return self.bvh
        
        # 移动平均平滑
        kernel = np.ones(window_size) / window_size
        
        for col in range(self.bvh.motion_data.shape[1]):
            # 边界处理
            padded = np.pad(
                self.bvh.motion_data[:, col],
                window_size // 2,
                mode='edge',
            )
            self.bvh.motion_data[:, col] = np.convolve(
                padded, kernel, mode='valid'
            )
        
        return self.bvh
```

### 3.1 加载 G1 模型

```python
#!/usr/bin/env python3
"""MuJoCo G1 仿真环境"""

import mujoco
import numpy as np
from pathlib import Path

class G1SimEnv:
    """G1 MuJoCo 仿真环境"""
    
    # G1 29DoF 关节名称
    JOINT_NAMES = [
        # 左腿
        'left_hip_pitch', 'left_hip_roll', 'left_hip_yaw',
        'left_knee', 'left_ankle_pitch', 'left_ankle_roll',
        # 右腿
        'right_hip_pitch', 'right_hip_roll', 'right_hip_yaw',
        'right_knee', 'right_ankle_pitch', 'right_ankle_roll',
        # 腰部
        'waist_yaw', 'waist_roll', 'waist_pitch',
        # 左臂
        'left_shoulder_pitch', 'left_shoulder_roll', 'left_shoulder_yaw',
        'left_elbow', 'left_wrist_roll', 'left_wrist_pitch', 'left_wrist_yaw',
        # 右臂
        'right_shoulder_pitch', 'right_shoulder_roll', 'right_shoulder_yaw',
        'right_elbow', 'right_wrist_roll', 'right_wrist_pitch', 'right_wrist_yaw',
    ]
    
    def __init__(self, model_path, freq=50):
        """
        Args:
            model_path: MJCF 模型文件路径
            freq: 仿真频率 Hz
        """
        # 加载模型
        self.model = mujoco.MjModel.from_xml_path(str(model_path))
        self.data = mujoco.MjData(self.model)
        
        # 仿真参数
        self.freq = freq
        self.dt = 1.0 / freq
        self.model.opt.timestep = self.dt
        
        # 获取关节 ID
        self.joint_ids = {}
        for name in self.JOINT_NAMES:
            try:
                self.joint_ids[name] = self.model.joint(name).id
            except:
                print(f"警告: 关节 '{name}' 未在模型中找到")
        
        self.num_joints = len(self.joint_ids)
        
        # 关节限位
        self.joint_limits = np.zeros((self.num_joints, 2))
        for i, name in enumerate(self.JOINT_NAMES):
            if name in self.joint_ids:
                jid = self.joint_ids[name]
                self.joint_limits[i] = [
                    self.model.jnt_range[jid][0],
                    self.model.jnt_range[jid][1],
                ]
        
        # 初始姿态
        self.reset()
        
        print(f"G1 仿真环境已初始化")
        print(f"  模型: {model_path}")
        print(f"  关节数: {self.num_joints}")
        print(f"  频率: {freq} Hz")
    
    def reset(self):
        """重置仿真"""
        mujoco.mj_resetData(self.model, self.data)
        
        # 设置初始姿态（站立）
        self._set_stand_pose()
        
        # 前进物理几步以稳定
        for _ in range(100):
            mujoco.mj_step(self.model, self.data)
    
    def _set_stand_pose(self):
        """设置站立姿态"""
        stand_pose = {
            'left_hip_pitch': -0.1,
            'right_hip_pitch': -0.1,
            'left_knee': 0.3,
            'right_knee': 0.3,
            'left_ankle_pitch': -0.2,
            'right_ankle_pitch': -0.2,
        }
        
        for name, angle in stand_pose.items():
            if name in self.joint_ids:
                jid = self.joint_ids[name]
                self.data.qpos[self.model.jnt_qposadr[jid]] = angle
    
    def set_joint_targets(self, targets):
        """
        设置关节目标角度
        
        Args:
            targets: dict {joint_name: angle_rad} 或 np.ndarray [num_joints]
        """
        if isinstance(targets, dict):
            for name, angle in targets.items():
                if name in self.joint_ids:
                    jid = self.joint_ids[name]
                    # 限位
                    lo, hi = self.joint_limits[list(self.JOINT_NAMES).index(name)]
                    angle = np.clip(angle, lo, hi)
                    
                    qpos_idx = self.model.jnt_qposadr[jid]
                    self.data.qpos[qpos_idx] = angle
        else:
            # 数组形式
            for i, name in enumerate(self.JOINT_NAMES):
                if name in self.joint_ids:
                    jid = self.joint_ids[name]
                    angle = np.clip(targets[i], 
                                   self.joint_limits[i, 0], 
                                   self.joint_limits[i, 1])
                    qpos_idx = self.model.jnt_qposadr[jid]
                    self.data.qpos[qpos_idx] = angle
    
    def step(self):
        """前进一步仿真"""
        mujoco.mj_step(self.model, self.data)
    
    def get_joint_positions(self):
        """获取当前关节角度"""
        positions = np.zeros(self.num_joints)
        for i, name in enumerate(self.JOINT_NAMES):
            if name in self.joint_ids:
                jid = self.joint_ids[name]
                qpos_idx = self.model.jnt_qposadr[jid]
                positions[i] = self.data.qpos[qpos_idx]
        return positions
    
    def get_joint_velocities(self):
        """获取关节速度"""
        velocities = np.zeros(self.num_joints)
        for i, name in enumerate(self.JOINT_NAMES):
            if name in self.joint_ids:
                jid = self.joint_ids[name]
                dof_idx = self.model.jnt_dofadr[jid]
                velocities[i] = self.data.qvel[dof_idx]
        return velocities
    
    def get_base_position(self):
        """获取基座位置"""
        return self.data.qpos[:3].copy()
    
    def get_base_orientation(self):
        """获取基座朝向（四元数）"""
        return self.data.qpos[3:7].copy()
    
    def render(self, camera_name='front', width=640, height=480):
        """渲染图像"""
        renderer = mujoco.Renderer(self.model, width, height)
        renderer.update_scene(self.data, camera=camera_name)
        return renderer.render()
```

### 3.2 BVH 到 G1 的运动重定向

```python
#!/usr/bin/env python3
"""BVH 到 G1 的运动重定向"""

import numpy as np
from scipy.spatial.transform import Rotation as R

class BVHToG1Retargeter:
    """BVH 到 G1 的运动重定向器"""
    
    # BVH 关节名到 G1 关节名的映射
    JOINT_MAPPING = {
        # BVH 关节名: G1 关节名
        'LeftUpLeg': ['left_hip_pitch', 'left_hip_roll', 'left_hip_yaw'],
        'LeftLeg': ['left_knee'],
        'LeftFoot': ['left_ankle_pitch', 'left_ankle_roll'],
        
        'RightUpLeg': ['right_hip_pitch', 'right_hip_roll', 'right_hip_yaw'],
        'RightLeg': ['right_knee'],
        'RightFoot': ['right_ankle_pitch', 'right_ankle_roll'],
        
        'Spine': ['waist_yaw', 'waist_roll', 'waist_pitch'],
        
        'LeftArm': ['left_shoulder_pitch', 'left_shoulder_roll', 'left_shoulder_yaw'],
        'LeftForeArm': ['left_elbow'],
        'LeftHand': ['left_wrist_roll', 'left_wrist_pitch', 'left_wrist_yaw'],
        
        'RightArm': ['right_shoulder_pitch', 'right_shoulder_roll', 'right_shoulder_yaw'],
        'RightForeArm': ['right_elbow'],
        'RightHand': ['right_wrist_roll', 'right_wrist_pitch', 'right_wrist_yaw'],
    }
    
    # 重定向比例因子
    SCALE_FACTORS = {

        self.num_joints = num_g1_joints
        self.preprocessor = BVHPreprocessor(bvh_data)
    
    def retarget_frame(self, frame_idx):
        """
        重定向单帧 BVH 数据到 G1 关节角度
        
        Args:
            frame_idx: 帧索引
        
        Returns:
            np.ndarray: [29] G1 关节角度 (rad)
        """
        # 提取 BVH 关节旋转
        bvh_rotations = self.preprocessor.extract_joint_rotations(frame_idx)
        
        # 初始化 G1 关节角度
        g1_angles = np.zeros(self.num_joints)
        
        # BVH 关节名列表（与 G1 JOINT_NAMES 对应）
        g1_joint_names = G1SimEnv.JOINT_NAMES
        
        # 遍历映射
        for bvh_name, g1_names in self.JOINT_MAPPING.items():
            if bvh_name not in bvh_rotations:
                continue
            
            bvh_rot = bvh_rotations[bvh_name]  # [3] euler angles (rad)
            
            # 映射到 G1 关节
            for i, g1_name in enumerate(g1_names):
                if g1_name not in g1_joint_names:
                    continue
                
                g1_idx = g1_joint_names.index(g1_name)
                
                # 简单映射：直接取对应轴
                if 'pitch' in g1_name:
                    angle = bvh_rot[0]  # X 旋转
                elif 'roll' in g1_name:
                    angle = bvh_rot[1]  # Y 旋转
                elif 'yaw' in g1_name:
                    angle = bvh_rot[2]  # Z 旋转
                else:
                    angle = bvh_rot[0]
                
                # 应用比例因子
                scale = self.SCALE_FACTORS.get(g1_name, 1.0)
                angle *= scale
                
                # 限位
                lo, hi = self.JOINT_LIMITS.get(g1_name, (-3.14, 3.14))
                angle = np.clip(angle, lo, hi)
                
                g1_angles[g1_idx] = angle
        
        return g1_angles
    
    def retarget_all(self):
        """重定向所有帧"""
        all_angles = np.zeros((self.bvh.num_frames, self.num_joints))
        
        for i in range(self.bvh.num_frames):
            all_angles[i] = self.retarget_frame(i)
        
        # 平滑
        all_angles = self._smooth_trajectory(all_angles)
        
        return all_angles
    
    def _smooth_trajectory(self, trajectory, window=5):
        """平滑轨迹"""
        if window < 3:
            return trajectory
        
        smoothed = np.copy(trajectory)
        kernel = np.ones(window) / window
        
        for j in range(trajectory.shape[1]):
            padded = np.pad(trajectory[:, j], window // 2, mode='edge')
            smoothed[:, j] = np.convolve(padded, kernel, mode='valid')
        
        return smoothed

        self.retargeter = BVHToG1Retargeter(self.bvh_data)

        print(f"  目标角度: {self.target_angles.shape}")

        self.viewer_enabled = viewer
        self.record_enabled = record
        self.output_dir = Path(output_dir)
        
        if record:
            self.output_dir.mkdir(parents=True, exist_ok=True)
            self.recorded_data = {
                'joint_positions': [],
                'joint_velocities': [],
                'base_positions': [],
                'images': [],
            }
        
        # 6. 可视化
        self.viewer = None
        if viewer:
            import mujoco
            self.viewer = mujoco.viewer.launch_passive(
                self.env.model, self.env.data
            )

        print("=" * 50)
        
        current_frame = 0
        
        try:
            while current_frame < num_frames:
                # 获取目标角度

                # 设置目标

                # 仿真步进
                self.env.step()
                
                # 录制数据
                if self.record_enabled:
                    self._record_frame(current_frame)
                
                # 更新可视化

                    self.viewer.sync()
                
                # 控制帧率
                time.sleep(self.dt)
                
                current_frame += 1
                
                # 进度
                if current_frame % 100 == 0:
                    print(f"  帧 {current_frame}/{num_frames}")

#!/usr/bin/env python3
"""多视图渲染"""

import mujoco
import numpy as np
import cv2

class MultiViewRenderer:
    """多视图渲染器"""
    
    def __init__(self, model, data, cameras=None, width=640, height=480):
        """
        Args:
            model: MuJoCo 模型
            data: MuJoCo 数据
            cameras: 相机名列表
            width: 图像宽度
            height: 图像高度
        """
        self.model = model
        self.data = data
        self.width = width
        self.height = height
        
        self.cameras = cameras or ['front', 'side', 'top', 'track']
        
        # 为每个相机创建渲染器
        self.renderers = {}
        for cam in self.cameras:
            self.renderers[cam] = mujoco.Renderer(model, width, height)
    
    def render_all(self):
        """渲染所有视图"""
        images = {}
        for cam_name, renderer in self.renderers.items():
            renderer.update_scene(self.data, camera=cam_name)
            images[cam_name] = renderer.render()
        return images
    
    def render_grid(self, cols=2):
        """
        将多个视图拼接成网格
        
        Returns:
            np.ndarray: 拼接后的图像
        """
        images = self.render_all()
        
        img_list = list(images.values())
        rows = (len(img_list) + cols - 1) // cols
        
        # 创建网格
        grid_rows = []
        for r in range(rows):
            row_imgs = img_list[r * cols : (r + 1) * cols]
            # 补齐
            while len(row_imgs) < cols:
                row_imgs.append(np.zeros_like(img_list[0]))
            grid_rows.append(np.hstack(row_imgs))
        
        grid = np.vstack(grid_rows)
        return grid
    
    def render_with_labels(self):
        """渲染带标签的视图"""
        images = self.render_all()
        
        labeled = {}
        for cam_name, img in images.items():
            # 添加标签
            labeled_img = img.copy()
            cv2.putText(
                labeled_img, cam_name,
                (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX,
                1.0, (0, 255, 0), 2,
            )
            labeled[cam_name] = labeled_img
        
        return labeled

#!/usr/bin/env python3
"""策略模型推理"""

import torch
import numpy as np
import mujoco

class PolicyInference:
    """策略推理器"""
    
    def __init__(self, model_path, device='cuda'):
        self.device = torch.device(device)
        
        # 加载模型
        checkpoint = torch.load(model_path, map_location=self.device)
        
        # 根据模型类型加载
        if 'policy_state_dict' in checkpoint:
            self.policy = self._build_policy(checkpoint['config'])
            self.policy.load_state_dict(checkpoint['policy_state_dict'])
        else:
            # 直接加载模型
            self.policy = checkpoint['model']
        
        self.policy.to(self.device)
        self.policy.eval()
        
        # 动作缓冲区
        self.action_buffer = []
        self.buffer_idx = 0
        
        # 配置
        self.horizon = checkpoint.get('horizon', 16)
        self.n_action_steps = checkpoint.get('n_action_steps', 8)
        
        print(f"策略模型已加载: {model_path}")
        print(f"  Horizon: {self.horizon}")
        print(f"  Action steps: {self.n_action_steps}")
    
    def _build_policy(self, config):
        """构建策略模型"""
        # 根据 config 构建模型
        # 这里使用简化的示例
        from torch import nn
        
        class SimplePolicy(nn.Module):
            def __init__(self, state_dim, action_dim, hidden=256):
                super().__init__()
                self.net = nn.Sequential(
                    nn.Linear(state_dim, hidden),
                    nn.ReLU(),
                    nn.Linear(hidden, hidden),
                    nn.ReLU(),
                    nn.Linear(hidden, action_dim),
                )
            
            def forward(self, x):
                return self.net(x)
        
        return SimplePolicy(
            state_dim=config['state_dim'],
            action_dim=config['action_dim'],
        )
    
    @torch.no_grad()
    def predict(self, observation):
        """
        预测动作
        
        Args:
            observation: dict
                - state: [state_dim] 关节状态
                - images: dict of camera_name -> [H, W, 3]
                - instruction: str
        
        Returns:
            action_chunk: [horizon, action_dim] 动作片段
        """
        # 预处理状态
        state = torch.from_numpy(observation['state']).float()
        state = state.unsqueeze(0).to(self.device)
        
        # 推理
        action_chunk = self.policy(state)
        
        return action_chunk.cpu().numpy()[0]
    
    def get_action(self, observation):
        """获取当前动作（带缓冲区）"""
        if self.buffer_idx >= len(self.action_buffer):
            self.action_buffer = self.predict(observation)
            self.buffer_idx = 0
        
        action = self.action_buffer[self.buffer_idx]
        self.buffer_idx += 1
        return action
    
    def reset(self):
        """重置缓冲区"""
        self.action_buffer = []
        self.buffer_idx = 0


# 仿真中的策略推理
class PolicySimRunner:
    """策略推理仿真运行器"""
    
    def __init__(self, model_path, sim_env, instruction="pick up the cup"):
        self.env = sim_env
        self.policy = PolicyInfernece(model_path)
        self.instruction = instruction
        
        # 可视化
        import mujoco
        self.viewer = mujoco.viewer.launch_passive(
            self.env.model, self.env.data
        )
    
    def run(self, max_steps=1000):
        """运行策略推理仿真"""
        self.env.reset()
        self.policy.reset()
        
        print(f"开始策略推理仿真")
        print(f"  指令: {self.instruction}")
        
        for step in range(max_steps):
            # 获取观测
            observation = {
                'state': self.env.get_joint_positions(),
                'images': {},  # 如果需要图像，在此获取
                'instruction': self.instruction,
            }
            
            # 获取动作
            action = self.policy.get_action(observation)
            
            # 执行动作
            self.env.set_joint_targets(action)
            self.env.step()
            
            # 更新可视化
            self.viewer.sync()
            
            # 控制帧率
            import time
            time.sleep(self.env.dt)
            
            if step % 100 == 0:
                print(f"  Step {step}/{max_steps}")
        
        print("策略推理仿真完成")

"""仿真数据录制器"""

import h5py
import numpy as np
import time
from pathlib import Path

class SimDataRecorder:
    """仿真数据录制器"""
    
    def __init__(
        self,
        output_path,
        num_joints=29,
        fps=30,
        camera_names=None,
        image_size=(480, 640, 3),
    ):
        """
        Args:
            output_path: 输出 HDF5 文件路径
            num_joints: 关节数
            fps: 录制帧率
            camera_names: 相机名列表
            image_size: 图像尺寸
        """
        self.output_path = Path(output_path)
        self.output_path.parent.mkdir(parents=True, exist_ok=True)
        
        self.num_joints = num_joints
        self.fps = fps
        self.camera_names = camera_names or ['front']
        self.image_size = image_size
        
        # 数据缓存
        self.frames = []
        
        print(f"数据录制器已初始化: {output_path}")
        print(f"  关节数: {num_joints}")
        print(f"  帧率: {fps} Hz")
        print(f"  相机: {camera_names}")
    
    def record_frame(self, joint_positions, joint_velocities, images=None, 
                     action=None, instruction=""):
        """录制一帧数据"""
        frame = {
            'timestamp': len(self.frames) / self.fps,
            'frame_index': len(self.frames),
            'joint_positions': joint_positions.copy(),
            'joint_velocities': joint_velocities.copy(),
            'instruction': instruction,
        }
        
        if action is not None:
            frame['action'] = action.copy()
        
        if images:
            frame['images'] = {k: v.copy() for k, v in images.items()}
        
        self.frames.append(frame)
    
    def save(self):
        """保存到 HDF5"""
        num_frames = len(self.frames)
        
        with h5py.File(self.output_path, 'w') as f:
            # 关节数据
            f.create_dataset(
                'joint_positions',
                data=np.array([fr['joint_positions'] for fr in self.frames]),
                compression='gzip',
            )
            f.create_dataset(
                'joint_velocities',
                data=np.array([fr['joint_velocities'] for fr in self.frames]),
                compression='gzip',
            )
            
            # 动作数据
            if 'action' in self.frames[0]:
                f.create_dataset(
                    'action',
                    data=np.array([fr['action'] for fr in self.frames]),
                    compression='gzip',
                )
            
            # 图像数据
            if 'images' in self.frames[0]:
                for cam_name in self.camera_names:
                    images = np.array([
                        fr['images'][cam_name] 
                        for fr in self.frames 
                        if 'images' in fr and cam_name in fr['images']
                    ])
                    if len(images) > 0:
                        f.create_dataset(
                            f'images/{cam_name}',
                            data=images,
                            compression='gzip',
                        )
            
            # 元数据
            f.attrs['num_frames'] = num_frames
            f.attrs['fps'] = self.fps
            f.attrs['num_joints'] = self.num_joints
            f.attrs['instruction'] = self.frames[0].get('instruction', '')
            f.attrs['timestamp'] = time.strftime('%Y-%m-%d %H:%M:%S')
        
        print(f"数据已保存: {self.output_path}")
        print(f"  总帧数: {num_frames}")
        print(f"  文件大小: {self.output_path.stat().st_size / 1e6:.1f} MB")
    
    def clear(self):
        """清空缓存"""
        self.frames = []
```

### 6.2 仿真数据回放

```python
#!/usr/bin/env python3
"""仿真数据回放器"""

import h5py
import numpy as np
import time

class SimDataPlayer:
    """仿真数据回放器"""
    
    def __init__(self, hdf5_path, sim_env):
        """
        Args:
            hdf5_path: HDF5 数据文件路径
            sim_env: G1SimEnv 实例
        """
        self.env = sim_env
        self.data = h5py.File(hdf5_path, 'r')
        
        self.num_frames = self.data.attrs['num_frames']
        self.fps = self.data.attrs['fps']
        
        print(f"加载数据: {hdf5_path}")
        print(f"  帧数: {self.num_frames}")
        print(f"  帧率: {self.fps} Hz")
    
    def play(self, start_frame=0, end_frame=None, speed=1.0):
        """
        回放数据
        
        Args:
            start_frame: 起始帧
            end_frame: 结束帧
            speed: 回放速度 (1.0 = 正常速度)
        """
        end_frame = end_frame or self.num_frames
        dt = 1.0 / self.fps / speed
        
        import mujoco
        viewer = mujoco.viewer.launch_passive(
            self.env.model, self.env.data
        )
        
        print(f"开始回放 ({start_frame}-{end_frame}, {speed}x)")
        
        for frame_idx in range(start_frame, end_frame):
            # 获取数据
            joint_pos = self.data['joint_positions'][frame_idx]
            
            # 设置到仿真
            self.env.set_joint_targets(joint_pos)
            self.env.step()
            
            # 更新可视化
            viewer.sync()
            
            # 控制帧率
            time.sleep(dt)
            
            if frame_idx % 100 == 0:
                print(f"  帧 {frame_idx}/{end_frame}")
        
        print("回放完成")
    
    def close(self):
        self.data.close()
```
