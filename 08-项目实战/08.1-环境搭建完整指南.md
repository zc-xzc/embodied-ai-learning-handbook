## 2. 基础环境安装

### 2.1 系统更新

```bash
# Ubuntu 系统更新
sudo apt update && sudo apt upgrade -y

# 安装基础开发工具
sudo apt install -y \
    build-essential \
    cmake \
    git \
    wget \
    curl \
    unzip \
    vim \
    htop \
    tree \
    terminator \
    python3-dev \
    python3-pip \
    python3-venv

# 安装媒体处理工具
sudo apt install -y \
    ffmpeg \
    libavcodec-dev \
    libavformat-dev \
    libswscale-dev

# 安装图像处理库
sudo apt install -y \
    libopencv-dev \
    libjpeg-dev \
    libpng-dev
```

### 2.2 Conda 安装

```bash
# 下载 Miniconda
wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh

# 安装
bash Miniconda3-latest-Linux-x86_64.sh -b -p $HOME/miniconda3

# 激活
source $HOME/miniconda3/bin/activate

# 初始化
conda init bash
source ~/.bashrc

# 配置镜像（国内加速）
conda config --add channels https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/main/
conda config --add channels https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/free/
conda config --set show_channel_urls yes

# 验证
conda --version
```

### 2.3 uv 安装（快速包管理器）

```bash
# 安装 uv
curl -LsSf https://astral.sh/uv/install.sh | sh

# 添加到 PATH
source $HOME/.local/bin/env

# 验证
uv --version

# 配置 pip 镜像
uv pip config set global.index-url https://pypi.tuna.tsinghua.edu.cn/simple
```

### 2.4 Git LFS 安装

```bash
# 安装 Git LFS
sudo apt install -y git-lfs

# 初始化
git lfs install

# 验证
git lfs version
```

### 2.5 CUDA 安装

```bash
# 检查 GPU 驱动
nvidia-smi

# 如果未安装驱动
sudo apt install -y nvidia-driver-535

# 安装 CUDA 12.1
wget https://developer.download.nvidia.com/compute/cuda/12.1.0/local_installers/cuda_12.1.0_530.30.02_linux.run
sudo sh cuda_12.1.0_530.30.02_linux.run

# 配置环境变量
echo 'export CUDA_HOME=/usr/local/cuda-12.1' >> ~/.bashrc
echo 'export PATH=$CUDA_HOME/bin:$PATH' >> ~/.bashrc
echo 'export LD_LIBRARY_PATH=$CUDA_HOME/lib64:$LD_LIBRARY_PATH' >> ~/.bashrc
source ~/.bashrc

# 验证
nvcc --version
```


## 3. Teleopit 完整环境搭建

### 3.1 项目概述

Teleopit 是一个完整的人形机器人遥操作框架，支持从 BVH 动作到 G1 机器人的运动重定向，包含 Sim2Sim 仿真和 Sim2Real 部署。

### 3.2 环境搭建步骤

```bash
# ============================================================
# Step 1: 克隆项目
# ============================================================
cd ~/projects
git clone https://github.com/your-org/Teleopit.git
cd Teleopit

# ============================================================
# Step 2: 创建 conda 环境
# ============================================================
conda create -n teleopit python=3.10 -y
conda activate teleopit

# ============================================================
# Step 3: 安装 PyTorch
# ============================================================
pip install torch torchvision torchaudio \
    --index-url https://download.pytorch.org/whl/cu121

# ============================================================
# Step 4: 安装 MuJoCo
# ============================================================
# 下载 MuJoCo 3.x
pip install mujoco

# 验证
python -c "import mujoco; print(f'MuJoCo: {mujoco.__version__}')"

# 安装 MuJoCo viewer
pip install mujoco-viewer

# ============================================================
# Step 5: 安装运动学求解器
# ============================================================
pip install mink          # MuJoCo IK 求解器
pip install qpsolvers    # QP 求解器
pip install pinocchio    # 运动学库（可选）

# ============================================================
# Step 6: 安装数据处理库
# ============================================================
pip install h5py          # HDF5 数据读写
pip install numpy scipy   # 数值计算
pip install opencv-python # 图像处理
pip install matplotlib    # 可视化

# ============================================================
# Step 7: 安装 BVH 处理库
# ============================================================
pip install bvh           # BVH 文件解析
pip install pybullet      # 物理仿真（可选）

# ============================================================
# Step 8: 安装 unitree_sdk2py
# ============================================================
pip install unitree_sdk2py

# 或从源码安装
# git clone https://github.com/unitreerobotics/unitree_sdk2py.git
# cd unitree_sdk2py && pip install -e .

# ============================================================
# Step 9: 安装项目依赖
# ============================================================
pip install -r requirements.txt

# ============================================================
# Step 10: 安装项目本身
# ============================================================
pip install -e .

# ============================================================
# Step 11: 下载 G1 MuJoCo 模型
# ============================================================
# 从 Unitree 官方获取 G1 MJCF 模型
mkdir -p models/g1
# 将 g1_29dof.xml 等模型文件放入 models/g1/

# ============================================================
# Step 12: 下载预训练策略模型（可选）
# ============================================================
# 如需运行策略推理，下载模型权重
mkdir -p checkpoints
# 将策略模型放入 checkpoints/

# ============================================================
# 验证安装
# ============================================================
python -c "
import mujoco
import mink
import h5py
import unitree_sdk2py
print('Teleopit 环境验证通过！')
print(f'  MuJoCo: {mujoco.__version__}')
"

# 运行最小仿真测试
python scripts/sim/run_sim.py --model g1 --bvh examples/bvh/walking.bvh
```

## 4. somehand 环境搭建

### 4.1 项目概述

somehand 是灵巧手运动重定向项目，支持 20+ 种灵巧手的统一配置和控制。

### 4.2 环境搭建步骤

```bash
# ============================================================
# Step 1: 克隆项目
# ============================================================
cd ~/projects
git clone https://github.com/your-org/somehand.git
cd somehand

# ============================================================
# Step 2: 创建 conda 环境
# ============================================================
conda create -n somehand python=3.10 -y
conda activate somehand

# ============================================================
# Step 3: 安装基础依赖
# ============================================================
pip install numpy scipy pyyaml

# ============================================================
# Step 4: 安装 MediaPipe（手部追踪）
# ============================================================
pip install mediapipe
pip install opencv-python

# 验证 MediaPipe
python -c "
import mediapipe as mp
hands = mp.solutions.hands.Hands()
print('MediaPipe 手部追踪已就绪')
"

# ============================================================
# Step 5: 安装 CAN 通信库
# ============================================================
# SocketCAN（Linux 内置）
sudo apt install -y can-utils

# Python CAN 库
pip install python-can

# 验证 CAN
python -c "import can; print('python-can 安装成功')"

# ============================================================
# Step 6: 安装串口通信库
# ============================================================
pip install pyserial

# ============================================================
# Step 7: 安装 MuJoCo（仿真用）
# ============================================================
pip install mujoco

# ============================================================
# Step 8: 安装灵巧手专用 SDK
# ============================================================
# LinkerHand SDK（如有）
# pip install linkerhand-sdk

# Unitree Dex5 SDK
pip install unitree_sdk2py

# ============================================================
# Step 9: 安装项目
# ============================================================
pip install -r requirements.txt
pip install -e .

# ============================================================
# Step 10: 配置 CAN 接口（如使用 CAN 通信）
# ============================================================
# 设置 CAN 接口
sudo ip link set can0 type can bitrate 1000000
sudo ip link set up can0

# 测试 CAN
candump can0  # 按 Ctrl+C 退出

# ============================================================
# 验证安装
# ============================================================
python -c "
import somehand
print(f'somehand 版本: {somehand.__version__}')

# 列出支持的灵巧手
from somehand.config import HandConfigLoader
loader = HandConfigLoader()
configs = loader.list_available()

    print(f'  - {c}')
"
```

## 5. pico-bridge 环境搭建

### 5.1 项目概述

pico-bridge 是 PICO 4 VR 头显与 PC 之间的数据桥接工具，支持手部追踪数据、头部位姿数据和视频流的传输。

### 5.2 PC 端环境搭建

```bash
# ============================================================
# Step 1: 克隆项目
# ============================================================
cd ~/projects
git clone https://github.com/your-org/pico-bridge.git
cd pico-bridge

# ============================================================
# Step 2: 创建 conda 环境
# ============================================================
conda create -n pico_bridge python=3.10 -y
conda activate pico_bridge

# ============================================================
# Step 3: 安装依赖
# ============================================================
# ZMQ 通信
pip install pyzmq

# 图像处理
pip install opencv-python numpy

# 数据序列化
pip install msgpack protobuf

# 视频解码
pip install av  # PyAV, 用于 H.264 解码

# ============================================================
# Step 4: 安装项目
# ============================================================
pip install -r requirements.txt
pip install -e .

# ============================================================
# Step 5: 配置网络
# ============================================================
# PICO 4 和 PC 需在同一局域网
# 查看PC IP
hostname -I
# 假设 PC IP: 192.168.1.100

# 配置防火墙（开放端口）
sudo ufw allow 5555/tcp  # ZMQ 数据端口
sudo ufw allow 5556/tcp  # ZMQ 视频端口
sudo ufw allow 5557/udp  # UDP 数据端口

# ============================================================
# 验证安装
# ============================================================
python -c "
import zmq
import cv2
import numpy as np
print('pico-bridge PC 端环境验证通过！')
print(f'  ZMQ 版本: {zmq.zmq_version()}')
print(f'  OpenCV: {cv2.__version__}')
"

# 启动接收端测试
python scripts/receiver_test.py --port 5555
```

### 5.3 PICO 头显端环境搭建

#### 5.3.1 Unity 开发环境

```
所需软件:
1. Unity Hub (最新版)
2. Unity 2022.3 LTS (或更高)
3. PICO SDK for Unity
4. Android Build Support (含 SDK & NDK)
```

```bash
# 详细步骤:

# 1. 安装 Unity Hub
# 下载: https://unity.com/download
# 安装后打开 Unity Hub

# 2. 安装 Unity 2022.3 LTS
# Unity Hub → Installs → Install Editor → 选择 2022.3 LTS
# 勾选:
#   - Android Build Support
#   - OpenJDK
#   - Android SDK & NDK Tools

# 3. 下载 PICO SDK
# 从 PICO 开发者平台下载:
# https://developer.pico-interactive.com/
# 下载 PICO Unity Integration SDK

# 4. 创建 Unity 项目
# Unity Hub → New Project → 3D (URP)
# 项目名: PicoBridge

# 5. 导入 PICO SDK
# Assets → Import Package → Custom Package
# 选择下载的 PICO SDK .unitypackage

# 6. 配置项目设置
# File → Build Settings → Platform: Android → Switch Platform
# Player Settings:
#   - Package Name: com.yourorg.picobridge
#   - Minimum API Level: Android 10 (API 29)
#   - Target API Level: Android 12 (API 31)
#   - Scripting Backend: IL2CPP
#   - Target Architectures: ARM64

# 7. 构建 APK
# File → Build Settings → Build
# 生成 PicoBridge.apk
```

#### 5.3.2 APK 安装到头显

```bash
# 方法1: ADB 安装
# 启用 PICO 开发者模式 → 设置 → 关于 → 连续点击版本号 7 次
# 启用 USB 调试 → 设置 → 开发者选项 → USB 调试

# 通过 USB 连接 PICO 到 PC
adb devices  # 确认设备已连接

# 安装 APK
adb install PicoBridge.apk

# 方法2: 通过 PICO 商店侧载
# 使用 SideQuest 或 PICO 助手安装
```

#### 5.3.3 PICO 端配置

```
在 PICO 头显中:
1. 启动 PicoBridge 应用
2. 在设置页面输入 PC IP 地址 (如 192.168.1.100)
3. 选择数据端口 (默认: 5555)
4. 选择视频端口 (默认: 5556)
5. 启用:
   - [x] 手部追踪
   - [x] 头部位姿
   - [x] 视频推流
6. 点击 "连接"
7. 等待 PC 端确认连接
```

## 6. OpenNeck 环境搭建

### 6.1 项目概述

OpenNeck 是一个开源的头颈控制项目，用于控制人形机器人的头部运动（俯仰、偏航），常用于配合遥操作实现自然的注视方向。

### 6.2 环境搭建步骤

```bash
# ============================================================
# Step 1: 克隆项目
# ============================================================
cd ~/projects
git clone https://github.com/your-org/OpenNeck.git
cd OpenNeck

# ============================================================
# Step 2: 创建 conda 环境
# ============================================================
conda create -n openneck python=3.10 -y
conda activate openneck

# ============================================================
# Step 3: 安装依赖
# ============================================================
# 基础
pip install numpy scipy pyyaml

# 串口通信（舵机控制）
pip install pyserial

# PID 控制
pip install simple-pid

# 可视化（可选）
pip install matplotlib

# ============================================================
# Step 4: 安装项目
# ============================================================
pip install -r requirements.txt
pip install -e .

# ============================================================
# Step 5: 配置硬件
# ============================================================
# OpenNeck 通常使用舵机驱动头部运动
# 连接方式: USB-TTL → 舵机

# 查看串口设备
ls /dev/ttyUSB*
# 通常为 /dev/ttyUSB0

# 设置串口权限
sudo chmod 666 /dev/ttyUSB0
# 或将用户添加到 dialout 组
sudo usermod -aG dialout $USER

# ============================================================
# Step 6: 配置参数
# ============================================================
# 编辑配置文件
cat > configs/openneck.yaml << 'EOF'
serial:
  port: /dev/ttyUSB0
  baudrate: 115200

servo:
  pitch_id: 1        # 俯仰舵机 ID
  yaw_id: 2          # 偏航舵机 ID
  pitch_range: [-45, 45]   # 度
  yaw_range: [-90, 90]     # 度
  speed: 60          # 舵机速度 (0-100)

control:
  freq: 50           # 控制频率 Hz
  pid_pitch: [1.0, 0.01, 0.1]
  pid_yaw: [1.5, 0.01, 0.1]
```

## 9. 常见问题排查表

### 9.1 安装问题

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| `CUDA out of memory` | 显存不足 | 降低 batch_size，启用梯度检查点，使用 LoRA |
| `ModuleNotFoundError: No module named 'mujoco'` | MuJoCo 未安装 | `pip install mujoco` |
| `ImportError: libGL.so.1` | 缺少 OpenGL 库 | `sudo apt install libgl1-mesa-glx` |
| `ImportError: libgthread-2.0.so.0` | 缺少 glib | `sudo apt install libglib2.0-0` |
| `can: No such device` | CAN 接口未初始化 | `sudo ip link set can0 type can bitrate 1000000 && sudo ip link set up can0` |
| `Permission denied: /dev/ttyUSB0` | 串口权限不足 | `sudo chmod 666 /dev/ttyUSB0` 或 `sudo usermod -aG dialout $USER` |
| `unitree_sdk2py import error` | DDS 未正确安装 | 重新安装 CycloneDDS 并 `sudo ldconfig` |
| `huggingface download timeout` | 网络问题 | 设置 HF 镜像: `export HF_ENDPOINT=https://hf-mirror.com` |
| `pip install timeout` | PyPI 网络问题 | 使用清华镜像: `pip install -i https://pypi.tuna.tsinghua.edu.cn/simple` |
| `git lfs: command not found` | Git LFS 未安装 | `sudo apt install git-lfs && git lfs install` |

### 9.2 运行问题

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| `MuJoCo: Model not found` | MJCF 路径错误 | 检查模型文件路径是否正确 |
| `DDS: No data received` | 网络配置错误 | 检查 IP 配置和 CycloneDDS XML 配置 |
| `PICO: Connection refused` | PC 端未启动接收 | 先启动 PC 端接收程序 |
| `MediaPipe: No hand detected` | 光线不足或手部超出画面 | 改善光线，确保手部在摄像头视野内 |
| `CAN: Bus error` | CAN 总线波特率不匹配 | 检查波特率设置 (通常 1000000) |
| `Robot: Motor overheat` | 电机温度过高 | 停止运行，等待降温，检查负载 |
| `Sim: Slow FPS` | GPU 渲染负担重 | 降低渲染分辨率，关闭不必要的渲染选项 |
| `Training: Loss not decreasing` | 学习率不当 | 调整学习率 (1e-4 → 1e-5)，检查数据质量 |

### 9.6 一键环境检查脚本

```python
#!/usr/bin/env python3
"""具身智能项目环境检查脚本"""

import subprocess
import sys
from pathlib import Path

class EnvChecker:
    """环境检查器"""
    
    def __init__(self):
        self.passed = 0
        self.failed = 0
        self.warnings = 0
    
    def check(self, name, condition, detail=""):
        """检查项"""
        if condition:
            print(f"  [PASS] {name}")
            self.passed += 1
        else:
            print(f"  [FAIL] {name} - {detail}")
            self.failed += 1
    
    def warn(self, name, detail=""):
        """警告项"""
        print(f"  [WARN] {name} - {detail}")
        self.warnings += 1
    
    def check_system(self):
        """检查系统"""
        print("\n=== 系统检查 ===")
        
        # Python
        self.check(
            "Python 3.10+",
            sys.version_info >= (3, 10),
            f"当前: {sys.version}",
        )
        
        # CUDA
        try:
            import torch
            self.check(
                "CUDA available",
                torch.cuda.is_available(),
                "未检测到 CUDA",
            )
            if torch.cuda.is_available():
                gpu_name = torch.cuda.get_device_name(0)
                gpu_mem = torch.cuda.get_device_properties(0).total_mem / 1e9
                self.check(
                    f"GPU: {gpu_name} ({gpu_mem:.1f} GB)",
                    gpu_mem >= 8,
                    "显存不足 8GB",
                )
        except ImportError:
            self.check("PyTorch", False, "未安装")
    
    def check_packages(self):
        """检查关键包"""
        print("\n=== 包检查 ===")
        
        packages = [
            ('numpy', 'numpy'),
            ('scipy', 'scipy'),
            ('opencv', 'cv2'),
            ('torch', 'torch'),
            ('mujoco', 'mujoco'),
            ('h5py', 'h5py'),
            ('zmq', 'zmq'),
            ('mediapipe', 'mediapipe'),
            ('transformers', 'transformers'),
            ('lerobot', 'lerobot'),
            ('unitree_sdk2py', 'unitree_sdk2py'),
        ]
        
        for name, module in packages:
            try:
                __import__(module)
                self.check(name, True)
            except ImportError:
                self.check(name, False, "未安装")
    
    def check_hardware(self):
        """检查硬件接口"""
        print("\n=== 硬件检查 ===")
        
        # CAN 接口
        import os
        can_exists = os.path.exists('/dev/can0') or os.path.exists('/sys/class/net/can0')
        if can_exists:
            self.check("CAN 接口", True)
        else:
            self.warn("CAN 接口", "未检测到 CAN 接口（仿真模式不需要）")
        
        # 串口
        usb_devices = list(Path('/dev').glob('ttyUSB*'))
        if usb_devices:
            self.check(f"串口 ({usb_devices[0]})", True)
        else:
            self.warn("串口", "未检测到 USB 串口设备")
        
        # 摄像头
        video_devices = list(Path('/dev').glob('video*'))
        if video_devices:
            self.check(f"摄像头 ({len(video_devices)} 个)", True)
        else:
            self.warn("摄像头", "未检测到摄像头设备")
    
    def run(self):
        """运行所有检查"""
        print("=" * 50)
        print("具身智能项目环境检查")
        print("=" * 50)
        
        self.check_system()
        self.check_packages()
        self.check_hardware()
        
        print("\n" + "=" * 50)
        print(f"结果: {self.passed} 通过, {self.failed} 失败, {self.warnings} 警告")
        print("=" * 50)
        
        return self.failed == 0


if __name__ == "__main__":
    checker = EnvChecker()
    success = checker.run()
    sys.exit(0 if success else 1)
```
