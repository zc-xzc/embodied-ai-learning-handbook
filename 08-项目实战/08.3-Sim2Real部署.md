```python
import subprocess
import time

    """Sim2Real 前置检查器"""

    def check_all(self):

            time.sleep(2)
            
            self._record("G1 DDS 连接", received[0])
            
            if received[0]:
                # 检查电池电量
                # 检查关节温度
                pass
        
        except Exception as e:
            self._record("G1 DDS 连接", False, str(e))
    
    def _check_network(self):
        """检查网络"""
        print("\n[Phase 3] 网络检查")
        
        # 检查网络延迟
        try:
            result = subprocess.run(
                ['ping', '-c', '5', '192.168.123.161'],
                capture_output=True, text=True, timeout=10,
            )
            if result.returncode == 0:
                # 解析延迟
                lines = result.stdout.split('\n')
                for line in lines:
                    if 'avg' in line:
                        parts = line.split('=')
                        if len(parts) >= 2:
                            avg_ms = float(parts[1].split('/')[0].strip())
                            self._record("网络延迟", avg_ms < 5, f"{avg_ms}ms")
            else:
                self._record("网络延迟", False, "Ping 失败")
        except Exception as e:
            self._record("网络延迟", False, str(e))
    
    def _check_software(self):
        """检查软件"""
        print("\n[Phase 4] 软件检查")
        
        checks = [
            ("unitree_sdk2py", "import unitree_sdk2py"),
            ("torch", "import torch"),
            ("numpy", "import numpy"),
            ("zmq", "import zmq"),
        ]
        
        for name, cmd in checks:
            try:
                exec(cmd)
                self._record(f"{name} 已安装", True)
            except ImportError:
                self._record(f"{name} 已安装", False)
    
    def _check_safety(self):
        """检查安全"""
        print("\n[Phase 5] 安全检查")
        
        # 这些需要人工确认
        print("  [需人工确认] 急停按钮可用")
        print("  [需人工确认] 保护垫已铺设")
        print("  [需人工确认] 安全区域已清理")
        print("  [需人工确认] 至少 2 人在场")
    
    def _record(self, name, passed, detail=""):
        if passed:
            print(f"  [PASS] {name}")
            self.results[name] = True
        else:
            print(f"  [FAIL] {name} - {detail}")
            self.results[name] = False
    
    def _print_summary(self):
        print("\n" + "=" * 60)
        passed = sum(self.results.values())
        total = len(self.results)
        print(f"结果: {passed}/{total} 通过")
        if passed == total:
            print("所有自动检查通过！请完成人工确认后开始部署。")
        else:
            print("存在未通过的检查项，请修复后再部署。")
        print("=" * 60)


if __name__ == "__main__":
    checker = Sim2RealChecker()
    checker.check_all()
```

### 2.1 G1 Bridge 概述

G1 Bridge 是连接高层策略与 G1 低层控制器的中间件，提供：
- 统一的关节控制接口
- 状态读取（位置、速度、力矩）
- 安全限位检查
- 紧急停止功能

### 2.3 G1 Bridge 服务端

```python
#!/usr/bin/env python3
"""G1 Bridge 服务端"""

import zmq
import json
import numpy as np
import threading
import time
from unitree_sdk2py.core.channel.channel_publisher import ChannelPublisher
from unitree_sdk2py.core.channel.channel_subscriber import ChannelSubscriber
from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowCmd_, unitree_hg_msg_dds__LowState_
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_
from unitree_sdk2py.utils.crc import CRC

class G1BridgeServer:
    """G1 Bridge 服务端"""
    
    NUM_JOINTS = 29
    
    def __init__(self, port=5555):
        self.crc = CRC()
        self.port = port
        self.running = False
        self.estop = False
        
        # DDS 通信
        self.low_cmd_pub = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.low_cmd_pub.Init()
        
        self.low_state_sub = ChannelSubscriber("rt/lowstate", LowState_)
        self.low_state_sub.Init()
        
        self.latest_state = None
        self.low_state_sub.RegisterCallback(self._state_callback)
        
        # ZMQ 服务
        self.context = zmq.Context()
        self.socket = self.context.socket(zmq.REP)
        self.socket.bind(f"tcp://*:{port}")
        
        # 当前命令
        self.target_positions = np.zeros(self.NUM_JOINTS)
        self.kp = 100.0
        self.kd = 3.0
        
        print(f"G1 Bridge 服务端启动 (端口: {port})")
    
    def _state_callback(self, msg: LowState_):
        self.latest_state = msg
    
    def start(self):
        """启动服务"""
        self.running = True
        
        # 启动控制循环线程
        control_thread = threading.Thread(target=self._control_loop)
        control_thread.daemon = True
        control_thread.start()
        
        # ZMQ 请求处理循环
        print("等待客户端连接...")
        
        while self.running:
            try:
                request = self.socket.recv_string()
                request = json.loads(request)
                response = self._handle_request(request)
                self.socket.send_string(json.dumps(response))
            except KeyboardInterrupt:
                break
            except Exception as e:
                print(f"错误: {e}")
        
        self.stop()
    
    def _handle_request(self, request):
        """处理客户端请求"""
        action = request.get('action')
        
        if action == 'get_state':
            return self._get_state_response()
        
        elif action == 'send_action':
            positions = request.get('target_positions', [])
            if len(positions) == self.NUM_JOINTS:
                self.target_positions = np.array(positions)
                return {'status': 'ok'}
            return {'status': 'error', 'message': '维度不匹配'}
        
        elif action == 'get_config':
            return {
                'num_joints': self.NUM_JOINTS,
                'joint_names': G1Controller.JOINT_NAMES,
            }
        
        elif action == 'enable':
            self.estop = False
            return {'status': 'ok'}
        
        elif action == 'disable':
            self.estop = True
            return {'status': 'ok'}
        
        elif action == 'estop':
            self.estop = True
            self.target_positions = self._get_current_positions()
            return {'status': 'estop'}
        
        return {'status': 'unknown_action'}
    
    def _get_state_response(self):
        """获取状态响应"""
        if self.latest_state is None:
            return {'status': 'no_data'}
        
        positions = []
        velocities = []
        torques = []
        
        for i in range(self.NUM_JOINTS):
            motor = self.latest_state.motor[i]
            positions.append(motor.q)
            velocities.append(motor.dq)
            torques.append(motor.torque)
        
        return {
            'status': 'ok',
            'positions': positions,
            'velocities': velocities,
            'torques': torques,
        }
    
    def _get_current_positions(self):
        """获取当前关节位置"""
        if self.latest_state is None:
            return np.zeros(self.NUM_JOINTS)
        
        return np.array([
            self.latest_state.motor[i].q 
            for i in range(self.NUM_JOINTS)
        ])
    
    def _control_loop(self):
        """控制循环（500Hz）"""
        freq = 500
        dt = 1.0 / freq
        
        while self.running:
            if self.estop:
                # 紧急停止：保持当前位置，高阻尼
                positions = self._get_current_positions()
                self._send_command(positions, kp=200, kd=10)
            else:
                self._send_command(self.target_positions, kp=self.kp, kd=self.kd)
            
            time.sleep(dt)
    
    def _send_command(self, positions, kp, kd):
        """发送低层命令"""
        cmd = unitree_hg_msg_dds__LowCmd_()
        cmd.head[0] = 0xFE
        cmd.head[1] = 0xEF
        
        for i in range(self.NUM_JOINTS):
            cmd.motor[i].mode = 1
            cmd.motor[i].q = float(positions[i])
            cmd.motor[i].dq = 0.0
            cmd.motor[i].tau = 0.0
            cmd.motor[i].kp = float(kp)
            cmd.motor[i].kd = float(kd)
        
        cmd.crc = self.crc.Crc(cmd)
        self.low_cmd_pub.Write(cmd)
    
    def stop(self):
        """停止"""
        self.running = False
        # 安全停止
        positions = self._get_current_positions()
        self._send_command(positions, kp=200, kd=10)
        time.sleep(0.5)
        self.socket.close()
        self.context.term()
        print("G1 Bridge 服务端已停止")


if __name__ == "__main__":
    server = G1BridgeServer(port=5555)
    server.start()
```

## 4. 安全模式设置

### 4.1 安全模式定义

```python
#!/usr/bin/env python3
"""安全模式管理器"""

from enum import Enum
import time
import numpy as np

class SafetyMode(Enum):
    """安全模式"""
    NORMAL = 0        # 正常运行
    DEGRADED = 1      # 降级运行（降低速度/力矩）
    PROTECTIVE_STOP = 2  # 保护性停止（保持当前位置）
    EMERGENCY_STOP = 3   # 紧急停止（立即停止）
    RECOVERY = 4      # 恢复模式

class SafetyManager:
    """安全管理器"""
    
    # 安全阈值
    THRESHOLDS = {
        'max_joint_velocity': 5.0,      # rad/s
        'max_joint_torque': 40.0,       # Nm
        'max_motor_temp': 70.0,         # °C
        'max_tracking_error': 0.5,      # rad
        'max_comm_latency': 0.01,       # s (10ms)
        'min_battery': 20,              # %
    }
    
    def __init__(self):
        self.mode = SafetyMode.NORMAL
        self.violations = []
        self.last_state = None
        self.last_command = None
        self.last_update = time.time()
    
    def check_state(self, state):
        """
        检查机器人状态安全性
        
        Args:
            state: dict with positions, velocities, torques, temperatures
        """
        self.last_state = state
        self.last_update = time.time()
        
        violations = []
        
        # 检查速度
        if 'velocities' in state:
            max_vel = np.max(np.abs(state['velocities']))
            if max_vel > self.THRESHOLDS['max_joint_velocity']:
                violations.append(
                    f"速度超限: {max_vel:.2f} > {self.THRESHOLDS['max_joint_velocity']}"
                )
        
        # 检查力矩
        if 'torques' in state:
            max_torque = np.max(np.abs(state['torques']))
            if max_torque > self.THRESHOLDS['max_joint_torque']:
                violations.append(
                    f"力矩超限: {max_torque:.2f} > {self.THRESHOLDS['max_joint_torque']}"
                )
        
        # 检查温度
        if 'temperatures' in state:
            max_temp = max(state['temperatures'])
            if max_temp > self.THRESHOLDS['max_motor_temp']:
                violations.append(
                    f"温度超限: {max_temp:.1f}°C > {self.THRESHOLDS['max_motor_temp']}°C"
                )
        
        # 根据违规情况调整安全模式
        if violations:
            self.violations.extend(violations)
            
            # 严重违规 → 紧急停止
            if any('温度' in v or '力矩' in v for v in violations):
                self.set_mode(SafetyMode.EMERGENCY_STOP)
            else:
                self.set_mode(SafetyMode.DEGRADED)
        
        return violations
    
    def check_command(self, command, current_state):
        """检查命令安全性"""
        if current_state is None:
            return False, "无状态数据"
        
        # 检查跟踪误差
        if 'positions' in current_state:
            error = np.abs(command - current_state['positions'])
            max_error = np.max(error)
            if max_error > self.THRESHOLDS['max_tracking_error']:
                self.set_mode(SafetyMode.PROTECTIVE_STOP)
                return False, f"跟踪误差过大: {max_error:.3f}"
        
        # 检查通信延迟
        latency = time.time() - self.last_update
        if latency > self.THRESHOLDS['max_comm_latency']:
            self.set_mode(SafetyMode.PROTECTIVE_STOP)
            return False, f"通信延迟过大: {latency*1000:.1f}ms"
        
        return True, "OK"
    
    def set_mode(self, mode):
        """设置安全模式"""
        if mode != self.mode:
            print(f"安全模式变更: {self.mode.name} → {mode.name}")
            self.mode = mode
    
    def get_safe_command(self, target, current_state):
        """根据安全模式获取安全命令"""
        if self.mode == SafetyMode.EMERGENCY_STOP:
            # 保持当前位置，高阻尼
            return current_state['positions'], 200, 10
        
        elif self.mode == SafetyMode.PROTECTIVE_STOP:
            # 保持当前位置
            return current_state['positions'], 150, 5
        
        elif self.mode == SafetyMode.DEGRADED:
            # 降级：降低速度（缩小目标与当前的差距）
            diff = target - current_state['positions']
            limited_target = current_state['positions'] + diff * 0.3
            return limited_target, 80, 3
        
        else:
            # 正常模式
            return target, 100, 3
    
    def is_safe(self):
        """是否安全运行"""
        return self.mode in [SafetyMode.NORMAL, SafetyMode.DEGRADED]
    
    def reset(self):
        """重置"""
        self.mode = SafetyMode.NORMAL
        self.violations = []
```

```python
class VelocityLimiter:
    """速度限制器"""
    
    def __init__(self, max_velocity=2.0, max_acceleration=10.0, freq=500):
        self.max_vel = max_velocity
        self.max_acc = max_acceleration
        self.dt = 1.0 / freq
        self.prev_target = None
    
    def limit(self, target, current):
        """
        限制目标位置的变化率
        
        Args:
            target: 目标位置 [num_joints]
            current: 当前位置 [num_joints]
        
        Returns:
            limited_target: 限制后的目标
        """
        if self.prev_target is None:
            self.prev_target = current.copy()
        
        # 计算速度
        velocity = (target - self.prev_target) / self.dt
        
        # 限制速度
        vel_norm = np.abs(velocity)
        scale = np.minimum(1.0, self.max_vel / (vel_norm + 1e-8))
        velocity_limited = velocity * scale
        
        # 限制加速度
        if hasattr(self, 'prev_velocity'):
            acc = (velocity_limited - self.prev_velocity) / self.dt
            acc_norm = np.abs(acc)
            acc_scale = np.minimum(1.0, self.max_acc / (acc_norm + 1e-8))
            velocity_limited = self.prev_velocity + acc * acc_scale * self.dt
        
        self.prev_velocity = velocity_limited
        self.prev_target = self.prev_target + velocity_limited * self.dt
        
        return self.prev_target.copy()
```

```python
#!/usr/bin/env python3
"""G1 低层控制器"""

import numpy as np
import time
import threading
from unitree_sdk2py.core.channel.channel_publisher import ChannelPublisher
from unitree_sdk2py.core.channel.channel_subscriber import ChannelSubscriber
from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowCmd_, unitree_hg_msg_dds__LowState_
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_
from unitree_sdk2py.utils.crc import CRC

class G1LowLevelController:
    """G1 低层控制器"""
    
    NUM_JOINTS = 29
    CONTROL_FREQ = 500  # Hz
    
    def __init__(self):
        self.crc = CRC()
        
        # DDS
        self.cmd_pub = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.cmd_pub.Init()
        
        self.state_sub = ChannelSubscriber("rt/lowstate", LowState_)
        self.state_sub.Init()
        
        self.latest_state = None
        self.state_sub.RegisterCallback(self._state_callback)
        
        # 控制参数
        self.target_positions = np.zeros(self.NUM_JOINTS)
        self.kp = np.full(self.NUM_JOINTS, 100.0)
        self.kd = np.full(self.NUM_JOINTS, 3.0)
        
        # 安全
        self.safety = SafetyManager()
        self.velocity_limiter = VelocityLimiter(
            max_velocity=3.0, max_acceleration=15.0, freq=self.CONTROL_FREQ
        )
        
        # 运行状态
        self.running = False
        self.estop = False
        
        # 等待状态
        print("等待 G1 状态...")
        for _ in range(50):
            if self.latest_state is not None:
                break
            time.sleep(0.1)
        
        if self.latest_state:
            self.target_positions = self.get_joint_positions()
            print("G1 低层控制器已就绪")
        else:
            print("警告: 未收到 G1 状态")
    
    def _state_callback(self, msg):
        self.latest_state = msg
    
    def get_joint_positions(self):
        """获取关节位置"""
        if self.latest_state is None:
            return np.zeros(self.NUM_JOINTS)
        return np.array([self.latest_state.motor[i].q for i in range(self.NUM_JOINTS)])
    
    def get_joint_velocities(self):
        """获取关节速度"""
        if self.latest_state is None:
            return np.zeros(self.NUM_JOINTS)
        return np.array([self.latest_state.motor[i].dq for i in range(self.NUM_JOINTS)])
    
    def get_joint_torques(self):
        """获取关节力矩"""
        if self.latest_state is None:
            return np.zeros(self.NUM_JOINTS)
        return np.array([self.latest_state.motor[i].torque for i in range(self.NUM_JOINTS)])
    
    def set_target(self, positions):
        """设置目标位置"""
        self.target_positions = np.array(positions)
    
    def start(self):
        """启动控制循环"""
        self.running = True
        self.estop = False
        
        self.control_thread = threading.Thread(target=self._control_loop)
        self.control_thread.daemon = True
        self.control_thread.start()
        
        print(f"低层控制循环已启动 ({self.CONTROL_FREQ} Hz)")
    
    def stop(self):
        """停止"""
        self.running = False
        if hasattr(self, 'control_thread'):
            self.control_thread.join(timeout=2)
        print("低层控制循环已停止")
    
    def emergency_stop(self):
        """紧急停止"""
        self.estop = True
        self.target_positions = self.get_joint_positions()
        print("紧急停止已触发！")
    
    def _control_loop(self):
        """控制循环"""
        dt = 1.0 / self.CONTROL_FREQ
        
        while self.running:
            if self.latest_state is None:
                time.sleep(dt)
                continue
            
            # 获取当前状态
            current_pos = self.get_joint_positions()
            current_vel = self.get_joint_velocities()
            current_torque = self.get_joint_torques()
            
            # 安全检查
            state = {
                'positions': current_pos,
                'velocities': current_vel,
                'torques': current_torque,
            }
            self.safety.check_state(state)
            
            # 获取安全命令
            safe_target, kp, kd = self.safety.get_safe_command(
                self.target_positions, state
            )
            
            # 速度限制
            if not self.estop:
                safe_target = self.velocity_limiter.limit(safe_target, current_pos)
            
            # 发送命令
            if self.estop:
                self._send_command(current_pos, kp=200, kd=10)
            else:
                self._send_command(safe_target, kp=kp, kd=kd)
            
            time.sleep(dt)
    
    def _send_command(self, positions, kp=100, kd=3):
        """发送 DDS 命令"""
        cmd = unitree_hg_msg_dds__LowCmd_()
        cmd.head[0] = 0xFE
        cmd.head[1] = 0xEF
        
        for i in range(self.NUM_JOINTS):
            cmd.motor[i].mode = 1
            cmd.motor[i].q = float(positions[i])
            cmd.motor[i].dq = 0.0
            cmd.motor[i].tau = 0.0
            cmd.motor[i].kp = float(kp if isinstance(kp, (int, float)) else kp[i])
            cmd.motor[i].kd = float(kd if isinstance(kd, (int, float)) else kd[i])
        
        cmd.crc = self.crc.Crc(cmd)
        self.cmd_pub.Write(cmd)
```

```python
        平滑移动到目标姿态
        
        Args:
```

```python
#!/usr/bin/env python3
"""实时监控面板"""

import time
import numpy as np
from collections import deque

class RealtimeMonitor:
    """实时监控器"""
    
    def __init__(self, controller, history_length=1000):
        self.controller = controller
        
        # 历史数据
        self.history_length = history_length
        self.position_history = deque(maxlen=history_length)
        self.velocity_history = deque(maxlen=history_length)
        self.torque_history = deque(maxlen=history_length)
        self.target_history = deque(maxlen=history_length)
        self.error_history = deque(maxlen=history_length)
        
        # 统计
        self.start_time = time.time()
        self.frame_count = 0
        
        # 告警
        self.alerts = []
    
    def update(self):
        """更新监控数据"""
        pos = self.controller.get_joint_positions()
        vel = self.controller.get_joint_velocities()
        torque = self.controller.get_joint_torques()
        target = self.controller.target_positions
        
        error = np.abs(target - pos)
        
        self.position_history.append(pos.copy())
        self.velocity_history.append(vel.copy())
        self.torque_history.append(torque.copy())
        self.target_history.append(target.copy())
        self.error_history.append(error.copy())
        
        self.frame_count += 1
        
        # 检查异常
        self._check_anomalies(pos, vel, torque, error)
    
    def _check_anomalies(self, pos, vel, torque, error):
        """检查异常"""
        # 大跟踪误差
        max_error = np.max(error)
        if max_error > 0.3:
            self._add_alert('WARNING', f'大跟踪误差: {max_error:.3f} rad')
        
        # 高速度
        max_vel = np.max(np.abs(vel))
        if max_vel > 4.0:
            self._add_alert('WARNING', f'高速度: {max_vel:.2f} rad/s')
        
        # 高力矩
        max_torque = np.max(np.abs(torque))
        if max_torque > 30.0:
            self._add_alert('ERROR', f'高力矩: {max_torque:.1f} Nm')
    
    def _add_alert(self, level, message):
        """添加告警"""
        alert = {
            'time': time.time() - self.start_time,
            'level': level,
            'message': message,
        }
        self.alerts.append(alert)
        if level == 'ERROR':
            print(f"[{level}] {message}")
    
    def print_status(self):
        """打印状态"""
        if not self.position_history:
            return
        
        pos = self.position_history[-1]
        vel = self.velocity_history[-1]
        torque = self.torque_history[-1]
        error = self.error_history[-1]
        
        elapsed = time.time() - self.start_time
        fps = self.frame_count / elapsed if elapsed > 0 else 0
        
        print(f"\r[{elapsed:.1f}s] FPS:{fps:.0f} | "
              f"MaxErr:{np.max(error):.3f} | "
              f"MaxVel:{np.max(np.abs(vel)):.2f} | "
              f"MaxTau:{np.max(np.abs(torque)):.1f} | "
              f"Alerts:{len(self.alerts)}", end='', flush=True)
    
    def get_summary(self):
        """获取摘要"""
        if not self.position_history:
            return {}
        
        return {
            'duration': time.time() - self.start_time,
            'frames': self.frame_count,
            'avg_error': np.mean(self.error_history),
            'max_error': np.max(self.error_history),
            'max_velocity': np.max(np.abs(self.velocity_history)),
            'max_torque': np.max(np.abs(self.torque_history)),
            'num_alerts': len(self.alerts),
            'num_errors': sum(1 for a in self.alerts if a['level'] == 'ERROR'),
```

```python
#!/usr/bin/env python3
"""紧急停止系统"""

import threading

class EmergencyStopSystem:
    """紧急停止系统"""
```

```python
#!/usr/bin/env python3
"""TWIST2 Sim2Real 部署"""

import numpy as np
import time
import threading

class TWIST2Sim2Real:
    """TWIST2 Sim2Real 部署器"""
    
    def __init__(self, robot_ip='192.168.123.161'):
        # 初始化双臂控制器
        self.left_arm = G1LowLevelController()  # 或其他臂控制器
        # self.right_arm = ...  # 右臂控制器
        
        # VR 输入
        self.vr_receiver = None  # VR 数据接收器
        
        # 策略模型
        self.policy = None  # 策略推理器
        
        # 安全
        self.safety = SafetyManager()
        self.estop = False
        
        # 运行状态
        self.mode = 'teleop'  # 'teleop' or 'policy'
        self.running = False
    
    def setup_teleop_mode(self, vr_port=5555):
        """设置遥操作模式"""
        print("设置遥操作模式...")
        
        # 连接 VR
        import zmq
        self.vr_context = zmq.Context()
        self.vr_socket = self.vr_context.socket(zmq.SUB)
        self.vr_socket.connect(f"tcp://localhost:{vr_port}")
        self.vr_socket.subscribe(b'')
        
        print(f"VR 接收器已连接 (端口: {vr_port})")
        
        self.mode = 'teleop'
    
    def setup_policy_mode(self, policy_path):
        """设置策略推理模式"""
        print("设置策略推理模式...")
        
        # 加载策略
        import torch
        self.policy = torch.load(policy_path)
        
        print(f"策略模型已加载: {policy_path}")
        
        self.mode = 'policy'
    
    def run(self, max_steps=10000):
        """运行"""
        self.running = True
        self.estop = False
        
        # 启动低层控制器
        self.left_arm.start()
        time.sleep(2)
        
        print(f"开始 TWIST2 Sim2Real (模式: {self.mode})")
        
        try:
            for step in range(max_steps):
                if self.estop:
                    break
                
                if self.mode == 'teleop':
                    self._teleop_step()
                else:
                    self._policy_step()
                
                time.sleep(1.0 / 30)
                
                if step % 100 == 0:
                    print(f"  Step {step}/{max_steps}")
        
        except KeyboardInterrupt:
            print("\n用户中断")
        
        finally:
            self.shutdown()
    
    def _teleop_step(self):
        """遥操作步骤"""
        # 接收 VR 数据
        try:
            data = self.vr_socket.recv_json(flags=zmq.NOBLOCK)
            
            # 解析双臂目标
            left_target = np.array(data['left_arm'])
            right_target = np.array(data['right_arm'])
            
            # 安全检查
            current = self.left_arm.get_joint_positions()
            state = {
                'positions': current,
                'velocities': self.left_arm.get_joint_velocities(),
                'torques': self.left_arm.get_joint_torques(),
            }
            self.safety.check_state(state)
            
            # 发送命令
            safe_target, kp, kd = self.safety.get_safe_command(left_target, state)
            self.left_arm.set_target(safe_target)
            
        except:
            pass
    
    def _policy_step(self):
        """策略推理步骤"""
        # 获取状态
        state = self.left_arm.get_joint_positions()
        
        # 策略推理
        action = self.policy.get_action(state)
        
        # 发送命令
        self.left_arm.set_target(action)
    
    def emergency_stop(self):
        """紧急停止"""
        self.estop = True
        self.left_arm.emergency_stop()
        print("TWIST2 紧急停止！")
    
    def shutdown(self):
        """关闭"""
        self.running = False
        self.left_arm.stop()
        print("TWIST2 已关闭")
```

## 9. 常见失败模式与排查

### 9.1 失败模式表

| 失败模式 | 症状 | 可能原因 | 解决方案 |
|----------|------|----------|----------|
| **通信中断** | 机器人突然不动 | DDS 连接断开 | 检查网络、重启 DDS |
| **关节抖动** | 关节快速振荡 | PD 增益过高 | 降低 kp，增加 kd |
| **跟踪滞后** | 动作跟不上目标 | 速度限制过低 | 增大速度限制 |
| **关节过冲** | 超过目标位置 | 阻尼不足 | 增加 kd |
| **力矩超限** | 电机报错 | 负载过大或卡住 | 减小负载、检查障碍 |
| **温度过高** | 关节发烫 | 长时间高负载 | 降低频率、休息冷却 |
| **摔倒** | 机器人失去平衡 | 策略失效或外力 | 立即急停、检查策略 |
| **碰撞** | 机器人撞到物体 | 感知不足或规划错误 | 增加碰撞检测、急停 |
| **动作发散** | 动作越来越大 | 策略不稳定 | 停止、检查模型 |
| **相机丢失** | 图像卡顿或黑屏 | USB 带宽不足或驱动问题 | 检查 USB 连接、降低分辨率 |

```python
# 领域随机化示例

class DomainRandomizer:
    """仿真领域随机化"""
    
    def __init__(self, model):
        self.model = model
    
    def randomize(self):
        """随机化物理参数"""
        # 随机化摩擦
        for i in range(self.model.geom_friction.shape[0]):
            self.model.geom_friction[i] = np.random.uniform(0.3, 1.5, 3)
        
        # 随机化质量
        for i in range(self.model.body_mass.shape[0]):
            original_mass = self.model.body_mass[i]
            self.model.body_mass[i] = original_mass * np.random.uniform(0.8, 1.2)
        
        # 随机化阻尼
        for i in range(self.model.dof_damping.shape[0]):
            self.model.dof_damping[i] *= np.random.uniform(0.5, 2.0)
        
        # 随机化控制延迟
        self.control_delay = np.random.uniform(0, 0.02)  # 0-20ms
```

### 9.4 日志记录与分析

```python
#!/usr/bin/env python3
"""Sim2Real 日志记录器"""

import json
import time
import numpy as np
from pathlib import Path

class Sim2RealLogger:
    """Sim2Real 日志记录器"""
    
    def __init__(self, log_dir='./logs'):
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(parents=True, exist_ok=True)
        
        self.session_id = time.strftime('%Y%m%d_%H%M%S')
        self.log_file = self.log_dir / f"sim2real_{self.session_id}.jsonl"
        
        self.start_time = time.time()
        self.frame_count = 0
        
        print(f"日志记录到: {self.log_file}")
    
    def log_frame(self, state, action, target, metadata=None):
        """记录一帧"""
        entry = {
            'timestamp': time.time() - self.start_time,
            'frame': self.frame_count,
            'state': state.tolist() if isinstance(state, np.ndarray) else state,
            'action': action.tolist() if isinstance(action, np.ndarray) else action,
            'target': target.tolist() if isinstance(target, np.ndarray) else target,
            'tracking_error': float(np.max(np.abs(target - state))),
        }
        
        if metadata:
            entry['metadata'] = metadata
        
        with open(self.log_file, 'a') as f:
            f.write(json.dumps(entry) + '\n')
        
        self.frame_count += 1
    
    def log_event(self, event_type, message, data=None):
        """记录事件"""
        entry = {
            'timestamp': time.time() - self.start_time,
            'event': event_type,
            'message': message,
        }
        
        if data:
            entry['data'] = data
        
        with open(self.log_file, 'a') as f:
            f.write(json.dumps(entry) + '\n')
        
        print(f"[{event_type}] {message}")
    
    def analyze(self):
        """分析日志"""
        entries = []
        with open(self.log_file, 'r') as f:
            for line in f:
                entries.append(json.loads(line))
        
        # 统计
        frames = [e for e in entries if 'frame' in e]
        events = [e for e in entries if 'event' in e]
        
        if frames:
            errors = [f['tracking_error'] for f in frames]
            print(f"\n日志分析 ({self.session_id})")
            print(f"  总帧数: {len(frames)}")
            print(f"  总事件: {len(events)}")
            print(f"  平均跟踪误差: {np.mean(errors):.4f}")
            print(f"  最大跟踪误差: {np.max(errors):.4f}")
            print(f"  持续时间: {frames[-1]['timestamp']:.1f}s")
        
        return entries
```
