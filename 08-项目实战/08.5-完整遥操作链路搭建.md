| 设备 | IP 地址 | 网段 | 接口 | 用途 |

### 3.4 网络配置命令

```bash
# ============================================================
# 策略 PC 网络配置（双网卡）
# ============================================================

# eth0: 机器人网
sudo ifconfig eth0 192.168.123.222 netmask 255.255.255.0

# wlan0: VR 网
sudo ifconfig wlan0 192.168.1.100 netmask 255.255.255.0

# 路由配置
sudo route add -net 192.168.123.0/24 dev eth0
sudo route add -net 192.168.1.0/24 dev wlan0

# DNS
echo "nameserver 8.8.8.8" | sudo tee -a /etc/resolv.conf

# 防火墙
sudo ufw allow from 192.168.123.0/24
sudo ufw allow from 192.168.1.0/24
sudo ufw allow 5555:5560/tcp
sudo ufw allow 5557/udp

# DDS 配置
export CYCLONEDDS_URI=file:///path/to/cyclonedds_g1.xml

# 验证
ping 192.168.123.161  # G1 主控
ping 192.168.1.1      # 路由器
```

### 4.6 Step 5: pico-bridge VR 链路搭建

```bash
# ============================================================
# Step 5: pico-bridge VR 链路搭建
# ============================================================

echo "=== Step 5: pico-bridge VR 链路搭建 ==="

conda activate pico_bridge

# 5.1 启动 VR 数据接收端
echo "[5.1] 启动 VR 数据接收端"
python scripts/receiver.py --port 5556 &
VR_PID=$!
sleep 2

# 5.2 PICO 端操作
echo "[5.2] PICO 端操作"
echo "  1. 在 PICO 4 上启动 PicoBridge 应用"
echo "  2. 输入 PC IP: $(hostname -I | awk '{print $1}')"
echo "  3. 端口: 5556"
echo "  4. 启用手部追踪"
echo "  5. 点击连接"
echo ""
echo "  等待 PICO 连接..."
echo "  PICO 已连接？(y/n)"
read -r confirm

if [ "$confirm" = "y" ]; then
    # 5.3 验证数据
    echo "[5.3] 验证 VR 数据"
    python -c "
    import zmq
    import json
    import time
    
    ctx = zmq.Context()
    sock = ctx.socket(zmq.SUB)
    sock.connect('tcp://localhost:5556')
    sock.subscribe(b'')
    
    received = False
    start = time.time()
    while time.time() - start < 5:
        try:
            data = sock.recv_string(flags=zmq.NOBLOCK)
            msg = json.loads(data)
            print(f'  收到 VR 数据: {list(msg.keys())}')
            received = True
            break
        except:
            time.sleep(0.1)
    
    if received:
        print('  VR 数据接收正常！')
    else:
        print('  错误: 未收到 VR 数据')
    "
fi

echo ""
echo "VR 链路搭建完成？(y/n)"
read -r confirm
[ "$confirm" = "y" ] || exit 1
```

#!/usr/bin/env python3
"""延迟优化工具"""

import time
import numpy as np
from collections import deque

class LatencyOptimizer:
    """延迟优化器"""
    
    def __init__(self):
        self.latency_records = {}
    
    def measure_latency(self, name, func, *args, **kwargs):
        """测量函数延迟"""
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = (time.perf_counter() - start) * 1000  # ms
        
        if name not in self.latency_records:
            self.latency_records[name] = deque(maxlen=1000)
        self.latency_records[name].append(elapsed)
        
        return result
    
    def get_stats(self):
        """获取延迟统计"""
        stats = {}
        for name, records in self.latency_records.items():
            if records:
                stats[name] = {
                    'mean': np.mean(records),
                    'std': np.std(records),
                    'min': np.min(records),
                    'max': np.max(records),
                    'p99': np.percentile(records, 99),
                    'count': len(records),
                }
        return stats
    
    def print_report(self):
        """打印报告"""
        stats = self.get_stats()
        
        print("\n延迟报告")
        print("-" * 60)
        print(f"{'组件':<20} {'均值(ms)':<10} {'P99(ms)':<10} {'最大(ms)':<10}")
        print("-" * 60)
        
        for name, s in stats.items():
            print(f"{name:<20} {s['mean']:<10.2f} {s['p99']:<10.2f} {s['max']:<10.2f}")
        
        total = sum(s['mean'] for s in stats.values())
        print("-" * 60)
        print(f"{'总延迟':<20} {total:<10.2f}")

## 7. 故障排除手册

### 7.1 故障分类

```
故障级别:
  L1 - 致命: 系统无法运行，需立即处理
  L2 - 严重: 功能受损，需尽快处理
  L3 - 一般: 性能下降，可继续运行
  L4 - 轻微: 体验问题，可延后处理
```

### 7.2 故障排查表

| 故障 | 级别 | 症状 | 原因 | 解决方案 |
|------|------|------|------|----------|
| G1 无响应 | L1 | DDS 无数据 | 网络断开 | 检查网线、IP 配置 |
| G1 急停 | L1 | 机器人不动 | 急停按下 | 释放急停按钮 |
| G1 通信中断 | L1 | DDS 突然断开 | 网络抖动 | 重启 DDS、检查网络 |
| VR 无数据 | L2 | 无手部追踪 | WiFi 断开 | 检查 WiFi、重启 PICO |
| VR 延迟高 | L3 | 手部追踪卡顿 | WiFi 干扰 | 切换信道、靠近路由器 |
| 灵巧手不动 | L2 | CAN 无响应 | CAN 断开 | 检查 CAN 接线 |
| 灵巧手抖动 | L3 | 手指振荡 | PID 不当 | 调整 kp/kd |
| 机器人摔倒 | L1 | 失去平衡 | 策略失败 | 立即急停 |
| 关节过热 | L2 | 温度 > 70°C | 过载 | 停止、冷却 |
| 数据丢失 | L3 | 录制缺帧 | I/O 瓶颈 | 异步录制 |
| 延迟过高 | L3 | 动作滞后 | 多种原因 | 逐环节排查 |

### 7.3 故障排除流程

```
发现故障
    │
    ▼
[1] 立即急停 (如果是 L1)
    │
    ▼
[2] 记录故障现象
    ├── 时间

    │
    ▼
[4] 逐环节排查
    ├── VR → 重定向 → G1 → 灵巧手
    └── 在每个环节测量延迟和正确性
    │
    ▼

    └── 修复后重新运行集成测试
```

                '3. 等待温度降至 50°C 以下',
                '4. 检查负载是否过大',
                '5. 降低控制频率或力矩限制',
                '6. 重新开始',
            ],
        },
        'vr_loss': {
            'name': 'VR 连接丢失',
            'level': 'L2',
            'steps': [
                '1. 机器人自动保持当前姿态',
                '2. 检查 PICO 电量和 WiFi',
                '3. 重启 PICO 应用',
                '4. 重新连接',
                '5. 验证 VR 数据恢复',
            ],
        },
    }
    
    @classmethod
    def get_plan(cls, scenario):
        """获取应急预案"""
        plan = cls.PLANS.get(scenario)
        if plan:
            print(f"\n应急预案: {plan['name']} (级别: {plan['level']})")
            print("-" * 40)
            for step in plan['steps']:
                print(f"  {step}")
        else:
            print(f"未知场景: {scenario}")
    
    @classmethod
    def list_scenarios(cls):
        """列出所有场景"""
        print("可用应急预案:")
        for key, plan in cls.PLANS.items():
            print(f"  {key}: {plan['name']} ({plan['level']})")


if __name__ == "__main__":
    EmergencyPlan.list_scenarios()
    
    print("\n输入场景名查看预案 (或按 Enter 退出):")
    scenario = input().strip()
    if scenario:
        EmergencyPlan.get_plan(scenario)
