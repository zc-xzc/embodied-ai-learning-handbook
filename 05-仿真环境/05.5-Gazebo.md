## 1. Gazebo 安装（ROS 集成）

### 1.1 Gazebo 简介

Gazebo 是一个功能丰富的机器人仿真器，由 Open Robotics 开发和维护。它是 ROS 生态中最常用的仿真器之一。当前有两个主要版本：

- **Gazebo Classic（Gazebo 11）**：传统版本，基于 ODE/Bullet 物理引擎和 OGRE 渲染引擎
- **Gazebo（Gazebo Sim / Ignition）**：新一代版本，模块化架构，支持现代渲染技术

### 1.2 版本选择

| Gazebo 版本 | 对应 ROS 版本 | 状态 | 说明 |
|:---:|:---:|:---:|---|
| Gazebo Classic 11 | ROS 1 Noetic | 维护模式 | 传统稳定版 |
| Gazebo Sim Fortress | ROS 2 Galactic | EOL | 早期 ROS 2 集成 |
| Gazebo Sim Harmonic | ROS 2 Humble/Iron | 当前推荐 | 稳定 LTS |
| Gazebo Sim Ionic | ROS 2 Jazzy | 最新 | 最新功能 |

### 1.3 安装 Gazebo Classic 11（ROS 1 Noetic）

```bash
# === Ubuntu 20.04 + ROS 1 Noetic ===

# 安装 ROS 1 Noetic（如果尚未安装）
sudo sh -c 'echo "deb http://packages.ros.org/ros/ubuntu $(lsb_release -sc) main" > /etc/apt/sources.list.d/ros-latest.list'
sudo apt install curl
curl -s https://raw.githubusercontent.com/ros/rosdistro/master/ros.asc | sudo apt-key add -
sudo apt update
sudo apt install ros-noetic-desktop-full

# 安装 Gazebo 11
sudo apt install gazebo11

# 安装 ROS-Gazebo 集成包
sudo apt install ros-noetic-gazebo11-ros-pkgs ros-noetic-gazebo11-ros-control

# 验证安装
gazebo --version
rospack find gazebo_ros

# 测试运行
roslaunch gazebo_ros empty_world.launch
```

## 2. SDF 模型格式

### 2.1 SDF 基础结构

SDF（Simulation Description Format）是 Gazebo 使用的模型描述格式，基于 XML：

```xml
<!-- 文件: my_robot.sdf -->
<?xml version="1.0" ?>
<sdf version="1.9">
  <model name="my_robot">
    
    <!-- 模型属性 -->
    <static>false</static>
    <self_collide>true</self_collide>
    
    <!-- 模型配置 -->
    <pose>0 0 0 0 0 0</pose>  <!-- x y z roll pitch yaw -->
    
    <!-- === 链接（Link）=== -->
    <!-- 基座 -->
    <link name="base_link">
      <pose>0 0 0.1 0 0 0</pose>
      
      <!-- 惯性属性 -->
      <inertial>
        <mass>5.0</mass>
        <inertia>
          <ixx>0.1</ixx> <ixy>0</ixy> <ixz>0</ixz>
          <iyy>0.1</iyy> <iyz>0</iyz>
          <izz>0.1</izz>
        </inertia>
      </inertial>
      
      <!-- 碰撞几何 -->
      <collision name="base_collision">
        <geometry>
          <box>
            <size>0.4 0.3 0.1</size>
          </box>
        </geometry>
        <surface>
          <friction>
            <ode>
              <mu>0.5</mu>
              <mu2>0.5</mu2>
            </ode>
          </friction>
        </surface>
      </collision>
      
      <!-- 可视几何 -->
      <visual name="base_visual">
        <geometry>
          <box>
            <size>0.4 0.3 0.1</size>
          </box>
        </geometry>
        <material>
          <ambient>0.2 0.2 0.8 1</ambient>
          <diffuse>0.2 0.2 0.8 1</diffuse>
          <specular>0.1 0.1 0.1 1</specular>
        </material>
      </visual>
    </link>
    
    <!-- 左轮 -->
    <link name="left_wheel">
      <pose>0.1 0.15 0.05 0 1.5708 0</pose>
      
      <inertial>
        <mass>1.0</mass>
        <inertia>
          <ixx>0.005</ixx> <ixy>0</ixy> <ixz>0</ixz>
          <iyy>0.005</iyy> <iyz>0</iyz>
          <izz>0.005</izz>
        </inertia>
      </inertial>
      
      <collision name="left_wheel_collision">
        <geometry>
          <cylinder>
            <radius>0.05</radius>
            <length>0.02</length>
          </cylinder>
        </geometry>
        <surface>
          <friction>
            <ode>
              <mu>1.0</mu>
              <mu2>1.0</mu2>
            </ode>
          </friction>
        </surface>
      </collision>
      
      <visual name="left_wheel_visual">
        <geometry>
          <cylinder>
            <radius>0.05</radius>
            <length>0.02</length>
          </cylinder>
        </geometry>
        <material>
          <ambient>0.1 0.1 0.1 1</ambient>
          <diffuse>0.1 0.1 0.1 1</diffuse>
        </material>
      </visual>
    </link>
    
    <!-- 右轮（类似左轮，位置相反） -->
    <link name="right_wheel">
      <pose>0.1 -0.15 0.05 0 1.5708 0</pose>
      <!-- ... 同左轮 ... -->
    </link>
    
    <!-- 激光雷达传感器链接 -->
    <link name="lidar_link">
      <pose>0 0 0.15 0 0 0</pose>
      
      <sensor name="lidar" type="gpu_lidar">
        <pose>0 0 0 0 0 0</pose>
        <update_rate>10</update_rate>
        <lidar>
          <scan>
            <horizontal>
              <samples>640</samples>
              <resolution>1</resolution>
              <min_angle>-3.14159</min_angle>
              <max_angle>3.14159</max_angle>
            </horizontal>
            <vertical>
              <samples>16</samples>
              <resolution>1</resolution>
              <min_angle>-0.2618</min_angle>
              <max_angle>0.2618</max_angle>
            </vertical>
          </scan>
          <range>
            <min>0.1</min>
            <max>30.0</max>
            <resolution>0.01</resolution>
          </range>
          <noise>
            <type>gaussian</type>
            <mean>0.0</mean>
            <stddev>0.01</stddev>
          </noise>
        </lidar>
      </sensor>
    </link>
    
    <!-- === 关节（Joint）=== -->
    <!-- 左轮关节 -->
    <joint name="left_wheel_joint" type="revolute">
      <parent>base_link</parent>
      <child>left_wheel</child>
      <axis>
        <xyz>0 1 0</xyz>
        <limit>
          <lower>-1e16</lower>
          <upper>1e16</upper>
        </limit>
        <dynamics>
          <damping>0.1</damping>
          <friction>0.0</friction>
        </dynamics>
      </axis>
    </joint>
    
    <!-- 右轮关节 -->
    <joint name="right_wheel_joint" type="revolute">
      <parent>base_link</parent>
      <child>right_wheel</child>
      <axis>
        <xyz>0 1 0</xyz>
        <limit>
          <lower>-1e16</lower>
          <upper>1e16</upper>
        </limit>
      </axis>
    </joint>
    
    <!-- 雷达固定关节 -->
    <joint name="lidar_joint" type="fixed">
      <parent>base_link</parent>
      <child>lidar_link</child>
    </joint>
    
  </model>
</sdf>
```

### 2.2 SDF vs URDF 对比

| 特性 | SDF | URDF |
|------|:---:|:---:|
| 支持传感器 | 是 | 否（需 Gazebo 扩展标签） |
| 支持多个模型 | 是 | 否 |
| 支持世界定义 | 是 | 否 |
| 支持插件 | 是 | 否（需 Gazebo 扩展） |
| ROS 原生支持 | 部分 | 是 |
| 格式复杂度 | 较高 | 较低 |
| 语义化 | 更强 | 一般 |

### 2.3 URDF 转 SDF

```bash
# 使用 gz sdf 命令转换
gz sdf -p robot.urdf > robot.sdf

# 或在 ROS 2 中使用
ros2 run sdformat_urdf sdformat_urdf robot.urdf > robot.sdf
```

## 3. 插件系统

### 3.1 插件类型

Gazebo 使用插件系统扩展功能，主要插件类型包括：

| 插件类型 | 说明 | 典型用途 |
|---------|------|---------|
| World Plugin | 世界级插件 | 环境控制、全局事件 |
| Model Plugin | 模型级插件 | 机器人控制、自定义行为 |
| Sensor Plugin | 传感器插件 | 自定义传感器数据处理 |
| System Plugin | 系统插件 | 物理引擎、渲染控制 |
| Visual Plugin | 可视化插件 | 自定义渲染效果 |

### 3.3 自定义插件

```cpp
// 文件: my_model_plugin.cpp
#include <gazebo/gazebo.hh>
#include <gazebo/physics/physics.hh>
#include <ros/ros.h>

namespace gazebo {
class MyModelPlugin : public ModelPlugin {
public:
    void Load(physics::ModelPtr _model, sdf::ElementPtr _sdf) override {
        // 保存模型指针
        this->model = _model;
        
        // 获取参数
        if (_sdf->HasElement("velocity")) {
            this->velocity = _sdf->Get<double>("velocity");
        }
        
        // 初始化 ROS 节点
        int argc = 0;
        char** argv = nullptr;
        ros::init(argc, argv, "my_model_plugin");
        
        this->rosNode = std::make_unique<ros::NodeHandle>();
        
        // 订阅速度命令
        this->cmdSub = this->rosNode->subscribe(
            "/cmd_vel", 10, &MyModelPlugin::OnCmdVel, this
        );
        
        // 绑定更新回调
        this->updateConnection = event::Events::ConnectWorldUpdateBegin(
            std::bind(&MyModelPlugin::OnUpdate, this)
        );
        
        ROS_INFO("MyModelPlugin loaded for model: %s", 
                 this->model->GetName().c_str());
    }
    
    void OnCmdVel(const geometry_msgs::Twist::ConstPtr& msg) {
        this->linearVel = msg->linear.x;
        this->angularVel = msg->angular.z;
    }
    
    void OnUpdate() {
        // 设置模型速度
        this->model->SetLinearVel(
            ignition::math::Vector3d(linearVel, 0, 0)
        );
        this->model->SetAngularVel(
            ignition::math::Vector3d(0, 0, angularVel)
        );
        
        // ROS 回调
        ros::spinOnce();
    }

private:
    physics::ModelPtr model;
    double velocity = 0.0;
    double linearVel = 0.0;
    double angularVel = 0.0;
    std::unique_ptr<ros::NodeHandle> rosNode;
    ros::Subscriber cmdSub;
    event::ConnectionPtr updateConnection;
};

// 注册插件
GZ_REGISTER_MODEL_PLUGIN(MyModelPlugin)
}  // namespace gazebo
```

## 4. 与 ROS/ROS2 集成

### 4.1 ROS 1 集成

#### 启动文件

```xml
<!-- 文件: launch/gazebo_world.launch -->
<launch>
  
  <!-- Gazebo 参数 -->
  <arg name="world_name" default="empty"/>
  <arg name="paused" default="false"/>
  <arg name="use_sim_time" default="true"/>
  <arg name="gui" default="true"/>
  <arg name="headless" default="false"/>
  <arg name="debug" default="false"/>
  
  <!-- 启动 Gazebo -->
  <include file="$(find gazebo_ros)/launch/empty_world.launch">
    <arg name="world_name" value="$(find my_package)/worlds/$(arg world_name).world"/>
    <arg name="paused" value="$(arg paused)"/>
    <arg name="use_sim_time" value="$(arg use_sim_time)"/>
    <arg name="gui" value="$(arg gui)"/>
    <arg name="headless" value="$(arg headless)"/>
    <arg name="debug" value="$(arg debug)"/>
  </include>
  
  <!-- 加载机器人 URDF -->
  <param name="robot_description" 
         command="$(find xacro)/xacro $(find my_package)/urdf/robot.urdf.xacro"/>
  
  <!-- 在 Gazebo 中生成机器人 -->
  <node name="spawn_urdf" pkg="gazebo_ros" type="spawn_model"
        args="-param robot_description -urdf -model my_robot -z 0.1"/>
  
  <!-- 控制器配置 -->
  <rosparam file="$(find my_package)/config/controllers.yaml" command="load"/>
  
  <!-- 启动控制器 -->
  <node name="controller_spawner" pkg="controller_manager" type="spawner"
        args="--shutdown-timeout 1.0
              joint_state_controller
              mobile_base_controller"/>
  
  <!-- 机器人状态发布 -->
  <node name="robot_state_publisher" pkg="robot_state_publisher" type="robot_state_publisher">
    <param name="publish_frequency" value="50"/>
  </node>
  
</launch>
```

### 4.2 ROS 2 集成

#### 启动文件

```python
# 文件: launch/gazebo_world.launch.py
import os
from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, IncludeLaunchDescription
from launch.substitutions import LaunchConfiguration
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch_ros.actions import Node
from ament_index_python.packages import get_package_share_directory

def generate_launch_description():
    
    # 获取包路径
    pkg_share = get_package_share_directory('my_robot_pkg')
    gz_share = get_package_share_directory('ros_gz_sim')
    
    # 参数
    world_file = LaunchConfiguration('world')
    urdf_file = LaunchConfiguration('urdf')
    
    declare_world = DeclareLaunchArgument(
        'world',
        default_value=os.path.join(pkg_share, 'worlds', 'default.sdf'),
    )
    
    declare_urdf = DeclareLaunchArgument(
        'urdf',
        default_value=os.path.join(pkg_share, 'urdf', 'robot.urdf.xacro'),
    )
    
    # Gazebo Sim 启动
    gz_sim = IncludeLaunchDescription(
        PythonLaunchDescriptionSource(
            os.path.join(gz_share, 'launch', 'gz_sim.launch.py')
        ),
        launch_arguments={
            'gz_args': ['-r ', world_file],
            'on_exit_shutdown': 'true',
        }.items(),
    )
    
    # 生成机器人
    spawn_robot = Node(
        package='ros_gz_sim',
        executable='create',
        arguments=[
            '-name', 'my_robot',
            '-topic', '/robot_description',
            '-z', '0.1',
        ],
        output='screen',
    )
    
    # 机器人状态发布
    robot_state_publisher = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        parameters=[{
            'robot_description': open(urdf_file).read(),
            'use_sim_time': True,
        }],
    )
    
    # ROS-Gazebo 桥接
    bridge = Node(
        package='ros_gz_bridge',
        executable='parameter_bridge',
        parameters=[{
            'config_file': os.path.join(pkg_share, 'config', 'bridge.yaml'),
        }],
        remappings=[],
    )
    
    return LaunchDescription([
        declare_world,
        declare_urdf,
        gz_sim,
        spawn_robot,
        robot_state_publisher,
        bridge,
    ])
```

### 4.3 Python 控制示例

```python
# 文件: control_robot.py
"""ROS 2 控制 Gazebo 中的机器人"""
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist
from nav_msgs.msg import Odometry
from sensor_msgs.msg import LaserScan
import numpy as np

class RobotController(Node):
    def __init__(self):
        super().__init__('robot_controller')
        
        # 速度发布
        self.cmd_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        
        # 里程计订阅
        self.odom_sub = self.create_subscription(
            Odometry, '/odom', self.odom_callback, 10
        )
        
        # 激光雷达订阅
        self.scan_sub = self.create_subscription(
            LaserScan, '/scan', self.scan_callback, 10
        )
        
        # 控制定时器
        self.timer = self.create_timer(0.1, self.control_loop)
        
        self.current_pos = np.array([0.0, 0.0])
        self.current_yaw = 0.0
        self.scan_data = None
        
        # 目标位置
        self.target_pos = np.array([5.0, 0.0])
        
    def odom_callback(self, msg):
        self.current_pos = np.array([
            msg.pose.pose.position.x,
            msg.pose.pose.position.y,
        ])
        # 从四元数提取 yaw
        q = msg.pose.pose.orientation
        self.current_yaw = np.arctan2(
            2 * (q.w * q.z + q.x * q.y),
            1 - 2 * (q.y * q.y + q.z * q.z)
        )
    
    def scan_callback(self, msg):
        self.scan_data = msg.ranges
    
    def control_loop(self):
        cmd = Twist()
        
        # 计算到目标的方向和距离
        direction = self.target_pos - self.current_pos
        distance = np.linalg.norm(direction)
        target_angle = np.arctan2(direction[1], direction[0])
        angle_error = target_angle - self.current_yaw
        
        # 归一化角度
        angle_error = np.arctan2(np.sin(angle_error), np.cos(angle_error))
        
        # 简单避障
        if self.scan_data:
            front_dist = min(self.scan_data[len(self.scan_data)//2 - 10:
                                           len(self.scan_data)//2 + 10])
            if front_dist < 0.5:
                cmd.linear.x = 0.0
                cmd.angular.z = 0.5  # 转向避障
            else:
                if abs(angle_error) > 0.1:
                    cmd.angular.z = 0.5 * angle_error
                    cmd.linear.x = 0.1
                else:
                    cmd.linear.x = min(0.5, distance * 0.3)
                    cmd.angular.z = 0.0
        else:
            cmd.linear.x = 0.0
            cmd.angular.z = 0.0
        
        self.cmd_pub.publish(cmd)
        
        if distance < 0.1:
            self.get_logger().info("到达目标!")
            cmd.linear.x = 0.0
            cmd.angular.z = 0.0
            self.cmd_pub.publish(cmd)

def main():
    rclpy.init()
    controller = RobotController()
    rclpy.spin(controller)
    controller.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```
