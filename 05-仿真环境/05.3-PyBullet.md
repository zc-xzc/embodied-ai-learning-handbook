## 1. PyBullet 安装与基础 API

### 1.1 PyBullet 简介

PyBullet 是基于 Bullet 物理引擎的 Python 机器人仿真包，由 Erwin Coumans 开发。它具有以下特点：

- **零依赖安装**：`pip install pybullet` 即可，无需编译
- **跨平台**：支持 Windows、Linux、macOS
- **快速刚体动力学**：基于 Bullet 3.x 引擎
- **内置渲染**：基于 OpenGL 的 CPU/GPU 渲染
- **丰富示例**：包含大量机器人模型和 RL 环境
- **VR 支持**：原生支持 HTC Vive、Oculus 等 VR 设备

### 1.3 基础 API

```python
import pybullet as p
import pybullet_data
import time

# === 连接物理引擎 ===
# GUI 模式（带可视化窗口）
physicsClient = p.connect(p.GUI)

# DIRECT 模式（无界面，用于训练）
# physicsClient = p.connect(p.DIRECT)

# SHARED_MEMORY 模式（共享内存，更快）
# physicsClient = p.connect(p.SHARED_MEMORY)

# 设置资源路径
p.setAdditionalSearchPath(pybullet_data.getDataPath())

# === 基本设置 ===
# 设置重力
p.setGravity(0, 0, -9.81)

# 设置时间步长
p.setTimeStep(1.0 / 240)  # 240Hz

# === 加载模型 ===
# 加载地面平面
planeId = p.loadURDF("plane.urdf")

# 加载机器人
robotId = p.loadURDF("franka_panda/panda.urdf", basePosition=[0, 0, 0], useFixedBase=True)

# === 仿真循环 ===
for i in range(10000):
    # 推进仿真
    p.stepSimulation()
    
    # 获取机器人状态
    joint_states = p.getJointStates(robotId, range(p.getNumJoints(robotId)))
    
    if i % 240 == 0:  # 每秒打印一次
        print(f"Step {i}")
        for j, state in enumerate(joint_states):
            print(f"  Joint {j}: pos={state[0]:.3f}, vel={state[1]:.3f}")
    
    time.sleep(1.0 / 240)  # 实时同步

# 断开连接
p.disconnect()
```

### 1.4 核心 API 速查表

| API | 说明 | 示例 |
|-----|------|------|

## 3. 机器人环境搭建

### 3.1 关节控制

```python
import pybullet as p
import numpy as np

class RobotController:
    """机器人控制器"""
    
    def __init__(self, robot_id: int):
        self.robot_id = robot_id
        self.joint_info = self._get_joint_info()
        self.controllable_joints = self.joint_info['controllable']
        self.num_joints = len(self.controllable_joints)
    
    def _get_joint_info(self):
        """获取关节信息"""
        info = {
            'controllable': [],
            'joint_names': {},
            'joint_limits': {},
            'joint_types': {},
        }
        
        num = p.getNumJoints(self.robot_id)
        for i in range(num):
            joint_info = p.getJointInfo(self.robot_id, i)
            joint_name = joint_info[1].decode('utf-8')
            joint_type = joint_info[2]
            
            if joint_type != p.JOINT_FIXED:  # 非固定关节
                info['controllable'].append(i)
                info['joint_names'][i] = joint_name
                info['joint_limits'][i] = (joint_info[8], joint_info[9])  # (lower, upper)
                info['joint_types'][i] = joint_type
        
        return info
    
    def set_joint_positions(self, target_positions, control_mode=p.POSITION_CONTROL):
        """
        设置关节位置
        
        参数:
            target_positions: 目标位置列表
            control_mode: 控制模式
        """
        for i, joint_idx in enumerate(self.controllable_joints):
            if control_mode == p.POSITION_CONTROL:
                p.setJointMotorControl2(
                    bodyUniqueId=self.robot_id,
                    jointIndex=joint_idx,
                    controlMode=p.POSITION_CONTROL,
                    targetPosition=target_positions[i],
                    force=100.0,  # 最大力
                    maxVelocity=0.5,  # 最大速度
                )
            elif control_mode == p.VELOCITY_CONTROL:
                p.setJointMotorControl2(
                    bodyUniqueId=self.robot_id,
                    jointIndex=joint_idx,
                    controlMode=p.VELOCITY_CONTROL,
                    targetVelocity=target_positions[i],
                    force=100.0,
                )
            elif control_mode == p.TORQUE_CONTROL:
                p.setJointMotorControl2(
                    bodyUniqueId=self.robot_id,
                    jointIndex=joint_idx,
                    controlMode=p.TORQUE_CONTROL,
                    force=target_positions[i],
                )
    
    def get_joint_states(self):
        """获取关节状态"""
        states = p.getJointStates(self.robot_id, self.controllable_joints)
        
        positions = [s[0] for s in states]
        velocities = [s[1] for s in states]
        forces = [s[3] for s in states]
        
        return {
            'positions': np.array(positions),
            'velocities': np.array(velocities),
            'forces': np.array(forces),
        }
    
    def get_link_state(self, link_index: int):
        """获取连杆状态"""
        state = p.getLinkState(
            self.robot_id, link_index,
            computeLinkVelocity=1,
            computeForwardKinematics=1,
        )
        
        return {
            'position': np.array(state[0]),        # 世界坐标位置
            'orientation': np.array(state[1]),      # 世界坐标姿态（四元数）
            'linear_velocity': np.array(state[6]),  # 线速度
            'angular_velocity': np.array(state[7]), # 角速度
        }
    
    def get_end_effector_state(self, ee_link_name: str = None):
        """获取末端执行器状态"""
        if ee_link_name:
            # 通过名称查找
            for i in range(p.getNumJoints(self.robot_id)):
                info = p.getJointInfo(self.robot_id, i)
                if info[12].decode('utf-8') == ee_link_name:
                    return self.get_link_state(i)
        
        # 默认使用最后一个连杆
        return self.get_link_state(p.getNumJoints(self.robot_id) - 1)
    
    def compute_ik(self, target_pos, target_orn=None, ee_link_index=None):
        """
        计算逆运动学
        
        参数:
            target_pos: 目标位置 [x, y, z]
            target_orn: 目标姿态四元数 [x, y, z, w]
            ee_link_index: 末端执行器连杆索引
        """
        if ee_link_index is None:
            ee_link_index = p.getNumJoints(self.robot_id) - 1
        
        # 使用数值 IK
        joint_poses = p.calculateInverseKinematics(
            bodyUniqueId=self.robot_id,
            endEffectorLinkIndex=ee_link_index,
            targetPosition=target_pos,
            targetOrientation=target_orn,
            lowerLimits=[self.joint_info['joint_limits'][j][0] for j in self.controllable_joints],
            upperLimits=[self.joint_info['joint_limits'][j][1] for j in self.controllable_joints],
            jointRanges=[u - l for l, u in [self.joint_info['joint_limits'][j] for j in self.controllable_joints]],
            restPoses=[0.0] * self.num_joints,  # 参考姿态
            maxNumIterations=100,
            residualThreshold=1e-5,
        )
        
        return np.array(joint_poses[:self.num_joints])
    
    def pd_control(self, target_pos, target_vel=None, kp=100.0, kd=5.0):
        """
        PD 控制
        
        参数:
            target_pos: 目标关节位置
            target_vel: 目标关节速度
            kp: 比例增益
            kd: 微分增益
        """
        if target_vel is None:
            target_vel = np.zeros(self.num_joints)
        
        current_state = self.get_joint_states()
        q = current_state['positions']
        dq = current_state['velocities']
        
        # PD 力矩
        tau = kp * (target_pos - q) + kd * (target_vel - dq)
        
        # 应用力矩控制
        self.set_joint_positions(tau, control_mode=p.TORQUE_CONTROL)
        
        return tau
```

### 3.2 场景搭建

```python
class SceneBuilder:
    """场景构建器"""
    
    def __init__(self):
        self.objects = []
    
    def create_ground(self):
        """创建地面"""
        import pybullet_data
        p.setAdditionalSearchPath(pybullet_data.getDataPath())
        ground = p.loadURDF("plane.urdf")
        self.objects.append(('ground', ground))
        return ground
    
    def create_table(self, pos=[0.5, 0, 0], size=[0.8, 0.8, 0.05]):
        """创建桌子"""
        col_shape = p.createCollisionShape(p.GEOM_BOX, halfExtents=[s/2 for s in size])
        vis_shape = p.createVisualShape(
            p.GEOM_BOX, halfExtents=[s/2 for s in size],
            rgbaColor=[0.6, 0.4, 0.2, 1.0]
        )
        table = p.createMultiBody(
            baseMass=0,  # 静态
            baseCollisionShapeIndex=col_shape,
            baseVisualShapeIndex=vis_shape,
            basePosition=[pos[0], pos[1], pos[2] + size[2]/2],
        )
        self.objects.append(('table', table))
        return table
    
    def create_object(self, obj_type='cube', pos=[0.5, 0, 0.1], size=0.05, mass=0.1):
        """创建可操作物体"""
        if obj_type == 'cube':
            col_shape = p.createCollisionShape(p.GEOM_BOX, halfExtents=[size/2]*3)
            vis_shape = p.createVisualShape(
                p.GEOM_BOX, halfExtents=[size/2]*3,
                rgbaColor=[0.8, 0.2, 0.2, 1.0]
            )
        elif obj_type == 'sphere':
            col_shape = p.createCollisionShape(p.GEOM_SPHERE, radius=size/2)
            vis_shape = p.createVisualShape(
                p.GEOM_SPHERE, radius=size/2,
                rgbaColor=[0.2, 0.8, 0.2, 1.0]
            )
        elif obj_type == 'cylinder':
            col_shape = p.createCollisionShape(p.GEOM_CYLINDER, radius=size/2, height=size)
            vis_shape = p.createVisualShape(
                p.GEOM_CYLINDER, radius=size/2, length=size,
                rgbaColor=[0.2, 0.2, 0.8, 1.0]
            )
        
        obj = p.createMultiBody(
            baseMass=mass,
            baseCollisionShapeIndex=col_shape,
            baseVisualShapeIndex=vis_shape,
            basePosition=pos,
        )
        self.objects.append((obj_type, obj))
        return obj
    
    def create_wall(self, pos=[0, 0, 0.5], size=[2.0, 0.1, 1.0]):
        """创建墙"""
        col_shape = p.createCollisionShape(p.GEOM_BOX, halfExtents=[s/2 for s in size])
        vis_shape = p.createVisualShape(
            p.GEOM_BOX, halfExtents=[s/2 for s in size],
            rgbaColor=[0.5, 0.5, 0.5, 1.0]
        )
        wall = p.createMultiBody(
            baseMass=0,
            baseCollisionShapeIndex=col_shape,
            baseVisualShapeIndex=vis_shape,
            basePosition=pos,
        )
        self.objects.append(('wall', wall))
        return wall
    
    def reset_objects(self):
        """重置所有物体"""
        for name, obj_id in self.objects:
            if name not in ['ground', 'table', 'wall']:
                # 随机放置可操作物体
                pos = [np.random.uniform(-0.3, 0.3), np.random.uniform(-0.3, 0.3), 0.1]
                p.resetBasePositionAndOrientation(obj_id, pos, [0, 0, 0, 1])
```

## 4. 与 Gymnasium 集成

### 4.1 自定义 Gymnasium 环境

```python
# 文件: pybullet_gym_env.py
import pybullet as p
import pybullet_data
import numpy as np
import gymnasium as gym
from gymnasium import spaces
import time


class PyBulletRobotEnv(gym.Env):
    """PyBullet 机器人 Gymnasium 环境"""
    
    metadata = {"render_modes": ["human", "rgb_array"], "render_fps": 30}
    
    def __init__(self, render_mode=None, robot_urdf="franka_panda/panda.urdf"):
        super().__init__()
        
        self.render_mode = render_mode
        self.robot_urdf = robot_urdf
        
        # 仿真参数
        self.sim_dt = 1.0 / 240
        self.control_dt = 1.0 / 30  # 30Hz 控制
        self.sim_steps_per_control = int(self.control_dt / self.sim_dt)
        
        # 动作空间：7个关节 + 1个夹爪
        self.action_space = spaces.Box(
            low=-1.0, high=1.0, shape=(8,), dtype=np.float32
        )
        
        # 观测空间
        obs_dim = 7 + 7 + 3 + 3 + 3  # 关节位置+速度+末端位置+物体位置+目标位置
        self.observation_space = spaces.Box(
            low=-np.inf, high=np.inf, shape=(obs_dim,), dtype=np.float32
        )
        
        # 初始化仿真
        self.physics_client = -1
        self._init_simulation()
    
    def _init_simulation(self):
        """初始化仿真环境"""
        if self.render_mode == "human":
            self.physics_client = p.connect(p.GUI)
        else:
            self.physics_client = p.connect(p.DIRECT)
        
        p.setAdditionalSearchPath(pybullet_data.getDataPath())
        p.setGravity(0, 0, -9.81)
        p.setTimeStep(self.sim_dt)
        
        # 加载场景
        self.plane = p.loadURDF("plane.urdf")
        self.robot = p.loadURDF(self.robot_urdf, useFixedBase=True)
        
        # 创建桌子
        self.table = self._create_table()
        
        # 创建可操作物体
        self.object = self._create_object()
        
        # 获取关节信息
        self._setup_joints()
        
        # 设置相机（用于渲染）
        self.view_matrix = p.computeViewMatrixFromYawPitchRoll(
            cameraTargetPosition=[0.5, 0, 0.3],
            distance=1.5,
            yaw=45,
            pitch=-30,
            roll=0,
            upAxisIndex=2,
        )
        self.proj_matrix = p.computeProjectionMatrixFOV(
            fov=60, aspect=1.0, nearVal=0.1, farVal=10.0
        )
    
    def _setup_joints(self):
        """设置关节索引"""
        self.controllable_joints = []
        self.joint_limits_lower = []
        self.joint_limits_upper = []
        
        num_joints = p.getNumJoints(self.robot)
        for i in range(num_joints):
            info = p.getJointInfo(self.robot, i)
            if info[2] != p.JOINT_FIXED:
                self.controllable_joints.append(i)
                self.joint_limits_lower.append(info[8])
                self.joint_limits_upper.append(info[9])
        
        self.num_joints = len(self.controllable_joints)
        self.ee_link_index = num_joints - 1  # 末端连杆
        
        # 关闭默认电机，使用力矩控制
        for joint_idx in self.controllable_joints:
            p.setJointMotorControl2(
                self.robot, joint_idx, p.VELOCITY_CONTROL, force=0
            )
    
    def _create_table(self):
        """创建桌子"""
        col_shape = p.createCollisionShape(p.GEOM_BOX, halfExtents=[0.4, 0.4, 0.025])
        vis_shape = p.createVisualShape(
            p.GEOM_BOX, halfExtents=[0.4, 0.4, 0.025],
            rgbaColor=[0.6, 0.4, 0.2, 1.0]
        )
        return p.createMultiBody(0, col_shape, vis_shape, [0.5, 0, 0.025])
    
    def _create_object(self):
        """创建物体"""
        col_shape = p.createCollisionShape(p.GEOM_BOX, halfExtents=[0.025]*3)
        vis_shape = p.createVisualShape(
            p.GEOM_BOX, halfExtents=[0.025]*3,
            rgbaColor=[0.8, 0.2, 0.2, 1.0]
        )
        return p.createMultiBody(0.1, col_shape, vis_shape, [0.5, 0, 0.1])
    
    def _get_obs(self):
        """获取观测"""
        # 关节状态
        joint_states = p.getJointStates(self.robot, self.controllable_joints)
        joint_pos = np.array([s[0] for s in joint_states])
        joint_vel = np.array([s[1] for s in joint_states])
        
        # 末端执行器位置
        ee_state = p.getLinkState(self.robot, self.ee_link_index)
        ee_pos = np.array(ee_state[0])
        
        # 物体位置
        obj_pos, _ = p.getBasePositionAndOrientation(self.object)
        obj_pos = np.array(obj_pos)
        
        # 目标位置（随机生成或固定）
        target_pos = self.target_pos if hasattr(self, 'target_pos') else np.array([0.7, 0.2, 0.1])
        
        obs = np.concatenate([
            joint_pos, joint_vel, ee_pos, obj_pos, target_pos
        ]).astype(np.float32)
        
        return obs
    
    def _apply_action(self, action):
        """应用动作"""
        # action: [delta_joint_0, ..., delta_joint_6, gripper]
        
        # 获取当前关节位置
        joint_states = p.getJointStates(self.robot, self.controllable_joints)
        current_pos = np.array([s[0] for s in joint_states])
        
        # 计算目标位置
        target_pos = current_pos + action[:7] * 0.1  # 缩放
        
        # 限位
        target_pos = np.clip(
            target_pos,
            self.joint_limits_lower,
            self.joint_limits_upper,
        )
        
        # PD 控制
        kp = 100.0
        kd = 5.0
        for i, joint_idx in enumerate(self.controllable_joints):
            q = current_pos[i]
            dq = joint_states[i][1]
            tau = kp * (target_pos[i] - q) - kd * dq
            p.setJointMotorControl2(
                self.robot, joint_idx, p.TORQUE_CONTROL, force=tau
            )
        
        # 推进仿真
        for _ in range(self.sim_steps_per_control):
            p.stepSimulation()
    
    def _compute_reward(self, obs):
        """计算奖励"""
        ee_pos = obs[14:17]
        obj_pos = obs[17:20]
        target_pos = obs[20:23]
        
        # 到达物体的奖励
        dist_ee_to_obj = np.linalg.norm(ee_pos - obj_pos)
        reaching_reward = -dist_ee_to_obj
        
        # 物体到目标的奖励
        dist_obj_to_target = np.linalg.norm(obj_pos - target_pos)
        placing_reward = -dist_obj_to_target
        
        # 物体提升奖励
        height_reward = 0
        if obj_pos[2] > 0.15:
            height_reward = 1.0
        
        total_reward = reaching_reward + placing_reward + height_reward
        
        return total_reward
    
    def _is_done(self, obs):
        """判断是否结束"""
        obj_pos = obs[17:20]
        target_pos = obs[20:23]
        
        # 到达目标
        if np.linalg.norm(obj_pos - target_pos) < 0.05:
            return True
        
        # 物体掉落
        if obj_pos[2] < -0.1:
            return True
        
        # 超时
        if self.current_step >= 300:
            return True
        
        return False
    
    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        
        # 重置仿真
        p.resetSimulation()
        p.setGravity(0, 0, -9.81)
        p.setTimeStep(self.sim_dt)
        
        # 重新加载场景
        self.plane = p.loadURDF("plane.urdf")
        self.robot = p.loadURDF(self.robot_urdf, useFixedBase=True)
        self.table = self._create_table()
        self.object = self._create_object()
        self._setup_joints()
        
        # 随机化目标位置
        self.target_pos = np.array([
            np.random.uniform(0.4, 0.7),
            np.random.uniform(-0.2, 0.2),
            0.1,
        ])
        
        self.current_step = 0
        
        obs = self._get_obs()
        info = {}
        
        return obs, info
    
    def step(self, action):
        self._apply_action(action)
        self.current_step += 1
        
        obs = self._get_obs()
        reward = self._compute_reward(obs)
        done = self._is_done(obs)
        truncated = self.current_step >= 300
        info = {
            'ee_pos': obs[14:17],
            'obj_pos': obs[17:20],
            'target_pos': obs[20:23],
        }
        
        return obs, reward, done, truncated, info
    
    def render(self):
        if self.render_mode == "rgb_array":
            img_arr = p.getCameraImage(
                width=640, height=480,
                viewMatrix=self.view_matrix,
                projectionMatrix=self.proj_matrix,
            )
            rgb = img_arr[2]  # RGB array
            return np.array(rgb, dtype=np.uint8)
        return None
    
    def close(self):
        if self.physics_client >= 0:
            p.disconnect()
            self.physics_client = -1
```

## 5. 代码示例：机械臂 Pick-Place 环境

### 5.1 完整 Pick-Place 环境

```python
# 文件: franka_pickplace.py
"""Franka Panda 机械臂 Pick-Place 完整示例"""
import pybullet as p
import pybullet_data
import numpy as np
import time


class FrankaPickPlace:
    """Franka 机械臂 Pick-Place 环境"""
    
    def __init__(self, render=True):
        # 连接引擎
        self.client = p.connect(p.GUI if render else p.DIRECT)
        p.setAdditionalSearchPath(pybullet_data.getDataPath())
        p.setGravity(0, 0, -9.81)
        p.setTimeStep(1.0 / 240)
        
        # 加载场景
        self.plane = p.loadURDF("plane.urdf")
        self.robot = p.loadURDF(
            "franka_panda/panda.urdf",
            basePosition=[0, 0, 0],
            useFixedBase=True,
        )
        
        # 创建桌子和物体
        self._create_scene()
        
        # 获取关节信息
        self._setup_joints()
        
        # 设置初始姿态
        self.home_pos = np.array([0, -0.5, 0, -2.5, 0, 2.0, 0.7])
        self._move_to_joints(self.home_pos, duration=2.0)
        
        print("环境初始化完成")
    
    def _create_scene(self):
        """创建场景"""
        # 桌子
        table_shape = p.createCollisionShape(
            p.GEOM_BOX, halfExtents=[0.4, 0.4, 0.025]
        )
        table_vis = p.createVisualShape(
            p.GEOM_BOX, halfExtents=[0.4, 0.4, 0.025],
            rgbaColor=[0.6, 0.4, 0.2, 1.0]
        )
        self.table = p.createMultiBody(
            0, table_shape, table_vis, [0.5, 0, 0.025]
        )
        
        # 可抓取物体
        obj_shape = p.createCollisionShape(
            p.GEOM_BOX, halfExtents=[0.025, 0.025, 0.025]
        )
        obj_vis = p.createVisualShape(
            p.GEOM_BOX, halfExtents=[0.025, 0.025, 0.025],
            rgbaColor=[0.8, 0.2, 0.2, 1.0]
        )
        self.object = p.createMultiBody(
            0.1, obj_shape, obj_vis, [0.5, 0, 0.1]
        )
        
        # 目标位置标记
        target_shape = p.createVisualShape(
            p.GEOM_BOX, halfExtents=[0.03, 0.03, 0.001],
            rgbaColor=[0.2, 0.8, 0.2, 0.5]
        )
        self.target = p.createMultiBody(
            0, -1, target_shape, [0.5, 0.3, 0.051]
        )
    
    def _setup_joints(self):
        """设置关节"""
        self.arm_joints = []
        self.gripper_joints = []
        
        for i in range(p.getNumJoints(self.robot)):
            info = p.getJointInfo(self.robot, i)
            joint_name = info[1].decode('utf-8')
            
            if 'panda_joint' in joint_name:
                self.arm_joints.append(i)
            elif 'panda_finger' in joint_name:
                self.gripper_joints.append(i)
        
        # 关闭默认电机
        for j in self.arm_joints + self.gripper_joints:
            p.setJointMotorControl2(self.robot, j, p.VELOCITY_CONTROL, force=0)
        
        # 末端连杆
        self.ee_link = 11  # panda_hand link
    
    def _move_to_joints(self, target_joints, duration=1.0):
        """平滑移动到目标关节位置"""
        steps = int(duration * 240)
        
        # 获取当前位置
        states = p.getJointStates(self.robot, self.arm_joints)
        current = np.array([s[0] for s in states])
        
        for i in range(steps):
            alpha = i / steps
            # 三次多项式插值
            alpha_smooth = 3 * alpha**2 - 2 * alpha**3
            
            target = current + (target_joints - current) * alpha_smooth
            
            # PD 控制
            for j, joint_idx in enumerate(self.arm_joints):
                state = p.getJointState(self.robot, joint_idx)
                q = state[0]
                dq = state[1]
                tau = 100 * (target[j] - q) - 5 * dq
                p.setJointMotorControl2(
                    self.robot, joint_idx, p.TORQUE_CONTROL, force=tau
                )
            
            # 夹爪打开
            for j in self.gripper_joints:
                p.setJointMotorControl2(
                    self.robot, j, p.POSITION_CONTROL,
                    targetPosition=0.04, force=10
                )
            
            p.stepSimulation()
            time.sleep(1.0 / 240)
    
    def move_to_position(self, target_pos, target_orn=None):
        """通过 IK 移动到位置"""
        if target_orn is None:
            target_orn = p.getQuaternionFromEuler([np.pi, 0, 0])
        
        # 计算 IK
        joint_poses = p.calculateInverseKinematics(
            self.robot, self.ee_link, target_pos, target_orn,
            maxNumIterations=200,
            residualThreshold=1e-5,
        )
        
        # 移动到目标
        self._move_to_joints(np.array(joint_poses[:7]), duration=1.0)
    
    def close_gripper(self):
        """关闭夹爪"""
        for _ in range(120):  # 0.5秒
            for j in self.gripper_joints:
                p.setJointMotorControl2(
                    self.robot, j, p.POSITION_CONTROL,
                    targetPosition=0.0, force=20
                )
            p.stepSimulation()
            time.sleep(1.0 / 240)
    
    def open_gripper(self):
        """打开夹爪"""
        for _ in range(120):
            for j in self.gripper_joints:
                p.setJointMotorControl2(
                    self.robot, j, p.POSITION_CONTROL,
                    targetPosition=0.04, force=10
                )
            p.stepSimulation()
            time.sleep(1.0 / 240)
    
    def pick_and_place(self):
        """执行完整的 pick-place 动作"""
        # 获取物体位置
        obj_pos, _ = p.getBasePositionAndOrientation(self.object)
        
        print(f"物体位置: {obj_pos}")
        
        # 1. 移动到物体上方
        print("1. 移动到物体上方...")
        self.move_to_position([obj_pos[0], obj_pos[1], obj_pos[2] + 0.15])
        
        # 2. 打开夹爪
        print("2. 打开夹爪...")
        self.open_gripper()
        
        # 3. 下降到物体
        print("3. 下降到物体...")
        self.move_to_position([obj_pos[0], obj_pos[1], obj_pos[2] + 0.03])
        
        # 4. 关闭夹爪（抓取）
        print("4. 抓取物体...")
        self.close_gripper()
        
        # 创建约束（模拟抓取）
        constraint = p.createConstraint(
            self.robot, self.ee_link,
            self.object, -1,
            p.JOINT_POINT2POINT,
            [0, 0, 0],
            [0, 0, 0],
            [0, 0, 0],
        )
        p.changeConstraint(constraint, maxForce=50)
        
        # 5. 提起物体
        print("5. 提起物体...")
        self.move_to_position([obj_pos[0], obj_pos[1], obj_pos[2] + 0.3])
        
        # 6. 移动到目标位置
        print("6. 移动到目标位置...")
        target_pos = [0.5, 0.3, 0.15]
        self.move_to_position(target_pos)
        
        # 7. 下降放置
        print("7. 下降放置...")
        self.move_to_position([0.5, 0.3, 0.08])
        
        # 8. 释放
        print("8. 释放物体...")
        p.removeConstraint(constraint)
        self.open_gripper()
        
        # 9. 回到初始位置
        print("9. 回到初始位置...")
        self.move_to_position([0.5, 0, 0.3])
        
        print("Pick-Place 完成!")
    
    def run(self):
        """运行仿真"""
        try:
            self.pick_and_place()
            
            # 保持窗口打开
            while p.isConnected():
                p.stepSimulation()
                time.sleep(1.0 / 240)
        except KeyboardInterrupt:
            print("仿真中断")
        finally:
            p.disconnect()


# 运行
if __name__ == "__main__":
    env = FrankaPickPlace(render=True)
    env.run()
```
