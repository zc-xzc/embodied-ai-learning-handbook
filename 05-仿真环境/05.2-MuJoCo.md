## 1. MuJoCo 安装与基础

### 1.1 MuJoCo 简介

MuJoCo（Multi-Joint dynamics with Contact）是一个高性能的物理仿真引擎，最初由 Emo Todorov 在华盛顿大学开发，后被 Google DeepMind 收购并开源。它的核心特点包括：

- **快速接触动力学**：使用凸优化求解接触力，支持软接触和硬接触
- **精确的物理模型**：支持肌肉、肌腱、执行器建模
- **确定性仿真**：相同输入产生相同输出，便于调试和复现
- **C 核心引擎**：底层用 C 编写，Python 绑定提供完整功能
- **轻量高效**：单线程即可实现实时仿真，资源占用低

### 1.3 基础概念

```python
import mujoco
import numpy as np

# 加载模型
model = mujoco.MjModel.from_xml_string("""
<mujoco>
  <worldbody>
    <body name="box" pos="0 0 0.5">
      <freejoint/>
      <geom type="box" size="0.1 0.1 0.1" mass="1.0"/>
    </body>
  </worldbody>
</mujoco>
""")

# 创建数据结构
data = mujoco.MjData(model)

# 仿真循环
for i in range(1000):
    mujoco.mj_step(model, data)
    print(f"Step {i}: box pos = {data.body('box').xpos}")
```

### 1.4 核心对象

| 对象 | 说明 | 关键属性 |
|------|------|---------|
| `MjModel` | 静态模型定义 | `nq`(自由度数), `nv`(速度数), `nu`(执行器数), `nbody`(刚体数) |
| `MjData` | 动态仿真数据 | `qpos`(位置), `qvel`(速度), `ctrl`(控制), `xpos`(世界坐标) |
| `MjViewer` | 交互式查看器 | 渲染场景、鼠标交互 |
| `mj_step` | 推进一步仿真 | 更新 `MjData` 中的状态 |

## 2. MJCF 模型格式详解

### 2.1 MJCF 基础结构

MJCF（MuJoCo XML）是 MuJoCo 的模型描述格式：

```xml
<!-- 文件: simple_arm.xml -->
<mujoco model="simple_arm">
  
  <!-- 编译器选项 -->
  <compiler angle="radian" coordinate="local" autolimits="true"/>
  
  <!-- 选项 -->
  <option timestep="0.002" gravity="0 0 -9.81" integrator="RK4">
    <flag warmstart="enable"/>
  </option>
  
  <!-- 默认值 -->
  <default>
    <joint armature="0.01" damping="0.5" limited="true"/>
    <geom contype="1" conaffinity="1" friction="1 0.1 0.1"/>
    <motor ctrlrange="-1 1"/>
  </default>
  
  <!-- 资源 -->
  <asset>
    <material name="mat_robot" rgba="0.7 0.7 0.7 1"/>
    <texture name="tex_grid" type="2d" builtin="checker" width="512" height="512"
             rgb1="0.2 0.3 0.4" rgb2="0.3 0.4 0.5"/>
    <material name="mat_grid" texture="tex_grid" texrepeat="5 5" reflectance="0.2"/>
  </asset>
  
  <!-- 世界体 -->
  <worldbody>
    <!-- 地面 -->
    <geom name="floor" type="plane" size="0 0 0.05" material="mat_grid"/>
    
    <!-- 灯光 -->
    <light name="top" pos="0 0 3" dir="0 0 -1" diffuse="0.8 0.8 0.8"/>
    
    <!-- 机器人 -->
    <body name="base" pos="0 0 0">
      <inertial pos="0 0 0" mass="1.0" diaginertia="0.01 0.01 0.01"/>
      <geom name="base_geom" type="cylinder" size="0.1 0.05" material="mat_robot"/>
      
      <body name="link1" pos="0 0 0.05">
        <joint name="joint1" type="hinge" axis="0 0 1" range="-3.14 3.14"/>
        <inertial pos="0.1 0 0" mass="0.5" diaginertia="0.005 0.005 0.001"/>
        <geom name="link1_geom" type="capsule" fromto="0 0 0 0.2 0 0" size="0.03" material="mat_robot"/>
        
        <body name="link2" pos="0.2 0 0">
          <joint name="joint2" type="hinge" axis="0 1 0" range="-3.14 3.14"/>
          <inertial pos="0.1 0 0" mass="0.3" diaginertia="0.003 0.003 0.001"/>
          <geom name="link2_geom" type="capsule" fromto="0 0 0 0.2 0 0" size="0.025" material="mat_robot"/>
          
          <!-- 末端执行器 -->
          <body name="end_effector" pos="0.2 0 0">
            <geom name="ee_geom" type="sphere" size="0.03" rgba="0.8 0.2 0.2 1"/>
          </body>
        </body>
      </body>
    </body>
  </worldbody>
  
  <!-- 执行器 -->
  <actuator>
    <motor name="motor1" joint="joint1" gear="1" ctrlrange="-5 5"/>
    <motor name="motor2" joint="joint2" gear="1" ctrlrange="-5 5"/>
  </actuator>
  
  <!-- 传感器 -->
  <sensor>
    <jointpos name="joint1_pos" joint="joint1"/>
    <jointvel name="joint1_vel" joint="joint1"/>
    <framepos name="ee_pos" objtype="site" objname="end_effector"/>
  </sensor>
  
</mujoco>
```

## 3. 关节定义与自由度配置

### 3.1 关节类型

| 类型 | `type` 值 | 自由度 | 说明 |
|------|----------|:---:|------|
| 铰链关节 | `hinge` | 1 | 绕轴旋转 |
| 滑动关节 | `slide` | 1 | 沿轴平移 |
| 球关节 | `ball` | 3 | 三轴旋转 |
| 自由关节 | `free` | 6 | 三轴平移+三轴旋转 |
| 固定 | 无 joint | 0 | 无运动 |

## 4. 仿真循环与物理参数

### 4.1 基本仿真循环

```python
import mujoco
import numpy as np
import time

class MuJoCoSimulation:
    """MuJoCo 仿真管理器"""
    
    def __init__(self, model_path: str, dt: float = 0.002):
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.model.opt.timestep = dt
        self.data = mujoco.MjData(self.model)
        
        # 控制频率（比仿真频率低）
        self.sim_dt = dt
        self.control_dt = 0.02  # 50Hz 控制
        self.control_steps = int(self.control_dt / self.sim_dt)  # 10步
        
        print(f"仿真步长: {self.sim_dt}s ({1/self.sim_dt:.0f}Hz)")
        print(f"控制步长: {self.control_dt}s ({1/self.control_dt:.0f}Hz)")
        print(f"每控制步仿真步数: {self.control_steps}")
    
    def reset(self, qpos=None, qvel=None):
        """重置仿真"""
        mujoco.mj_resetData(self.model, self.data)
        if qpos is not None:
            self.data.qpos[:] = qpos
        if qvel is not None:
            self.data.qvel[:] = qvel
        mujoco.mj_forward(self.model, self.data)
    
    def set_control(self, ctrl):
        """设置控制输入"""
        self.data.ctrl[:] = ctrl
    
    def step(self):
        """推进一步仿真"""
        mujoco.mj_step(self.model, self.data)
    
    def step_control(self, ctrl):
        """推进一个控制步"""
        self.set_control(ctrl)
        for _ in range(self.control_steps):
            self.step()
    
    def get_state(self):
        """获取当前状态"""
        return {
            'qpos': self.data.qpos.copy(),
            'qvel': self.data.qvel.copy(),
            'qacc': self.data.qacc.copy(),
            'ctrl': self.data.ctrl.copy(),
            'time': self.data.time,
        }
    
    def get_body_state(self, body_name: str):
        """获取指定刚体状态"""
        body_id = mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_BODY, body_name)
        return {
            'pos': self.data.xpos[body_id].copy(),       # 世界坐标位置
            'quat': self.data.xquat[body_id].copy(),     # 世界坐标姿态 (w,x,y,z)
            'vel': self.data.cvel[body_id].copy(),       # 速度
        }
    
    def run_realtime(self, duration: float, control_callback=None):
        """实时运行仿真"""
        n_steps = int(duration / self.sim_dt)
        for i in range(n_steps):
            start_time = time.time()
            
            # 控制回调
            if control_callback and i % self.control_steps == 0:
                ctrl = control_callback(self.get_state())
                self.set_control(ctrl)
            
            self.step()
            
            # 实时同步
            elapsed = time.time() - start_time
            sleep_time = self.sim_dt - elapsed
            if sleep_time > 0:
                time.sleep(sleep_time)
```

## 5. MuJoCo Viewer 使用

### 5.1 基础查看器

```python
import mujoco
import mujoco.viewer
import numpy as np
import time

# 加载模型
model = mujoco.MjModel.from_xml_path("g1.xml")
data = mujoco.MjData(model)

# 启动交互式查看器
with mujoco.viewer.launch_passive(model, data) as viewer:
    
    # 仿真循环
    while viewer.is_running():
        
        # 设置控制（例如：保持站立）
        data.ctrl[:] = 0.0  # 所有关节回到默认位置
        
        # 推进仿真
        mujoco.mj_step(model, data)
        
        # 更新查看器
        viewer.sync()
        
        # 实时控制
        time.sleep(0.002)  # 500Hz
        
        # 打印状态
        if int(data.time * 10) % 10 == 0:
            print(f"Time: {data.time:.1f}s, COM: {data.body('pelvis').xpos}")
```

### 5.3 多视图渲染

```python
class MultiViewRenderer:
    """多视图渲染器"""
    
    def __init__(self, model, width=480, height=360):
        self.model = model
        self.renderer = mujoco.Renderer(model, height=height, width=width)
        self.views = ["front", "side", "top", "tracking"]
    
    def render_views(self, data, body_name="pelvis"):
        """渲染多个视角"""
        frames = {}
        
        for view in self.views:
            if view == "tracking":
                # 跟踪相机：跟随指定刚体
                body_id = mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_BODY, body_name)
                self.renderer.update_scene(data, camera=body_id, options={
                    "fixedcam": False,
                })
            else:
                self.renderer.update_scene(data, camera=view)
            
            frames[view] = self.renderer.render()
        
        return frames
    
    def render_grid(self, data):
        """渲染2x2网格视图"""
        frames = self.render_views(data)
        
        # 拼接为2x2网格
        top = np.hstack([frames["front"], frames["side"]])
        bottom = np.hstack([frames["top"], frames["tracking"]])
        grid = np.vstack([top, bottom])
        
        return grid
```

## 6. 与 mink/qpsolvers 集成做运动学求解

```python
import mujoco
import numpy as np
import mink

# 加载模型
model = mujoco.MjModel.from_xml_path("g1.xml")
data = mujoco.MjData(model)

# 创建配置
configuration = mink.Configuration(model)

# 定义任务
# 任务1：末端执行器位置跟踪
ee_task = mink.FrameTask(
    frame_name="left_ee_site",    # 末端执行器 site 名
    frame_type="site",
    position_cost=1.0,            # 位置误差权重
    orientation_cost=0.1,         # 姿态误差权重
    lm_damping=1e-3,             # Levenberg-Marquardt 阻尼
)

# 任务2：姿态保持（保持重心稳定）
posture_task = mink.PostureTask(
    cost=1e-2,  # 较低权重
)

# 任务3：重心保持
com_task = mink.ComTask(
    cost=1.0,
)

# 定义约束
# 关节限位约束
joint_limit_constraint = mink.JointLimitConstraint(
    model=model,
    gain=0.5,
)

# 创建求解器
solver = mink.solve_qp

# 仿真循环
dt = 0.02  # 控制周期
for i in range(1000):
    # 更新配置
    configuration.update(data.qpos)
    
    # 设置目标
    # 让左臂末端到达指定位置
    target_pos = np.array([0.3, 0.3, 1.0])
    target_quat = np.array([1, 0, 0, 0])  # w, x, y, z
    ee_task.set_target(target_pos, target_quat)
    
    # 保持初始姿态
    posture_task.set_target(model.keyframe("home").qpos)
    
    # 保持重心在支撑面上方
    com_task.set_target(np.array([0.0, 0.0, 0.5]))
    
    # 构建 QP 问题并求解
    vel = mink.solve_ik(
        configuration,
        tasks=[ee_task, posture_task, com_task],
        constraints=[joint_limit_constraint],
        dt=dt,
        solver="osqp",
    )
    
    # 将速度转换为关节位置目标
    target_qpos = data.qpos.copy()
    # 跳过自由关节（前7个：3位置+4四元数）
    target_qpos[7:] += vel * dt
    
    # 设置控制
    data.ctrl[:] = target_qpos[7:]  # position 控制器
    
    # 推进仿真
    mujoco.mj_step(model, data)
    
    if i % 50 == 0:
        ee_pos = data.site("left_ee_site").xpos
        print(f"Step {i}: EE pos = {ee_pos}, target = {target_pos}")
```

### 6.3 使用 qpsolvers 直接求解

```python
import qpsolvers
import numpy as np

def solve_ik_qp(J, error, q_max=None, q_min=None, dt=0.02):
    """
    使用 QP 求解逆运动学
    
    min 0.5 * ||J*q_dot - error||^2 + 0.5 * ||q_dot||^2 * reg
    s.t. q_min <= q + q_dot*dt <= q_max
    
    参数:
        J: 雅可比矩阵 [m, n]
        error: 误差向量 [m]
        q_max: 关节上限 [n]
        q_min: 关节下限 [n]
        dt: 时间步长
    """
    n = J.shape[1]
    
    # 目标函数: min 0.5 * x^T * P * x + q^T * x
    # x = q_dot
    # P = J^T * J + reg * I
    reg = 1e-4  # 正则化
    P = J.T @ J + reg * np.eye(n)
    
    # q = -J^T * error
    q = -J.T @ error
    
    # 约束: lb <= A * x <= ub
    # 关节限位: q_min <= q + q_dot * dt <= q_max
    A = np.eye(n) * dt
    
    lb = None
    ub = None
    if q_min is not None and q_max is not None:
        lb = q_min - np.zeros(n)  # 当前位置（简化）
        ub = q_max - np.zeros(n)
    
    # 求解
    q_dot = qpsolvers.solve_qp(
        P=P, q=q, A=A, lb=lb, ub=ub,
        solver="osqp",
    )
    
    return q_dot

# 使用示例
def ik_control(model, data, target_pos, body_name="left_ee_site", dt=0.02):
    """IK 控制器"""
    # 获取当前末端位置
    body_id = mujoco.mj_name2id(model, mujoco.mjtObj.mjOBJ_SITE, body_name)
    ee_pos = data.site_xpos[body_id]
    
    # 计算误差
    error_pos = target_pos - ee_pos
    
    # 计算雅可比
    jacp = np.zeros((3, model.nv))  # 位置雅可比
    jacr = np.zeros((3, model.nv))  # 旋转雅可比
    mujoco.mj_jacSite(model, data, jacp, jacr, body_id)
    
    # 只使用手臂关节（跳过自由关节和腿部）
    # 假设手臂关节从索引7开始
    arm_jacp = jacp[:, 7:7+7]  # 7个手臂关节
    
    # 求解
    q_dot = solve_ik_qp(arm_jacp, error_pos, dt=dt)
    
    return q_dot
```

### 7.1 完整 G1 仿真示例

```python
# 文件: g1_simulation.py
"""Unitree G1 机器人 MuJoCo 仿真完整示例"""
import mujoco
import mujoco.viewer
import numpy as np
import time
import argparse

class G1Simulation:
    """G1 机器人仿真管理器"""
    
    def __init__(self, model_path: str = "g1.xml"):
        # 加载模型
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.data = mujoco.MjData(self.model)
        
        # 设置仿真参数
        self.model.opt.timestep = 0.002  # 500Hz 仿真
        self.control_dt = 0.02  # 50Hz 控制
        self.control_steps = int(self.control_dt / self.model.opt.timestep)
        
        # 获取关键关节索引
        self._setup_joint_indices()
        
        # PD 控制器参数
        self.kp = np.full(self.model.nu, 100.0)  # 比例增益
        self.kd = np.full(self.model.nu, 5.0)    # 微分增益
        
        print(f"G1 模型加载完成")
        print(f"  自由度: {self.model.nq}")
        print(f"  执行器: {self.model.nu}")
        print(f"  刚体数: {self.model.nbody}")
    
    def _setup_joint_indices(self):
        """设置关节索引"""
        # 获取 keyframe
        self.home_keyframe = mujoco.mj_name2id(
            self.model, mujoco.mjtObj.mjOBJ_KEY, "home"
        )
        if self.home_keyframe >= 0:
            self.home_qpos = self.model.key_qpos[self.home_keyframe].copy()
        else:
            self.home_qpos = np.zeros(self.model.nq)
            # 设置自由关节（站立姿态）
            self.home_qpos[2] = 0.75  # 高度
            self.home_qpos[6] = 1.0   # 四元数 w
            # 设置关节默认位置
            self.home_qpos[7:] = 0.0
        
        # 关节名列表
        self.joint_names = []
        for i in range(self.model.njnt):
            name = mujoco.mj_id2name(self.model, mujoco.mjtObj.mjOBJ_JOINT, i)
            self.joint_names.append(name)
        
        # 腿部关节
        self.leg_joints = [j for j in self.joint_names if 'hip' in j or 'knee' in j or 'ankle' in j]
        # 手臂关节
        self.arm_joints = [j for j in self.joint_names if 'shoulder' in j or 'elbow' in j or 'wrist' in j]
        # 躯干关节
        self.torso_joints = [j for j in self.joint_names if 'waist' in j or 'torso' in j]
    
    def reset(self):
        """重置到默认姿态"""
        mujoco.mj_resetData(self.model, self.data)
        self.data.qpos[:] = self.home_qpos
        self.data.qvel[:] = 0
        mujoco.mj_forward(self.model, self.data)
    
    def pd_control(self, q_target, dq_target=None):
        """
        PD 位置控制
        
        参数:
            q_target: 目标关节位置 [nu]
            dq_target: 目标关节速度 [nu]，默认为0
        """
        if dq_target is None:
            dq_target = np.zeros(self.model.nu)
        
        # 当前关节位置和速度（跳过自由关节）
        q_current = self.data.qpos[7:]  # 跳过自由关节的7个值
        dq_current = self.data.qvel[6:]  # 跳过自由关节的6个值
        
        # PD 控制
        q_error = q_target[:self.model.nu] - q_current[:self.model.nu]
        dq_error = dq_target - dq_current[:self.model.nu]
        
        # 计算力矩
        torque = self.kp * q_error + self.kd * dq_error
        
        # 设置控制（位置执行器模式）
        self.data.ctrl[:] = q_target[:self.model.nu]
        
        return torque
    
    def get_com(self):
        """获取重心位置"""
        return self.data.subtree_com[1].copy()  # 第一个 subtree 是根
    
    def get_pelvis_state(self):
        """获取骨盆状态"""
        pelvis_id = mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_BODY, "pelvis")
        return {
            'pos': self.data.xpos[pelvis_id].copy(),
            'quat': self.data.xquat[pelvis_id].copy(),
            'vel': self.data.cvel[pelvis_id].copy(),
        }
    
    def apply_external_force(self, body_name, force, torque=None):
        """施加外力"""
        body_id = mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_BODY, body_name)
        self.data.xfrc_applied[body_id, :3] = force
        if torque is not None:
            self.data.xfrc_applied[body_id, 3:] = torque
    
    def run_with_viewer(self, duration: float = 60.0, control_mode: str = "hold"):
        """使用查看器运行仿真"""
        self.reset()
        
        with mujoco.viewer.launch_passive(self.model, self.data) as viewer:
            
            start_time = time.time()
            n_steps = int(duration / self.model.opt.timestep)
            
            for i in range(n_steps):
                if not viewer.is_running():
                    break
                
                # 控制逻辑
                if i % self.control_steps == 0:
                    if control_mode == "hold":
                        # 保持默认姿态
                        ctrl_target = self.home_qpos[7:7+self.model.nu]
                        self.pd_control(ctrl_target)
                    
                    elif control_mode == "wave":
                        # 挥手动作
                        t = self.data.time
                        ctrl_target = self.home_qpos[7:7+self.model.nu].copy()
                        # 找到右肩关节并设置正弦运动
                        for j, name in enumerate(self.joint_names):
                            if 'right_shoulder_pitch' in name:
                                ctrl_target[j] += 0.5 * np.sin(t * 2)
                            elif 'right_shoulder_roll' in name:
                                ctrl_target[j] += 0.5
                            elif 'right_elbow' in name:
                                ctrl_target[j] += 0.5 + 0.3 * np.sin(t * 2)
                        self.pd_control(ctrl_target)
                
                # 推进仿真
                mujoco.mj_step(self.model, self.data)
                
                # 更新查看器
                viewer.sync()
                
                # 实时控制
                elapsed = time.time() - start_time
                expected = (i + 1) * self.model.opt.timestep
                if elapsed < expected:
                    time.sleep(expected - elapsed)
                
                # 打印状态
                if i % 500 == 0:  # 每1秒
                    pelvis = self.get_pelvis_state()
                    com = self.get_com()
                    print(f"Time: {self.data.time:.1f}s | "
                          f"Pelvis: [{pelvis['pos'][0]:.2f}, {pelvis['pos'][1]:.2f}, {pelvis['pos'][2]:.2f}] | "
                          f"COM: [{com[0]:.2f}, {com[1]:.2f}, {com[2]:.2f}]")
    
    def record_video(self, output_path: str, duration: float = 10.0, fps: int = 30):
        """录制仿真视频"""
        import mediapy as media
        
        self.reset()
        renderer = mujoco.Renderer(self.model, height=480, width=640)
        
        frames = []
        n_steps = int(duration / self.model.opt.timestep)
        frame_interval = int(1.0 / (fps * self.model.opt.timestep))
        
        for i in range(n_steps):
            # 保持站立
            ctrl_target = self.home_qpos[7:7+self.model.nu]
            self.pd_control(ctrl_target)
            
            mujoco.mj_step(self.model, self.data)
            
            if i % frame_interval == 0:
                renderer.update_scene(self.data, camera="front")
                frames.append(renderer.render())
        
        media.write_video(output_path, frames, fps=fps)
        print(f"视频已保存到 {output_path}")


# === 运行 ===
if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="G1 MuJoCo 仿真")
    parser.add_argument("--model", type=str, default="g1.xml", help="模型文件路径")
    parser.add_argument("--mode", type=str, default="hold", 
                        choices=["hold", "wave"], help="控制模式")
    parser.add_argument("--duration", type=float, default=60.0, help="运行时长(秒)")
    parser.add_argument("--record", type=str, default=None, help="录制视频路径")
    args = parser.parse_args()
    
    sim = G1Simulation(args.model)
    
    if args.record:
        sim.record_video(args.record, duration=args.duration)
    else:
        sim.run_with_viewer(duration=args.duration, control_mode=args.mode)
```

```python
"""Teleopit MuJoCo 仿真后端"""
import mujoco
import mujoco.viewer
import numpy as np
from pathlib import Path

class TeleopitSim:

        将 BVH 帧数据应用到机器人
        
        参数:
            bvh_frame_data: 包含各关节旋转角度的字典
        """
        # 设置根关节位置
        root_pos = bvh_frame_data.get("root_position", np.zeros(3))
        root_quat = bvh_frame_data.get("root_rotation", np.array([1, 0, 0, 0]))
        
        self.data.qpos[0:3] = root_pos
        self.data.qpos[3:7] = root_quat

        for robot_joint, bvh_joint in self.bvh_to_robot_map.items():
            joint_id = mujoco.mj_name2id(
                self.model, mujoco.mjtObj.mjOBJ_JOINT, robot_joint
            )
            if joint_id >= 0:
                # 从 BVH 数据中提取对应角度
                angle = bvh_frame_data.get(robot_joint, 0.0)
                qpos_addr = self.model.jnt_qposadr[joint_id]
                self.data.qpos[qpos_addr] = angle
        
        # 前向运动学计算
        mujoco.mj_forward(self.model, self.data)
    
    def run_bvh_replay(self, bvh_file: str, loop: bool = True):
        """回放 BVH 动作"""
        from bvh_parser import BVHParser
        
        bvh = BVHParser(bvh_file)
        frames = bvh.get_frames()
        
        with mujoco.viewer.launch_passive(self.model, self.data) as viewer:
            while viewer.is_running():
                for frame in frames:
                    if not viewer.is_running():
                        break
                    
                    self.apply_bvh_frame(frame)
                    viewer.sync()
                    time.sleep(1.0 / bvh.frame_rate)
                
                if not loop:
                    break
```

### 8.2 somehand 中的 MuJoCo 使用

somehand 项目使用 MuJoCo 进行灵巧手运动重定向仿真：

```python
# 文件: somehand_mujoco.py
"""somehand 灵巧手 MuJoCo 仿真"""
import mujoco
import numpy as np
import yaml

class DexterousHandSim:
    """灵巧手仿真环境"""
    
    # 支持的灵巧手模型
    SUPPORTED_HANDS = [
        "linker_hand_l20", "linker_hand_l25", "linker_hand_l21",
        "unitree_dex5", "inspire_hand", "shadow_hand",
        "ability_hand", "allegro_hand", "leap_hand",
        "qhand", "pasa_hand", "scar_hand",
        # ... 更多手型
    ]
    
    def __init__(self, hand_type: str, config_path: str = None):
        """
        参数:
            hand_type: 灵巧手类型
            config_path: YAML 配置文件路径
        """
        self.hand_type = hand_type
        self.config = self._load_config(config_path)
        
        # 加载模型
        model_path = self.config["model_path"]
        self.model = mujoco.MjModel.from_xml_path(model_path)
        self.data = mujoco.MjData(self.model)
        
        # 关节映射
        self._setup_joint_mapping()
    
    def _load_config(self, config_path: str) -> dict:
        """加载 YAML 配置"""
        if config_path is None:
            config_path = f"configs/{self.hand_type}.yaml"
        
        with open(config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        return config
    
    def _setup_joint_mapping(self):
        """设置关节映射（MediaPipe -> 机器人关节）"""
        self.mediapipe_to_robot = {}
        
        for mp_joint, robot_joints in self.config["joint_mapping"].items():
            self.mediapipe_to_robot[mp_joint] = robot_joints
    
    def apply_hand_pose(self, mediapipe_landmarks: np.ndarray):
        """
        将 MediaPipe 手部关键点转换为关节角度并应用
        
        参数:
            mediapipe_landmarks: [21, 3] MediaPipe 21个关键点
        """
        # 计算各手指关节角度
        joint_angles = self._compute_joint_angles(mediapipe_landmarks)
        
        # 应用到机器人
        for joint_name, angle in joint_angles.items():
            joint_id = mujoco.mj_name2id(
                self.model, mujoco.mjtObj.mjOBJ_JOINT, joint_name
            )
            if joint_id >= 0:
                qpos_addr = self.model.jnt_qposadr[joint_id]
                self.data.qpos[qpos_addr] = angle
        
        mujoco.mj_forward(self.model, self.data)
    
    def _compute_joint_angles(self, landmarks: np.ndarray) -> dict:
        """从 MediaPipe 关键点计算关节角度"""
        angles = {}
        
        # 拇指
        angles.update(self._compute_finger_angles(
            landmarks, "thumb", [1, 2, 3, 4]
        ))
        # 食指
        angles.update(self._compute_finger_angles(
            landmarks, "index", [5, 6, 7, 8]
        ))
        # 中指
        angles.update(self._compute_finger_angles(
            landmarks, "middle", [9, 10, 11, 12]
        ))
        # 无名指
        angles.update(self._compute_finger_angles(
            landmarks, "ring", [13, 14, 15, 16]
        ))
        # 小指
        angles.update(self._compute_finger_angles(
            landmarks, "pinky", [17, 18, 19, 20]
        ))
        
        return angles
    
    def _compute_finger_angles(self, landmarks, finger_name, indices):
        """计算单根手指的关节角度"""
        angles = {}
        
        # 计算各关节弯曲角度（简化的2D角度计算）
        for i in range(len(indices) - 1):
            p1 = landmarks[indices[i]]
            p2 = landmarks[indices[i + 1]]
            
            # 计算弯曲角度（简化版）
            angle = np.arctan2(p2[1] - p1[1], abs(p2[0] - p1[0]))
            
            joint_name = f"{finger_name}_joint_{i+1}"
            if joint_name in self.mediapipe_to_robot:
                for robot_joint in self.mediapipe_to_robot[joint_name]:
                    angles[robot_joint] = angle
        
        return angles
```
