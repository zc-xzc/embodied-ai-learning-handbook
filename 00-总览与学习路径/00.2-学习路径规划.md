```python
import numpy as np

def forward_kinematics_2dof(theta1, theta2, l1=1.0, l2=1.0):
    """2-DOF 机械臂正运动学"""
    x = l1 * np.cos(theta1) + l2 * np.cos(theta1 + theta2)
    y = l1 * np.sin(theta1) + l2 * np.sin(theta1 + theta2)
    return np.array([x, y])

def inverse_kinematics_2dof(x, y, l1=1.0, l2=1.0):
    """2-DOF 机械臂逆运动学（解析解）"""
    cos_theta2 = (x**2 + y**2 - l1**2 - l2**2) / (2 * l1 * l2)
    if abs(cos_theta2) > 1:
        raise ValueError("目标位置超出工作空间")
    theta2 = np.arccos(cos_theta2)
    theta1 = np.arctan2(y, x) - np.arctan2(l2 * np.sin(theta2), l1 + l2 * np.cos(theta2))
    return np.array([theta1, theta2])

# 测试
target = np.array([1.5, 0.5])
angles = inverse_kinematics_2dof(*target)
pos = forward_kinematics_2dof(*angles)
print(f"目标位置: {target}")
print(f"逆解角度: {np.degrees(angles)} 度")
print(f"验证位置: {pos}")
print(f"误差: {np.linalg.norm(pos - target):.6f}")
```

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class SelfAttention(nn.Module):
    def __init__(self, embed_dim, num_heads):
        super().__init__()
        self.embed_dim = embed_dim
        self.num_heads = num_heads
        self.head_dim = embed_dim // num_heads
        
        self.q_proj = nn.Linear(embed_dim, embed_dim)
        self.k_proj = nn.Linear(embed_dim, embed_dim)
        self.v_proj = nn.Linear(embed_dim, embed_dim)
        self.out_proj = nn.Linear(embed_dim, embed_dim)
    
    def forward(self, x):
        B, N, C = x.shape
        q = self.q_proj(x).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(x).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(x).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        
        attn = (q @ k.transpose(-2, -1)) / math.sqrt(self.head_dim)
        attn = F.softmax(attn, dim=-1)
        out = (attn @ v).transpose(1, 2).contiguous().view(B, N, C)
        return self.out_proj(out)

# 测试
attn = SelfAttention(embed_dim=256, num_heads=8)
x = torch.randn(2, 10, 256)  # batch=2, seq_len=10, dim=256
out = attn(x)
print(f"输入形状: {x.shape}, 输出形状: {out.shape}")
```

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
import numpy as np

class BCDataset(Dataset):
    """模仿学习数据集"""
    def __init__(self, observations, actions):
        self.observations = torch.FloatTensor(observations)
        self.actions = torch.FloatTensor(actions)
    
    def __len__(self):
        return len(self.observations)
    
    def __getitem__(self, idx):
        return self.observations[idx], self.actions[idx]

class BCPolicy(nn.Module):
    """简单的 BC 策略网络"""
    def __init__(self, obs_dim, action_dim, hidden_dim=256):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim)
        )
    
    def forward(self, obs):
        return self.net(obs)

def train_bc(observations, actions, epochs=100, lr=1e-3):
    """训练 BC 策略"""
    dataset = BCDataset(observations, actions)
    dataloader = DataLoader(dataset, batch_size=64, shuffle=True)
    
    policy = BCPolicy(obs_dim=observations.shape[1], action_dim=actions.shape[1])
    optimizer = optim.Adam(policy.parameters(), lr=lr)
    criterion = nn.MSELoss()
    
    for epoch in range(epochs):
        total_loss = 0
        for obs_batch, action_batch in dataloader:
            pred_actions = policy(obs_batch)
            loss = criterion(pred_actions, action_batch)
            
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            total_loss += loss.item()
        
        if (epoch + 1) % 10 == 0:
            print(f"Epoch {epoch+1}/{epochs}, Loss: {total_loss/len(dataloader):.6f}")
    
    return policy

# 模拟数据训练
np.random.seed(42)
obs = np.random.randn(1000, 10)  # 1000条观测，维度10
actions = np.random.randn(1000, 4)  # 1000条动作，维度4
policy = train_bc(obs, actions)
print("BC 训练完成！")
```

**ACT 核心思想理解**：
```python
import torch
import torch.nn as nn

class ActionChunker:
    """Action Chunking 概念示例"""
    def __init__(self, chunk_size=10):
        self.chunk_size = chunk_size
    
    def predict_chunk(self, policy, obs):
        """一次性预测 chunk_size 步动作"""
        with torch.no_grad():
            # 策略输出一个动作序列而非单步动作
            action_chunk = policy(obs)  # shape: (chunk_size, action_dim)
        return action_chunk
    
    def execute_with_receding_horizon(self, policy, env, total_steps=100):
        """使用 receding horizon 执行"""
        obs = env.reset()
        executed = 0
        while executed < total_steps:
            action_chunk = self.predict_chunk(policy, obs)
            # 只执行前 k 步，然后重新预测
            k = self.chunk_size // 2  # 执行一半后重新预测
            for i in range(k):
                obs, reward, done, info = env.step(action_chunk[i])
                executed += 1
                if done or executed >= total_steps:
                    break
        return executed
```

```python
"""
端到端 VLA 项目模板
1. 加载预训练 VLA 模型
2. 在仿真环境中运行
3. 记录成功率和轨迹质量
"""
import numpy as np
from collections import deque

class VLAEvaluationPipeline:
    def __init__(self, model, processor, env):
        self.model = model
        self.processor = processor
        self.env = env
        self.success_count = 0
        self.total_episodes = 0
    
    def run_episode(self, instruction, max_steps=200):
        """运行一个 episode"""
        obs = self.env.reset()
        for step in range(max_steps):
            # 获取当前帧图像
            image = self.env.render()
            
            # VLA 预测动作
            prompt = f"In: What action should the robot take to {instruction}?\nOut:"
            inputs = self.processor(prompt, image).to("cuda")
            action = self.model.predict_action(**inputs, do_sample=False)
            
            # 执行动作
            obs, reward, done, info = self.env.step(action)
            
            if done:
                self.success_count += 1
                return True
        
        return False
    
    def evaluate(self, instruction, num_episodes=20):
        """评估多个 episode"""
        self.success_count = 0
        self.total_episodes = num_episodes
        for i in range(num_episodes):
            success = self.run_episode(instruction)
            print(f"Episode {i+1}: {'成功' if success else '失败'}")
        
        success_rate = self.success_count / self.total_episodes
        print(f"\n成功率: {success_rate:.2%} ({self.success_count}/{self.total_episodes})")
        return success_rate
```

```python
import numpy as np

class GraspingReward:
    """抓取任务奖励函数设计"""
    def __init__(self, target_pos, reach_threshold=0.05):
        self.target_pos = np.array(target_pos)
        self.reach_threshold = reach_threshold
    
    def compute(self, ee_pos, object_pos, gripper_state, success):
        """
        ee_pos: 末端执行器位置 (3,)
        object_pos: 物体位置 (3,)
        gripper_state: 夹爪状态 (0=open, 1=closed)
        success: 是否成功抓取
        """
        reward = 0.0
        
        # 1. 接近物体奖励（密集奖励）
        dist_to_object = np.linalg.norm(ee_pos - object_pos)
        reach_reward = -dist_to_object  # 距离越近奖励越高
        reward += reach_reward * 1.0
        
        # 2. 接触奖励
        if dist_to_object < self.reach_threshold:
            reward += 2.0  # 接触奖励
            # 3. 闭合夹爪奖励
            if gripper_state > 0.5:
                reward += 5.0
        
        # 4. 抓起物体奖励
        if success:
            reward += 20.0  # 成功奖励
        
        # 5. 时间惩罚
        reward -= 0.01  # 鼓励快速完成
        
        return reward

# 课程学习示例
class CurriculumGrasping:
    """渐进式课程学习"""
    def __init__(self):
        self.level = 0
        self.success_history = []
    
    def get_task_params(self):
        """根据当前等级返回任务参数"""
        params = {
            0: {"object_distance": 0.1, "noise": 0.0},    # 近距离，无噪声
            1: {"object_distance": 0.2, "noise": 0.01},   # 中距离，小噪声
            2: {"object_distance": 0.3, "noise": 0.05},   # 远距离，中噪声
            3: {"object_distance": 0.5, "noise": 0.1},    # 随机位置，大噪声
        }
        return params[min(self.level, 3)]
    
    def update_level(self, success_rate):
        """根据成功率更新难度"""
        self.success_history.append(success_rate)
        if len(self.success_history) > 10:
            recent_rate = np.mean(self.success_history[-10:])
            if recent_rate > 0.7 and self.level < 3:
                self.level += 1
                print(f"课程升级！当前等级: {self.level}")
                self.success_history = []
```

```python
import numpy as np

class DomainRandomizer:
    """域随机化配置"""
    def __init__(self):
        self.params = {
            # 物理参数随机化
            "friction": (0.3, 1.5),       # 摩擦系数
            "mass_scale": (0.8, 1.2),     # 质量缩放
            "damping": (0.1, 0.5),        # 阻尼
            "stiffness": (50, 150),       # 刚度
            
            # 视觉参数随机化
            "light_pos": lambda: np.random.uniform(-2, 2, 3),
            "light_intensity": (0.5, 1.5),
            "texture_color": lambda: np.random.rand(3),
            "camera_pos_noise": 0.02,
            
            # 动力学参数
            "control_delay": (0, 3),      # 控制延迟（步数）
            "action_noise": 0.05,         # 动作噪声
        }
    
    def sample(self):
        """随机采样一组参数"""
        config = {}
        for key, val in self.params.items():
            if callable(val):
                config[key] = val()
            elif isinstance(val, tuple):
                config[key] = np.random.uniform(*val)
            else:
                config[key] = val
        return config
```

**Teacher-Student 框架**：
```python
"""
Teacher-Student 蒸馏框架
- Teacher: 在仿真中有特权信息（privilege info），如物体精确位置
- Student: 部署时只有本体感知和视觉，通过蒸馏学习 Teacher 的策略
"""
import torch
import torch.nn as nn

class TeacherPolicy(nn.Module):
    """Teacher: 使用特权信息"""
    def __init__(self, obs_dim, priv_dim, action_dim):
        super().__init__()
        # 输入 = 本体感知 + 特权信息
        input_dim = obs_dim + priv_dim
        self.net = nn.Sequential(
            nn.Linear(input_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 256),
            nn.ReLU(),
            nn.Linear(256, action_dim)
        )
    
    def forward(self, obs, priv_info):
        x = torch.cat([obs, priv_info], dim=-1)
        return self.net(x)

class StudentPolicy(nn.Module):
    """Student: 仅使用可观测信息"""
    def __init__(self, obs_dim, action_dim):
        super().__init__()
        # 输入 = 仅本体感知（可能含历史信息）
        self.net = nn.Sequential(
            nn.Linear(obs_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 256),
            nn.ReLU(),
            nn.Linear(256, action_dim)
        )
    
    def forward(self, obs):
        return self.net(obs)

def distill(teacher, student, dataloader, epochs=100):
    """蒸馏训练"""
    optimizer = torch.optim.Adam(student.parameters(), lr=1e-3)
    criterion = nn.MSELoss()
    
    for epoch in range(epochs):
        for obs, priv_info in dataloader:
            with torch.no_grad():
                teacher_action = teacher(obs, priv_info)
            
            student_action = student(obs)
            loss = criterion(student_action, teacher_action)
            
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
        
        if (epoch + 1) % 10 == 0:
            print(f"Epoch {epoch+1}, Distill Loss: {loss.item():.6f}")
```

```python
简化的人形机器人运动跟踪策略
跟踪目标关节角度、速度、角速度
"""
import torch
import torch.nn as nn

class MotionTrackingPolicy(nn.Module):
    """运动跟踪策略（Student 版本）"""
    def __init__(self, obs_dim=72, action_dim=12):
        super().__init__()
        # 观测: 本体感知(关节角, 速度, 角速度) + 历史信息
        self.actor = nn.Sequential(
            nn.Linear(obs_dim, 512),
            nn.ELU(),
            nn.Linear(512, 256),
            nn.ELU(),
            nn.Linear(256, 128),
            nn.ELU(),
            nn.Linear(128, action_dim)
        )
        # 输出动作的 log_std 用于 SAC
        self.log_std = nn.Parameter(torch.zeros(action_dim))
    
    def forward(self, obs):
        mean = self.actor(obs)
        std = torch.exp(self.log_std)
        return mean, std

# 运动跟踪奖励
def motion_tracking_reward(current_state, target_state):
    """
    current_state: 当前关节状态 (角度, 速度)
    target_state: 目标关节状态（来自 mocap 数据）
    """
    joint_pos_error = torch.sum((current_state[:, :6] - target_state[:, :6])**2, dim=1)
    joint_vel_error = torch.sum((current_state[:, 6:] - target_state[:, 6:])**2, dim=1)
    
    reward = torch.exp(-2.0 * joint_pos_error) + 0.5 * torch.exp(-0.1 * joint_vel_error)
    return reward
```

```python
"""
策略部署框架：从仿真到真机
"""
import numpy as np
import time

class PolicyDeployer:
    """策略部署器"""
    def __init__(self, policy, robot_interface, safety_checker):
        self.policy = policy
        self.robot = robot_interface
        self.safety = safety_checker
        self.action_history = []
        self.obs_history = []
        self.history_len = 10  # 历史观测长度
    
    def get_observation(self):
        """获取观测（含历史）"""
        obs = self.robot.get_state()
        self.obs_history.append(obs)
        if len(self.obs_history) > self.history_len:
            self.obs_history.pop(0)
        
        # 拼接历史观测
        obs_with_history = np.concatenate(self.obs_history)
        return obs_with_history
    
    def predict_action(self, obs):
        """策略预测动作"""
        obs_tensor = torch.FloatTensor(obs).unsqueeze(0).to("cuda")
        with torch.no_grad():
            mean, std = self.policy(obs_tensor)
        action = mean.cpu().numpy().flatten()
        return action
    
    def safety_filter(self, action):
        """安全过滤"""
        # 1. 动作限幅
        action = np.clip(action, self.robot.action_low, self.robot.action_high)
        
        # 2. 动作平滑（低通滤波）
        if len(self.action_history) > 0:
            alpha = 0.7  # 平滑系数
            action = alpha * action + (1 - alpha) * self.action_history[-1]
        
        # 3. 速度限制
        if len(self.action_history) > 0:
            max_change = 0.1  # 每步最大变化
            delta = action - self.action_history[-1]
            delta = np.clip(delta, -max_change, max_change)
            action = self.action_history[-1] + delta
        
        # 4. 紧急停止检查
        if self.safety.check_emergency(self.robot.get_state()):
            return None  # 返回 None 表示紧急停止
        
        return action
    
    def run(self, max_steps=1000):
        """运行策略"""
        for step in range(max_steps):
            obs = self.get_observation()
            action = self.predict_action(obs)
            action = self.safety_filter(action)
            
            if action is None:
                print(f"Step {step}: 紧急停止！")
                break
            
            self.robot.send_action(action)
            self.action_history.append(action)
            time.sleep(0.02)  # 50Hz 控制频率
        
        print("部署结束")
```

```python
import numpy as np
import torch
from torchvision import transforms

class DataPreprocessor:
    """数据预处理管线"""
    def __init__(self, action_dim=6, image_size=(224, 224)):
        self.action_dim = action_dim
        self.image_size = image_size
        
        # 图像增强（仅训练时使用）
        self.train_transform = transforms.Compose([
            transforms.Resize(image_size),
            transforms.ColorJitter(brightness=0.2, contrast=0.2, saturation=0.2, hue=0.1),
            transforms.RandomGrayscale(p=0.1),
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
        ])
        
        # 推理时的变换（无随机增强）
        self.eval_transform = transforms.Compose([
            transforms.Resize(image_size),
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
        ])
    
    def normalize_actions(self, actions, stats=None):
        """动作归一化到 [-1, 1]"""
        if stats is None:
            stats = {
                "min": actions.min(dim=0)[0],
                "max": actions.max(dim=0)[0]
            }
        normalized = 2 * (actions - stats["min"]) / (stats["max"] - stats["min"] + 1e-8) - 1
        return normalized, stats
    
    def denormalize_actions(self, normalized_actions, stats):
        """反归一化"""
        return (normalized_actions + 1) / 2 * (stats["max"] - stats["min"]) + stats["min"]
    
    def smooth_actions(self, actions, window_size=5):
        """动作平滑（移动平均）"""
        if len(actions) < window_size:
            return actions
        kernel = torch.ones(1, 1, window_size) / window_size
        smoothed = torch.zeros_like(actions)
        for i in range(actions.shape[-1]):
            smoothed[:, i] = torch.nn.functional.conv1d(
                actions[:, i].unsqueeze(0).unsqueeze(0),
                kernel, padding=window_size // 2
            ).squeeze()
        return smoothed

def filter_episodes(episodes, min_length=20, max_jerk=5.0):
    """过滤异常 episode"""
    valid_episodes = []
    for ep in episodes:
        # 1. 检查长度
        if len(ep["actions"]) < min_length:
            continue
        
        # 2. 检查动作平滑度（jerk 过大说明有异常跳变）
        actions = ep["actions"]
        jerk = np.diff(actions, n=2, axis=0)
        max_jerk_val = np.max(np.abs(jerk))
        if max_jerk_val > max_jerk:
            continue
        
        valid_episodes.append(ep)
    
    print(f"过滤前: {len(episodes)} episodes, 过滤后: {len(valid_episodes)} episodes")
    return valid_episodes
```

```python
def evaluate_act(policy, env, num_episodes=20):
    """评估 ACT 策略"""
    successes = 0
    total_steps = 0
    
    for ep in range(num_episodes):
        obs = env.reset()
        done = False
        steps = 0
        
        while not done and steps < 200:
            # ACT 预测一个 action chunk
            with torch.no_grad():
                action_chunk = policy.predict(obs)
            
            # 执行 chunk（receding horizon）
            for action in action_chunk[:5]:  # 执行前 5 步
                obs, reward, done, info = env.step(action)
                steps += 1
                if done:
                    break
        
        if info.get("success", False):
            successes += 1
        total_steps += steps
    
    success_rate = successes / num_episodes
    avg_steps = total_steps / num_episodes
    print(f"成功率: {success_rate:.2%}")
    print(f"平均完成步数: {avg_steps:.1f}")
    return success_rate, avg_steps
```

```python
class PolicyEvaluator:
    """策略评估器"""
    def __init__(self, env, num_episodes=50, max_steps=300):
        self.env = env
        self.num_episodes = num_episodes
        self.max_steps = max_steps
    
    def evaluate(self, policy):
        """全面评估策略"""
        results = {
            "success_rate": 0,
            "avg_completion_time": 0,
            "avg_path_length": 0,
            "failure_modes": {},
            "per_episode": []
        }
        
        successes = 0
        completion_times = []
        path_lengths = []
        
        for ep in range(self.num_episodes):
            obs = self.env.reset()
            done = False
            steps = 0
            path_length = 0
            prev_pos = obs.get("robot_pos", None)
            
            while not done and steps < self.max_steps:
                action = policy.predict(obs)
                obs, reward, done, info = self.env.step(action)
                steps += 1
                
                # 路径长度
                curr_pos = obs.get("robot_pos", None)
                if prev_pos is not None and curr_pos is not None:
                    path_length += np.linalg.norm(curr_pos - prev_pos)
                prev_pos = curr_pos
            
            success = info.get("success", False)
            if success:
                successes += 1
                completion_times.append(steps)
                path_lengths.append(path_length)
            else:
                # 记录失败模式
                failure_reason = self._classify_failure(info, steps, obs)
                results["failure_modes"][failure_reason] = \
                    results["failure_modes"].get(failure_reason, 0) + 1
            
            results["per_episode"].append({
                "episode": ep,
                "success": success,
                "steps": steps,
                "path_length": path_length,
            })
        
        results["success_rate"] = successes / self.num_episodes
        results["avg_completion_time"] = np.mean(completion_times) if completion_times else 0
        results["avg_path_length"] = np.mean(path_lengths) if path_lengths else 0
        
        self._print_report(results)
        return results
    
    def _classify_failure(self, info, steps, obs):
        """分类失败原因"""
        if steps >= self.max_steps:
            return "timeout"
        if info.get("collision", False):
            return "collision"
        if info.get("dropped_object", False):
            return "dropped_object"
        return "other"
    
    def _print_report(self, results):
        """打印评估报告"""
        print("=" * 50)
        print("策略评估报告")
        print("=" * 50)
        print(f"成功率: {results['success_rate']:.2%}")
        print(f"平均完成步数: {results['avg_completion_time']:.1f}")
        print(f"平均路径长度: {results['avg_path_length']:.3f}")
        print("\n失败模式分布:")
        for mode, count in results["failure_modes"].items():
            print(f"  {mode}: {count} 次 ({count/self.num_episodes:.1%})")
```

```python
"""
数据飞轮流程：
1. 用当前策略自主执行任务
2. 人在旁边监督，失败时介入干预（intervention）
3. 将成功 episode 加入数据集
4. 重新训练策略
5. 重复
"""

class DataFlywheel:
    def __init__(self, policy, env, dataset_path):
        self.policy = policy
        self.env = env
        self.dataset_path = dataset_path
        self.iteration = 0
    
    def collect_with_intervention(self, num_episodes=20):
        """带干预的数据采集"""
        new_data = []
        
        for ep in range(num_episodes):
            obs = self.env.reset()
            trajectory = {"observations": [], "actions": []}
            done = False
            steps = 0
            interventions = 0
            
            while not done and steps < 300:
                # 策略预测动作
                action = self.policy.predict(obs)
                
                # 检查是否需要干预
                if self._needs_intervention(obs, action):
                    # 人工接管
                    human_action = self._get_human_action()
                    action = human_action
                    interventions += 1
                
                trajectory["observations"].append(obs)
                trajectory["actions"].append(action)
                
                obs, _, done, info = self.env.step(action)
                steps += 1
            
            # 只保存成功的 episode
            if info.get("success", False):
                new_data.append(trajectory)
                print(f"Episode {ep}: 成功 (干预 {interventions} 次)")
            else:
                print(f"Episode {ep}: 失败")
        
        print(f"\n收集了 {len(new_data)} 个成功 episode")
        return new_data
    
    def _needs_intervention(self, obs, action):
        """判断是否需要干预"""
        # 简单的启发式规则
        # 1. 动作过大
        if np.max(np.abs(action)) > 0.95:
            return True
        # 2. 动作变化过大
        # 3. 位置接近边界
        # ... 实际中需要更精细的判断
        return False
    
    def _get_human_action(self):
        """获取人工干预动作"""
```
