## 1. 灵巧手概述与分类

### 1.1 什么是灵巧手

灵巧手是模仿人手结构和功能的机器人末端执行器，具备多指、多自由度的精细操作能力。与简单的二指夹爪相比，灵巧手可以执行更复杂的操作任务，如：

- 精细抓取（捏取、侧握、包裹握持）
- 手内操作（In-Hand Manipulation）
- 工具使用（使用螺丝刀、剪刀等）
- 手势表达

### 1.2 分类维度

#### 按自由度分类

| 类型 | 自由度 | 特点 | 代表产品 |
|------|--------|------|----------|
| 简易夹爪 | 1-2 DoF | 结构简单、成本低 | Robotiq 2F-85 |
| 三指手 | 3-10 DoF | 适度灵活、工业应用 | Robotiq 3F |
| 五指欠驱动 | 5-15 DoF | 自适应抓取 | Inspire RH56 |
| 五指全驱动 | 16-24 DoF | 精细操作、研究 | Shadow Hand |
| 拟人化手 | 20+ DoF | 最接近人手 | LinkerHand L20 |

#### 按驱动方式分类

| 驱动方式 | 说明 | 优点 | 缺点 |
|----------|------|------|------|
| 电机直驱 | 电机内置关节 | 结构紧凑 | 力量有限 |
| 腱传动 | 钢丝绳远程驱动 | 拟人化高 | 维护复杂 |
| 气动 | 气动人工肌肉 | 柔顺性好 | 控制精度低 |
| 液压 | 液压驱动 | 力量大 | 体积大 |

#### 按感知能力分类

| 类型 | 传感器 | 特点 |
|------|--------|------|
| 无感知 | 无 | 仅位置控制 |
| 力感知 | 关节力矩传感器 | 力控精度高 |
| 触觉感知 | 指尖压力传感器 | 可感知接触 |
| 视触觉 | 视觉触觉传感器（如 GelSight） | 高分辨率触觉成像 |

### 1.3 人手参考结构

```
人手骨骼结构 (参考):

     指尖 (Distal Phalanx)
      │
      │  DIP 关节 (远端指间) ─ 1 DoF
      │
     中节 (Middle Phalanx)
      │
      │  PIP 关节 (近端指间) ─ 1 DoF
      │
     近节 (Proximal Phalanx)
      │
      │  MCP 关节 (掌指) ─ 2 DoF (屈伸 + 侧摆)
      │
     掌骨 (Metacarpal)

每根手指: 4 DoF (2+1+1)
拇指特殊: 5 DoF (CMC 2 + MCP 2 + IP 1)
总计: 4 × 4 + 5 = 21 DoF (单手)
```

```python
#!/usr/bin/env python3
"""LinkerHand 控制示例"""

import numpy as np
import time

class LinkerHandController:
    """LinkerHand 灵巧手控制器"""
    
    # L20 手指名称 (20 DoF)
    FINGER_NAMES = [
        # 拇指 (5 DoF)
        'thumb_cmc_abduction', 'thumb_cmc_flexion',
        'thumb_mcp_abduction', 'thumb_mcp_flexion', 'thumb_ip',
        # 食指 (4 DoF)
        'index_mcp_abduction', 'index_mcp_flexion',
        'index_pip', 'index_dip',
        # 中指 (4 DoF)
        'middle_mcp_abduction', 'middle_mcp_flexion',
        'middle_pip', 'middle_dip',
        # 无名指 (4 DoF)
        'ring_mcp_abduction', 'ring_mcp_flexion',
        'ring_pip', 'ring_dip',
        # 小指 (3 DoF)
        'pinky_mcp_flexion', 'pinky_pip', 'pinky_dip',
    ]
    
    NUM_DOFS = 20
    
    def __init__(self, hand_type='right', interface='can0'):
        """
        Args:
            hand_type: 'left' 或 'right'
            interface: CAN 接口名
        """
        self.hand_type = hand_type
        self.interface = interface
        
        # 初始化 CAN 通信
        # from linkerhand import LinkerHandAPI
        # self.hand = LinkerHandAPI(hand_type=hand_type, interface=interface)
        
        print(f"LinkerHand {hand_type} 已初始化 (接口: {interface})")
        
        # 当前位置
        self.current_pos = np.zeros(self.NUM_DOFS)
    
    def send_positions(self, positions, speed=0.5):
        """
        发送手指位置命令
        
        Args:
            positions: [NUM_DOFS] 弧度值 [0, pi/2]
            speed: 运动速度 [0, 1]
        """
        assert len(positions) == self.NUM_DOFS, \
            f"位置维度应为 {self.NUM_DOFS}，实际为 {len(positions)}"
        
        # 限位
        positions = np.clip(positions, 0, np.pi / 2)
        
        # 发送命令
        # self.hand.set_positions(positions.tolist())
        self.current_pos = positions
    
    def get_positions(self):
        """获取当前手指位置"""
        # return np.array(self.hand.get_positions())
        return self.current_pos
    
    def get_torques(self):
        """获取力矩反馈"""
        # return np.array(self.hand.get_torques())
        return np.zeros(self.NUM_DOFS)
    
    def open_hand(self, speed=0.5):
        """张开手掌"""
        target = np.zeros(self.NUM_DOFS)
        self.move_to(target, speed=speed)
    
    def close_hand(self, speed=0.5):
        """握拳"""
        target = np.full(self.NUM_DOFS, np.pi / 2 * 0.8)
        self.move_to(target, speed=speed)
    
    def pinch_grasp(self, speed=0.5):
        """捏取姿态（拇指 + 食指）"""
        target = np.zeros(self.NUM_DOFS)
        # 拇指弯曲
        target[1] = 0.8  # thumb_cmc_flexion
        target[4] = 0.9  # thumb_ip
        # 食指弯曲
        target[6] = 0.7  # index_mcp_flexion
        target[7] = 0.8  # index_pip
        target[8] = 0.7  # index_dip
        self.move_to(target, speed=speed)
    
    def move_to(self, target, speed=0.5, freq=100):
        """平滑移动到目标位置"""
        start = self.current_pos.copy()
        num_steps = int(50 / speed)
        dt = 1.0 / freq
        
        for i in range(num_steps + 1):
            alpha = i / num_steps
            alpha_smooth = 0.5 * (1 - np.cos(np.pi * alpha))
            pos = start + (target - start) * alpha_smooth
            self.send_positions(pos)
            time.sleep(dt)
    
    def emergency_stop(self):
        """紧急停止"""
        self.send_positions(self.current_pos, speed=1.0)
        print("LinkerHand 紧急停止")
```

Unitree Dex5 是宇树科技推出的五指灵巧手，采用内置电机直驱方式。

| 参数 | Dex5 |
|------|------|

```python
#!/usr/bin/env python3
"""Unitree Dex5 灵巧手控制"""

import numpy as np
from unitree_sdk2py.core.channel.channel_publisher import ChannelPublisher
from unitree_sdk2py.core.channel.channel_subscriber import ChannelSubscriber
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import HandCmd_, HandState_

class Dex5Controller:
    """Unitree Dex5 灵巧手控制器"""
    
    NUM_DOFS = 16  # 单手
    
    def __init__(self, hand_side='left'):
        self.hand_side = hand_side
        
        # DDS Topic
        topic_cmd = f"rt/handcmd_{hand_side}"
        topic_state = f"rt/handstate_{hand_side}"
        
        self.cmd_pub = ChannelPublisher(topic_cmd, HandCmd_)
        self.cmd_pub.Init()
        
        self.state_sub = ChannelSubscriber(topic_state, HandState_)
        self.state_sub.Init()
        
        self.latest_state = None
        self.state_sub.RegisterCallback(self._state_callback)
    
    def _state_callback(self, msg):
        self.latest_state = msg
    
    def send_positions(self, positions):
        """发送手指位置命令"""
        cmd = HandCmd_()
        for i in range(self.NUM_DOFS):
            cmd.motor_cmd[i].q = float(positions[i])
            cmd.motor_cmd[i].kp = 50.0
            cmd.motor_cmd[i].kd = 2.0
        self.cmd_pub.Write(cmd)
    
    def get_state(self):
        if self.latest_state is None:
            return None
        positions = [self.latest_state.motor_state[i].q 
                     for i in range(self.NUM_DOFS)]
        return np.array(positions)
```

```python
#!/usr/bin/env python3
"""Inspire 灵巧手控制"""

import serial
import numpy as np

class InspireHandController:
    """Inspire RH56 灵巧手控制器"""
    
    NUM_ACTUATORS = 6  # 6 个主动驱动
    
    def __init__(self, port='/dev/ttyUSB0', baudrate=115200):
        self.serial = serial.Serial(port, baudrate, timeout=1.0)
        print(f"Inspire 手已连接: {port}")
    
    def send_positions(self, positions):
        """
        发送位置命令
        
        Args:
            positions: [6] 角度值 [0, 1000]
                0: 拇指对掌
                1: 食指弯曲
                2: 中指弯曲
                3: 无名指弯曲
                4: 小指弯曲
                5: 拇指弯曲
        """
        # 构建串口命令帧
        cmd = self._build_command(positions)
        self.serial.write(cmd)
    
    def _build_command(self, positions):
        """构建命令帧"""
        # Inspire 协议帧格式
        frame = bytearray()
        frame.append(0xEB)  # 帧头
        frame.append(0x90)  # 帧头
        frame.append(0x04)  # 数据长度
        frame.append(0x02)  # 命令类型: 位置控制
        
        for pos in positions:
            pos_int = int(np.clip(pos, 0, 1000))
            frame.append((pos_int >> 8) & 0xFF)  # 高字节
            frame.append(pos_int & 0xFF)          # 低字节
        
        # 校验和
        checksum = sum(frame[2:]) & 0xFF
        frame.append(checksum)
        
        return bytes(frame)
    
    def close(self):
        self.serial.close()
```

## 3. 运动重定向方法

### 3.1 什么是运动重定向

运动重定向（Motion Retargeting）是将人体手部运动映射到机器人灵巧手的过程。由于人手和机器人手的骨骼结构、自由度数量和关节范围不同，需要通过重定向算法进行适配。

```
人手运动 → [重定向算法] → 机器人灵巧手运动
                  │
                  ├── 骨骼映射 (Skeleton Mapping)
                  ├── 关节角度映射 (Joint Angle Mapping)
                  ├── 比例缩放 (Scaling)
                  └── 范围裁剪 (Clipping)
```

### 3.2 重定向方法分类

#### 3.2.1 基于关键点的重定向

```python
#!/usr/bin/env python3
"""基于关键点的运动重定向"""

import numpy as np

class KeypointRetargeter:
    """基于 MediaPipe 21 关键点的重定向"""
    
    # MediaPipe 21 个手部关键点
    HAND_LANDMARKS = [
        'wrist',
        'thumb_cmc', 'thumb_mcp', 'thumb_ip', 'thumb_tip',
        'index_mcp', 'index_pip', 'index_dip', 'index_tip',
        'middle_mcp', 'middle_pip', 'middle_dip', 'middle_tip',
        'ring_mcp', 'ring_pip', 'ring_dip', 'ring_tip',
        'pinky_mcp', 'pinky_pip', 'pinky_dip', 'pinky_tip',
    ]
    
    def __init__(self, hand_config):
        """
        Args:
            hand_config: 灵巧手配置（自由度、关节范围等）
        """
        self.config = hand_config
        self.num_dofs = hand_config['num_dofs']
        self.joint_limits = hand_config['joint_limits']
    
    def retarget(self, landmarks):
        """
        将 MediaPipe 关键点重定向为灵巧手关节角度
        
        Args:
            landmarks: [21, 3] 关键点 3D 坐标
        
        Returns:
            joint_angles: [num_dofs] 关节角度 (rad)
        """
        joint_angles = np.zeros(self.num_dofs)
        
        # 计算每根手指的弯曲角度
        # 拇指
        joint_angles[0] = self._calc_thumb_abduction(landmarks)
        joint_angles[1] = self._calc_finger_curvature(
            landmarks, [1, 2, 3, 4]  # thumb_cmc -> tip
        )
        
        # 食指
        joint_angles[2] = self._calc_finger_spread(landmarks, 'index')
        joint_angles[3] = self._calc_finger_curvature(
            landmarks, [5, 6, 7, 8]  # index_mcp -> tip
        )
        
        # 中指
        joint_angles[4] = self._calc_finger_spread(landmarks, 'middle')
        joint_angles[5] = self._calc_finger_curvature(
            landmarks, [9, 10, 11, 12]
        )
        
        # 无名指
        joint_angles[6] = self._calc_finger_spread(landmarks, 'ring')
        joint_angles[7] = self._calc_finger_curvature(
            landmarks, [13, 14, 15, 16]
        )
        
        # 小指
        joint_angles[8] = self._calc_finger_spread(landmarks, 'pinky')
        joint_angles[9] = self._calc_finger_curvature(
            landmarks, [17, 18, 19, 20]
        )
        
        # 范围裁剪
        for i in range(self.num_dofs):
            lo, hi = self.joint_limits[i]
            joint_angles[i] = np.clip(joint_angles[i], lo, hi)
        
        return joint_angles
    
    def _calc_finger_curvature(self, landmarks, indices):
        """计算手指弯曲度（基于关节角度）"""
        p0 = landmarks[indices[0]]
        p1 = landmarks[indices[1]]
        p2 = landmarks[indices[2]]
        p3 = landmarks[indices[3]]
        
        # 计算 MCP 和 PIP 的关节角度
        angle_mcp = self._calc_angle(p0, p1, p2)
        angle_pip = self._calc_angle(p1, p2, p3)
        
        # 总弯曲度
        curvature = (np.pi - angle_mcp) + (np.pi - angle_pip)
        
        return curvature
    
    def _calc_angle(self, p1, p2, p3):
        """计算三点之间的角度"""
        v1 = p1 - p2
        v2 = p3 - p2
        cos_angle = np.dot(v1, v2) / (
            np.linalg.norm(v1) * np.linalg.norm(v2) + 1e-8
        )
        return np.arccos(np.clip(cos_angle, -1, 1))
    
    def _calc_thumb_abduction(self, landmarks):
        """计算拇指外展角度"""
        wrist = landmarks[0]
        thumb_mcp = landmarks[2]
        index_mcp = landmarks[5]
        
        v1 = thumb_mcp - wrist
        v2 = index_mcp - wrist
        angle = np.arccos(
            np.dot(v1, v2) / (np.linalg.norm(v1) * np.linalg.norm(v2) + 1e-8)
        )
        return angle - 0.5  # 偏移
    
    def _calc_finger_spread(self, landmarks, finger_name):
        """计算手指展开角度"""
        # 简化实现
        return 0.0
```

#### 3.2.2 基于 BVH 的重定向

```python
#!/usr/bin/env python3
"""基于 BVH 的运动重定向"""

import numpy as np

class BVHRetargeter:
    """从 BVH 动画数据重定向到灵巧手"""
    
    def __init__(self, bvh_joint_map, hand_config):
        """
        Args:
            bvh_joint_map: BVH 关节名到灵巧手关节的映射
            hand_config: 灵巧手配置
        """
        self.joint_map = bvh_joint_map
        self.config = hand_config
    
    def retarget_bvh_frame(self, bvh_frame_data):
        """
        重定向单帧 BVH 数据
        
        Args:
            bvh_frame_data: dict, BVH 关节名 -> 旋转角度
        
        Returns:
            joint_angles: [num_dofs] 灵巧手关节角度
        """
        joint_angles = np.zeros(self.config['num_dofs'])
        
        for bvh_joint, (hand_joint_idx, axis, scale) in self.joint_map.items():
            if bvh_joint in bvh_frame_data:
                # 提取指定轴的旋转
                rotation = bvh_frame_data[bvh_joint]
                angle = rotation[axis] * scale
                
                # 映射到灵巧手关节
                joint_angles[hand_joint_idx] = angle
        
        # 裁剪到安全范围
        limits = self.config['joint_limits']
        for i in range(len(joint_angles)):
            joint_angles[i] = np.clip(
                joint_angles[i], limits[i][0], limits[i][1]
            )
        
        return joint_angles
```

```python
                (robot_keypoints - target_keypoints) ** 2

            objective,
```

### 3.3 somehand 的重定向架构

somehand 项目采用模块化的重定向架构：

```
输入源 (BVH/MediaPipe/VR)
       │
```

### 4.2 抓取规划

```python
#!/usr/bin/env python3
"""灵巧手抓取规划器"""

import numpy as np
from enum import Enum

class GraspType(Enum):
    PINCH = "pinch"          # 捏取
    LATERAL = "lateral"      # 侧握
    TRIPOD = "tripod"        # 三指捏
    CYLINDRICAL = "cylindrical"  # 圆柱握
    SPHERICAL = "spherical"  # 球握
    HOOK = "hook"            # 钩握
    FIST = "fist"            # 拳握

class GraspPlanner:
    """抓取规划器"""
    
    def __init__(self, hand_controller):
        self.hand = hand_controller
        self.num_dofs = hand_controller.NUM_DOFS
    
    def plan_grasp(self, object_info):
        """
        根据物体信息规划抓取策略
        
        Args:
            object_info: dict
                - shape: 'cylinder', 'sphere', 'box', 'flat'
                - size: [length, width, height] (m)
                - weight: kg
                - fragility: 'low', 'medium', 'high'
        
        Returns:
            grasp_type: GraspType
            joint_angles: [num_dofs] 目标关节角度
        """
        shape = object_info['shape']
        size = object_info['size']
        weight = object_info.get('weight', 0.1)
        fragility = object_info.get('fragility', 'medium')
        
        # 选择抓取类型
        if shape == 'cylinder' and size[0] > 0.05:
            grasp_type = GraspType.CYLINDRICAL
        elif shape == 'sphere' and size[0] > 0.04:
            grasp_type = GraspType.SPHERICAL
        elif shape == 'box' and size[1] < 0.02:
            grasp_type = GraspType.LATERAL
        elif size[0] < 0.03 and size[1] < 0.03:
            grasp_type = GraspType.PINCH
        else:
            grasp_type = GraspType.CYLINDRICAL
        
        # 生成关节角度
        joint_angles = self._generate_grasp_pose(grasp_type, object_info)
        
        return grasp_type, joint_angles
    
    def _generate_grasp_pose(self, grasp_type, object_info):
        """生成抓取姿态"""
        pose = np.zeros(self.num_dofs)
        
        if grasp_type == GraspType.PINCH:
            # 捏取：拇指 + 食指
            pose = self._pinch_pose()
        
        elif grasp_type == GraspType.LATERAL:
            # 侧握
            pose = self._lateral_pose()
        
        elif grasp_type == GraspType.TRIPOD:
            # 三指捏
            pose = self._tripod_pose()
        
        elif grasp_type == GraspType.CYLINDRICAL:
            # 圆柱握
            pose = self._cylindrical_pose(object_info['size'])
        
        elif grasp_type == GraspType.SPHERICAL:
            # 球握
            pose = self._spherical_pose(object_info['size'])
        
        elif grasp_type == GraspType.HOOK:
            # 钩握
            pose = self._hook_pose()
        
        elif grasp_type == GraspType.FIST:
            # 拳握
            pose = self._fist_pose()
        
        return pose
    
    def _pinch_pose(self):
        """捏取姿态"""
        pose = np.zeros(self.num_dofs)
        # 假设 LinkerHand L20 的关节索引
        # 拇指弯曲
        pose[1] = 1.0  # thumb_cmc_flexion
        pose[3] = 1.2  # thumb_mcp_flexion
        pose[4] = 1.0  # thumb_ip
        # 食指弯曲
        pose[6] = 0.8  # index_mcp_flexion
        pose[7] = 1.0  # index_pip
        pose[8] = 0.8  # index_dip
        return pose
    
    def _cylindrical_pose(self, size):
        """圆柱握姿态"""
        pose = np.zeros(self.num_dofs)
        # 根据直径调整弯曲度
        diameter = size[0]
        bend = np.clip(0.5 + diameter * 5, 0.5, 1.3)
        
        # 所有手指弯曲
        for i in range(5, self.num_dofs):
            pose[i] = bend
        # 拇指也弯曲包裹
        pose[1] = bend * 0.8
        pose[3] = bend * 0.9
        pose[4] = bend * 0.7
        return pose
    
    def _spherical_pose(self, size):
        """球握姿态"""
        pose = np.zeros(self.num_dofs)
        radius = size[0] / 2
        bend = np.clip(0.3 + radius * 8, 0.3, 1.0)
        
        # 所有手指中等弯曲，形成杯状
        for i in range(5, self.num_dofs):
            pose[i] = bend
        pose[1] = bend * 0.6
        pose[3] = bend * 0.7
        pose[4] = bend * 0.5
        return pose
    
    def _lateral_pose(self):
        """侧握姿态"""
        pose = np.zeros(self.num_dofs)
        # 拇指内收
        pose[0] = 0.8  # thumb_cmc_abduction
        pose[3] = 0.6  # thumb_mcp_flexion
        # 食指微弯
        pose[6] = 0.3
        pose[7] = 0.4
        return pose
    
    def _tripod_pose(self):
        """三指捏姿态"""
        pose = np.zeros(self.num_dofs)
        # 拇指
        pose[1] = 0.9
        pose[4] = 1.0
        # 食指
        pose[6] = 0.7
        pose[7] = 0.9
        # 中指
        pose[10] = 0.7
        pose[11] = 0.9
        return pose
    
    def _hook_pose(self):
        """钩握姿态"""
        pose = np.zeros(self.num_dofs)
        # 四指弯曲，拇指放松
        for i in range(6, self.num_dofs):
            pose[i] = 1.2
        return pose
    
    def _fist_pose(self):
        """握拳姿态"""
        pose = np.full(self.num_dofs, 1.3)
        return pose
    
    def execute_grasp(self, joint_angles, speed=0.5):
        """执行抓取"""
        self.hand.move_to(joint_angles, speed=speed)
    
    def release(self, speed=0.5):
        """释放"""
        self.hand.open_hand(speed=speed)
```

### 4.3 力控抓取

```python
#!/usr/bin/env python3
"""力控抓取策略"""

import numpy as np
import time

class ForceControlledGrasp:
    """基于力反馈的抓取控制"""
    
    def __init__(self, hand_controller):
        self.hand = hand_controller
        self.num_dofs = hand_controller.NUM_DOFS
    
    def grasp_with_force_control(
        self,
        target_pose,
        max_force=5.0,        # 最大抓取力 (N)
        force_threshold=1.0,   # 接触检测阈值 (N)
        kp=50.0,               # 位置增益
        ki=0.1,                # 积分增益
        freq=100,              # 控制频率
    ):
        """
        力控抓取
        
        Args:
            target_pose: 目标关节角度
            max_force: 最大允许力
            force_threshold: 检测到接触的力阈值
            kp: 位置控制增益
            ki: 力积分增益
            freq: 控制频率
        """
        dt = 1.0 / freq
        current_pos = self.hand.get_positions()
        integral = np.zeros(self.num_dofs)
        
        for step in range(1000):
            # 获取当前力矩
            torques = self.hand.get_torques()
            current_force = np.abs(torques)
            
            # 检测接触
            contact = current_force > force_threshold
            
            # 位置-力混合控制
            target = target_pose.copy()
            for i in range(self.num_dofs):
                if contact[i]:
                    # 接触后：力控模式
                    error = max_force - current_force[i]
                    integral[i] += error * dt
                    target[i] = current_pos[i] + ki * integral[i]
                else:
                    # 未接触：位置控制
                    target[i] = target_pose[i]
            
            # 发送命令
            self.hand.send_positions(target)
            current_pos = target
            
            # 检查是否所有手指都接触
            if np.all(contact):
                print("所有手指已接触物体，抓取完成")
                break
            
            time.sleep(dt)
        
        return current_pos
```

## 5. somehand 项目 20+ 灵巧手配置详解

### 5.1 somehand 项目概述

somehand 是一个开源的灵巧手运动重定向项目，支持 20+ 种灵巧手的统一配置和控制。其核心设计理念是：

- **统一接口**：不同灵巧手使用统一的控制接口
- **YAML 配置**：通过 YAML 文件描述每种灵巧手的参数
- **模块化重定向**：可插拔的重定向算法
- **多输入源支持**：BVH、MediaPipe、VR 手柄等

```python
#!/usr/bin/env python3
"""somehand 配置加载器"""

import yaml
from pathlib import Path
from dataclasses import dataclass, field
from typing import List, Dict, Optional
import numpy as np

@dataclass
class JointConfig:
    """关节配置"""
    name: str
    index: int
    type: str  # revolute / prismatic
    axis: List[float]
    limits: List[float]  # [min, max] in rad
    default: float

@dataclass
class HandConfig:
    """灵巧手配置"""
    name: str
    manufacturer: str
    type: str
    hand_side: str
    num_dofs: int
    drive_type: str
    joints: List[JointConfig] = field(default_factory=list)
    finger_groups: Dict[str, List[int]] = field(default_factory=dict)
    presets: Dict[str, np.ndarray] = field(default_factory=dict)
    retarget_mapping: Dict = field(default_factory=dict)
    communication: Dict = field(default_factory=dict)
    safety: Dict = field(default_factory=dict)
    simulation: Optional[Dict] = None

class HandConfigLoader:
    """灵巧手配置加载器"""
    
    def __init__(self, config_dir='configs'):
        self.config_dir = Path(config_dir)
    
    def load(self, config_name):
        """
        加载灵巧手配置
        
        Args:
            config_name: 配置名 (不含 .yaml 后缀)
                        如 'linkerhand_l20', 'unitree_dex5'
        
        Returns:
            HandConfig 对象
        """
        config_path = self.config_dir / f"{config_name}.yaml"
        
        if not config_path.exists():
            raise FileNotFoundError(f"配置文件不存在: {config_path}")
        
        with open(config_path, 'r', encoding='utf-8') as f:
            data = yaml.safe_load(f)
        
        # 解析基本信息
        info = data['hand_info']
        
        # 解析关节
        joints = []
        for j in data.get('joints', []):
            joints.append(JointConfig(
                name=j['name'],
                index=j['index'],
                type=j.get('type', 'revolute'),
                axis=j.get('axis', [1, 0, 0]),
                limits=j.get('limits', [0, 1.57]),
                default=j.get('default', 0.0),
            ))
        
        # 解析预设姿态
        presets = {}
        for name, preset in data.get('presets', {}).items():
            presets[name] = np.array(preset['positions'])
        
        # 解析手指分组
        finger_groups = {}
        for name, group in data.get('finger_groups', {}).items():
            finger_groups[name] = group['indices']
        
        return HandConfig(
            name=info['name'],
            manufacturer=info.get('manufacturer', ''),
            type=info.get('type', 'dexterous'),
            hand_side=info.get('hand_side', 'right'),
            num_dofs=info['num_dofs'],
            drive_type=info.get('drive_type', 'motor'),
            joints=joints,
            finger_groups=finger_groups,
            presets=presets,
            retarget_mapping=data.get('retarget_mapping', {}),
            communication=data.get('communication', {}),
            safety=data.get('safety', {}),
            simulation=data.get('simulation'),
        )
    
    def list_available(self):
        """列出所有可用配置"""
        configs = []
        for p in self.config_dir.glob("*.yaml"):
            configs.append(p.stem)
        return sorted(configs)
    
    def get_joint_limits(self, config):
        """获取关节限位矩阵"""
        limits = np.zeros((config.num_dofs, 2))
        for joint in config.joints:
            limits[joint.index] = joint.limits
        return limits


# 使用示例
if __name__ == "__main__":
    loader = HandConfigLoader(config_dir='configs')
    
    # 列出可用配置
    print("可用灵巧手配置:")
    for name in loader.list_available():
        print(f"  - {name}")
    
    # 加载配置
    config = loader.load('linkerhand_l20')
    print(f"\n已加载: {config.name}")
    print(f"  自由度: {config.num_dofs}")
    print(f"  驱动方式: {config.drive_type}")
    print(f"  关节数: {len(config.joints)}")
    print(f"  预设姿态: {list(config.presets.keys())}")
    print(f"  手指分组: {list(config.finger_groups.keys())}")
```

```python
#!/usr/bin/env python3
"""somehand 统一驱动接口"""

import numpy as np
from abc import ABC, abstractmethod

class HandDriverBase(ABC):
    """灵巧手驱动基类"""
    
    def __init__(self, config):
        self.config = config
        self.num_dofs = config.num_dofs
        self.current_pos = np.zeros(self.num_dofs)
    
    @abstractmethod
    def connect(self):
        """连接硬件"""
        pass
    
    @abstractmethod
    def send_positions(self, positions):
        """发送位置命令"""
        pass
    
    @abstractmethod
    def get_positions(self):
        """获取当前位置"""
        pass
    
    @abstractmethod
    def get_torques(self):
        """获取力矩"""
        pass
    
    @abstractmethod
    def disconnect(self):
        """断开连接"""
        pass
    
    def apply_preset(self, preset_name, speed=0.5):
        """应用预设姿态"""
        if preset_name in self.config.presets:
            target = self.config.presets[preset_name]
            self.move_to(target, speed=speed)
        else:
            raise ValueError(f"未知预设: {preset_name}")
    
    def move_to(self, target, speed=0.5, freq=100):
        """平滑移动到目标位置"""
        target = np.clip(
            target,
            [j.limits[0] for j in self.config.joints],
            [j.limits[1] for j in self.config.joints],
        )
        
        start = self.current_pos.copy()
        num_steps = max(1, int(50 / speed))
        dt = 1.0 / freq
        
        import time
        for i in range(num_steps + 1):
            alpha = i / num_steps
            alpha_smooth = 0.5 * (1 - np.cos(np.pi * alpha))
            pos = start + (target - start) * alpha_smooth
            self.send_positions(pos)
            time.sleep(dt)
        
        self.current_pos = target


class CANDriver(HandDriverBase):
    """CAN 总线驱动"""
    
    def connect(self):
        import can
        self.bus = can.interface.Bus(
            interface='socketcan',
            channel=self.config.communication.get('interface', 'can0'),
            bitrate=self.config.communication.get('baudrate', 1000000),
        )
        print(f"CAN 连接: {self.config.communication.get('interface')}")
    
    def send_positions(self, positions):
        import can
        # 构建 CAN 帧
        can_id = self.config.communication.get('can_id_base', 0x141)
        data = bytes([int(p * 1000 / 1.57) & 0xFF for p in positions[:8]])
        msg = can.Message(
            arbitration_id=can_id,
            data=data,
            is_extended_id=False,
        )
        self.bus.send(msg)
        self.current_pos = positions
    
    def get_positions(self):
        return self.current_pos
    
    def get_torques(self):
        return np.zeros(self.num_dofs)
    
    def disconnect(self):
        if hasattr(self, 'bus'):
            self.bus.shutdown()


class SerialDriver(HandDriverBase):
    """串口驱动"""
    
    def connect(self):
        import serial
        self.serial = serial.Serial(
            port=self.config.communication.get('interface', '/dev/ttyUSB0'),
            baudrate=self.config.communication.get('baudrate', 115200),
            timeout=1.0,
        )
        print(f"串口连接: {self.config.communication.get('interface')}")
    
    def send_positions(self, positions):
        # 构建串口帧
        frame = self._build_frame(positions)
        self.serial.write(frame)
        self.current_pos = positions
    
    def _build_frame(self, positions):
        frame = bytearray()
        frame.append(0xEB)
        frame.append(0x90)
        for p in positions:
            val = int(np.clip(p * 1000 / 1.57, 0, 1000))
            frame.append((val >> 8) & 0xFF)
            frame.append(val & 0xFF)
        checksum = sum(frame[2:]) & 0xFF
        frame.append(checksum)
        return bytes(frame)
    
    def get_positions(self):
        return self.current_pos
    
    def get_torques(self):
        return np.zeros(self.num_dofs)
    
    def disconnect(self):
        if hasattr(self, 'serial'):
            self.serial.close()


class SimDriver(HandDriverBase):
    """仿真驱动（MuJoCo）"""
    
    def connect(self):
        import mujoco
        mjcf_path = self.config.simulation.get('mjcf_path')
        if mjcf_path:
            self.model = mujoco.MjModel.from_xml_path(mjcf_path)
            self.data = mujoco.MjData(self.model)
            print(f"MuJoCo 仿真加载: {mjcf_path}")
    
    def send_positions(self, positions):
        if hasattr(self, 'model'):
            # 设置 MuJoCo 关节目标
            for joint in self.config.joints:
                idx = self.model.joint(joint.name).qposadr[0]
                self.data.qpos[idx] = positions[joint.index]
            import mujoco
            mujoco.mj_step(self.model, self.data)
        self.current_pos = positions
    
    def get_positions(self):
        return self.current_pos
    
    def get_torques(self):
        return np.zeros(self.num_dofs)
    
    def disconnect(self):
        pass


class HandDriverFactory:
    """驱动工厂"""
    
    @staticmethod
    def create(config):
        """根据配置创建驱动"""
        comm_type = config.communication.get('type', 'sim')
        
        if comm_type == 'can':
            return CANDriver(config)
        elif comm_type == 'serial':
            return SerialDriver(config)
        elif comm_type == 'sim':
            return SimDriver(config)
        elif comm_type == 'ethercat':
            # EtherCAT 驱动实现...
            raise NotImplementedError("EtherCAT 驱动尚未实现")
        else:
            raise ValueError(f"未知通信类型: {comm_type}")


# 使用示例
if __name__ == "__main__":
    # 加载配置
    loader = HandConfigLoader(config_dir='configs')
    config = loader.load('linkerhand_l20')
    
    # 创建驱动
    driver = HandDriverFactory.create(config)
    driver.connect()
    
    # 使用预设
    driver.apply_preset('open')
    driver.apply_preset('pinch')
    driver.apply_preset('fist')
    
    # 发送自定义位置
    custom_pos = np.full(config.num_dofs, 0.5)
    driver.move_to(custom_pos, speed=0.3)
    
    # 断开
    driver.disconnect()
```

```python
#!/usr/bin/env python3
"""
灵巧手完整控制示例
包含：配置加载 → 驱动连接 → MediaPipe 追踪 → 重定向 → 控制
"""

import numpy as np
import time
import cv2

class DexterousHandControlPipeline:
    """灵巧手控制完整流水线"""
    
    def __init__(self, hand_config_name='linkerhand_l20'):
        # 1. 加载配置
        from pathlib import Path
        loader = HandConfigLoader(config_dir='configs')
        self.config = loader.load(hand_config_name)
        print(f"已加载配置: {self.config.name} ({self.config.num_dofs} DoF)")
        
        # 2. 创建驱动
        self.driver = HandDriverFactory.create(self.config)
        self.driver.connect()
        
        # 3. 初始化重定向器
        self.retargeter = KeypointRetargeter({
            'num_dofs': self.config.num_dofs,
            'joint_limits': loader.get_joint_limits(self.config),
        })
        
        # 4. 初始化平滑器
        self.smoother = ExponentialSmoother(alpha=0.3)
        
        # 5. 状态
        self.is_running = False
    
    def start_mediapipe_tracking(self, camera_index=0):
        """启动 MediaPipe 手部追踪"""
        import mediapipe as mp
        
        self.cap = cv2.VideoCapture(camera_index)
        self.mp_hands = mp.solutions.hands
        self.hands = self.mp_hands.Hands(
            static_image_mode=False,
            max_num_hands=1,
            min_detection_confidence=0.7,
            min_tracking_confidence=0.5,
        )
        self.mp_draw = mp.solutions.drawing_utils
        
        self.is_running = True
        print("MediaPipe 手部追踪已启动")
    
    def run(self):
        """运行控制循环"""
        while self.is_running:
            # 1. 获取图像
            ret, frame = self.cap.read()
            if not ret:
                continue
            
            # 2. MediaPipe 检测
            rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            results = self.hands.process(rgb)
            
            if results.multi_hand_landmarks:
                landmarks = results.multi_hand_landmarks[0]
                
                # 提取 3D 关键点
                points = np.array([
                    [lm.x, lm.y, lm.z] for lm in landmarks.landmark
                ])
                
                # 3. 重定向
                joint_angles = self.retargeter.retarget(points)
                
                # 4. 平滑
                smooth_angles = self.smoother.smooth(joint_angles)
                
                # 5. 发送到硬件
                self.driver.send_positions(smooth_angles)
                
                # 6. 可视化
                self.mp_draw.draw_landmarks(
                    frame, landmarks, self.mp_hands.HAND_CONNECTIONS
                )
                
                # 显示关节角度
                cv2.putText(
                    frame, f"Joints: {smooth_angles[:5]}", 
                    (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1
                )
            
            cv2.imshow('Hand Tracking', frame)
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break
    
    def stop(self):
        """停止"""
        self.is_running = False
        self.driver.apply_preset('open')
        self.driver.disconnect()
        self.cap.release()
        cv2.destroyAllWindows()
        print("已停止")


class ExponentialSmoother:
    """指数平滑器"""
    
    def __init__(self, alpha=0.3):
        self.alpha = alpha
        self.prev = None
    
    def smooth(self, value):
        if self.prev is None:
            self.prev = value.copy()
        else:
            self.prev = self.alpha * value + (1 - self.alpha) * self.prev
        return self.prev.copy()
    
    def reset(self):
        self.prev = None


# 主程序
if __name__ == "__main__":
    pipeline = DexterousHandControlPipeline(hand_config_name='linkerhand_l20')
    pipeline.start_mediapipe_tracking(camera_index=0)
    
    try:
        pipeline.run()
    except KeyboardInterrupt:
        pass
    finally:
        pipeline.stop()
```

```python
#!/usr/bin/env python3
"""双手协调控制"""

import numpy as np
import time

class BimanualHandController:
    """双手协调控制器"""
    
    def __init__(self, left_config_name, right_config_name):
        # 加载左右手配置
        loader = HandConfigLoader(config_dir='configs')
        
        self.left_config = loader.load(left_config_name)
        self.right_config = loader.load(right_config_name)
        
        # 创建驱动
        self.left_driver = HandDriverFactory.create(self.left_config)
        self.right_driver = HandDriverFactory.create(self.right_config)
        
        self.left_driver.connect()
        self.right_driver.connect()
        
        print("双手控制器已初始化")
    
    def synchronize_grasp(self, grasp_pose, speed=0.5):
        """同步双手抓取"""
        import threading
        
        def move_left():
            self.left_driver.move_to(grasp_pose, speed=speed)
        
        def move_right():
            self.right_driver.move_to(grasp_pose, speed=speed)
        
        # 并行执行
        t1 = threading.Thread(target=move_left)
        t2 = threading.Thread(target=move_right)
        
        t1.start()
        t2.start()
        
        t1.join()
        t2.join()
        
        print("双手同步动作完成")
    
    def bimanual_manipulation(self, task='pass_object'):
        """双手操作任务"""
        if task == 'pass_object':
            # 左手张开
            self.left_driver.apply_preset('open')
            time.sleep(1)
            
            # 左手握住物体
            self.left_driver.apply_preset('cylindrical')
            time.sleep(1)
            
            # 右手张开
            self.right_driver.apply_preset('open')
            time.sleep(1)
            
            # 双手靠近（需配合机械臂运动）
            
            # 右手握住物体
            self.right_driver.apply_preset('cylindrical')
            time.sleep(1)
            
            # 左手释放
            self.left_driver.apply_preset('open')
            time.sleep(1)
            
            print("物体传递完成")
    
    def disconnect(self):
        self.left_driver.apply_preset('open')
        self.right_driver.apply_preset('open')
        self.left_driver.disconnect()
        self.right_driver.disconnect()
```
