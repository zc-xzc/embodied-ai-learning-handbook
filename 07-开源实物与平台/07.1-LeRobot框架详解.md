## 1. LeRobot 框架架构

### 1.1 概述

LeRobot 是由 Hugging Face 开发的开源机器人学习框架，旨在将最先进的机器人学习算法 democratize（民主化），让研究者和爱好者都能轻松使用。其核心理念是"像训练语言模型一样训练机器人"。

### 1.2 架构总览

```
┌──────────────────────────────────────────────────┐
│              LeRobot 框架架构                      │
├──────────────────────────────────────────────────┤
│                                                   │
│  ┌─────────────┐  ┌─────────────┐  ┌──────────┐ │
│  │  数据集系统   │  │  策略模型    │  │  评估    │ │
```

### 1.3 核心设计原则

| 原则 | 说明 |
|------|------|
| Hugging Face 生态 | 与 HF Hub、Transformers、Datasets 深度集成 |
| 端到端流程 | 数据采集→训练→评估→部署 全链路支持 |
| 可复现性 | 配置驱动的实验管理，版本化的数据和模型 |
| 模块化 | 策略、数据集、机器人接口均可独立使用和扩展 |
| 社区驱动 | 数据集和模型可在 HF Hub 上共享 |

## 5. 评估与可视化

### 5.1 评估

```python
from lerobot.common.policies import Policy
from lerobot.common.envs.factory import make_env

# 加载训练好的策略
policy = Policy.from_pretrained("./outputs/diffusion/checkpoints/last")

# 创建评估环境
env = make_env(
    env_type="aloha",
    task="AlohaInsertion-v0",
    n_envs=1,
)

# 评估
results = []
for episode in range(10):
    obs, info = env.reset()
    episode_reward = 0
    episode_length = 0
    success = False
    
    for step in range(300):
        # 策略预测
        action = policy.predict(obs)
        
        # 执行
        obs, reward, terminated, truncated, info = env.step(action)
        episode_reward += reward
        episode_length += 1
        
        if terminated or truncated:
            success = info.get('success', False)
            break
    
    results.append({
        'episode': episode,
        'reward': episode_reward,
        'length': episode_length,
        'success': success,
    })

# 打印统计
success_rate = sum(r['success'] for r in results) / len(results)
avg_reward = np.mean([r['reward'] for r in results])
avg_length = np.mean([r['length'] for r in results])

print(f"成功率: {success_rate:.2%}")
print(f"平均奖励: {avg_reward:.2f}")
print(f"平均步数: {avg_length:.1f}")
```

### 5.2 可视化

```python
import matplotlib.pyplot as plt
import imageio

class Visualizer:
    """LeRobot 可视化工具"""
    
    @staticmethod
    def visualize_predictions(dataset, policy, episode_idx=0, output_path="predictions.mp4"):
        """可视化策略预测"""
        frames = []
        
        from_idx = dataset.episode_data_index["from"][episode_idx]
        to_idx = dataset.episode_data_index["to"][episode_idx]
        
        for i in range(from_idx, to_idx):
            frame = dataset[i]
            
            # 获取观测
            obs = {k: v.unsqueeze(0) for k, v in frame.items() if k.startswith('observation')}
            
            # 策略预测
            with torch.no_grad():
                action = policy.predict(obs)
            
            # 可视化
            fig, axes = plt.subplots(1, 3, figsize=(15, 5))
            
            # 图像
            if 'observation.images.top' in frame:
                img = frame['observation.images.top'].permute(1, 2, 0).numpy()
                axes[0].imshow(img)
                axes[0].set_title('Top Camera')
            
            # 动作对比
            gt_action = frame['action'].numpy()
            pred_action = action[0].numpy()
            
            axes[1].bar(range(len(gt_action)), gt_action, alpha=0.5, label='Ground Truth')
            axes[1].bar(range(len(pred_action)), pred_action, alpha=0.5, label='Predicted')
            axes[1].legend()
            axes[1].set_title('Action Comparison')
            
            # 状态
            if 'observation.state' in frame:
                state = frame['observation.state'].numpy()
                axes[2].bar(range(len(state)), state)
                axes[2].set_title('Robot State')
            
            plt.tight_layout()
            
            # 保存为图像
            fig.canvas.draw()
            img = np.frombuffer(fig.canvas.tostring_rgb(), dtype=np.uint8)
            img = img.reshape(fig.canvas.get_width_height()[::-1] + (3,))
            frames.append(img)
            plt.close(fig)
        
        # 保存视频
        imageio.mimsave(output_path, frames, fps=dataset.fps)
        print(f"可视化已保存: {output_path}")
```

## 7. Sim2Real 工作流

### 7.1 Sim2Real 流程

```
仿真训练
    │
    ├── 数据采集（仿真）
    │     ├── 随机策略探索
    │     ├── 专家策略示教
    │     └── 域随机化
    │
    ├── 策略训练
    │     ├── Diffusion Policy / ACT
    │     ├── 域随机化训练
    │     └── 模型评估（仿真）
    │
    ├── Sim2Real 迁移
    │     ├── 域适配
    │     ├── 真实数据微调
    │     └── 真实环境评估
    │
    └── 部署
          ├── ONNX 导出
          ├── 实时推理
          └── 持续学习
```

### 7.2 Sim2Real 代码

```python
# 文件: sim2real.py
"""LeRobot Sim2Real 工作流"""
import torch
import numpy as np
from lerobot.common.policies import Policy

class Sim2RealPipeline:
    """Sim2Real 迁移管线"""
    
    def __init__(self, sim_policy_path: str, robot_config: dict):
        # 加载仿真训练的策略
        self.sim_policy = Policy.from_pretrained(sim_policy_path)
        
        # 连接真实机器人
        self.robot = self._connect_robot(robot_config)
        
        # 域适配参数
        self.image_mean = None  # 仿真图像均值
        self.image_std = None   # 仿真图像标准差
        self.real_mean = None   # 真实图像均值
        self.real_std = None    # 真实图像标准差
    
    def _connect_robot(self, config):
        """连接真实机器人"""
        if config['type'] == 'so100':
            from lerobot.common.robots.so100 import SO100Robot
            return SO100Robot(config)
        elif config['type'] == 'aloha':
            from lerobot.common.robots.aloha import AlohaRobot
            return AlohaRobot(config)
    
    def adapt_domain(self, real_images: list):
        """域适配：计算图像归一化参数"""
        real_data = np.stack(real_images)
        self.real_mean = real_data.mean(axis=(0, 1, 2))
        self.real_std = real_data.std(axis=(0, 1, 2))
        
        print(f"真实域: mean={self.real_mean}, std={self.real_std}")
    
    def deploy(self, task: str, max_steps: int = 300):
        """部署策略到真实机器人"""
        print(f"开始 Sim2Real 部署: {task}")
        
        obs = self.robot.get_observation()
        
        for step in range(max_steps):
            # 域适配：归一化图像
            if self.real_mean is not None:
                for key in obs:
                    if 'image' in key:
                        obs[key] = (obs[key] - self.real_mean) / self.real_std
                        obs[key] = obs[key] * self.image_std + self.image_mean
            
            # 策略推理
            with torch.no_grad():
                action = self.sim_policy.predict(obs)
            
            # 执行
            self.robot.send_action(action)
            
            # 获取新观测
            obs = self.robot.get_observation()
            
            if step % 50 == 0:
                print(f"Step {step}")
        
        print("部署完成")
    
    def export_onnx(self, output_path: str):
        """导出 ONNX 模型"""
        dummy_input = {
            'observation.state': torch.zeros(1, 6),
            'observation.images.top': torch.zeros(1, 3, 480, 640),
        }
        
        torch.onnx.export(
            self.sim_policy,
            dummy_input,
            output_path,
            input_names=list(dummy_input.keys()),
            output_names=['action'],
            dynamic_axes={
                'observation.state': {0: 'batch'},
                'observation.images.top': {0: 'batch'},
                'action': {0: 'batch'},
            },
        )
        print(f"ONNX 模型已导出: {output_path}")
```

```python
        self.config = self._load_config(config_path)
        
        # 设备
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        logger.info(f"使用设备: {self.device}")
    
    def _load_config(self, config_path):
        """加载配置"""
        if config_path is None:
            return {
                'dataset': {
                    'repo_id': 'lerobot/aloha_sim_insertion_human_image',
                },
                'policy': {
                    'type': 'diffusion',
                    'num_inference_steps': 100,
                    'horizon': 16,
                    'n_action_steps': 8,
                },
                'train': {
                    'batch_size': 64,
                    'learning_rate': 1e-4,
                    'num_train_steps': 50000,
                    'save_freq': 5000,
                    'eval_freq': 5000,
                },
                'output_dir': './outputs/training',
            }
        # 从 YAML 加载
        from omegaconf import OmegaConf
        return OmegaConf.load(config_path)
    
    def prepare_data(self):
        """准备数据"""
        logger.info("加载数据集...")
        
        self.dataset = LeRobotDataset(
            repo_id=self.config['dataset']['repo_id'],
        )
        
        # 创建 DataLoader
        from torch.utils.data import DataLoader
        
        self.dataloader = DataLoader(
            self.dataset,
            batch_size=self.config['train']['batch_size'],
            num_workers=self.config['train'].get('num_workers', 4),
            shuffle=True,
            pin_memory=True,
            drop_last=True,
        )
        
        logger.info(f"数据集: {self.dataset.num_episodes} episodes, "
                    f"{self.dataset.num_frames} frames")
        logger.info(f"特征: {list(self.dataset.features.keys())}")
        
        return self.dataset
    
    def build_policy(self):
        """构建策略模型"""
        logger.info("构建策略模型...")
        
        policy_type = self.config['policy']['type']
        
        if policy_type == 'diffusion':
            policy_config = DiffusionConfig(
                num_inference_steps=self.config['policy'].get('num_inference_steps', 100),
                horizon=self.config['policy'].get('horizon', 16),
                n_action_steps=self.config['policy'].get('n_action_steps', 8),
                batch_size=self.config['train']['batch_size'],
                learning_rate=self.config['train']['learning_rate'],
            )
            self.policy = DiffusionPolicy(
                policy_config,
                dataset_stats=self.dataset.stats,
            )
        
        elif policy_type == 'act':
            from lerobot.common.policies.act import ACTPolicy
            from lerobot.common.policies.act.configuration_act import ACTConfig
            policy_config = ACTConfig(
                chunk_size=self.config['policy'].get('chunk_size', 100),
                batch_size=self.config['train']['batch_size'],
                learning_rate=self.config['train']['learning_rate'],
            )
            self.policy = ACTPolicy(
                policy_config,
                dataset_stats=self.dataset.stats,
            )
        
        self.policy.to(self.device)
        
        # 优化器
        self.optimizer = torch.optim.AdamW(
            self.policy.parameters(),
            lr=self.config['train']['learning_rate'],
            weight_decay=1e-6,
        )
        
        # 学习率调度
        from torch.optim.lr_scheduler import CosineAnnealingLR
        self.scheduler = CosineAnnealingLR(
            self.optimizer,
            T_max=self.config['train']['num_train_steps'],
        )
        
        logger.info(f"策略参数量: {sum(p.numel() for p in self.policy.parameters()):,}")
        
        return self.policy
    
    def train(self):
        """训练"""
        logger.info("开始训练...")
        
        output_dir = Path(self.config['output_dir'])
        output_dir.mkdir(parents=True, exist_ok=True)
        
        step = 0
        total_loss = 0
        log_interval = 100
        
        # 无限数据迭代
        from itertools import cycle
        
        for batch in cycle(self.dataloader):
            self.policy.train()
            
            # 准备数据
            batch = {k: v.to(self.device) for k, v in batch.items()}
            
            # 前向 + 计算损失
            loss = self.policy.compute_loss(batch)
            
            # 反向传播
            self.optimizer.zero_grad()
            loss.backward()
            torch.nn.utils.clip_grad_norm_(self.policy.parameters(), 1.0)
            self.optimizer.step()
            self.scheduler.step()
            
            total_loss += loss.item()
            step += 1
            
            # 日志
            if step % log_interval == 0:
                avg_loss = total_loss / log_interval
                lr = self.scheduler.get_last_lr()[0]
                logger.info(f"Step {step}/{self.config['train']['num_train_steps']} | "
                           f"Loss: {avg_loss:.6f} | LR: {lr:.2e}")
                total_loss = 0
            
            # 保存检查点
            if step % self.config['train']['save_freq'] == 0:
                self._save_checkpoint(step, output_dir)
            
            # 评估
            if step % self.config['train']['eval_freq'] == 0:
                self._evaluate(step)
            
            # 结束
            if step >= self.config['train']['num_train_steps']:
                break
        
        # 保存最终模型
        self._save_checkpoint(step, output_dir, final=True)
        logger.info(f"训练完成！模型已保存到 {output_dir}")
    
    def _save_checkpoint(self, step, output_dir, final=False):
        """保存检查点"""
        checkpoint_dir = output_dir / "checkpoints" / ("final" if final else f"step_{step}")
        checkpoint_dir.mkdir(parents=True, exist_ok=True)
        
        self.policy.save_pretrained(checkpoint_dir)
        
        # 保存训练状态
        torch.save({
            'step': step,
            'optimizer': self.optimizer.state_dict(),
            'scheduler': self.scheduler.state_dict(),
        }, checkpoint_dir / "training_state.pt")
        
        logger.info(f"检查点已保存: {checkpoint_dir}")
    
    def _evaluate(self, step):
        """评估"""
        self.policy.eval()
        
        # 在数据集上评估
        total_loss = 0
        num_batches = 0
        
        with torch.no_grad():
            for batch in self.dataloader:
                batch = {k: v.to(self.device) for k, v in batch.items()}
                loss = self.policy.compute_loss(batch)
                total_loss += loss.item()
                num_batches += 1
                if num_batches >= 10:
                    break
        
        avg_loss = total_loss / num_batches
        logger.info(f"评估 (Step {step}): Loss = {avg_loss:.6f}")
    
    def run(self):
        """运行完整 Pipeline"""
        self.prepare_data()
        self.build_policy()
        self.train()


# === 运行 ===
if __name__ == "__main__":
    pipeline = LeRobotTrainingPipeline()
    pipeline.run()
```

### 9.1 数据格式转换

```python
# 文件: lerobot_to_groot.py
"""LeRobot Dataset → GR00T 数据格式转换"""
import json
import numpy as np
from pathlib import Path
from lerobot.common.datasets.lerobot_dataset import LeRobotDataset

class LeRobotToGR00TConverter:
    """LeRobot 数据集转换为 GR00T 格式"""
    
    def __init__(self, lerobot_dataset_id: str, output_dir: str):
        self.dataset = LeRobotDataset(lerobot_dataset_id)
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)
    
    def convert(self):
        """执行转换"""
        # GR00T 数据格式
        # 每条数据包含:
        # - state: 机器人状态
        # - action: 动作
        # - images: 多视角图像
        # - language: 语言指令
        
        groot_data = []
        
        for episode_idx in range(self.dataset.num_episodes):
            from_idx = self.dataset.episode_data_index["from"][episode_idx]
            to_idx = self.dataset.episode_data_index["to"][episode_idx]
            
            for frame_idx in range(from_idx, to_idx):
                frame = self.dataset[frame_idx]
                
                groot_frame = {
                    # 元数据
                    'episode_id': episode_idx,
                    'frame_id': frame_idx - from_idx,
                    'timestamp': (frame_idx - from_idx) / self.dataset.fps,
                    
                    # 状态
                    'state': frame.get('observation.state', np.array([])).tolist(),
                    
                    # 动作
                    'action': frame.get('action', np.array([])).tolist(),
                    
                    # 图像（保存路径）
                    'images': {},
                    
                    # 语言指令
                    'language': self._get_task_description(episode_idx),
                }
                
                # 处理图像
                for key in frame:
                    if 'image' in key:
                        img = frame[key]
                        if hasattr(img, 'numpy'):
                            img = img.numpy()
                        
                        # 保存图像
                        img_path = self.output_dir / "images" / key
                        img_path.mkdir(parents=True, exist_ok=True)
                        img_file = img_path / f"ep{episode_idx:04d}_f{frame_idx:06d}.jpg"
                        
                        from PIL import Image
                        Image.fromarray(img).save(img_file)
                        
                        groot_frame['images'][key] = str(img_file)
                
                groot_data.append(groot_frame)
        
        # 保存元数据
        metadata = {
            'num_episodes': self.dataset.num_episodes,
            'num_frames': len(groot_data),
            'fps': self.dataset.fps,
            'features': self.dataset.features,
            'source': 'lerobot',
        }
        
        with open(self.output_dir / "metadata.json", 'w') as f:
            json.dump(metadata, f, indent=2)
        
        # 保存数据索引
        with open(self.output_dir / "data_index.jsonl", 'w') as f:
            for frame in groot_data:
                f.write(json.dumps(frame) + '\n')
        
        print(f"转换完成: {len(groot_data)} 帧")
        print(f"输出目录: {self.output_dir}")
    
    def _get_task_description(self, episode_idx):
        """获取任务描述"""
        if hasattr(self.dataset, 'episodes'):
            for ep in self.dataset.episodes:
                if ep.get('index') == episode_idx:
                    return ep.get('task', 'unknown')
        return 'unknown'
```
