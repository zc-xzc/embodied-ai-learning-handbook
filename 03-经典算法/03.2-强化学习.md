## 一、MDP 与 POMDP 基础

### 1.1 MDP 定义

```python
"""
马尔可夫决策过程 (Markov Decision Process, MDP)

五元组 (S, A, P, R, γ):
  S: 状态空间 (State Space)
  A: 动作空间 (Action Space)
  P: 转移概率 P(s'|s,a) (Transition Probability)
  R: 奖励函数 R(s,a) (Reward Function)
  γ: 折扣因子 γ ∈ [0,1] (Discount Factor)

马尔可夫性质：未来只依赖于当前状态和动作，不依赖于历史
  P(s_{t+1} | s_0, a_0, ..., s_t, a_t) = P(s_{t+1} | s_t, a_t)

目标：找到策略 π(a|s) 最大化累积折扣奖励
  J(π) = E[Σ γ^t R(s_t, a_t)]
"""
```

### 1.2 POMDP

```python
"""
部分可观测马尔可夫决策过程 (POMDP)

在 MDP 基础上增加：
  O: 观测空间 (Observation Space)
  Z: 观测概率 Z(o|s) (Observation Function)

机器人通常是 POMDP：
  - 相机只能看到部分环境
  - 关节编码器有噪声
  - 物体被遮挡时不可见

处理方法：
  1. 历史信息：用 RNN/Transformer 处理观测序列
  2. 状态估计：滤波器（卡尔曼/粒子）
  3. 隐变量模型：学习隐状态表示
"""
```

### 1.3 值函数与贝尔曼方程

```python
import numpy as np

"""
值函数定义：

状态值函数 V(s)：
  V(s) = E[Σ γ^t R(s_t, a_t) | s_0 = s, π]

动作值函数 Q(s,a)：
  Q(s,a) = E[Σ γ^t R(s_t, a_t) | s_0 = s, a_0 = a, π]

贝尔曼方程：
  V(s) = Σ_a π(a|s) [R(s,a) + γ Σ_s' P(s'|s,a) V(s')]
  Q(s,a) = R(s,a) + γ Σ_s' P(s'|s,a) V(s')

最优值函数：
  Q*(s,a) = R(s,a) + γ Σ_s' P(s'|s,a) max_a' Q*(s',a')
"""

# 简单的值迭代示例（网格世界）
class GridWorld:
    """4x4 网格世界"""
    def __init__(self, size=4):
        self.size = size
        self.terminal_states = [(0, 0), (size-1, size-1)]
    
    def get_reward(self, state):
        if state in self.terminal_states:
            return 0
        return -1  # 每步 -1 奖励
    
    def get_next_state(self, state, action):
        """执行动作后的下一个状态"""
        moves = {'up': (-1, 0), 'down': (1, 0), 'left': (0, -1), 'right': (0, 1)}
        dr, dc = moves[action]
        new_r = max(0, min(self.size-1, state[0] + dr))
        new_c = max(0, min(self.size-1, state[1] + dc))
        return (new_r, new_c)

def value_iteration(env, gamma=0.9, theta=1e-6):
    """值迭代算法"""
    V = np.zeros((env.size, env.size))
    actions = ['up', 'down', 'left', 'right']
    
    while True:
        delta = 0
        for r in range(env.size):
            for c in range(env.size):
                state = (r, c)
                if state in env.terminal_states:
                    continue
                
                v = V[r, c]
                # 计算每个动作的 Q 值
                q_values = []
                for action in actions:
                    next_state = env.get_next_state(state, action)
                    reward = env.get_reward(state)
                    q = reward + gamma * V[next_state]
                    q_values.append(q)
                
                V[r, c] = max(q_values)
                delta = max(delta, abs(v - V[r, c]))
        
        if delta < theta:
            break
    
    return V

# 运行
env = GridWorld(4)
V = value_iteration(env)
print("状态值函数 V(s):")
print(np.round(V, 2))
```

## 二、值函数方法：DQN

### 2.1 DQN 核心思想

```
DQN (Deep Q-Network):
  用神经网络近似 Q(s, a)

关键组件：
  1. Experience Replay: 存储经验 (s, a, r, s') 随机采样训练
  2. Target Network: 独立的目标网络，定期更新，稳定训练
  3. ε-greedy 探索: 以 ε 概率随机探索

损失函数：
  L(θ) = E[(r + γ max_a' Q_target(s', a'; θ⁻) - Q(s, a; θ))²]
```

### 2.2 DQN 实现

```python
import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
from collections import deque
import random

class QNetwork(nn.Module):
    """Q 网络"""
    def __init__(self, obs_dim, action_dim, hidden_dim=128):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim)
        )
    
    def forward(self, x):
        return self.net(x)

class ReplayBuffer:
    """经验回放缓冲区"""
    def __init__(self, capacity=10000):
        self.buffer = deque(maxlen=capacity)
    
    def push(self, state, action, reward, next_state, done):
        self.buffer.append((state, action, reward, next_state, done))
    
    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)
        return (np.array(states), np.array(actions), np.array(rewards),
                np.array(next_states), np.array(dones))
    
    def __len__(self):
        return len(self.buffer)

class DQN:
    """DQN 算法"""
    def __init__(self, obs_dim, action_dim, lr=1e-3, gamma=0.99,
                 epsilon_start=1.0, epsilon_end=0.01, epsilon_decay=500):
        self.action_dim = action_dim
        self.gamma = gamma
        
        self.q_net = QNetwork(obs_dim, action_dim)
        self.target_net = QNetwork(obs_dim, action_dim)
        self.target_net.load_state_dict(self.q_net.state_dict())
        
        self.optimizer = optim.Adam(self.q_net.parameters(), lr=lr)
        self.buffer = ReplayBuffer(10000)
        
        self.epsilon = epsilon_start
        self.epsilon_end = epsilon_end
        self.epsilon_decay = epsilon_decay
        self.steps = 0
    
    def select_action(self, state):
        """ε-greedy 动作选择"""
        self.epsilon = self.epsilon_end + (1.0 - self.epsilon_end) * \
            np.exp(-1.0 * self.steps / self.epsilon_decay)
        self.steps += 1
        
        if random.random() < self.epsilon:
            return random.randrange(self.action_dim)
        
        with torch.no_grad():
            state_tensor = torch.FloatTensor(state).unsqueeze(0)
            q_values = self.q_net(state_tensor)
            return q_values.argmax().item()
    
    def update(self, batch_size=64):
        """更新 Q 网络"""
        if len(self.buffer) < batch_size:
            return
        
        states, actions, rewards, next_states, dones = self.buffer.sample(batch_size)
        states = torch.FloatTensor(states)
        actions = torch.LongTensor(actions)
        rewards = torch.FloatTensor(rewards)
        next_states = torch.FloatTensor(next_states)
        dones = torch.FloatTensor(dones)
        
        # 当前 Q 值
        q_values = self.q_net(states).gather(1, actions.unsqueeze(1)).squeeze(1)
        
        # 目标 Q 值
        with torch.no_grad():
            next_q_values = self.target_net(next_states).max(dim=1)[0]
            target_q = rewards + self.gamma * next_q_values * (1 - dones)
        
        loss = nn.MSELoss()(q_values, target_q)
        
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
        
        return loss.item()
    
    def update_target(self):
        """更新目标网络"""
        self.target_net.load_state_dict(self.q_net.state_dict())
```

## 三、策略梯度：PPO

### 3.1 PPO 核心思想

```
PPO (Proximal Policy Optimization):
  最常用的 on-policy RL 算法

核心思想：
  限制策略更新幅度，避免灾难性更新

Clipped Surrogate Objective:
  L(θ) = E[min(r(θ) * A, clip(r(θ), 1-ε, 1+ε) * A)]
  
  其中 r(θ) = π_θ(a|s) / π_θ_old(a|s) 是新旧策略的概率比
  A 是优势函数 (Advantage)
  ε 是裁剪参数（通常 0.2）
```

### 3.2 PPO 实现要点

```python
class PPOPolicy(nn.Module):
    """PPO 策略网络（Actor-Critic）"""
    def __init__(self, obs_dim, action_dim, hidden_dim=64):
        super().__init__()
        # Actor
        self.actor = nn.Sequential(
            nn.Linear(obs_dim, hidden_dim),
            nn.Tanh(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.Tanh(),
            nn.Linear(hidden_dim, action_dim),
            nn.Tanh()  # 连续动作 [-1, 1]
        )
        # Critic
        self.critic = nn.Sequential(
            nn.Linear(obs_dim, hidden_dim),
            nn.Tanh(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.Tanh(),
            nn.Linear(hidden_dim, 1)
        )
        # 动作标准差（可学习）
        self.log_std = nn.Parameter(torch.zeros(action_dim))
    
    def forward(self, obs):
        mean = self.actor(obs)
        value = self.critic(obs)
        return mean, value
    
    def get_action(self, obs):
        mean, value = self.forward(obs)
        std = torch.exp(self.log_std)
        dist = torch.distributions.Normal(mean, std)
        action = dist.sample()
        log_prob = dist.log_prob(action).sum(dim=-1)
        return action, log_prob, value
    
    def evaluate(self, obs, actions):
        mean, value = self.forward(obs)
        std = torch.exp(self.log_std)
        dist = torch.distributions.Normal(mean, std)
        log_prob = dist.log_prob(actions).sum(dim=-1)
        entropy = dist.entropy().sum(dim=-1)
        return log_prob, entropy, value
```

## 四、Actor-Critic：SAC 和 TD3

### 4.1 SAC（Soft Actor-Critic）

```python
"""
SAC 核心特点：
  1. 最大熵 RL：在奖励中加入熵项，鼓励探索
  2. Off-policy：使用经验回放，数据效率高
  3. 随机策略：输出动作分布，适合探索

目标：max E[Σ γ^t (r_t + α H(π(·|s_t)))]
  其中 H 是策略的熵，α 是温度参数
"""
```

### 4.2 TD3（Twin Delayed DDPG）

```python
"""
TD3 核心改进（相比 DDPG）：
  1. Twin Q networks: 取两个 Q 网络的最小值，减少过估计
  2. Delayed policy update: 策略更新频率低于 Q 网络
  3. Target policy smoothing: 给目标动作加噪声，防止利用 Q 函数误差
"""
```

### 4.3 算法对比

| 特性 | DQN | PPO | SAC | TD3 |
|------|-----|-----|-----|-----|
| 动作空间 | 离散 | 连续 | 连续 | 连续 |
| On/Off-policy | Off | On | Off | Off |
| 随机策略 | 否 | 是 | 是 | 否 |
| 探索方式 | ε-greedy | 策略熵 | 最大熵 | OU噪声 |
| 数据效率 | 中 | 低 | 高 | 高 |
| 稳定性 | 中 | 高 | 高 | 高 |
| 适用场景 | 离散控制 | 通用 | 机器人控制 | 机器人控制 |

### 6.2 PPO 训练 CartPole

```python
from stable_baselines3 import PPO
from stable_baselines3.common.evaluation import evaluate_policy
from stable_baselines3.common.monitor import Monitor
from gymnasium import make
import numpy as np

# 创建环境
env = Monitor(make("CartPole-v1"))

# 创建 PPO 模型
model = PPO(
    "MlpPolicy",
    env,
    learning_rate=3e-4,
    n_steps=2048,        # 每次更新收集的步数
    batch_size=64,
    n_epochs=10,         # 每次更新的 epoch 数
    gamma=0.99,
    gae_lambda=0.95,
    clip_range=0.2,
    ent_coef=0.01,       # 熵正则化系数
    verbose=1,
    tensorboard_log="./ppo_cartpole/"
)

# 训练
print("开始训练...")
model.learn(total_timesteps=50000)
print("训练完成！")

# 评估
print("\n评估...")
mean_reward, std_reward = evaluate_policy(model, env, n_eval_episodes=20)
print(f"平均奖励: {mean_reward:.2f} +/- {std_reward:.2f}")

# 保存
model.save("ppo_cartpole")
env.close()

# 可视化训练
```

## 七、课程学习与课程设计

### 7.1 课程学习原理

```python
"""
课程学习 (Curriculum Learning):
  从简单任务逐步过渡到复杂任务

类比：人类学习
  先学加法 → 再学乘法 → 最后学微积分

在 RL 中的应用：
  1. 初始：简单任务（近距离、少障碍、少噪声）
  2. 中期：中等任务（增加距离、障碍、噪声）
  3. 后期：困难任务（随机位置、多干扰、复杂场景）

关键：在当前难度上达到一定成功率后，再提升难度
"""
```

### 7.2 课程学习实现

```python
class CurriculumManager:
    """课程学习管理器"""
    def __init__(self, num_levels=5, success_threshold=0.7, window=20):
        self.num_levels = num_levels
        self.threshold = success_threshold
        self.window = window
        self.current_level = 0
        self.success_history = []
    
    def get_task_params(self):
        """获取当前难度的任务参数"""
        params = {
            0: {"difficulty": 0.1, "noise": 0.0, "random_init": False},
            1: {"difficulty": 0.3, "noise": 0.01, "random_init": False},
            2: {"difficulty": 0.5, "noise": 0.05, "random_init": True},
            3: {"difficulty": 0.7, "noise": 0.1, "random_init": True},
            4: {"difficulty": 1.0, "noise": 0.2, "random_init": True},
        }
        return params[self.current_level]
    
    def update(self, success):
        """记录成功/失败，可能升级"""
        self.success_history.append(success)
        
        if len(self.success_history) >= self.window:
            recent_rate = np.mean(self.success_history[-self.window:])
            
            if recent_rate >= self.threshold and self.current_level < self.num_levels - 1:
                self.current_level += 1
                self.success_history = []
                print(f"课程升级！当前等级: {self.current_level}")
            elif recent_rate < 0.3 and self.current_level > 0:
                self.current_level -= 1
                self.success_history = []
                print(f"课程降级！当前等级: {self.current_level}")

# 使用
curriculum = CurriculumManager(num_levels=5)

for episode in range(1000):
    params = curriculum.get_task_params()
    # 用 params 设置环境难度
    # env.set_difficulty(params["difficulty"], params["noise"])
    
    # 训练一个 episode
    success = True  # 假设训练结果
    curriculum.update(success)
    
    if episode % 100 == 0:
        print(f"Episode {episode}: Level {curriculum.current_level}")
```
