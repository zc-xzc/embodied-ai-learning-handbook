### 1.1 贝塞尔曲线

```python
import numpy as np
from scipy.special import comb

class BezierCurve:
    """贝塞尔曲线"""
    def __init__(self, control_points):
        """
        control_points: (n, dim) 控制点数组
        """
        self.control_points = np.array(control_points)
        self.n = len(control_points) - 1  # 阶数
    
    def evaluate(self, t):
        """计算参数 t 处的曲线点 (t ∈ [0, 1])"""
        n = self.n
        point = np.zeros(self.control_points.shape[1])
        
        for i in range(n + 1):
            # 伯恩斯坦基函数
            b = comb(n, i) * (t ** i) * ((1 - t) ** (n - i))
            point += b * self.control_points[i]
        
        return point
    
    def evaluate_derivative(self, t):
        """计算参数 t 处的一阶导数"""
        n = self.n
        if n == 0:
            return np.zeros(self.control_points.shape[1])
        
        # 导数的控制点
        deriv_controls = n * (self.control_points[1:] - self.control_points[:-1])
        deriv_bezier = BezierCurve(deriv_controls)
        return deriv_bezier.evaluate(t)
    
    def sample(self, num_points=100):
        """采样曲线"""
        t_values = np.linspace(0, 1, num_points)
        points = np.array([self.evaluate(t) for t in t_values])
        return points

# 使用示例
control_points = np.array([
    [0.0, 0.0, 0.0],   # 起点
    [0.2, 0.5, 0.1],   # 控制点1
    [0.8, 0.5, 0.1],   # 控制点2
    [1.0, 0.0, 0.0],   # 终点
])

bezier = BezierCurve(control_points)
trajectory = bezier.sample(50)
print(f"贝塞尔曲线轨迹: {trajectory.shape}")
print(f"起点: {trajectory[0]}, 终点: {trajectory[-1]}")

# 导数（速度）
t = 0.5
velocity = bezier.evaluate_derivative(t)
print(f"t=0.5 处速度: {velocity}")
```

```python
from scipy.interpolate import BSpline, splprep, splev

class BSplineTrajectory:
    """B 样条轨迹"""
    def __init__(self, waypoints, degree=3, smoothing=0.0):
        """
        waypoints: (n, dim) 路径点
        degree: 样条阶数（3 = 三次 B 样条）
        smoothing: 平滑参数
        """
        waypoints = np.array(waypoints)
        
        # 参数化 B 样条拟合
        # splprep 需要转置输入
        tck, u = splprep(waypoints.T, k=degree, s=smoothing)
        self.tck = tck
        self.u = u
    
    def evaluate(self, t):
        """计算参数 t 处的点 (t ∈ [0, 1])"""
        point = np.array(splev(t, self.tck))
        return point
    
    def evaluate_derivative(self, t, der=1):
        """计算导数"""
        deriv = np.array(splev(t, self.tck, der=der))
        return deriv
    
    def sample(self, num_points=100):
        """采样轨迹"""
        t_values = np.linspace(0, 1, num_points)
        points = np.array([self.evaluate(t) for t in t_values])
        return points

# 使用示例
waypoints = np.array([
    [0.0, 0.0, 0.0],
    [0.3, 0.2, 0.1],
    [0.6, 0.1, 0.2],
    [0.9, 0.3, 0.0],
    [1.0, 0.0, 0.0],
])

bspline = BSplineTrajectory(waypoints, degree=3, smoothing=0.1)
```

### 1.3 最小 Jerk 轨迹

```python
class MinimumJerkTrajectory:
    """
    最小 Jerk 轨迹：最小化加加速度（jerk）的平滑轨迹
    5 阶多项式：x(t) = a0 + a1*t + a2*t^2 + a3*t^3 + a4*t^4 + a5*t^5
    
    边界条件：位置、速度、加速度在起点和终点都已知
    """
    def __init__(self, start_pos, end_pos, duration):
        self.start = np.array(start_pos)
        self.end = np.array(end_pos)
        self.T = duration
    
    def evaluate(self, t):
        """计算时间 t 处的位置"""
        tau = t / self.T  # 归一化时间 [0, 1]
        
        # 最小 Jerk 多项式
        # x(tau) = 10*tau^3 - 15*tau^4 + 6*tau^5
        profile = 10 * tau**3 - 15 * tau**4 + 6 * tau**5
        
        pos = self.start + (self.end - self.start) * profile
        return pos
    
    def evaluate_velocity(self, t):
        """计算速度"""
        tau = t / self.T
        d_tau = 1 / self.T
        
        # 导数
        d_profile = (30 * tau**2 - 60 * tau**3 + 30 * tau**4) * d_tau
        
        vel = (self.end - self.start) * d_profile
        return vel
    
    def sample(self, num_points=100):
        """采样"""
        t_values = np.linspace(0, self.T, num_points)
        positions = np.array([self.evaluate(t) for t in t_values])
        velocities = np.array([self.evaluate_velocity(t) for t in t_values])
        return positions, velocities

# 使用
traj = MinimumJerkTrajectory(start_pos=[0, 0, 0], end_pos=[0.5, 0.1, 0.3], duration=2.0)
pos, vel = traj.sample(50)
print(f"最小 Jerk 轨迹: 位置 {pos.shape}, 速度 {vel.shape}")
print(f"起点速度: {vel[0]} (应该为 0)")
print(f"终点速度: {vel[-1]} (应该为 0)")
```

## 二、轨迹优化方法

```python
from scipy.optimize import minimize

class TrajectoryOptimizer:
    """轨迹优化器"""
    def __init__(self, start, goal, obstacles=None):
        self.start = np.array(start)
        self.goal = np.array(goal)
        self.obstacles = obstacles or []
    
    def cost_function(self, waypoints_flat, n_waypoints, dim):
        """代价函数"""
        waypoints = waypoints_flat.reshape(n_waypoints, dim)
        
        # 1. 路径长度
        path_length = sum(
            np.linalg.norm(waypoints[i+1] - waypoints[i])
            for i in range(n_waypoints - 1)
        )
        
        # 2. 平滑性（二阶差分）
        smoothness = 0
        for i in range(1, n_waypoints - 1):
            smoothness += np.linalg.norm(
                waypoints[i+1] - 2 * waypoints[i] + waypoints[i-1]
            )**2
        
        # 3. 避障
        obstacle_cost = 0
        for obs in self.obstacles:
            obs_pos, obs_radius = obs
            for wp in waypoints:
                dist = np.linalg.norm(wp - obs_pos)
                if dist < obs_radius:
                    obstacle_cost += (obs_radius - dist)**2 * 100
        
        return path_length + 0.1 * smoothness + obstacle_cost
    
    def optimize(self, n_waypoints=10):
        """优化轨迹"""
        dim = len(self.start)
        
        # 初始猜测：线性插值
        t = np.linspace(0, 1, n_waypoints)
        initial = np.array([
            self.start + (self.goal - self.start) * ti for ti in t
        ]).flatten()
        
        # 约束：起点和终点固定
        constraints = [
            {'type': 'eq', 'fun': lambda x: x[:dim] - self.start},
            {'type': 'eq', 'fun': lambda x: x[-dim:] - self.goal},
        ]
        
        result = minimize(
            self.cost_function, initial,
            args=(n_waypoints, dim),
            method='SLSQP',
            constraints=constraints
        )
        
        return result.x.reshape(n_waypoints, dim)

# 使用
optimizer = TrajectoryOptimizer(
    start=[0, 0, 0.5],
    goal=[0.5, 0.3, 0.5],
    obstacles=[([0.25, 0.15, 0.5], 0.1)]  # (位置, 半径)
)
optimized_traj = optimizer.optimize(n_waypoints=10)
print(f"优化后轨迹: {optimized_traj.shape}")
```

## 三、MPC 模型预测控制

```python
import numpy as np
from scipy.optimize import minimize

class MPCController:
    """MPC 模型预测控制"""
    def __init__(self, horizon=10, dt=0.1):
        self.horizon = horizon
        self.dt = dt
    
    def dynamics(self, state, control):
        """简化动力学模型（二阶积分器）"""
        # state: [x, y, vx, vy]
        # control: [ax, ay]
        x, y, vx, vy = state
        ax, ay = control
        
        new_x = x + vx * self.dt
        new_y = y + vy * self.dt
        new_vx = vx + ax * self.dt
        new_vy = vy + ay * self.dt
        
        return np.array([new_x, new_y, new_vx, new_vy])
    
    def cost_function(self, controls_flat, current_state, reference):
        """MPC 代价函数"""
        controls = controls_flat.reshape(self.horizon, 2)
        state = current_state.copy()
        cost = 0
        
        for t in range(self.horizon):
            # 跟踪误差
            pos_error = np.linalg.norm(state[:2] - reference[:2])
            cost += 10.0 * pos_error**2
            
            # 控制能量
            cost += 0.1 * np.sum(controls[t]**2)
            
            # 控制变化率（平滑性）
            if t > 0:
                cost += 1.0 * np.sum((controls[t] - controls[t-1])**2)
            
            # 前向预测
            state = self.dynamics(state, controls[t])
        
        return cost
    
    def compute_control(self, current_state, reference):
        """计算最优控制"""
        # 初始猜测
        x0 = np.zeros(self.horizon * 2)
        
        # 控制约束
        bounds = [(-5, 5)] * (self.horizon * 2)  # 加速度限制
        
        result = minimize(
            self.cost_function, x0,
            args=(current_state, reference),
            method='SLSQP',
            bounds=bounds
        )
        
        controls = result.x.reshape(self.horizon, 2)
        return controls[0]  # 返回第一步控制

# 仿真测试
mpc = MPCController(horizon=10, dt=0.1)

current_state = np.array([0.0, 0.0, 0.0, 0.0])  # 初始状态
reference = np.array([1.0, 0.5, 0.0, 0.0])       # 目标

print("MPC 仿真:")
for step in range(50):
    control = mpc.compute_control(current_state, reference)
    current_state = mpc.dynamics(current_state, control)
    
    if step % 10 == 0:
        pos_error = np.linalg.norm(current_state[:2] - reference[:2])
        print(f"  Step {step}: pos={np.round(current_state[:2], 3)}, "
              f"error={pos_error:.4f}, control={np.round(control, 3)}")
```

## 四、阻抗控制实现

```python
class CartesianImpedanceController:
    """笛卡尔空间阻抗控制"""
    def __init__(self, K_pos=100, K_rot=50, D_pos=20, D_rot=10):
        """
        K_pos, K_rot: 位置和旋转刚度
        D_pos, D_rot: 位置和旋转阻尼
        """
        self.K_pos = K_pos * np.eye(3)
        self.K_rot = K_rot * np.eye(3)
        self.D_pos = D_pos * np.eye(3)
        self.D_rot = D_rot * np.eye(3)
    
    def compute(self, current_pos, current_vel, target_pos, target_vel,
                current_rot=None, target_rot=None, current_omega=None):
        """
        计算阻抗控制力
        
        F = K * (x_d - x) + D * (dx_d - dx)
        """
        # 位置阻抗
        pos_error = target_pos - current_pos
        vel_error = target_vel - current_vel
        F_pos = self.K_pos @ pos_error + self.D_pos @ vel_error
        
        # 旋转阻抗（简化）
        F_rot = np.zeros(3)
        if current_rot is not None and target_rot is not None:
            # 使用旋转矩阵的差异
            rot_error_mat = target_rot @ current_rot.T
            # 转为旋转向量
            from scipy.spatial.transform import Rotation
            rot_error = Rotation.from_matrix(rot_error_mat).as_rotvec()
            omega_error = -current_omega if current_omega is not None else np.zeros(3)
            F_rot = self.K_rot @ rot_error + self.D_rot @ omega_error
        
        return F_pos, F_rot

# 仿真测试
controller = CartesianImpedanceController(K_pos=200, D_pos=30)

current_pos = np.array([0.0, 0.0, 0.0])
current_vel = np.array([0.0, 0.0, 0.0])
target_pos = np.array([0.5, 0.0, 0.3])
target_vel = np.array([0.0, 0.0, 0.0])

# 模拟环境刚度
env_stiffness = 500  # 环境刚度

print("阻抗控制仿真:")
dt = 0.01
for step in range(200):
    F_pos, _ = controller.compute(current_pos, current_vel, target_pos, target_vel)
    
    # 简化动力学：m * a = F
    m = 1.0  # 虚拟质量
    acc = F_pos / m
    current_vel += acc * dt
    current_pos += current_vel * dt
    
    if step % 50 == 0:
        error = np.linalg.norm(target_pos - current_pos)
        print(f"  Step {step}: pos={np.round(current_pos, 4)}, error={error:.4f}")
```

## 五、运动重定向（Retargeting）

### 5.1 什么是 Retargeting

```
运动重定向 (Retargeting):
  将人体运动捕捉数据映射到机器人关节空间

应用场景：
  1. 从人体动作学习机器人步态
  2. 从人类演示学习操作策略
  3. 创建自然的人形机器人运动

挑战：
  1. 自由度不同（人体 >> 机器人）
  2. 关节限位不同
  3. 运动学结构不同
  4. 动力学约束
```

### 5.2 Retargeting 实现

```python
import numpy as np
from scipy.optimize import minimize

class MotionRetargeter:
    """运动重定向：人体 → 机器人"""
    def __init__(self, robot_joint_limits, robot_link_lengths):
        """
        robot_joint_limits: [(min, max), ...] 每个关节的角度限制
        robot_link_lengths: [l1, l2, ...] 连杆长度
        """
        self.joint_limits = robot_joint_limits
        self.link_lengths = robot_link_lengths
        self.num_joints = len(robot_joint_limits)
    
    def retarget(self, human_keypoints, human_to_robot_map):
        """
        将人体关键点重定向到机器人关节角度
        
        human_keypoints: dict {joint_name: 3D position}
        human_to_robot_map: dict {human_joint: robot_joint_indices}
        """
        def objective(robot_angles_flat):
            robot_angles = robot_angles_flat.reshape(self.num_joints, -1)
            
            # 计算机器人末端位置
            robot_positions = self.forward_kinematics(robot_angles)
            
            # 计算与人体关键点的差异
            error = 0
            for human_joint, robot_idx in human_to_robot_map.items():
                if human_joint in human_keypoints:
                    human_pos = human_keypoints[human_joint]
                    robot_pos = robot_positions[robot_idx]
                    error += np.sum((robot_pos - human_pos)**2)
            
            return error
        
        # 约束：关节限位
        bounds = self.joint_limits
        
        # 初始猜测
        x0 = np.zeros(self.num_joints * 3)  # 假设每关节3个参数
        
        result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds)
        
        return result.x.reshape(self.num_joints, -1)
    
    def forward_kinematics(self, joint_angles):

                self.link_lengths[i] * np.cos(angle),
                self.link_lengths[i] * np.sin(angle),
                0
            ])
            positions[i] = pos
        
        return positions
```
