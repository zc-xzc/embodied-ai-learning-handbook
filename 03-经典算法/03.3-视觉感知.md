```python
from ultralytics import YOLO
import cv2

class ObjectDetector:
    """目标检测器"""
    def __init__(self, model_path="yolov8n.pt"):
        self.model = YOLO(model_path)
    
    def detect(self, image_path, conf_threshold=0.5):
        """检测图像中的物体"""
        results = self.model(image_path, conf=conf_threshold)
        
        detections = []
        for result in results:
            boxes = result.boxes
            for box in boxes:
                detection = {
                    "class": self.model.names[int(box.cls)],
                    "confidence": float(box.conf),
                    "bbox": box.xyxy[0].tolist(),  # [x1, y1, x2, y2]
                }
                detections.append(detection)
        
        return detections
    
    def detect_for_robot(self, image_path, target_class=None):
        """为机器人操作提取目标信息"""
        detections = self.detect(image_path)
        
        if target_class:
            detections = [d for d in detections if d["class"] == target_class]
        
        if not detections:
            return None
        
        # 选择置信度最高的
        best = max(detections, key=lambda d: d["confidence"])
        
        # 计算中心点
        x1, y1, x2, y2 = best["bbox"]
        center_x = (x1 + x2) / 2
        center_y = (y1 + y2) / 2
        
        return {
            "class": best["class"],
            "confidence": best["confidence"],
            "center": (center_x, center_y),
            "bbox": best["bbox"],
            "size": (x2 - x1, y2 - y1)
        }
```

### 1.3 语义分割

```python
"""
语义分割：为图像中每个像素分配类别标签

在机器人中的应用：
  - 精确的物体边界（用于抓取规划）
  - 可通行区域识别（导航）
  - 场景理解（任务规划）

常用模型：
  - Mask2Former: 通用分割
  - SAM (Segment Anything): 零样本分割
  - FastSAM: 轻量级 SAM
"""
from transformers import SamModel, SamProcessor
from PIL import Image
import torch
import numpy as np

class SceneSegmentor:
    """使用 SAM 进行场景分割"""
    def __init__(self):
        self.model = SamModel.from_pretrained("facebook/sam-vit-base")
        self.processor = SamProcessor.from_pretrained("facebook/sam-vit-base")
    
    def segment_point(self, image, point):
        """根据点提示分割物体"""
        inputs = self.processor(image, input_points=[point], return_tensors="pt")
        
        with torch.no_grad():
            outputs = self.model(**inputs)
        
        masks = self.processor.image_processor.post_process_masks(
            outputs.pred_masks.cpu(),
            inputs["original_sizes"].cpu(),
            inputs["reshaped_input_sizes"].cpu()
        )
        
        return masks[0]  # 返回分割掩码

# 语义分割在机器人操作中的价值
```

### 2.1 深度图

```python
import numpy as np

class DepthImageProcessor:
    """深度图处理"""
    def __init__(self, camera_intrinsics=None):
        """
        camera_intrinsics: 相机内参 [fx, fy, cx, cy]
        """
        self.fx = 525.0  # 默认值（RealSense 类似）
        self.fy = 525.0
        self.cx = 320.0
        self.cy = 240.0
    
    def depth_to_pointcloud(self, depth_image):
        """将深度图转换为点云"""
        h, w = depth_image.shape
        
        # 创建像素坐标网格
        u = np.arange(w)
        v = np.arange(h)
        u, v = np.meshgrid(u, v)
        
        # 有效深度
        valid = depth_image > 0
        
        # 计算 3D 坐标
        z = depth_image[valid]
        x = (u[valid] - self.cx) * z / self.fx
        y = (v[valid] - self.cy) * z / self.fy
        
        pointcloud = np.stack([x, y, z], axis=-1)
        return pointcloud
    
    def compute_object_distance(self, depth_image, bbox):
        """计算 bbox 区域内物体的平均距离"""
        x1, y1, x2, y2 = map(int, bbox)
        roi = depth_image[y1:y2, x1:x2]
        valid = roi > 0
        if valid.any():
            return roi[valid].mean()
        return None
```

### 2.2 点云基础

```python
"""
点云处理常用工具

安装：
  pip install open3d
"""

class PointCloudUtils:
    """点云处理工具"""
    
    @staticmethod
    def downsample(points, voxel_size=0.01):
        """体素下采样"""
        # 量化到体素网格
        voxel_indices = np.floor(points / voxel_size).astype(int)
        # 每个体素内取平均
        unique_voxels, inverse = np.unique(voxel_indices, axis=0, return_inverse=True)
        downsampled = np.zeros((len(unique_voxels), 3))
        counts = np.zeros(len(unique_voxels))
        np.add.at(downsampled, inverse, points)
        np.add.at(counts, inverse, 1)
        downsampled /= counts[:, np.newaxis]
        return downsampled
    
    @staticmethod
    def remove_outliers(points, k=20, threshold=2.0):
        """移除离群点（统计方法）"""
        from scipy.spatial import KDTree
        tree = KDTree(points)
        distances, _ = tree.query(points, k=k+1)
        mean_dist = distances[:, 1:].mean(axis=1)
        std_dist = distances[:, 1:].std(axis=1)
        z_scores = np.abs((mean_dist - mean_dist.mean()) / (mean_dist.std() + 1e-8))
        return points[z_scores < threshold]
    
    @staticmethod
    def compute_centroid(points):
        """计算点云质心"""
        return points.mean(axis=0)
    
    @staticmethod
    def compute_bbox(points):
        """计算轴对齐包围盒"""
        min_corner = points.min(axis=0)
        max_corner = points.max(axis=0)
        center = (min_corner + max_corner) / 2
        size = max_corner - min_corner
        return {"center": center, "size": size, 
                "min": min_corner, "max": max_corner}
```

## 三、6D 姿态估计

### 3.1 什么是 6D 姿态

```python
"""
6D 姿态 = 3D 位置 (x, y, z) + 3D 旋转 (roll, pitch, yaw) 或四元数

在机器人操作中的重要性：
  - 知道物体在哪里（位置）
  - 知道物体怎么放（朝向）
  - 决定从哪个角度抓取

表示方法：
  1. 旋转矩阵 R (3x3)
  2. 四元数 q = [w, x, y, z]
  3. 欧拉角 [roll, pitch, yaw]
  4. 轴角表示
  5. 6D 表示 [R[:2, 0], R[:2, 1]] (Procrustes, 更适合学习)
"""
```

### 3.2 6D 姿态估计方法

```python
"""
6D 姿态估计的主要方法：

1. 基于 PnP 的方法
   - 检测 2D 关键点
   - 用 PnP 算法从 2D-3D 对应恢复 6D 位姿
   - 代表：PVNet, PVN3D

2. 基于点云配准的方法
   - 将观测点云与模型点云对齐
   - 代表：ICP, DeepICP

3. 基于深度学习的直接回归
   - 直接从图像/点云回归 6D 位姿
   - 代表：PoseCNN, PoseNet

4. 基于 Transformer 的方法
   - 用 Transformer 处理点云/图像
   - 代表：Transformer6D

5. 基于扩散模型的方法
   - 用扩散模型生成位姿分布
   - 适合多模态位姿（对称物体）
"""
```

### 3.3 姿态估计在抓取中的应用

```python
class GraspPlanner:
    """基于位姿估计的抓取规划"""
    def __init__(self):
        pass
    
    def plan_grasp(self, object_pose, object_size):
        """
        根据物体位姿规划抓取
        
        object_pose: 4x4 齐次变换矩阵
        object_size: (width, height, depth) 物体尺寸
        """
        # 物体位置
        object_pos = object_pose[:3, 3]
        
        # 计算抓取位置（物体中心上方）
        grasp_pos = object_pos + object_pose[:3, 2] * (object_size[2] / 2 + 0.1)
        
        # 计算抓取方向（从上往下）
        # 末端 Z 轴朝下 = 物体 Z 轴朝上
        grasp_rot = object_pose[:3, :3].copy()
        grasp_rot[:, 2] *= -1  # 反转 Z 轴
        
        # 夹爪开口宽度
        gripper_width = object_size[0] * 1.2  # 略大于物体宽度
        
        return {
            "position": grasp_pos,
            "orientation": grasp_rot,
            "gripper_width": gripper_width,
        }
```

## 四、视觉特征提取用于机器人控制

### 4.1 视觉作为策略输入

```python
import torch
import torch.nn as nn

class VisualFeatureExtractor(nn.Module):
    """视觉特征提取器（用于策略输入）"""
    def __init__(self, feature_dim=256, pretrained=True):
        super().__init__()
        
        # 选项1: 简单 CNN
        self.cnn = nn.Sequential(
            nn.Conv2d(3, 32, 8, stride=4), nn.ReLU(),
            nn.Conv2d(32, 64, 4, stride=2), nn.ReLU(),
            nn.Conv2d(64, 64, 3, stride=1), nn.ReLU(),
            nn.Flatten(),
            nn.Linear(64 * 7 * 7, feature_dim), nn.ReLU(),
        )
        
        # 选项2: 使用预训练 CLIP（推荐）
        # from transformers import CLIPModel
        # self.clip = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
        # # 冻结 CLIP 参数
        # for param in self.clip.parameters():
        #     param.requires_grad = False
    
    def forward(self, image):
        """
        image: (B, 3, H, W) 归一化后的图像
        return: (B, feature_dim) 视觉特征
        """
        features = self.cnn(image)
        return features

class VisionConditionedPolicy(nn.Module):
    """视觉条件策略"""
    def __init__(self, visual_dim=256, action_dim=7):
        super().__init__()
        self.visual_encoder = VisualFeatureExtractor(feature_dim=visual_dim)
        
        # 策略网络
        self.policy = nn.Sequential(
            nn.Linear(visual_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 256),
            nn.ReLU(),
            nn.Linear(256, action_dim),
            nn.Tanh()
        )
    
    def forward(self, image):
        features = self.visual_encoder(image)
        action = self.policy(features)
        return action
```

## 五、主动视觉概念

### 5.1 什么是主动视觉

```python
"""
主动视觉 (Active Vision):
  机器人主动改变相机视角以获取更好的视觉信息

与传统视觉的区别：
  传统视觉：被动接收图像，在固定视角下处理
  主动视觉：主动移动相机/机器人，选择最佳观测角度

在机器人操作中的应用：
  1. 遮挡处理：从另一个角度看被遮挡的物体
  2. 精确感知：靠近物体获取高分辨率图像
  3. 探索：在未知环境中主动搜索目标
  4. 视觉伺服：用视觉反馈引导运动

关键问题：
  - 什么时候需要改变视角？
  - 改变到哪个视角？
  - 如何平衡感知和操作的 trade-off？
"""
```

### 5.2 主动视觉策略

```python
class ActiveVisionPolicy:
    """主动视觉策略框架"""
    def __init__(self):
        self.confidence_threshold = 0.7
        self.view_history = []
    
    def should_reposition(self, current_detection):
        """判断是否需要重新定位视角"""
        if current_detection is None:
            return True  # 没检测到，需要找
        
        if current_detection["confidence"] < self.confidence_threshold:
            return True  # 置信度低，需要更好的视角
        
        if current_detection.get("occluded", False):
            return True  # 被遮挡，需要换角度
        
        return False
    
    def select_next_viewpoint(self, current_state, target_info):
        """选择下一个观测视角"""
        if target_info is None:
            # 搜索模式：旋转寻找目标
            return {"action": "rotate", "angle": 30}
        
        # 根据物体位置计算最佳观测角度
        target_pos = target_info["center_3d"]
        
        # 选择能看到物体的角度
        # 简化：从物体上方斜 45 度观察
        viewpoint = {
            "position": target_pos + np.array([0.3, 0.0, 0.3]),
            "look_at": target_pos,
        }
        return viewpoint
```
